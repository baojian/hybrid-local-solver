"""Package the audited optional grid adapter with explicit factory injection."""
import ast
import hashlib
import json
from pathlib import Path

ROOT=Path(__file__).resolve().parent
OUT=ROOT/'deliverables'/'deterministic-rppr'
PACKAGE=OUT/'deterministic_rppr'


def dump(node):
    return ast.dump(node,include_attributes=False)


def main():
    source=(ROOT/'mass_scaled_grid_rppr.py').read_bytes()
    source_hash=hashlib.sha256(source).hexdigest()
    audit=json.loads((ROOT/'mass_scaled_grid_independent_verification.json').read_text())
    assert audit['passed'] and audit['source_sha256']['mass_scaled_grid_rppr.py']==source_hash
    tree=ast.parse(source)
    replacements={'combined_certified_dyadic_rppr':'_combined',
                  'direct_exception_integer_rppr':'_direct',
                  'deterministic_rppr_solver':'_types'}
    imports=0
    for node in tree.body:
        if isinstance(node,ast.ImportFrom) and node.module in replacements:
            if node.module=='direct_exception_integer_rppr':
                assert [a.name for a in node.names]==['DirectExceptionIntegerCorrector','_continuation']
                node.names=node.names[:1]
            node.level,node.module=1,replacements[node.module]
            imports+=1
    assert imports==3
    reference_import=ast.parse('from . import _integer as reference').body[0]
    last_import=max(k for k,node in enumerate(tree.body) if isinstance(node,(ast.Import,ast.ImportFrom)))
    tree.body.insert(last_import+1,reference_import)
    calls=0
    original_func=ast.parse('_continuation(factory)').body[0].value
    for node in ast.walk(tree):
        if isinstance(node,ast.Call) and dump(node.func)==dump(original_func):
            assert not node.keywords
            assert [dump(a) for a in node.args]==[dump(ast.Name(id=name,ctx=ast.Load()))
                                                for name in ('oracle','seed','alpha','rho','epsilon')]
            node.func=ast.Attribute(value=ast.Name(id='reference',ctx=ast.Load()),
                                    attr='solve_rppr_integer',ctx=ast.Load())
            node.keywords=[ast.keyword(arg='corrector_class',value=ast.Name(id='factory',ctx=ast.Load()))]
            calls+=1
    assert calls==1
    rendered=ast.unparse(ast.fix_missing_locations(tree))+'\n'
    compile(rendered,'<packaged mass-scaled grid>','exec')
    # Independently reverse only the prescribed transformations before writing.
    restored=ast.parse(rendered)
    restored.body=[n for n in restored.body if dump(n)!=dump(reference_import)]
    reverse={new:old for old,new in replacements.items()}
    for node in restored.body:
        if isinstance(node,ast.ImportFrom) and node.level==1:
            node.module,node.level=reverse[node.module],0
            if node.module=='direct_exception_integer_rppr':
                node.names.append(ast.alias(name='_continuation'))
    for node in ast.walk(restored):
        if isinstance(node,ast.Call) and dump(node.func)==dump(ast.parse('reference.solve_rppr_integer').body[0].value):
            assert len(node.keywords)==1 and node.keywords[0].arg=='corrector_class'
            assert dump(node.keywords[0].value)==dump(ast.Name(id='factory',ctx=ast.Load()))
            node.func=original_func
            node.keywords=[]
    assert dump(restored)==dump(ast.parse(source))
    previous={p.name:hashlib.sha256(p.read_bytes()).hexdigest()
              for p in PACKAGE.glob('*.py') if p.name!='_mass_grid.py'}
    (PACKAGE/'_mass_grid.py').write_text(rendered)
    manifest=dict(source='mass_scaled_grid_rppr.py',source_sha256=source_hash,
        package_sha256=hashlib.sha256(rendered.encode()).hexdigest(),
        imports=replacements, removed_research_import='_continuation',
        added_import='from . import _integer as reference',
        factory_rewrite=dict(old='_continuation(factory)(oracle, seed, alpha, rho, epsilon)',
            new='reference.solve_rppr_integer(oracle, seed, alpha, rho, epsilon, corrector_class=factory)',
            changed_calls=calls),
        approved_by='mass_scaled_grid_independent_verification.json',
        approval_sha256=hashlib.sha256((ROOT/'mass_scaled_grid_independent_verification.json').read_bytes()).hexdigest(),
        existing_module_sha256_at_build=previous,
        note='Entire module AST otherwise identical. No FunctionType cloning, numerical step/repair change, export/default change, or finalizer modification is made by this builder.')
    (OUT/'MASS_GRID_TRANSCRIPTION.json').write_text(json.dumps(manifest,indent=2)+'\n')
    print(json.dumps(dict(source_sha256=source_hash,package_sha256=manifest['package_sha256'],
                         relative_imports=imports,factory_rewrites=calls),indent=2))


if __name__=='__main__':main()
