"""Fixed deterministic manifest for the final optional combined backend.

Run only after composition is declared stable. Manifest generation does not
import or inspect the combined implementation. No whole virtual graph is
constructed. Every completed output is independently certified separately.
"""
import argparse
import hashlib
import importlib
import json
import platform
import signal
import sys
from dataclasses import asdict
from fractions import Fraction as F
from pathlib import Path
from time import perf_counter,process_time
from statistics import median

from benchmark_practical_rppr import CASES,Case,CountingOracle,VirtualGraph,json_value
from independent_sparse_certificate import certify_sparse_output

ROOT=Path(__file__).resolve().parent
LOCAL_NAMES=('star-billion','star-huge-labels','path-local','cycle-local','grid-local','cube-local',
             'tree-clique-boundary','path-alpha-1e3')
MANIFEST=tuple(next(case for case in CASES if case.name==name) for name in LOCAL_NAMES)+tuple(
    Case(f'path-sweep-alpha-1over{den}','path',4096,0,1,F(1,den),F(1,256),F(1,10**8),'sweep')
    for den in (4,16,64,256,1024,4096))
MATCHED=('grid-local','path-sweep-alpha-1over64')
SOURCE_NAMES=('study_final_combined_rppr.py','combined_certified_dyadic_rppr.py',
    'binomial_block_dyadic_rppr.py','early_stop_source_energy_rppr.py',
    'source_energy_dyadic_rppr.py','integer_dyadic_rppr_solver.py',
    'integer_clipped_reporter.py','independent_sparse_certificate.py',
    'deterministic_avl_containers.py','benchmark_practical_rppr.py')


def write_manifest(args):
    keys=[(c.family,c.size,c.auxiliary,c.seed,c.alpha,c.rho,c.epsilon) for c in MANIFEST]
    assert len(keys)==len(set(keys))==14
    result=dict(deterministic=True,cases=[json_value(asdict(c)) for c in MANIFEST],
        matched_abba_cases=list(MATCHED),large_graphs_implicit=True,
        tree_clique_interpretation='one depth-5 binary tree, every leaf attached to a distinct port of one shared 16384-vertex clique',
        sweep_interpretation='4096-vertex endpoint-seeded path; fixed rho=1/256 and epsilon=1e-8',
        guards=dict(solver_wall_seconds=args.seconds_per_case,first_oracle_adjacency_entries=args.entry_limit,
                    verification_wall_seconds=args.verification_seconds,verification_oracle_adjacency_entries=args.entry_limit),
        guard_scope='Entry limits cover actual oracle-supplied entries; repeated cached work is measured and subject to the wall-time guard.')
    (ROOT/'final_combined_rppr_manifest.json').write_text(json.dumps(result,indent=2)+'\n')
    return result


def timed_call(call,seconds):
    def expired(signum,frame):
        raise TimeoutError('Explicit wall-time guard reached')
    old=signal.signal(signal.SIGALRM,expired)
    signal.setitimer(signal.ITIMER_REAL,seconds)
    start_wall,start_cpu=perf_counter(),process_time()
    result,error=None,None
    try:
        result=call()
    except (TimeoutError,RuntimeError) as exc:
        error=dict(type=type(exc).__name__,message=str(exc))
    finally:
        cpu,wall=process_time()-start_cpu,perf_counter()-start_wall
        signal.setitimer(signal.ITIMER_REAL,0)
        signal.signal(signal.SIGALRM,old)
    return result,cpu,wall,error


def verify(case,output,args):
    graph=VirtualGraph(case.family,case.size,case.auxiliary)
    oracle=CountingOracle(graph,args.entry_limit)
    certificate,cpu,wall,error=timed_call(lambda:certify_sparse_output(oracle,case.seed,case.alpha,case.rho,output),args.verification_seconds)
    result=dict(cpu_seconds=cpu,wall_seconds=wall,oracle_counts=oracle.snapshot())
    if error is not None:
        result.update(status='guarded_failure',error=error)
        return result
    mass=sum((value*degree for _,value,degree in output),F(0))
    volume=sum(degree for _,value,degree in output)
    assert volume<=1/case.rho and 0<=mass<=1
    result.update(status='certified' if certificate.meets(case.epsilon) else 'bound_inconclusive',
        certificate=json_value(asdict(certificate)),bound_meets_epsilon=certificate.meets(case.epsilon),
        output_volume=volume,output_mass=str(mass),
        provenance='Independent recomputation of a valid subgradient norm; strong-convexity gap bound, with upward integer squared-norm rounding.')
    return result


def run(case,backend,functions,args):
    graph=VirtualGraph(case.family,case.size,case.auxiliary)
    oracle=CountingOracle(graph,args.entry_limit)
    answer,cpu,wall,error=timed_call(lambda:functions[backend](oracle,case.seed,case.alpha,case.rho,case.epsilon),args.seconds_per_case)
    row=dict(case=json_value(asdict(case)),backend=backend,graph_vertices=str(graph.n),
        solver_cpu_seconds=cpu,solver_wall_seconds=wall,solver_oracle_counts=oracle.snapshot(),
        provenance='Fresh sequential run by this study driver')
    if error is not None:
        row.update(status='guarded_failure',error=error)
        return row
    assert 0<=answer.objective_gap_bound<=case.epsilon
    checkpoints=[stage.corrector_metrics.get('checkpoints',{}) for stage in answer.stages]
    fields={key for item in checkpoints for key in item}
    checkpoint_totals={key:sum(item.get(key,0) for item in checkpoints) for key in sorted(fields)}
    numeric_fields={key for stage in answer.stages for key,value in stage.corrector_metrics.items() if type(value) is int}
    numeric_totals={key:sum(stage.corrector_metrics.get(key,0) for stage in answer.stages) for key in sorted(numeric_fields)}
    reporter_fields={key for stage in answer.stages for key in stage.corrector_metrics.get('reporter',{})}
    reporter_totals={key:sum(stage.corrector_metrics.get('reporter',{}).get(key,0) for stage in answer.stages) for key in sorted(reporter_fields)}
    corrector_reads=sum(stage.corrector_metrics['scanned_adjacency_entries'] for stage in answer.stages)
    pg_reads=answer.metrics.repair_scanned_adjacency_entries
    checkpoint_reads=checkpoint_totals.get('cached_adjacency_entries',0)
    exact=json_value(asdict(answer))
    row.update(status='completed',iterations=sum(stage.iterations for stage in answer.stages),stages=len(answer.stages),
        output_records=len(answer.output),output=json_value(answer.output),
        returned_objective_bound=str(answer.objective_gap_bound),
        exact_solver_result=exact,signature_sha256=hashlib.sha256(json.dumps(exact,sort_keys=True).encode()).hexdigest(),
        corrector_integer_metric_sums=numeric_totals,checkpoint_totals=checkpoint_totals,reporter_totals=reporter_totals,
        adjacency_work=dict(corrector=corrector_reads,terminal_pg=pg_reads,checkpoint_cached=checkpoint_reads,
                            total=corrector_reads+pg_reads+checkpoint_reads),
        checkpoint_scalar_point_requests=sum(checkpoint_totals.get(key,0) for key in
            ('checker_point_lookups','checker_point_writes','cache_degree_lookups','cache_row_lookups','candidate_materialization_scalar_lookups')),
        independent_verification=verify(case,answer.output,args))
    return row


def main():
    parser=argparse.ArgumentParser()
    parser.add_argument('--manifest-only',action='store_true')
    parser.add_argument('--seconds-per-case',type=int,default=120)
    parser.add_argument('--verification-seconds',type=int,default=30)
    parser.add_argument('--entry-limit',type=int,default=1000000)
    parser.add_argument('--case',action='append')
    parser.add_argument('--reuse-hub-record')
    parser.add_argument('--output',default='final_combined_rppr_study.json')
    args=parser.parse_args()
    manifest=write_manifest(args)
    if args.manifest_only:
        print('Fixed 14-case manifest saved; combined code was not imported.')
        return
    combined=importlib.import_module('combined_certified_dyadic_rppr')
    fixed=importlib.import_module('source_energy_dyadic_rppr')
    functions=dict(combined=combined.solve_rppr_combined,fixed_source=fixed.solve_rppr_source_energy)
    hashes={name:hashlib.sha256((ROOT/name).read_bytes()).hexdigest() for name in SOURCE_NAMES}
    report=dict(manifest=manifest,python=sys.version,platform=platform.platform(),source_sha256=hashes,
        timing_scope='Solve plus shared counting oracle; graph construction, independent verification and serialization excluded.',
        accounting_note='Checkpoint cached reads are included in total adjacency work. Scalar logical requests/record visits are reported separately; no claim to a literal instruction trace.',
        measurements_are_not_runtime_asymptotics=True,runs=[],comparisons=[])
    output=ROOT/args.output
    def save():output.write_text(json.dumps(report,indent=2)+'\n')
    chosen=[case for case in MANIFEST if not args.case or case.name in args.case]
    for case in chosen:
        if case.name=='star-billion' and args.reuse_hub_record:
            loaded=json.loads(Path(args.reuse_hub_record).read_text())
            for reused in (loaded if isinstance(loaded,list) else [loaded]):
                assert reused['case']==json_value(asdict(case)) and reused['backend']=='combined'
                reconstructed=tuple((int(i),F(value),int(d)) for i,value,d in reused['output'])
                reused['independent_verification']=verify(case,reconstructed,args)
                report['runs'].append(reused);save()
            print('REUSED and independently verified star-billion',flush=True)
            continue
        modes=('fixed_source','combined','combined','fixed_source') if case.name in MATCHED else ('combined',)
        rows=[]
        for mode in modes:
            print(f'START {case.name} {mode}',flush=True)
            row=run(case,mode,functions,args)
            rows.append(row);report['runs'].append(row);save()
            print(json.dumps(dict(case=case.name,backend=mode,status=row['status'],cpu=row['solver_cpu_seconds'],wall=row['solver_wall_seconds'],iterations=row.get('iterations'),adjacency=row.get('adjacency_work'),verification=row.get('independent_verification',{}).get('status'))),flush=True)
        if case.name in MATCHED and all(row['status']=='completed' for row in rows):
            for mode in ('fixed_source','combined'):
                signatures=[row['signature_sha256'] for row in rows if row['backend']==mode]
                assert signatures[0]==signatures[1]
            med={mode:{clock:median(row[clock] for row in rows if row['backend']==mode) for clock in ('solver_cpu_seconds','solver_wall_seconds')} for mode in ('fixed_source','combined')}
            report['comparisons'].append(dict(case=case.name,medians=med,
                ratio_fixed_over_combined={clock:med['fixed_source'][clock]/med['combined'][clock] for clock in ('solver_cpu_seconds','solver_wall_seconds')}))
            save()
    report['finished']=True
    report['source_hashes_still_match']=all(hashlib.sha256((ROOT/name).read_bytes()).hexdigest()==digest for name,digest in hashes.items())
    save()

if __name__=='__main__':main()
