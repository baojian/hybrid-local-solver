"""Audit a frozen review ZIP and run only extracted data/help commands."""
import argparse
import base64
import csv
import hashlib
import io
import json
import re
import stat
import subprocess
import sys
import zipfile
from pathlib import Path, PurePosixPath
from urllib.parse import unquote, urlsplit

ROOT=Path(__file__).resolve().parent


def digest(data):
    return hashlib.sha256(data).hexdigest()


RUNNER=r'''
import builtins,json,runpy,sys
from pathlib import Path
bundle=Path(sys.argv[1]).resolve()
target=bundle/sys.argv[2]
sys.argv=[str(target),*sys.argv[3:]]
allowed=set(sys.stdlib_module_names)|set(sys.builtin_module_names)|{'deterministic_rppr','tests','__main__'}
original=builtins.__import__
def guarded(name,globals=None,locals=None,fromlist=(),level=0):
    if level==0 and name.split('.')[0] not in allowed:
        raise AssertionError('Research or external import: '+name)
    return original(name,globals,locals,fromlist,level)
builtins.__import__=guarded
def no_solver(frame,event,arg):
    if event=='call' and '/deterministic_rppr/' in frame.f_code.co_filename:
        if frame.f_code.co_name.startswith('solve_') or frame.f_code.co_name=='step':
            raise AssertionError('Unexpected solver invocation during data/help audit')
sys.setprofile(no_solver)
try:
    runpy.run_path(str(target),run_name='__main__')
finally:
    sys.setprofile(None)
    origins={}
    for name,module in tuple(sys.modules.items()):
        if name=='deterministic_rppr' or name.startswith('deterministic_rppr.') or name=='tests' or name.startswith('tests.'):
            path=Path(module.__file__).resolve()
            assert path.is_relative_to(bundle),(name,path)
            origins[name]=str(path.relative_to(bundle))
    assert len([n for n in origins if n=='deterministic_rppr' or n.startswith('deterministic_rppr.')])==20
    print('ISOLATION='+json.dumps(dict(origins=origins,solver_calls=0,research_imports_blocked=True)))
'''


def main():
    parser=argparse.ArgumentParser()
    parser.add_argument('--name',default='conjecture2_review_bundle')
    parser.add_argument('--audit-dir',default='review_bundle_audit')
    parser.add_argument('--expected-sha',required=True)
    args=parser.parse_args()
    bundle=ROOT/'deliverables'/args.name
    archive=bundle.with_suffix('.zip')
    out=ROOT/args.audit_dir
    assert not out.exists()
    assert bundle.is_dir() and not bundle.is_symlink()
    archive_bytes=archive.read_bytes()
    assert digest(archive_bytes)==args.expected_sha
    report=dict(archive=dict(path=str(archive),sha256=digest(archive_bytes),bytes=len(archive_bytes)),
                defects=[],commands=[],scope='Integrity, provenance, navigation, and extracted data/help only; no solver suites rerun')
    out.mkdir()
    extraction=out/'extracted'
    extraction.mkdir()
    with zipfile.ZipFile(io.BytesIO(archive_bytes)) as z:
        infos=z.infolist()
        names=[i.filename for i in infos]
        assert len(names)==len(set(names))==len({n.casefold() for n in names})
        assert z.testzip() is None
        for info in infos:
            path=PurePosixPath(info.filename)
            assert not path.is_absolute() and path.parts[0]==args.name
            assert all(part not in ('','.','..') for part in path.parts)
            assert '\\' not in info.filename and ':' not in info.filename and '\x00' not in info.filename
            assert str(path)==info.filename and not info.is_dir()
            mode=info.external_attr>>16
            assert stat.S_ISREG(mode) and not stat.S_ISLNK(mode)
            assert not info.flag_bits&1
            target=extraction.joinpath(*path.parts)
            assert target.resolve().is_relative_to(extraction.resolve())
            target.parent.mkdir(parents=True,exist_ok=True)
            target.write_bytes(z.read(info))
        extracted=extraction/args.name
        manifest_bytes=z.read(args.name+'/BUNDLE_MANIFEST.json')
        manifest=json.loads(manifest_bytes)
        assert manifest['default_solver']=='solve_fast' and manifest['no_randomized_algorithm']
        assert manifest['source_material_copied'] is False
        expected=set(manifest['files'])|{'BUNDLE_MANIFEST.json'}
        assert set(names)=={args.name+'/'+n for n in expected}
        disk_files={str(p.relative_to(bundle)) for p in bundle.rglob('*') if p.is_file()}
        assert disk_files==expected
        for name in expected:
            payload=z.read(args.name+'/'+name)
            assert (bundle/name).read_bytes()==payload==(extracted/name).read_bytes()
            assert not (bundle/name).is_symlink()
            if name!='BUNDLE_MANIFEST.json':
                item=manifest['files'][name]
                assert len(payload)==item['bytes'] and digest(payload)==item['sha256'],name
                if item['source']!='generated by build_review_bundle.py':
                    assert Path(item['source']).resolve().is_relative_to(ROOT)
        report['bundle_manifest']=dict(sha256=digest(manifest_bytes),hashed_files=len(manifest['files']),
            total_files=len(expected),uncompressed_bytes=sum(i.file_size for i in infos),safe_regular_paths=True,
            crc_checked=True,directory_zip_extraction_identical=True)

    package=extracted/'deliverables'/'deterministic-rppr'
    pm=json.loads((package/'PACKAGE_MANIFEST.json').read_text())
    current=ROOT/'deliverables'/'deterministic-rppr'
    assert (package/'PACKAGE_MANIFEST.json').read_bytes()==(current/'PACKAGE_MANIFEST.json').read_bytes()
    assert len(pm['module_sha256'])==20 and pm['default']=='solve_fast'
    for name,expected in pm['module_sha256'].items():
        assert digest((package/name).read_bytes())==expected==digest((current/name).read_bytes())
    for name,expected in pm['validation_sha256'].items():
        assert digest((extracted/name).read_bytes())==expected==digest((ROOT/name).read_bytes())
        record=json.loads((extracted/name).read_text())
        assert record.get('passed') is True or record.get('successful') is True or record.get('status')=='passed'
    report['package']=dict(modules=20,referenced_validation_hashes=len(pm['validation_sha256']),default='solve_fast')
    wheels=list((extracted/'install').glob('*.whl'))
    assert len(wheels)==1
    wheel=wheels[0]
    wheel_verification=json.loads((extracted/'final_offline_wheel_independent_verification.json').read_text())
    assert digest(wheel.read_bytes())==wheel_verification['wheel']['sha256']
    with zipfile.ZipFile(wheel) as z:
        assert z.testzip() is None
        names=z.namelist()
        assert len(names)==len(set(names))
        assert {n for n in names if n.endswith('.py')}==set(pm['module_sha256'])
        for name,expected in pm['module_sha256'].items():
            assert digest(z.read(name))==expected
        record='deterministic_rppr-0.1.0.dist-info/RECORD'
        rows=list(csv.reader(io.StringIO(z.read(record).decode())))
        assert len(rows)==len({r[0] for r in rows}) and {r[0] for r in rows}==set(names)
        for name,checksum,size in rows:
            data=z.read(name)
            if name==record:
                assert checksum==size==''
            else:
                assert checksum=='sha256='+base64.urlsafe_b64encode(hashlib.sha256(data).digest()).decode().rstrip('=')
                assert size==str(len(data))
        report['wheel']=dict(sha256=digest(wheel.read_bytes()),modules=20,record_checksums=len(rows)-1)

    compiled=json.loads((extracted/'proof_compile_manifest.json').read_text())
    visual=json.loads((extracted/'proof_visual_review.json').read_text())
    texnames=('deterministic_conjecture2.tex','practical_refinements_appendix.tex','practical_schedules_appendix.tex')
    for name in texnames:
        assert digest((extracted/name).read_bytes())==compiled['source_sha256'][name]==digest((ROOT/name).read_bytes())
    pdf=extracted/'output/pdf/deterministic_conjecture2.pdf'
    assert digest(pdf.read_bytes())==compiled['pdf_sha256']==visual['pdf_sha256']
    assert pdf.stat().st_size==compiled['pdf_bytes']
    assert compiled['status']=='compiled_without_warnings' and compiled['warnings']==[]
    assert compiled['compile_passes']==2 and visual['status']=='passed'
    assert visual['pages_reviewed']==list(range(1,compiled['pages']+1))
    assert digest((ROOT/'output/pdf/deterministic_conjecture2.log').read_bytes())==compiled['log_sha256']
    for name,expected in visual['page_png_sha256'].items():
        assert digest((ROOT/'output/rendered_final'/name).read_bytes())==expected
    from pypdf import PdfReader
    assert len(PdfReader(pdf).pages)==compiled['pages']==20
    inputs=re.findall(r'\\input\{([^}]+)\}',(extracted/texnames[0]).read_text())
    assert [name+'.tex' for name in inputs]==list(texnames[1:])
    report['proof']=dict(tex_hashes=compiled['source_sha256'],pdf_sha256=compiled['pdf_sha256'],pages=20,
        compile_log_hash_matches=True,reviewed_render_hashes=20,
        render_scope='Matches current fresh compile log and all reviewed page PNG hashes; no new compilation or visual judgment')

    links=[]
    for path in extracted.rglob('*.md'):
        text=path.read_text()
        targets=re.findall(r'\]\(([^)]+)\)',text)+re.findall(r'^\[[^\]]+\]:\s*(\S+)',text,re.M)
        for target in targets:
            parsed=urlsplit(target.strip('<>'))
            if parsed.scheme or target.startswith('#'):
                continue
            resolved=(path.parent/unquote(parsed.path)).resolve()
            okay=resolved.is_relative_to(extracted.resolve()) and resolved.exists()
            item=dict(document=str(path.relative_to(extracted)),target=target,portable=okay)
            links.append(item)
            if not okay:
                report['defects'].append(dict(kind='nonportable_markdown_link',**item))
    labels=set()
    for name in texnames:
        labels.update(re.findall(r'\\label\{([^}]+)\}',(extracted/name).read_text()))
    maptext=(extracted/'proof_to_implementation_map.md').read_text()
    references=set(re.findall(r'\b(?:eq|lem|app):[A-Za-z0-9-]+',maptext))
    assert references<=labels,sorted(references-labels)
    report['navigation']=dict(explicit_relative_links=links,proof_map_tex_labels_checked=len(references),
        key_documents=('START_HERE.md','proof_to_implementation_map.md'),
        historical_commands='START_HERE explicitly distinguishes original-workspace historical commands from bundled portable commands')

    runner=out/'isolated_runner.py'
    runner.write_text(RUNNER)
    tasks=[('data',['deliverables/deterministic-rppr/validation/run_full.py','--check-data','--output',str(out/'manifest-check.json')]),
           ('validation-help',['deliverables/deterministic-rppr/validation/run_full.py','--help']),
           ('benchmark-help',['deliverables/deterministic-rppr/benchmarks/run.py','--help'])]
    for name,arguments in tasks:
        command=[sys.executable,'-I','-S','-B',str(runner),str(extracted),*arguments]
        result=subprocess.run(command,cwd='/',capture_output=True,text=True,timeout=30)
        (out/(name+'.log')).write_text(result.stdout+result.stderr)
        assert result.returncode==0,result.stdout+result.stderr
        isolation=[line for line in result.stdout.splitlines() if line.startswith('ISOLATION=')]
        assert len(isolation)==1
        data=json.loads(isolation[0].split('=',1)[1])
        assert data['solver_calls']==0 and data['research_imports_blocked']
        report['commands'].append(dict(argv=command,returncode=0,isolation=data))
    data=json.loads((out/'manifest-check.json').read_text())
    assert data['status']=='data_verified' and data['solver_cases_run']==0
    assert data['manifest_case_count']==183 and data['completed']==[]
    assert data['source_hashes_before']==data['source_hashes_after']
    report['data_check']=dict(status=data['status'],manifest_case_count=183,solver_cases_run=0,
                              manifest_sha256=data['manifest_sha256'])
    assert archive.read_bytes()==archive_bytes
    report['passed']=not report['defects']
    report['integrity_and_execution_passed']=True
    report['audit_source_sha256']=digest(Path(__file__).read_bytes())
    (out/'verification.json').write_text(json.dumps(report,indent=2)+'\n')
    (ROOT/'review_bundle_independent_audit.json').write_text(json.dumps(report,indent=2)+'\n')
    print(json.dumps(report,indent=2))


if __name__=='__main__':
    main()
