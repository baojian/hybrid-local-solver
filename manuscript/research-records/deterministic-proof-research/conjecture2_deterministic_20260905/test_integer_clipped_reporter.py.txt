"""Independent exact dense tests for the integer two-tree reporter."""

import json
import unittest
from dataclasses import asdict
from fractions import Fraction as F
from itertools import permutations, product
from pathlib import Path
from unittest.mock import patch

import integer_clipped_reporter as production


LOG = dict(projections=0, binding_caps=0, upper_binding_cases=0,
           mutation_states=0, checked_nodes=0, fixtures=[])


def dense_projection(records, scale, denominator, shift, upper, cap, grid):
    raw = [(int(vertex), F((scale if kind == 0 else 1)*n, denominator*d)-shift, d)
           for vertex, (kind, n, d) in records]

    def mass(multiplier):
        return sum((d*min(upper, max(F(0), value-multiplier))
                    for _, value, d in raw), F(0))

    active = mass(F(0)) > cap
    multiplier = F(0)
    if active:
        breaks = sorted({F(0)} | {point for _, value, _ in raw
                                 for point in (value, value-upper) if point > 0})
        for index, high in enumerate(breaks):
            high_mass = mass(high)
            if high_mass == cap:
                multiplier = high
                break
            if high_mass < cap:
                low = breaks[index-1]
                low_mass = mass(low)
                slope = (low_mass-high_mass)/(high-low)
                multiplier = low+(low_mass-cap)/slope
                break
        else:
            raise AssertionError('Dense breakpoint scan did not find its root')
    ideals = {i:min(upper, max(F(0), value-multiplier)) for i, value, _ in raw}
    counts = {}
    for i, value in ideals.items():
        quotient = value/grid
        count = quotient.numerator//quotient.denominator
        if count:
            counts[i] = count
    return counts, ideals, active, raw, mass


def validate(reporter):
    seen = []
    for kind, tree in ((0, reporter.base), (1, reporter.exception)):
        def visit(node):
            if node is None:
                return [], 0, 0, 0
            left, lh, ld, ln = visit(node.left)
            right, rh, rd, rn = visit(node.right)
            record = (node.vertex, (kind, node.numerator, node.degree))
            ordered = left+[record]+right
            expected = sorted(ordered, key=lambda item:(F(item[1][1], item[1][2]), item[0]))
            assert ordered == expected
            assert node.count == len(ordered)
            assert node.height == 1+max(lh, rh)
            assert abs(lh-rh) <= 1
            assert node.degree_sum == ld+node.degree+rd
            assert node.numerator_sum == ln+node.numerator+rn
            assert all(isinstance(value, int) for value in (
                node.vertex, node.numerator, node.degree, node.count,
                node.height, node.degree_sum, node.numerator_sum))
            LOG['checked_nodes'] += 1
            return ordered, node.height, node.degree_sum, node.numerator_sum
        records, _, _, _ = visit(tree.root)
        seen.extend(records)
        for rank, record in enumerate(records):
            n, d, vertex = tree.rank_record(rank)
            assert (vertex, (kind, n, d)) == record
    assert sorted(seen) == list(reporter.records.items())
    assert len({int(vertex) for vertex, _ in seen}) == len(reporter)
    LOG['mutation_states'] += 1


class HashForbiddenInt(int):
    def __hash__(self):
        raise AssertionError('Reporter hashed a vertex ID')


class IntegerReporterTests(unittest.TestCase):
    def check_projection(self, records, scale=3, denominator=7, shift=F(1, 11),
                         upper=F(3, 7), cap=F(1), grid=F(1, 64), reporter=None):
        if reporter is None:
            reporter = production.IntegerClippedReporter()
            for vertex, (kind, numerator, degree) in records:
                (reporter.set_base if kind == 0 else reporter.set_exception)(vertex, numerator, degree)
        expected, ideals, active, raw, mass = dense_projection(
            records, scale, denominator, shift, upper, cap, grid)
        old_emissions = reporter.metrics.emissions
        result = reporter.project_counts(base_scale=scale, common_denominator=denominator,
                                         shift=shift, upper=upper, cap=cap, grid=grid)
        actual = {int(vertex):count for vertex, count, degree in result.entries}
        self.assertEqual(actual, expected)
        self.assertEqual(len(result.entries), len(actual))
        self.assertEqual(result.cap_active, active)
        self.assertGreaterEqual(result.multiplier, 0)
        self.assertEqual(reporter.metrics.emissions-old_emissions, len(actual))
        self.assertEqual(mass(result.multiplier), cap if active else mass(F(0)))
        self.assertTrue(all(0 <= ideals[i]-actual.get(i, 0)*grid < grid for i in ideals))
        self.assertTrue(all(isinstance(count, int) and count > 0 for _, count, _ in result.entries))
        self.assertTrue(all(0 < count*grid <= upper for _, count, _ in result.entries))
        self.assertLessEqual(sum(d*actual.get(i, 0)*grid for i, _, d in raw), cap)
        LOG['projections'] += 1
        LOG['binding_caps'] += int(active)
        LOG['upper_binding_cases'] += int(any(value == upper for value in ideals.values()))
        return reporter, result

    def test_exhaustive_small_dense_projection(self):
        globals_list = [
            (1, 3, F(0), F(1, 2), F(0), F(1, 16)),
            (3, 7, F(1, 11), F(3, 7), F(1), F(1, 64)),
            (5, 2, F(-1, 3), F(3, 8), F(5, 16), F(1, 32)),
            (2, 1, F(1, 4), F(1, 4), F(1, 2), F(1, 16)),
            (7, 13, F(3, 5), F(2, 3), F(1, 7), F(3, 64)),
        ]
        for first, second in product(range(-2, 5), repeat=2):
            for degrees in ((1, 1), (2, 3), (3, 2)):
                for kinds in product((0, 1), repeat=2):
                    records = [(0, (kinds[0], first, degrees[0])),
                               (1, (kinds[1], second, degrees[1]))]
                    for parameters in globals_list:
                        self.check_projection(records, *parameters)
        for numerators in product((-2, 0, 1, 3, 6), repeat=3):
            for kinds in product((0, 1), repeat=3):
                records = [(i, (kinds[i], numerators[i], (1, 3, 7)[i])) for i in range(3)]
                self.check_projection(records, scale=3, denominator=5, cap=F(5, 8))

    def test_ties_plateaus_and_disparate_degrees(self):
        # Binding cap gives ideal densities exactly h and h/2; only the h
        # coordinate is emitted. They lie in different trees.
        self.check_projection([(0, (0, 24, 3)), (1, (1, 91, 13))],
                              scale=1, denominator=32, shift=F(0), upper=F(1),
                              cap=F(19, 32), grid=F(1, 16))
        # A plateau mass target, upper clipping, duplicated ratio keys,
        # and an empty component are all included explicitly.
        self.check_projection([(0, (0, 10, 1)), (1, (1, 1, 1))],
                              scale=1, denominator=1, shift=F(0), upper=F(1), cap=F(1))
        self.check_projection([(0, (0, 3, 3)), (1, (1, 7, 7)), (2, (0, 11, 11))],
                              scale=1, denominator=1, shift=F(0), upper=F(1, 3), cap=F(1, 2))
        degrees = (101, 103, 107, 109, (1 << 61)-1, (1 << 127)-1)
        ids = [HashForbiddenInt((1 << 2048)+i if i % 2 else -(1 << 2048)-i) for i in range(6)]
        records = [(ids[i], (i % 2, degrees[i]*(i+1)+(-1)**i, degrees[i])) for i in range(6)]
        self.check_projection(records, scale=11, denominator=37, upper=F(7, 13),
                              cap=F(17, 19), grid=F(1, 256))
        self.check_projection([])

    def test_integer_only_updates_moves_and_removals(self):
        def forbidden_fraction(*args, **kwargs):
            raise AssertionError('Point update constructed a Fraction')

        for order in permutations(range(5)):
            reporter = production.IntegerClippedReporter()
            with patch.object(production, 'F', forbidden_fraction):
                for i in order:
                    reporter.set_base(i, i*i-3, i+1)
                    validate(reporter)
                for i in reversed(order):
                    reporter.set_exception(i, 7-2*i, i+1)
                    validate(reporter)
                for i in order:
                    reporter.set_base(i, -i, 2*i+1)
                    reporter.set_base(i, -i, 2*i+1)
                    validate(reporter)
                for i in reversed(order):
                    reporter.remove(i)
                    self.assertFalse(reporter.discard(i))
                    validate(reporter)
                reporter.clear()
            self.assertEqual(len(reporter), 0)
            self.assertEqual(reporter.metrics.moves, 10)

    def test_rejected_subgrid_tail_is_not_enumerated(self):
        reporter = production.IntegerClippedReporter()
        total = 8192
        for i in range(total):
            reporter.set_base(HashForbiddenInt((1 << 2048)+i), i+1, 1000003+2*i)
        selected = HashForbiddenInt(-(1 << 2048)-9)
        reporter.set_exception(selected, 1 << 20, 1)
        records = list(reporter.records.items())
        before = reporter.metrics.tree_visits
        _, answer = self.check_projection(records, scale=7, denominator=1 << 40,
                                          shift=F(0), upper=F(1), cap=F(1),
                                          grid=F(1, 1 << 20), reporter=reporter)
        visits = reporter.metrics.tree_visits-before
        self.assertEqual(answer.entries, ((selected, 1, 1),))
        self.assertLess(visits, 200)
        LOG['fixtures'].append(dict(label='8192 rejected positive sub-grid records',
                                    query_tree_visits=visits, **asdict(reporter.metrics)))


if __name__ == '__main__':
    result = unittest.TextTestRunner(verbosity=2).run(
        unittest.defaultTestLoader.loadTestsFromTestCase(IntegerReporterTests))
    LOG.update(tests_run=result.testsRun, successful=result.wasSuccessful())
    Path(__file__).with_name('integer_clipped_reporter_verification.json').write_text(
        json.dumps(LOG, indent=2)+'\n')
    print(json.dumps(LOG, indent=2))
    raise SystemExit(not result.wasSuccessful())
