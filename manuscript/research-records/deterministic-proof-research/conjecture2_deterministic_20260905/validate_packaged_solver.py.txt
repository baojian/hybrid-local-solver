PACKAGE_ROOT = ROOT/'deliverables'/'deterministic-rppr'

def rooted_representatives(n):
    edges = list(itertools.combinations(range(1, n+1), 2))
    edge_index = {edge: k for k, edge in enumerate(edges)}
    permutations = [(1,)+p for p in itertools.permutations(range(2, n+1))]
    transforms = []
    for permutation in permutations:
        transforms.append([edge_index[tuple(sorted((permutation[i-1], permutation[j-1])))]
                           for i, j in edges])
    answer = []
    for mask in range(1 << len(edges)):
        rows = graph(n, [edge for k, edge in enumerate(edges) if mask & (1 << k)])
        reached, stack = {1}, [1]
        while stack:
            for j in rows[stack.pop()]:
                if j not in reached:
                    reached.add(j)
                    stack.append(j)
        if len(reached) < n:
            continue
        canonical = True
        for transform in transforms:
            other = sum(1 << transform[k] for k in range(len(edges)) if mask & (1 << k))
            if other < mask:
                canonical = False
                break
        if canonical:
            answer.append((mask, rows))
    return answer


def make_manifest():
    cases, representatives = [], {}

    def add(name, rows, seed, alpha, rho, epsilon, **extra):
        cases.append(dict(name=name, rows=rows, seed=seed, alpha=alpha, rho=rho,
                          epsilon=epsilon, **extra))

    for n in range(2, 6):
        representatives[n] = rooted_representatives(n)
        for mask, rows in representatives[n]:
            degree = len(rows[1])
            add(f'rooted-n{n}-m{mask}-A', rows, 1, F(1, 7), F(3, 8*degree), F(1, 10000))
            add(f'rooted-n{n}-m{mask}-B', rows, 1, F(1, 16), F(1, 16*degree), F(1, 10**6))
    assert {n: len(rows) for n, rows in representatives.items()} == {2: 1, 3: 3, 4: 11, 5: 58}

    path = graph(5, [(1, 2), (2, 3), (3, 4), (4, 5)])
    star = graph(5, [(1, 2), (1, 3), (1, 4), (1, 5)])
    clique = graph(5, itertools.combinations(range(1, 6), 2))
    tail = graph(5, [(1, 2), (2, 3), (3, 1), (3, 4), (4, 5)])
    for name, rows, seed in [('path', path, 1), ('star', star, 2),
                              ('clique', clique, 1), ('tail', tail, 5)]:
        for alpha in (F(63, 1024), F(1, 16), F(65, 1024)):
            add(f'momentum-{name}-{alpha}', rows, seed, alpha,
                F(3, 10*len(rows[seed])), F(1, 10**6))
    edge = graph(2, [(1, 2)])
    for alpha in (F(1, 4), F(1, 7)):
        for direction in (-1, 0, 1):
            add(f'contact-{alpha}-{direction}', edge, 1, alpha,
                (1-alpha)/2+direction*F(1, 2**18), F(1, 2**60))
    for name, rows, seed in [('path', path, 1), ('star', star, 1)]:
        for direction in (-1, 0, 1):
            add(f'zero-{name}-{direction}', rows, seed, F(1, 7),
                (1+direction*F(1, 2**20))/len(rows[seed]), F(1, 2**60))
    for name, rows, seed in [('edge', edge, 1), ('star-center', star, 1), ('star-leaf', star, 2)]:
        add(f'alpha-one-{name}', rows, seed, F(1), F(3, 7*len(rows[seed])), F(1, 2**60))
    for mask, rows in representatives[5][:8]:
        rename = {i: 2**100+37*i for i in rows}
        renamed = {rename[i]: tuple(rename[j] for j in row) for i, row in rows.items()}
        add(f'labels-n5-m{mask}', renamed, rename[1], F(1, 16),
            F(1, 16*len(rows[1])), F(1, 10**6), reverse=True,
            compare_to=f'rooted-n5-m{mask}-B', inverse={v: k for k, v in rename.items()})
    for leaves in (10**6, 2**100):
        cases.append(dict(name=f'access-star-{leaves}', star_leaves=leaves,
                          alpha=F(1, 100), rho=F(1, 64), epsilon=F(1, 10**6)))
    assert len(cases) == 183
    return cases


def serialize(value):
    if isinstance(value, F):
        return str(value)
    if isinstance(value, dict):
        return {str(k): serialize(v) for k, v in value.items()}
    if isinstance(value, (tuple, list)):
        return [serialize(v) for v in value]
    return value


def star_case(case):
    leaves, alpha, rho, epsilon = (case[k] for k in ('star_leaves', 'alpha', 'rho', 'epsilon'))
    seed = leaves+1

    class StarOracle:
        def __init__(self):
            self.degree_calls = self.entries = 0
        def degree(self, i):
            assert i in (1, seed)
            self.degree_calls += 1
            return leaves if i == 1 else 1
        def neighbors(self, i):
            assert i == seed, 'Inactive hub adjacency was scanned'
            self.entries += 1
            yield 1

    oracle = StarOracle()
    answer = solve(oracle, seed, alpha, rho, epsilon)
    exact = alpha*(1-rho)/((1+alpha)/2)
    assert (1-alpha)*exact/(2*leaves) <= alpha*rho
    assert len(answer.output) == 1
    vertex, value, degree = answer.output[0]
    assert vertex == seed and degree == 1 and 0 <= value <= exact
    gap = (1+alpha)*(value-exact)**2/4
    assert 0 <= gap <= answer.objective_gap_bound <= epsilon
    for stage in answer.stages:
        assert stage.corrector_metrics['rebase_adjacency_entries'] == 0
    return answer, dict(actual_gap=gap, degree_calls=oracle.degree_calls,
                        first_adjacency_entries=oracle.entries,
                        iterations=sum(stage.iterations for stage in answer.stages),
                        stages=len(answer.stages), volume=1)


def main():
    cases = make_manifest()
    hashes = {str(path.relative_to(PACKAGE_ROOT)): hashlib.sha256(path.read_bytes()).hexdigest()
              for path in sorted((PACKAGE_ROOT/'deterministic_rppr').glob('*.py'))}
    target = ROOT/'packaged_solver_validation.json'
    evidence = dict(status='running', deterministic=True, case_count=len(cases),
                    package_hashes=hashes, manifest=serialize(cases), completed=[])
    target.write_text(json.dumps(evidence, indent=2)+'\n')
    previous_results, total_iterations, total_entries = {}, 0, 0
    for number, case in enumerate(cases, 1):
        start = time.perf_counter()
        if 'star_leaves' in case:
            answer, checked = star_case(case)
        else:
            answer, checked = validate_case(case['rows'], case['seed'], case['alpha'],
                case['rho'], case['epsilon'], solve_function=solve,
                reverse=case.get('reverse', False))
        if 'compare_to' in case:
            inverse = case['inverse']
            normalized = tuple((inverse[i], value, degree) for i, value, degree in answer.output)
            old = previous_results[case['compare_to']]
            assert normalized == old.output
            assert answer.objective_gap_bound == old.objective_gap_bound
        previous_results[case['name']] = answer
        total_iterations += checked['iterations']
        total_entries += checked['first_adjacency_entries']
        if total_iterations > 250000 or total_entries > 10000000:
            raise RuntimeError('Fixed integration-suite resource guard reached; suite is incomplete')
        evidence['completed'].append(serialize(dict(name=case['name'], passed=True,
            elapsed_seconds=time.perf_counter()-start, checks=checked,
            certificate=answer.objective_gap_bound, output=answer.output)))
        evidence.update(total_iterations=total_iterations, total_first_entries=total_entries)
        target.write_text(json.dumps(evidence, indent=2)+'\n')
        print(f'PASS {number}/{len(cases)} {case["name"]} '
              f'({checked["iterations"]} iterations)', flush=True)
    evidence['status'] = 'passed'
    target.write_text(json.dumps(evidence, indent=2)+'\n')
    print(f'ALL PASSED: {len(cases)} cases, {total_iterations} iterations', flush=True)


if __name__ == '__main__':
    main()
