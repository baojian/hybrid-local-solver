"""Independent, bounded, deterministic certificate audit; no solver calls."""
from fractions import Fraction as F
from hashlib import sha256
from pathlib import Path
import json

from independent_sparse_certificate import certify_sparse_output
from audit_virtual_graphs_and_verifier import (
    VirtualGraph, structural_check, exact_optimum, independent_objective,
)


class RowOracle:
    def __init__(self, labels, rows):
        self.labels, self.rows = list(labels), list(rows)
        self.degree_calls, self.row_calls, self.entries = [], [], 0

    def degree(self, i):
        self.degree_calls.append(i)
        return len(self.rows[self.labels.index(i)])

    def neighbors(self, i):
        self.row_calls.append(i)
        for j in self.rows[self.labels.index(i)]:
            self.entries += 1
            yield j


def direct_norm(rows, seed, alpha, rho, values):
    """Exact density-gradient norm, without the certificate's integer formula."""
    norm, nonzero = F(0), 0
    for i, row in rows.items():
        density = values.get(i, F(0))
        g = ((1+alpha)/2*density
             - (1-alpha)/2*sum((values.get(j, F(0)) for j in row), F(0))/len(row)
             - alpha*int(i == seed)/len(row)+alpha*rho)
        selected = g if density > 0 else min(g, 0)
        norm += len(row)*selected*selected
        nonzero += bool(selected)
    return norm, nonzero


def main():
    report = dict(certificate_sha256=sha256(Path(__file__).with_name(
        'independent_sparse_certificate.py').read_bytes()).hexdigest(),
        deterministic=True, solver_called=False, graph_parameter_cases=0,
        output_checks=0, exact_gap_checks=0, strict_rounding_cases=0,
        inconclusive_accurate_cases=0, malformed_inputs_rejected=0,
        families={})
    fixtures = [('path', 5, 0), ('cycle', 5, 0), ('star', 5, 0),
                ('grid', 2, 0), ('cube', 2, 0), ('barbell', 3, 0),
                ('tree_clique', 1, 2)]
    for family, size, aux in fixtures:
        graph = VirtualGraph(family, size, aux)
        rows = structural_check(graph)
        family_checks = 0
        for seed in (1, graph.n):
            for alpha in (F(1, 7), F(1, 4), F(1)):
                for rho in (F(1, 4*len(rows[seed])), F(1, len(rows[seed]))):
                    optimum = exact_optimum(rows, seed, alpha, rho)
                    optimal_objective = independent_objective(rows, seed, alpha, rho, optimum)
                    floor_optimum = {i: F((v*4096).__floor__(), 4096)
                                     for i, v in optimum.items() if (v*4096).__floor__()}
                    variants = [{}, {seed: F(1, 8)}, floor_optimum,
                                {i: v+F(1, 4096) for i, v in floor_optimum.items()},
                                {i: F((i % 3)+1, 16) for i in rows},
                                {i: F(i, 4) for i in rows if i % 2}]
                    report['graph_parameter_cases'] += 1
                    for values in variants:
                        oracle = RowOracle(rows, rows.values())
                        output = [(i, v, len(rows[i])) for i, v in values.items()]
                        cert = certify_sparse_output(oracle, seed, alpha, rho, output)
                        norm, nonzero = direct_norm(rows, seed, alpha, rho, values)
                        gap = independent_objective(rows, seed, alpha, rho, values)-optimal_objective
                        assert 0 <= gap <= norm/(2*alpha) <= cert.objective_gap_bound
                        excess = cert.subgradient_norm_squared_upper-norm
                        assert excess >= 0
                        if nonzero:
                            assert excess < cert.norm_squared_rounding_excess_bound
                        else:
                            assert excess == cert.norm_squared_rounding_excess_bound == 0
                        report['strict_rounding_cases'] += excess > 0
                        affected = {seed, *values}
                        for i in values: affected.update(rows[i])
                        assert sorted(oracle.row_calls) == sorted(values)
                        assert sorted(oracle.degree_calls) == sorted(affected)
                        assert cert.first_adjacency_entries == oracle.entries == sum(len(rows[i]) for i in values)
                        assert cert.accumulation_updates == oracle.entries
                        assert cert.degree_replies == cert.affected_records == len(affected)
                        assert cert.output_volume == oracle.entries
                        assert cert.output_records == len(values)
                        assert cert.nonzero_subgradient_records == nonzero
                        assert cert.meets(cert.objective_gap_bound+1)
                        if gap < cert.objective_gap_bound:
                            tolerance = (gap+cert.objective_gap_bound)/2
                            assert tolerance > 0 and not cert.meets(tolerance) and gap < tolerance
                            report['inconclusive_accurate_cases'] += 1
                        report['output_checks'] += 1
                        report['exact_gap_checks'] += 1
                        family_checks += 1
        report['families'][family] = family_checks

    # A huge unscanned hub is represented only by its degree and its label.
    class HugeStar:
        n = 10**12
        def __init__(self): self.degree_calls, self.row_calls = [], []
        def degree(self, i):
            self.degree_calls.append(i)
            return self.n-1 if i == 1 else 1
        def neighbors(self, i):
            self.row_calls.append(i)
            assert i == 2, 'Certificate attempted to scan the huge hub'
            yield 1
    star = HugeStar()
    cert = certify_sparse_output(star, 2, F(1, 1000), F(1, 10000), [(2, F(1, 4), 1)])
    assert star.row_calls == [2] and sorted(star.degree_calls) == [1, 2]
    assert cert.first_adjacency_entries == 1 and cert.affected_records == 2
    report['huge_star'] = dict(vertices=star.n, adjacency_entries=1, degree_replies=2)

    # Hashing an integer label is forbidden: every production registry is AVL.
    class NoHash(int):
        def __hash__(self): raise AssertionError('Label hashing is forbidden')
    labels = [NoHash(2**100+37*i) for i in range(3)]
    oracle = RowOracle(labels, [[labels[1]], [labels[0], labels[2]], [labels[1]]])
    cert = certify_sparse_output(oracle, labels[0], F(1, 7), F(1, 8),
                                 [(labels[0], F(3, 16), 1), (labels[2], F(1, 32), 1)])
    ordinary = RowOracle([1, 2, 3], [[2], [1, 3], [2]])
    expected = certify_sparse_output(ordinary, 1, F(1, 7), F(1, 8),
                                    [(1, F(3, 16), 1), (3, F(1, 32), 1)])
    assert cert == expected
    report['hash_forbidden_large_label_case'] = True

    def rejected(call):
        try: call()
        except ValueError: report['malformed_inputs_rejected'] += 1
        else: raise AssertionError('Malformed input was accepted')
    def oracle(): return RowOracle([1, 2], [[2], [1]])
    for bad in ([(1, F(0), 1)], [(1, F(-1), 1)], [(1, F(1), 0)],
                [(1, F(1), 2)], [('1', F(1), 1)],
                [(1, F(1, 2), 1), (1, F(1, 2), 1)],
                [(1, F(1, 2), 1), (2, F(1, 3), 1)]):
        rejected(lambda bad=bad: certify_sparse_output(oracle(), 1, F(1, 4), F(1, 8), bad))
    for alpha, rho in ((F(0), F(1)), (F(2), F(1)), (F(1), F(0))):
        rejected(lambda alpha=alpha, rho=rho: certify_sparse_output(oracle(), 1, alpha, rho, []))
    rejected(lambda: expected.meets(0))
    class WrongRow(RowOracle):
        def degree(self, i): return 2
    rejected(lambda: certify_sparse_output(WrongRow([1, 2], [[2], [1]]), 1,
                                           F(1, 4), F(1, 8), [(1, F(1, 8), 2)]))
    destination = Path(__file__).with_name('independent_sparse_certificate_audit_results.json')
    destination.write_text(json.dumps(report, indent=2)+'\n')
    print(json.dumps(report, indent=2))


if __name__ == '__main__': main()
