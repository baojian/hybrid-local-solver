"""Verify and record the final compiled proof snapshot (no source edits)."""
import hashlib
import json
import re
import subprocess
from datetime import datetime, timezone
from pathlib import Path

ROOT = Path(__file__).resolve().parent
SOURCES = ('deterministic_conjecture2.tex', 'practical_refinements_appendix.tex',
           'practical_schedules_appendix.tex')
PDFINFO = Path('/Users/baojian/.cache/codex-runtimes/codex-primary-runtime/dependencies/bin/override/pdfinfo')

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def main():
    pdf = ROOT/'output/pdf/deterministic_conjecture2.pdf'
    log = ROOT/'output/pdf/deterministic_conjecture2.log'
    text = log.read_text()
    bad = [line for line in text.splitlines() if re.search(r'Warning|Overfull|Underfull|undefined|^!', line)]
    assert not bad, bad
    assert all((ROOT/name).stat().st_mtime <= pdf.stat().st_mtime for name in SOURCES)
    info = subprocess.run([str(PDFINFO), str(pdf)], capture_output=True, text=True, check=True).stdout
    pages = int(re.search(r'^Pages:\s+(\d+)', info, re.M).group(1))
    match = re.search(r'Output written on .*\((\d+) pages,', text)
    assert match and int(match.group(1)) == pages
    report = dict(status='compiled_without_warnings', created_utc=datetime.now(timezone.utc).isoformat(),
                  source_sha256={name:sha(ROOT/name) for name in SOURCES},
                  pdf_sha256=sha(pdf), pdf_bytes=pdf.stat().st_size, pages=pages,
                  log_sha256=sha(log), compile_passes=2, warnings=[],
                  visual_review='See proof_visual_review.json; compilation alone does not establish visual quality.')
    (ROOT/'proof_compile_manifest.json').write_text(json.dumps(report, indent=2)+'\n')
    print(json.dumps(report, indent=2))

if __name__ == '__main__':
    main()
