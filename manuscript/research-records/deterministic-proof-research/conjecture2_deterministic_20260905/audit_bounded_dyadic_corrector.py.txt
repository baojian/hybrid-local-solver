"""Independent denominator/locality audit, separate from the dense suite.

No random input. This checks every tree mutation on a local graph with
large pairwise-coprime boundary degrees, and exposes a precise distinction
between reduced Fraction sizes and actual pre-reduction temporaries.
"""

import fractions
import json
import sys
from dataclasses import asdict
from fractions import Fraction as F
from pathlib import Path

from bounded_dyadic_corrector import (
    ArithmeticTracker, BoundedDyadicCorrector, DyadicMetrics, Tracked,
)


def rational(value):
    return value.value if isinstance(value, Tracked) else F(value)


class PrimeBoundaryOracle:
    """Seed joined to three hubs, each with its own private leaves.

    The full graph is defined by these parameters, but the diagnostic
    rejects any hub adjacency query: this instance's exact optimizer is
    supported on the seed alone. Huge hub degrees must not be scanned.
    """
    hub_degrees = (1000003, 1000033, 1000037)

    def __init__(self):
        self.degree_queries = []
        self.neighbor_queries = []

    def degree(self, i):
        self.degree_queries.append(i)
        if i == 0:
            return 3
        if 1 <= i <= 3:
            return self.hub_degrees[i-1]
        raise AssertionError('The solver exposed a private hub leaf')

    def neighbors(self, i):
        self.neighbor_queries.append(i)
        if i != 0:
            raise AssertionError('Inactive high-degree hub adjacency was scanned')
        return iter((1, 2, 3))


class DenominatorAuditedCorrector(BoundedDyadicCorrector):
    def __init__(self, *args, **kwargs):
        self.denominator_audits = 0
        self.audited_tree_nodes = 0
        super().__init__(*args, **kwargs)

    def _set_key(self, i, value):
        super()._set_key(i, value)
        self.audit_tree()

    def audit_tree(self):
        D, T, H = self.alpha.denominator, self.theta.denominator, self.grid.denominator
        S = self.sigma*H
        assert S.denominator == 1
        common = 2*D*H*(T+1)*S.numerator

        def visit(node):
            if node is None:
                return 0, F(0), 0, 0
            lw, ls, lc, lh = visit(node.left)
            rw, rs, rc, rh = visit(node.right)
            key = rational(node.key[0])
            weighted = key*node.weight
            total = weighted+ls+rs
            assert common % weighted.denominator == 0
            assert common % total.denominator == 0
            assert rational(node.weighted_sum) == total
            assert node.total_weight == lw+node.weight+rw
            assert getattr(node, 'count', 1) == lc+rc+1
            assert node.height == 1+max(lh, rh)
            assert abs(lh-rh) <= 1
            self.audited_tree_nodes += 1
            return node.total_weight, total, lc+rc+1, node.height

        visit(self.tree.root)
        self.denominator_audits += 1


def arithmetic_intermediate_fixture():
    metrics = DyadicMetrics()
    tracker = ArithmeticTracker(metrics)
    left = tracker.wrap(F(10402, 10403))
    right = tracker.wrap(F(1, 20604))
    trace_max = 0
    witnessed_t = None

    def trace(frame, event, arg):
        nonlocal trace_max, witnessed_t
        if frame.f_code.co_filename == fractions.__file__ and frame.f_code.co_name == '_add':
            for value in frame.f_locals.values():
                if isinstance(value, int):
                    trace_max = max(trace_max, value.bit_length())
            if frame.f_locals.get('t') == 2122111:
                witnessed_t = frame.f_locals['t']
        return trace

    previous = sys.gettrace()
    try:
        sys.settrace(trace)
        result = left+right
    finally:
        sys.settrace(previous)
    assert result.value == F(21011, 21012)
    assert metrics.max_numerator_bits == metrics.max_denominator_bits == 15
    assert witnessed_t == 2122111 and trace_max == 22
    envelope = 2*max(metrics.max_numerator_bits, metrics.max_denominator_bits)+1
    assert trace_max <= envelope
    return dict(result=str(result.value), reduced_max_bits=15,
                witnessed_pre_reduction_integer=witnessed_t,
                true_observed_integer_bits=trace_max,
                conservative_binary_arithmetic_envelope=envelope)


def local_denominator_fixture():
    oracle = PrimeBoundaryOracle()
    alpha, theta, rho = F(13, 97), F(1, 4), F(1, 8)
    baseline = {0:F(1, 64)}
    solver = DenominatorAuditedCorrector(
        oracle, 0, alpha, theta, rho, F(1, 4096), baseline)
    initial_source = dict(solver.source)
    raw_checks = 0
    zero_emission_steps = 0
    for _ in range(solver.target_iterations):
        # Independently reconstruct the full raw density from the represented
        # state, rather than from the lazy-key formula used by production.
        y = {i:(solver.sigma*solver.X.get(i, 0)+theta*solver.z.get(i, 0))/(1+theta)
             for i in solver.degree}
        for i in solver.degree:
            adjacent_y = sum((y.get(j, 0) for j in (1, 2, 3)), F(0)) if i == 0 else y[0]
            qy = solver.q0*y[i]-solver.c*adjacent_y/solver.degree[i]
            raw = solver.a*solver.z.get(i, 0)+theta*y[i]-(qy-solver.source.get(i, 0)+solver.lam)/theta
            represented = solver.sigma*rational(solver.keys[i])-solver.lam/theta
            assert raw == represented
            raw_checks += 1
        solver.step()
        zero_emission_steps += int(not solver.z)
        assert set(solver.X) <= {0} and set(solver.z) <= {0}
        assert solver.source == initial_source
        for i in solver.degree:
            expected_l = F(0) if i == 0 else solver.X.get(0, F(0))
            expected_m = F(0) if i == 0 else solver.z.get(0, F(0))
            assert solver.L.get(i, 0) == expected_l
            assert solver.M.get(i, 0) == expected_m
    result = dict((i, value) for i, value, _ in solver.output())
    # The exact seed-only optimum and all three hub KKT inequalities.
    optimum = alpha*(F(1, 3)-rho)/solver.q0
    for d in oracle.hub_degrees:
        assert solver.c*optimum/d <= solver.lam
    gap = F(3, 2)*solver.q0*(result[0]-optimum)**2
    assert 0 <= gap <= solver.objective_gap_bound < solver.tolerance
    assert oracle.neighbor_queries == [0]
    assert sorted(oracle.degree_queries) == [0, 1, 2, 3]
    assert solver.metrics.scanned_adjacency_entries == (
        solver.metrics.baseline_adjacency_entries+
        solver.metrics.selected_adjacency_entries+
        solver.metrics.rebase_adjacency_entries)
    assert solver.metrics.rebase_adjacency_entries == 3*solver.metrics.rebases
    return dict(steps=solver.target_iterations, raw_identities=raw_checks,
                denominator_audits=solver.denominator_audits,
                audited_tree_nodes=solver.audited_tree_nodes,
                pairwise_coprime_boundary_degrees=list(oracle.hub_degrees),
                queried_adjacency_rows=oracle.neighbor_queries,
                zero_emission_steps=zero_emission_steps,
                exact_gap=str(gap), metrics=asdict(solver.metrics))


if __name__ == '__main__':
    report = dict(success=True,
                  arithmetic_intermediate=arithmetic_intermediate_fixture(),
                  local_denominators=local_denominator_fixture())
    Path(__file__).with_name('bounded_dyadic_independent_verification.json').write_text(
        json.dumps(report, indent=2)+'\n')
    print(json.dumps(report, indent=2))
