"""Independent dense fixed-face checks, exact and deterministic throughout."""
import json
import sys
import unittest
from dataclasses import asdict
from fractions import Fraction as F
from pathlib import Path

import test_monotone_capped_solver as base
from guarded_monotone_capped_solver import GuardedMonotoneCappedSolver
from test_local_safe_solver import connected_graphs, dense_exact_obstacle_oracle, graph, path, star


class DenseGuardedReference(base.DenseReference):
    def __init__(self, adjacency, seed, theta, rho):
        super().__init__(adjacency, seed, theta, rho)
        # This reference knows the full fixture graph, unlike the tested solver.
        self.allowed = {i for i in self.vertices
                        if rho*(self.degrees[seed] +
                                (self.degrees[i] if i != seed else 0)) < 1}

    def project(self, raw):
        # First restrict the dense vector to the fixed face, then independently
        # water fill; no production guard, key, or AVL method is called.
        return super().project({i: raw[i] for i in self.allowed})


LOG = dict(trajectory_cases=0, checked_optima=0, fixtures=[])


class GuardedTests(unittest.TestCase):
    compare_step = base.MonotoneTests.compare_step
    set_state = base.MonotoneTests.set_state

    def assert_face(self, solver, reference, optimum):
        self.assertTrue(set(optimum) <= reference.allowed)
        self.assertTrue(set(solver.X) <= reference.allowed)
        self.assertTrue(set(solver.z) <= reference.allowed)
        self.assertTrue(set(solver.adjacency) <= reference.allowed)
        for i in solver.degree:
            if i not in reference.allowed:
                self.assertEqual(solver.keys[i], 0)

    def test_exhaustive_small_graphs_keep_optimum_and_match_dense(self):
        for n in (2, 3, 4):
            for mask, adjacency in connected_graphs(n):
                for seed in adjacency:
                    for theta in (F(1,2), F(1,4)):
                        for scale in (F(1,8), F(1,2), F(1)):
                            rho = scale/len(adjacency[seed])
                            reference = DenseGuardedReference(adjacency, seed, theta, rho)
                            optimum = dense_exact_obstacle_oracle(adjacency, seed, theta**2, rho)
                            oracle = base.CheckedOracle(adjacency)
                            solver = GuardedMonotoneCappedSolver(oracle, seed, theta, rho)
                            x, z = {}, {}
                            with self.subTest(n=n, mask=mask, seed=seed, theta=theta, rho=rho):
                                self.assert_face(solver, reference, optimum)
                                for _ in range(3):
                                    x,z = self.compare_step(solver, reference, x, z, optimum)
                                    self.assert_face(solver, reference, optimum)
                                self.assertEqual(len(oracle.neighbor_queries),
                                                 len(set(oracle.neighbor_queries)))
                                if not optimum:
                                    self.assertEqual(solver.metrics.first_adjacency_entries, 0)
                            LOG['trajectory_cases'] += 1
                            LOG['checked_optima'] += 1

    def test_old_transient_hub_is_now_forbidden(self):
        adjacency, seed, theta, rho = star(101), 1, F(1,4), F(1,100)
        reference = DenseGuardedReference(adjacency, seed, theta, rho)
        optimum = {seed:F(99,850)}
        gradient = reference.multiply(optimum)
        self.assertTrue(all(gradient[i]-reference.h[i] >= 0 for i in adjacency))
        self.assertEqual(gradient[seed]-reference.h[seed], 0)
        oracle = base.CheckedOracle(adjacency)
        solver = GuardedMonotoneCappedSolver(oracle, seed, theta, rho)
        x, z = {}, {}
        for _ in range(5):
            x,z = self.compare_step(solver, reference, x, z, optimum)
            self.assert_face(solver, reference, optimum)
        self.assertEqual(set(solver.adjacency), {seed})
        self.assertEqual(set(solver.degree), {0, seed})
        self.assertEqual(solver.keys[0], 0)
        self.assertNotEqual(solver.R[0], 0)  # Forbidden boundary response is retained.
        self.assertEqual(solver.metrics.first_adjacency_entries, 1)
        self.assertEqual(solver.metrics.scanned_adjacency_entries, 5)
        LOG['fixtures'].append(dict(label='guarded-degree100-star',
            support_volumes=[record[0] for record in solver.history],
            **asdict(solver.metrics)))

    def test_binding_cap_excludes_positive_forbidden_raw_coordinate(self):
        adjacency = graph(5, [(0,1), (1,2), (2,3), (2,4)])
        seed, theta, rho = 0, F(1,4), F(1,4)
        reference = DenseGuardedReference(adjacency, seed, theta, rho)
        optimum = dense_exact_obstacle_oracle(adjacency, seed, theta**2, rho)
        solver = GuardedMonotoneCappedSolver(base.CheckedOracle(adjacency), seed, theta, rho)
        x, z = {1:F(1,2)}, {seed:F(1,2)}
        self.set_state(solver, reference, x, z, optimum)
        # The ordinary unconstrained raw mirror value at the forbidden hub is 3/16.
        ordinary = base.DenseReference(adjacency, seed, theta, rho)
        _, ordinary_z, _, ordinary_active, _ = ordinary.step(x, z)
        self.assertTrue(ordinary_active)
        self.assertEqual(ordinary_z, {seed:F(61,64), 2:F(1,64)})
        x,z = self.compare_step(solver, reference, x, z, optimum)
        self.assertEqual(z, {seed:F(1)})
        self.assertEqual(solver.mass_z, 1)
        self.assertTrue(solver.history[-1][3])
        self.assertNotIn(2, solver.adjacency)
        self.assertEqual(solver.keys[2], 0)  # rho*(dseed+d_hub) equals one exactly.
        self.assert_face(solver, reference, optimum)
        LOG['fixtures'].append(dict(label='binding-cap-and-equality-rejection',
            fraction=str(solver.segment_fractions[-1]), **asdict(solver.metrics)))

    def test_warm_start_validation_and_valid_warm_recurrence(self):
        adjacency, seed, theta, rho = path(2), 0, F(1,2), F(1,2)
        reference = DenseGuardedReference(adjacency, seed, theta, rho)
        optimum = dense_exact_obstacle_oracle(adjacency, seed, theta**2, rho)
        solver = GuardedMonotoneCappedSolver(base.CheckedOracle(adjacency), seed, theta, rho)
        with self.assertRaisesRegex(ValueError, 'forbidden'):
            solver.initialize_from_primal({1:F(1,10)}, F(1))
        self.assertEqual(solver.metrics.degree_replies, 2)
        self.assertEqual(solver.metrics.first_adjacency_entries, 0)
        self.assertEqual(solver.keys[1], 0)
        primal = {i:value/2 for i,value in optimum.items()}
        solver = GuardedMonotoneCappedSolver(base.CheckedOracle(adjacency), seed, theta, rho)
        # A supplied zero at a forbidden vertex does not trigger its degree query.
        gap = reference.objective(primal)-reference.objective(optimum)
        solver.initialize_from_primal(primal, gap)
        x,z = primal.copy(), primal.copy()
        for _ in range(3):
            x,z = self.compare_step(solver, reference, x, z, optimum)
            self.assert_face(solver, reference, optimum)
        zero_solver = GuardedMonotoneCappedSolver(base.CheckedOracle(adjacency), seed, theta, rho)
        zero_solver.initialize_from_primal({1:F(0)}, F(1))
        self.assertEqual(zero_solver.metrics.degree_replies, 1)
        self.assertEqual(zero_solver.metrics.first_adjacency_entries, 0)

    def test_zero_optimum_and_seed_equality_need_no_adjacency(self):
        solver = GuardedMonotoneCappedSolver(base.CheckedOracle(path(2)), 0, F(1,2), F(1))
        self.assertEqual(solver.keys[0], 0)
        self.assertEqual(solver.run(F(1,100)), [])
        self.assertEqual(solver.metrics.iterations, 0)
        self.assertEqual(solver.metrics.degree_replies, 1)
        self.assertEqual(solver.metrics.first_adjacency_entries, 0)
        with self.assertRaisesRegex(ValueError, 'forbidden'):
            solver.initialize_from_primal({0:F(1,10)}, F(1))


if __name__ == '__main__':
    outcome = unittest.TextTestRunner(verbosity=2).run(
        unittest.defaultTestLoader.loadTestsFromTestCase(GuardedTests))
    LOG.update({key:value for key,value in base.LOG.items()
                if key not in ('fixtures', 'trajectory_cases', 'injected_states')})
    LOG['passed'], LOG['test_methods'] = outcome.wasSuccessful(), outcome.testsRun
    Path(__file__).with_name('guarded_monotone_capped_verification.json').write_text(
        json.dumps(LOG, indent=2)+'\n')
    print(json.dumps({key:value for key,value in LOG.items() if key != 'fixtures'}, indent=2))
    sys.exit(0 if outcome.wasSuccessful() else 1)
