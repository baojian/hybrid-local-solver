"""Independent streaming, exact-threshold, and interface audit; no source edits."""
import hashlib
import json
from dataclasses import asdict, dataclass
from fractions import Fraction as F
from pathlib import Path

from singleton_precheck_rppr import solve_rppr_with_singleton_precheck as solve
from independent_sparse_certificate import certify_sparse_output

ROOT = Path(__file__).resolve().parent
SOURCE = ROOT/'singleton_precheck_rppr.py'


class Label(int):
    def __hash__(self):
        raise AssertionError('A vertex label was hashed')


class Spokes:
    """Seed with disjoint star branches, represented without building leaves.

    Boundary vertex j has k_j-1 private leaves, so all chosen degrees are
    realized by a finite simple undirected graph. Only the seed row is legal
    for this audit; neither the precheck nor the singleton verifier needs
    any boundary row or private leaf label.
    """
    def __init__(self, degrees, fail_after=None):
        self.degrees = tuple(degrees)
        self.seed = Label(-(1 << 2051)-13)
        self.offset = 1 << 4099
        self.degree_queries = []
        self.row_queries = []
        self.entries = 0
        self.fail_after = fail_after

    def degree(self, i):
        self.degree_queries.append(i)
        if i == self.seed:
            return len(self.degrees)
        j = int(i)-self.offset
        assert 0 <= j < len(self.degrees)
        return self.degrees[j]

    def neighbors(self, i):
        assert i == self.seed, 'Boundary adjacency is forbidden'
        self.row_queries.append(i)
        for j in range(len(self.degrees)):
            if self.fail_after is not None and j >= self.fail_after:
                raise AssertionError('The failed trial consumed another entry')
            self.entries += 1
            yield Label(self.offset+j)


@dataclass
class Sentinel:
    output: tuple
    objective_gap_bound: F
    marker: object


def main():
    before = hashlib.sha256(SOURCE.read_bytes()).hexdigest()
    log = dict(threshold_cases=0, exact_zero_certificates=0,
               fallback_identity_cases=0, direct_cases=0,
               cutoff_cases=0, malformed_completed_rows=0,
               largest_parameter_bits=0, largest_degree_bits=0)
    for m, k, alpha in (
        (1, 1, F(1, 3)), (2, 2, F(2, 7)),
        (7, 19, F(1, (1 << 127)-1)),
        (32, 2, F((1 << 521)-1, (1 << 521)+1)),
        (33, 3, F(17, 97)),
    ):
        degrees = [((1 << (149+j))+2*j+1) for j in range(m-1)]+[k]
        threshold = (1-alpha)/(m*((1-alpha)+(1+alpha)*k))
        for direction in (-1, 0, 1):
            rho = threshold*(1+F(direction, 1 << 257))
            expected = 'failed_trial' if direction < 0 else 'singleton'
            oracle = Spokes(degrees)
            token = object()
            sentinel = Sentinel(((oracle.seed, F(7, 13), m),), F(11, 17), token)
            original_fields = vars(sentinel).copy()
            calls = []
            def fallback(*args):
                calls.append(args)
                return sentinel
            answer = solve(oracle, oracle.seed, alpha, rho, F(1, 10**30),
                           seed_degree_cutoff=33, fallback_solver=fallback)
            assert answer.outcome == expected
            metric = answer.trial_metrics
            assert metric.adjacency_entries == oracle.entries == m
            assert metric.neighbor_degree_queries == m
            assert len(oracle.degree_queries) == m+1
            assert len(oracle.row_queries) == 1
            if direction < 0:
                assert len(calls) == 1 and calls[0][0] is oracle
                assert calls[0][1] is oracle.seed
                assert calls[0][2:] == (alpha, rho, F(1, 10**30))
                assert answer.fallback_result is sentinel
                assert answer.output is sentinel.output
                assert answer.objective_gap_bound is sentinel.objective_gap_bound
                assert vars(sentinel) == original_fields and sentinel.marker is token
                assert metric.completed_row_length_checks == 0
                log['fallback_identity_cases'] += 1
            else:
                assert not calls and answer.fallback_result is None
                f = answer.output[0][1]
                q0, c = (1+alpha)/2, (1-alpha)/2
                assert q0*m*f-alpha+alpha*rho*m == 0
                residuals = [alpha*rho*di-c*f for di in degrees]
                assert min(residuals) >= 0
                assert (min(residuals) == 0) == (direction == 0)
                assert answer.objective_gap_bound == 0
                assert m*f+rho*m <= 1 and m < 1/rho
                verifier = Spokes(degrees)
                cert = certify_sparse_output(verifier, verifier.seed, alpha, rho, answer.output)
                assert cert.objective_gap_bound == 0
                assert cert.nonzero_subgradient_records == 0
                assert verifier.entries == m and len(verifier.degree_queries) == m+1
                log['exact_zero_certificates'] += 1
            log['threshold_cases'] += 1
            log['largest_parameter_bits'] = max(log['largest_parameter_bits'],
                *(x.bit_length() for x in (alpha.numerator, alpha.denominator,
                                           rho.numerator, rho.denominator)))
            log['largest_degree_bits'] = max(log['largest_degree_bits'],
                                            *(x.bit_length() for x in degrees))

    # This iterator is permitted to raise if even one extra entry is requested.
    oracle = Spokes([1]*32, fail_after=1)
    sentinel = Sentinel((), F(9), object())
    answer = solve(oracle, oracle.seed, F(1, 100), F(1, 1000000), F(1),
                   fallback_solver=lambda *args: sentinel)
    assert answer.outcome == 'failed_trial'
    assert answer.fallback_result is sentinel and oracle.entries == 1
    assert asdict(answer.trial_metrics) == dict(seed_degree_queries=1,
        neighbor_degree_queries=1, seed_row_requests=1, adjacency_entries=1,
        cutoff_comparisons=1, neighbor_inequality_checks=1,
        neighbor_inequality_failures=1, completed_row_length_checks=0,
        fallback_calls=1, final_output_words=0)
    log['first_failure_generator_guard'] = True

    def forbidden(*args):
        raise AssertionError('A direct branch called fallback')
    for alpha, rho, expected in ((F(1, 7), F(1, 33), 'zero'),
        (F(1), F(2, 33), 'zero'), (F(1), F(1, 100), 'alpha_one')):
        oracle = Spokes([2]*33, fail_after=0)
        answer = solve(oracle, oracle.seed, alpha, rho, F(1, 100),
                       seed_degree_cutoff=0, fallback_solver=forbidden)
        assert answer.outcome == expected
        assert len(oracle.degree_queries) == 1 and not oracle.row_queries
        assert answer.trial_metrics.cutoff_comparisons == 0
        assert answer.objective_gap_bound == 0
        log['direct_cases'] += 1

    for cutoff, m in ((0, 1), (32, 33)):
        oracle = Spokes([2]*m, fail_after=0)
        answer = solve(oracle, oracle.seed, F(1, 7), F(1, 1000), F(1),
                       seed_degree_cutoff=cutoff, fallback_solver=lambda *args: sentinel)
        assert answer.outcome == 'skipped_cutoff' and answer.fallback_result is sentinel
        assert answer.output is sentinel.output and not oracle.row_queries
        assert len(oracle.degree_queries) == 1
        log['cutoff_cases'] += 1

    class ShortRow(Spokes):
        def neighbors(self, i):
            assert i == self.seed
            return iter(())
    oracle = ShortRow([100])
    try:
        solve(oracle, oracle.seed, F(1, 3), F(1, 2), F(1), fallback_solver=forbidden)
    except ValueError as error:
        assert str(error) == 'Seed degree and adjacency disagree'
        log['malformed_completed_rows'] += 1
    else:
        raise AssertionError('Completed inconsistent row accepted')

    after = hashlib.sha256(SOURCE.read_bytes()).hexdigest()
    assert before == after
    result = dict(passed=True, source_sha256=before, source_unchanged=True,
                  evidence=log)
    (ROOT/'singleton_precheck_independent_verification.json').write_text(
        json.dumps(result, indent=2)+'\n')
    print(json.dumps(result, indent=2))


if __name__ == '__main__':
    main()
