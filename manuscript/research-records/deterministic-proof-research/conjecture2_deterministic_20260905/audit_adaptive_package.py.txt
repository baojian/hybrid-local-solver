"""Focused source/package equivalence and isolation for the optional wrapper."""
import ast
import hashlib
import inspect
import json
import shutil
import subprocess
import sys
import tempfile
from dataclasses import asdict
from fractions import Fraction as F
from pathlib import Path

sys.dont_write_bytecode=True
ROOT=Path(__file__).resolve().parent
PKGROOT=ROOT/'deliverables'/'deterministic-rppr'
PKG=PKGROOT/'deterministic_rppr'
LOG={}


def audit_ast():
    record=json.loads((PKGROOT/'ADAPTIVE_TRANSCRIPTION.json').read_text())
    source=(ROOT/record['source']).read_bytes()
    package=(PKG/'_adaptive.py').read_bytes()
    assert hashlib.sha256(source).hexdigest()==record['source_sha256']
    assert hashlib.sha256(package).hexdigest()==record['package_sha256']
    a,b=ast.parse(source),ast.parse(package)
    reverse={item['new'][1:]:item for item in record['transformations']}
    assert len(reverse)==6
    count=0
    for node in b.body:
        if isinstance(node,ast.ImportFrom) and node.level==1:
            expected=reverse[node.module]
            assert [n.name for n in node.names]==expected['names']
            assert all(n.asname is None for n in node.names)
            node.module,node.level=expected['old'],0
            count+=1
    assert count==6
    assert ast.dump(a,include_attributes=False)==ast.dump(b,include_attributes=False)
    unchanged=[]
    for name,expected in record['existing_module_sha256_at_build'].items():
        if name=='__init__.py':continue
        assert hashlib.sha256((PKG/name).read_bytes()).hexdigest()==expected
        unchanged.append(name)
    LOG['ast_import_relocations']=count
    LOG['unchanged_module_files']=sorted(unchanged)
    LOG['transcription']=record


def exact_comparisons():
    sys.path.insert(0,str(PKGROOT))
    import deterministic_rppr as package
    from adaptive_source_continuation_rppr import solve_rppr_adaptive as source
    from audit_singleton_package import Oracle
    from audit_singleton_precheck_independent import Spokes
    from deterministic_rppr._adaptive import solve_rppr_adaptive
    assert package.solve is package.solve_fast
    assert package.solve_adaptive is solve_rppr_adaptive
    assert 'solve_adaptive' in package.__all__
    assert inspect.signature(package.solve_adaptive)==inspect.signature(source)
    path=((1,),(0,2),(1,3),(2,4),(3,))
    asymmetric=((1,2),(0,2,3),(0,1),(1,))
    edge=((1,),(0,))
    cases=((path,0,F(13,97),F(1,64),F(1,1000000)),
           (asymmetric,3,F(1,16),F(1,32),F(1,1000000)),
           (edge,0,F(13,97),F((1<<20)-1,1<<20),F(1,16)),
           (edge,0,F(1),F(1,4),F(1,1000000)),
           (edge,0,F(1,3),F(1),F(1,1000000)))
    evidence=[]
    for rows,seed,alpha,rho,eps in cases:
        a,b=Oracle(rows),Oracle(rows)
        x,y=source(a,a.label(seed),alpha,rho,eps),package.solve_adaptive(b,b.label(seed),alpha,rho,eps)
        assert asdict(x)==asdict(y)
        assert a.degree_queries==b.degree_queries and a.row_queries==b.row_queries
        assert a.entries==b.entries
        certificate=package.certify_output(Oracle(rows),b.label(seed),alpha,rho,y.output)
        assert certificate.meets(eps)
        evidence.append(dict(stages=len(y.stages),iterations=sum(s.iterations for s in y.stages),
            source_passes=y.adaptive_metrics.source_passes,
            source_scans=y.adaptive_metrics.scanned_adjacency_entries,
            independent_gap_bound=str(certificate.objective_gap_bound),
            zero_stages=sum(s.iterations==0 for s in y.stages)))
    a,b=Spokes([10**30]),Spokes([10**30])
    args=a.seed,F(1,100),F(1,64),F(1,1000000)
    x,y=source(a,*args),package.solve_adaptive(b,*args)
    assert asdict(x)==asdict(y)
    assert a.degree_queries==b.degree_queries and a.row_queries==b.row_queries
    assert all(i==b.seed for i in b.row_queries)
    assert package.certify_output(Spokes([10**30]),*args[:3],y.output).meets(args[-1])
    evidence.append(dict(stages=len(y.stages),iterations=sum(s.iterations for s in y.stages),
        source_passes=y.adaptive_metrics.source_passes,
        source_scans=y.adaptive_metrics.scanned_adjacency_entries,
        inactive_hub_unscanned=True))
    LOG['exact_source_package_cases']=evidence
    LOG['no_hash_labels_in_all_cases']=True
    LOG['default_stays_fast']=True


def isolated():
    with tempfile.TemporaryDirectory(prefix='adaptive_package_',dir=ROOT) as temp:
        shutil.copytree(PKG,Path(temp)/'deterministic_rppr',ignore=shutil.ignore_patterns('__pycache__'))
        code=r'''
import builtins,json,sys
from fractions import Fraction as F
sys.path.insert(0,sys.argv[1])
original=builtins.__import__
def guarded(name,*args,**kwargs):
    if name in ('adaptive_source_continuation_rppr','combined_certified_dyadic_rppr',
                'direct_exception_integer_rppr','integer_dyadic_rppr_solver',
                'practical_dyadic_rppr_solver','deterministic_rppr_solver',
                'independent_sparse_certificate','deterministic_avl_containers'):
        raise AssertionError('Research import attempted: '+name)
    return original(name,*args,**kwargs)
builtins.__import__=guarded
import deterministic_rppr as package
assert package.solve is package.solve_fast
class NH(int):
    def __hash__(self):raise AssertionError('Vertex hashing forbidden')
offset=-(1<<2051)
class Path:
    def degree(self,i):
        j=int(i)-offset;assert 0<=j<5
        return 1 if j in (0,4) else 2
    def neighbors(self,i):
        j=int(i)-offset;assert 0<=j<5
        if j:yield NH(offset+j-1)
        if j<4:yield NH(offset+j+1)
seed=NH(offset)
result=package.solve_adaptive(Path(),seed,F(13,97),F(1,64),F(1,1000000))
check=package.certify_output(Path(),seed,F(13,97),F(1,64),result.output)
assert check.meets(F(1,1000000))
assert result.result.stages==result.stages
assert result.adaptive_metrics.source_passes==len(result.stages)-1
print(json.dumps(dict(successful=True,stages=len(result.stages),
    source_passes=result.adaptive_metrics.source_passes,
    certificate=str(check.objective_gap_bound))))
'''
        result=subprocess.run([sys.executable,'-I','-S','-B','-c',code,temp],cwd='/',
                              capture_output=True,text=True,check=True)
        LOG['isolated_package_only']=json.loads(result.stdout)


def main():
    audit_ast();exact_comparisons();isolated()
    LOG['passed']=LOG['successful']=True
    (ROOT/'adaptive_package_verification.json').write_text(json.dumps(LOG,indent=2)+'\n')
    print(json.dumps({k:v for k,v in LOG.items() if k!='transcription'},indent=2))


if __name__=='__main__':main()
