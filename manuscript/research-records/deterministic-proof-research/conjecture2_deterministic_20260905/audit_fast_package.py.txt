"""Independent audit of direct-transition package additions; no package edits."""
import ast,copy,hashlib,json,shutil,subprocess,sys,tempfile
from dataclasses import asdict
from fractions import Fraction as F
from pathlib import Path
sys.dont_write_bytecode=True
ROOT=Path(__file__).resolve().parent
PKGROOT=ROOT/'deliverables'/'deterministic-rppr'
PKG=PKGROOT/'deterministic_rppr'
from audit_accelerated_package import TinyOracle,HubOracle,NoHashInt

LOG=dict(direct_ast_equal=False,fast_wrapper_checked=False,prior_modules_unchanged=False,
         direct_exact_steps=0,fast_exact_steps=0,scalar_rebases=0,full_output_matches=0,
         full_stage_matches=0,checkpoints=0,zero_stages=0,direct_alpha_one=0,
         isolated_execution=False)

def dump(node):return ast.dump(node,include_attributes=False)

def audit_transcription():
    record=json.loads((PKGROOT/'DIRECT_TRANSCRIPTION.json').read_text())
    for name,value in record.items():
        assert hashlib.sha256((PKG/(name+'.py')).read_bytes()).hexdigest()==value['package_sha256']
    assert hashlib.sha256((ROOT/record['_direct']['source']).read_bytes()).hexdigest()==record['_direct']['source_sha256']
    assert hashlib.sha256((ROOT/record['_fast']['source_composition']).read_bytes()).hexdigest()==record['_fast']['source_composition_sha256']
    previous=json.loads((PKGROOT/'ACCELERATED_TRANSCRIPTION.json').read_text())
    all_prior={}
    for filename in ('TRANSCRIPTION.json','INTEGER_TRANSCRIPTION.json','ACCELERATED_TRANSCRIPTION.json'):
        all_prior.update(json.loads((PKGROOT/filename).read_text()))
    checked_prior=0
    for name,value in all_prior.items():
        if name=='__init__':continue
        expected=value.get('package_sha256',value.get('source_sha256'))
        assert hashlib.sha256((PKG/(name+'.py')).read_bytes()).hexdigest()==expected
        checked_prior+=1
    LOG['prior_module_hashes_checked']=checked_prior
    old_init=(PKG/'__init__.py').read_text().split('\n# Exact direct-transition composition.\n')[0]
    assert hashlib.sha256(old_init.encode()).hexdigest()==previous['__init__']['package_sha256']
    LOG['prior_modules_unchanged']=True
    source=ast.parse((ROOT/'direct_exception_integer_rppr.py').read_text())
    package=ast.parse((PKG/'_direct.py').read_text())
    reverse={'_avl':'deterministic_avl_containers','_source_energy':'source_energy_dyadic_rppr','_integer':'integer_dyadic_rppr_solver'}
    originals={node.name:node for node in source.body if isinstance(node,ast.FunctionDef)}
    new_nodes=[]
    for node in package.body:
        if isinstance(node,ast.ImportFrom) and node.level==1:
            if node.module is None:
                assert len(node.names)==1 and node.names[0].name=='_integer'
                node=ast.Import(names=[ast.alias(name=reverse['_integer'],asname=node.names[0].asname)])
            else:
                assert node.module in reverse
                node.module,node.level=reverse[node.module],0
        if isinstance(node,ast.FunctionDef) and node.name in ('solve_rppr_direct_integer','solve_rppr_direct_source'):
            cls='DirectExceptionIntegerCorrector' if node.name.endswith('_integer') else 'DirectSourceEnergyCorrector'
            expected=ast.parse(f'return reference.solve_rppr_integer(oracle, seed, alpha, rho, epsilon, corrector_class={cls})').body
            assert [dump(n) for n in node.body]==[dump(n) for n in expected]
            node.body=copy.deepcopy(originals[node.name].body)
        new_nodes.append(node)
    package.body=new_nodes
    source.body=[node for node in source.body if not
        (isinstance(node,ast.ImportFrom) and node.module=='types') and not
        (isinstance(node,ast.FunctionDef) and node.name=='_continuation') and not
        (isinstance(node,ast.Assign) and len(node.targets)==1 and isinstance(node.targets[0],ast.Name) and node.targets[0].id in ('_fixed','_source'))]
    assert dump(source)==dump(package)
    LOG['direct_ast_equal']=True
    fast=ast.parse((PKG/'_fast.py').read_text())
    imports=[node for node in fast.body if isinstance(node,ast.ImportFrom)]
    expected=ast.parse('from ._combined import CombinedDyadicCorrector\nfrom ._direct import DirectExceptionIntegerCorrector\nfrom ._integer import solve_rppr_integer').body
    assert [dump(n) for n in imports]==[dump(n) for n in expected]
    cls,=[node for node in fast.body if isinstance(node,ast.ClassDef)]
    composition=ast.parse((ROOT/'benchmark_direct_exception_transition.py').read_text())
    old,=[node for node in composition.body if isinstance(node,ast.ClassDef) and node.name=='DirectCombinedCorrector']
    assert [dump(n) for n in cls.bases]==[dump(n) for n in old.bases]
    assert not cls.keywords and not cls.decorator_list and len(cls.body)==1 and isinstance(cls.body[0],ast.Expr) and isinstance(cls.body[0].value,ast.Constant)
    wrapper,=[node for node in fast.body if isinstance(node,ast.FunctionDef)]
    assert wrapper.name=='solve_rppr_fast'
    expected=ast.parse('return solve_rppr_integer(oracle, seed, alpha, rho, epsilon, corrector_class=FastDyadicCorrector)').body[0]
    assert dump(wrapper.body[-1])==dump(expected)
    assert len(fast.body)==6 # doc,three imports,class,wrapper: no extra executable state
    LOG['fast_wrapper_checked']=True
    LOG['transcription_hashes']=record


def state(stage):
    return (stage.S,stage.H,[(name,list(getattr(stage,name).items())) for name in
             ('baseline','degree','adjacency','X','L','z','M','source')],list(stage.exceptions),
            list(stage.tree.records.items()),stage.last_step,stage.objective_gap_bound,stage.diagnostic_metrics)


def audit_behavior():
    sys.path.insert(0,str(PKGROOT))
    import deterministic_rppr as package
    from deterministic_rppr._direct import DirectExceptionIntegerCorrector as PD
    from direct_exception_integer_rppr import DirectExceptionIntegerCorrector as SD
    from benchmark_direct_exception_transition import DirectCombinedCorrector as SF,direct as source_fast
    assert package.solve is package.solve_source_energy
    assert 'solve_fast' in package.__all__ and 'FastDyadicCorrector' in package.__all__
    expected=['FastDyadicCorrector','CombinedDyadicCorrector','EarlyStopSourceEnergyCorrector','BinomialBlockDyadicCorrector','SourceEnergyDyadicCorrector','DirectExceptionIntegerCorrector','IntegerDyadicCorrector','object']
    assert [cls.__name__ for cls in package.FastDyadicCorrector.__mro__]==expected
    assert [cls.__name__ for cls in SF.__mro__][1:]==expected[1:]
    path8=((1,),(0,2),(1,3),(2,4),(3,5),(4,6),(5,7),(6,))
    for source,packaged,field in ((SD,PD,'direct_exact_steps'),(SF,package.FastDyadicCorrector,'fast_exact_steps')):
        a,b=TinyOracle(path8,1<<2048),TinyOracle(path8,1<<2048)
        args=(a.label(3),F(13,97),F(1,4),F(1,8),F(1,4096))
        x,y=source(a,*args),packaged(b,*args)
        limit=min(48,x.target_iterations,y.target_iterations)
        for k in range(limit):
            assert asdict(x.step())==asdict(y.step())
            assert state(x)==state(y)
            assert a.degree_queries==b.degree_queries and a.row_queries==b.row_queries
            LOG[field]+=1
        LOG['scalar_rebases']+=y.metrics.rebases
    path5=((1,),(0,2),(1,3),(2,4),(3,))
    asymmetric=((1,2),(0,2,3),(0,1),(1,4),(3,))
    cases=[(path5,0,F(1,100),F(1,64),F(1,1000000),0),
           (asymmetric,4,F(13,97),F(1,32),F(1,1000000),-(1<<2048)),
           (((1,),(0,)),0,F(13,97),F(4095,4096),F(1,16),0),
           (path5,2,F(13,97),F(1,2),F(1,1000000),0),
           (path5,0,F(1),F(1,4),F(1,1000000),0)]
    for rows,seed,alpha,rho,eps,base in cases:
        a,b=TinyOracle(rows,base),TinyOracle(rows,base)
        x=source_fast(a,a.label(seed),alpha,rho,eps)
        y=package.solve_fast(b,b.label(seed),alpha,rho,eps)
        assert asdict(x)==asdict(y)
        assert a.degree_queries==b.degree_queries and a.row_queries==b.row_queries
        assert package.certify_output(TinyOracle(rows,base),b.label(seed),alpha,rho,y.output).meets(eps)
        for stage in y.stages:
            cm=stage.corrector_metrics['checkpoints']
            assert cm['cache_misses']==0 and cm['checker_containers']==cm['checker_reclaimed_containers']
            LOG['checkpoints']+=cm['checkpoints'];LOG['full_stage_matches']+=1
            LOG['zero_stages']+=int(stage.iterations==0)
        if alpha==1:
            assert not b.row_queries;LOG['direct_alpha_one']+=1
        LOG['full_output_matches']+=1
    a,b=HubOracle(),HubOracle()
    args=(a.seed,F(1,100),F(1,64),F(1,1000000))
    x,y=source_fast(a,*args),package.solve_fast(b,*args)
    assert asdict(x)==asdict(y)
    assert a.degree_queries==b.degree_queries and a.row_queries==b.row_queries
    assert all(i==b.seed for i in b.row_queries)
    assert package.certify_output(HubOracle(),b.seed,F(1,100),F(1,64),y.output).meets(F(1,1000000))
    LOG['full_output_matches']+=1
    LOG['full_stage_matches']+=len(y.stages)
    LOG['checkpoints']+=sum(stage.corrector_metrics['checkpoints']['checkpoints'] for stage in y.stages)

        shutil.copytree(PKG,Path(temp)/'deterministic_rppr',ignore=shutil.ignore_patterns('__pycache__'))
        code=r'''

        result=subprocess.run([sys.executable,'-I','-S','-B','-c',code,temp],cwd='/',capture_output=True,text=True,check=True)

def main():
    audit_transcription();audit_behavior();isolated()
    LOG['successful']=True
    (ROOT/'fast_package_independent_verification.json').write_text(json.dumps(LOG,indent=2)+'\n')
    print(json.dumps({k:v for k,v in LOG.items() if k!='transcription_hashes'},indent=2))

if __name__=='__main__':main()
