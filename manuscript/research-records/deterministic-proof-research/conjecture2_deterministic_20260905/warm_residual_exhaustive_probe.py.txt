"""Exhaustive deterministic small-graph diagnostics. Warm pairs are rational exact.
Floating accelerated trajectories are diagnostics, not interval certificates.
"""
import argparse,json,itertools,math
from fractions import Fraction as F
import numpy as np

def solve(H,rhs):
 n=len(rhs);a=[list(row)+[rhs[i]] for i,row in enumerate(H)]
 for j in range(n):
  for i in range(j+1,n):
   if a[i][j]:
    v=a[i][j]/a[j][j]
    for k in range(j,n+1):a[i][k]-=v*a[j][k]
 x=[F(0)]*n
 for i in range(n-1,-1,-1):x[i]=(a[i][-1]-sum(a[i][j]*x[j] for j in range(i+1,n)))/a[i][i]
 return x

def warm(A,alpha,rho):
 n=len(A);d=[sum(row) for row in A];c=(1-alpha)/2;q0=(1+alpha)/2
 H=[[q0*d[i] if i==j else -c*A[i][j] for j in range(n)] for i in range(n)]
 def opt(reg):
  b=[-alpha*reg*d[i] for i in range(n)];b[0]+=alpha
  S={0} if b[0]>0 else set();f=[F(0)]*n
  while S:
   s=sorted(S);v=solve([[H[i][j] for j in s]for i in s],[b[i] for i in s]);f=[F(0)]*n
   for i,vi in zip(s,v):
    assert vi>0
    f[i]=vi
   add={i for i in range(n) if i not in S and sum(H[i][j]*f[j] for j in S)<b[i]}
   if not add:break
   S|=add
  return f
 return opt(rho),opt(2*rho)

def connected(A):
 seen={0};stack=[0]
 while stack:
  i=stack.pop()
  for j,v in enumerate(A[i]):
   if v and j not in seen:seen.add(j);stack.append(j)
 return len(seen)==len(A)

def trajectory(A,L,rho,fs,fo,variant,K):
 d=np.array([sum(row) for row in A],dtype=float);P=np.array(A,dtype=float)/d[None,:]
 th=1/L;al=th*th;a=1-th;c=(1-al)/2;lam=al*float(rho)
 us=np.array([float(x) for x in fs])*d;u=np.array([float(x) for x in fo])*d;V=th*u;old=u.copy()
 def res(v):return al*v+c*(v-P@v)
 r0=res(u-us);initial=np.dot(r0,r0/d)
 if initial==0:return None
 sums=np.zeros(3);peaks=np.zeros(3);ghost=cap=0
 for k in range(K):
  err=u-us;q=res(err);ll=c*(err-P@err);g=res(u)+lam*d;g[0]-=al;gm=np.minimum(u,g)
  vals=np.array([q@(q/d),ll@(ll/d),gm@(gm/d)]);sums+=vals;peaks=np.maximum(peaks,vals)
  if variant=='pg':
   y=u+a/(1+th)*(u-old);gr=res(y)+lam*d;gr[0]-=al;unew=np.maximum(y-gr,0);old=u;u=unew
  else:
   raw=a/2*(P@(u+V)-u+V);raw[0]+=al;raw-=lam*d;vn=np.maximum(raw,0)
   if vn.sum()>th:
    cap+=1;ix=np.flatnonzero(vn>0);order=ix[np.argsort(vn[ix]/d[ix])[::-1]]
    levels=(np.cumsum(vn[order])-th)/np.cumsum(d[order]);jj=np.flatnonzero(vn[order]/d[order]>levels)[-1];vn=np.maximum(raw-levels[jj]*d,0)
   if variant=='aligned':
    pg=a/2*(u+V+P@(u+V))-lam*d;pg[0]+=al;u=np.minimum(np.maximum(pg,0),a*u+vn)
   elif variant=='monotone':
    dire=vn-th*u;grad=res(u)+lam*d;grad[0]-=al;dot=grad@(dire/d);den=res(dire)@(dire/d);ss=np.clip(-dot/den,0,1) if den>0 else 0;u+=ss*dire
   else:u=a*u+vn
   V=vn
  ghost+=int(np.count_nonzero((u>0)&(us==0)))
 return dict(ratioQ=th*sums[0]/initial,ratioL=th*sums[1]/initial,targetL=th*sums[1]/(al*al*float(rho)),peakQ=peaks[0]/initial,initial=initial,ghost=ghost,cap=cap)

if __name__=='__main__':
 p=argparse.ArgumentParser();p.add_argument('--n',nargs='+',type=int,default=[2,3,4,5]);p.add_argument('--L',nargs='+',type=int,default=[8,32]);p.add_argument('--variant',nargs='+',default=['lazy','pg']);p.add_argument('--horizon',type=int,default=16);p.add_argument('--out',required=True);args=p.parse_args()
 best={};total=graphs=representatives=0;amplifications=[]
 with open(args.out,'w') as out:
  for n in args.n:
   pairs=list(itertools.combinations(range(n),2))
   for mask in range(1<<len(pairs)):
    A=[[0]*n for _ in range(n)];edges=[]
    for j,(u,v) in enumerate(pairs):
     if mask>>j&1:A[u][v]=A[v][u]=1;edges.append([u,v])
    if not connected(A):continue
    graphs+=1
    canonical=True
    for tail in itertools.permutations(range(1,n)):
     order=(0,)+tail
     remask=sum(A[order[i]][order[j]]<<k for k,(i,j) in enumerate(pairs))
     if remask<mask:canonical=False;break
    if not canonical:continue
    representatives+=1;vol=2*len(edges);ds=sum(A[0])
    for L in args.L:
     for rho in sorted({F(1,vol),F(2,vol),F(1,2*vol),F(1,2*ds),F(1,4*ds),F(1,8*ds)}):
      fs,fo=warm(A,F(1,L*L),rho)
      for variant in args.variant:
       r=trajectory(A,L,rho,fs,fo,variant,args.horizon*L)
       if r is None:continue
       total+=1;r.update(n=n,mask=mask,edges=edges,L=L,rho=str(rho),variant=variant,fs=[str(x) for x in fs],fo=[str(x) for x in fo])
       for key in ['ratioQ','ratioL','targetL','peakQ']:
        if key not in best or r[key]>best[key][key]:
         best[key]=r;out.write(json.dumps(dict(record='best',metric=key,case=r))+'\n');out.flush()
       if r['peakQ']>1.000001:
        amplifications.append(r);out.write(json.dumps(dict(record='amplification',case=r))+'\n');out.flush()
    if representatives%10==0:print(json.dumps(dict(graphs=graphs,representatives=representatives,total=total,best={k: v[k] for k,v in best.items()},amplifications=len(amplifications))),flush=True)
  summary=dict(record='summary',graphs=graphs,representatives=representatives,total=total,best=best,amplifications=len(amplifications));out.write(json.dumps(summary)+'\n');print(json.dumps(summary),flush=True)
