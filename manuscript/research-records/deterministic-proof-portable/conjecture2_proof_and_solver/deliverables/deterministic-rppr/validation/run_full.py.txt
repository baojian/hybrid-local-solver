def serialize(value):
    if isinstance(value, F):
        return str(value)
    if is_dataclass(value):
        return serialize(asdict(value))
    if isinstance(value, dict):
        return {str(k): serialize(v) for k,v in value.items()}
    if isinstance(value, (tuple,list)):
        return [serialize(v) for v in value]
    return value

def guard():
    if CASE_STARTED is not None and time.monotonic()-CASE_STARTED>GUARDS['per_case_seconds']:
        raise TimeoutError('Fixed per-case wall-time guard reached')
    if time.monotonic()-BUDGET['started'] > GUARDS['total_seconds']:
        raise RuntimeError('Fixed total wall-time guard reached')
    for actual,limit in [('iterations','total_iterations'),('entries','total_oracle_entries'),
                         ('degrees','total_degree_replies')]:
        if BUDGET[actual] > GUARDS[limit]:
            raise RuntimeError('Fixed resource guard reached: '+limit)

class StrictLocalOracle:
    def __init__(self, problem, reverse=False):
        self.problem, self.reverse = problem, reverse
        self.known = {problem.seed}
        self.phase = 'solver'
        self.counts = dict(degrees=0, rows=0, entries=0)
        self.phases = {}
        self.degree_vertices, self.row_vertices = {}, {}

    def increment(self, kind, vertex):
        self.counts[kind] += 1
        self.phases.setdefault(self.phase,dict(degrees=0,rows=0,entries=0))[kind] += 1
        if kind in ('degrees','entries'):
            BUDGET[kind] += 1
        if kind == 'degrees':
            self.degree_vertices[vertex] = self.degree_vertices.get(vertex,0)+1
        elif kind == 'rows':
            self.row_vertices[vertex] = self.row_vertices.get(vertex,0)+1
        guard()

    def degree(self, vertex):
        assert isinstance(vertex,int) and vertex in self.known, 'Degree before local discovery'
        self.increment('degrees',vertex)
        return self.problem.degree(vertex)

    def neighbors(self, vertex):
        assert isinstance(vertex,int) and vertex in self.known, 'Row before local discovery'
        self.increment('rows',vertex)
        row = self.problem.row(vertex)
        for neighbor in reversed(row) if self.reverse else row:
            self.increment('entries',vertex)
            self.known.add(neighbor)
            yield neighbor

class Problem:
    def __init__(self, case):
        self.case, self.alpha = case, case['alpha']
        self.star = 'star_leaves' in case
        if self.star:
            self.leaves = case['star_leaves']
            self.seed = self.leaves+1
            self.vertices = (1,self.seed)
            self.rows = None
        else:
            self.seed, self.rows = case['seed'],case['rows']
            self.vertices = tuple(sorted(self.rows))
            assert self.seed in self.rows
            for i,row in self.rows.items():
                assert isinstance(i,int) and row and i not in row and len(set(row))==len(row)
                assert all(j in self.rows and i in self.rows[j] for j in row)
            reached, stack = {self.seed},[self.seed]
            while stack:
                for j in self.rows[stack.pop()]:
                    if j not in reached:
                        reached.add(j); stack.append(j)
            assert reached==set(self.rows)
        self.optima={}

    def degree(self,i):
        if not self.star:
            return len(self.rows[i])
        assert i in self.vertices
        return self.leaves if i==1 else 1

    def row(self,i):
        if not self.star:
            return self.rows[i]
        assert i==self.seed, 'An inactive huge star hub was scanned'
        return (1,)

    def optimum(self,r):
        if r not in self.optima:
            if self.star:
                value=max(F(0),2*self.alpha*(1-r)/(1+self.alpha))
                values={self.seed:value} if value else {}
                assert (1-self.alpha)*value/(2*self.leaves)<=self.alpha*r
            else:
                values=exact_optimum(self.rows,self.seed,self.alpha,r)
            density=self.source(values)
            assert all(0<=v<=self.alpha*r for v in density.values())
            assert all(density[i]==self.alpha*r for i,v in values.items() if v)
            self.optima[r]=values
        return self.optima[r]

    def source(self,values):
        if not self.star:
            return source(self.rows,self.seed,self.alpha,values)
        assert set(values).issubset({self.seed})
        value=values.get(self.seed,F(0))
        return {self.seed:self.alpha-(1+self.alpha)*value/2,
                1:(1-self.alpha)*value/(2*self.leaves)}

    def objective(self,r,values):
        if not self.star:
            return objective(self.rows,self.seed,self.alpha,r,values)
        assert set(values).issubset({self.seed})
        value=values.get(self.seed,F(0))
        return (1+self.alpha)*value**2/4-self.alpha*value+self.alpha*r*value

    def mass(self,values):
        return sum((self.degree(i)*v for i,v in values.items()),F(0))

    def volume(self,values):
        return sum(self.degree(i) for i,v in values.items() if v)

    def unpack(self,output):
        values={}
        for i,value,degree in output:
            assert i not in values and i in self.vertices
            assert value>0 and degree==self.degree(i)
            values[i]=value
        return values

    def independent_gap_bound(self,r,values):
        s=self.source(values)
        square=F(0)
        for i,si in s.items():
            g=self.alpha*r-si
            if not values.get(i,0):
                g=min(g,F(0))
            square+=self.degree(i)*g*g
        # Every other leaf in the analytic star has zero output and positive
        # orthant gradient alpha*r, so contributes zero selected subgradient.
        return square/(2*self.alpha)

def observed_source(oracle,*args,**kwargs):
    before=dict(oracle.counts)
    previous_phase=oracle.phase
    oracle.phase='adaptive_source'
    try:
        returned=ORIGINAL_SOURCE(oracle,*args,**kwargs)
    finally:
        oracle.phase=previous_phase
    after=oracle.counts
    actual={key:after[key]-before[key] for key in before}
    maximum,vertex,metrics=returned
    assert actual['degrees']==metrics.degree_replies
    assert actual['entries']==metrics.first_adjacency_entries
    assert actual['rows']==metrics.first_row_requests
    CONTEXT['observed_source_passes'].append(dict(maximum=maximum,vertex=vertex,
                                                 actual_oracle=actual,metrics=asdict(metrics)))
    return returned

def initial_checks(c):
    eta,tau,r,h,T=c.mass_cap,c.tolerance,c.rho,c.grid,c.T
    assert c.mass_cap_hint==eta and c.source_mass==c.alpha*eta
    assert r<=eta<=1 and 0<tau<eta
    assert c.relaxed_initialization_tolerance==tau/eta
    assert c.gamma_bound==29*eta*h/c.theta<tau/8
    assert h<=min(F(1,8),c.theta*tau/(256*eta),c.theta*c.alpha*c.alpha*r/29)
    assert c.gamma_bound<=eta*c.alpha*c.alpha*r
    assert c.metrics.iterations==c.metrics.completed_blocks==0
    assert c._accepted_bound is None and c._accepted_iteration is None
    assert c._next_checkpoint==T and not c._checker_containers
    assert c.checkpoint_metrics.checkpoints==0
    assert (c.block_power_numerator,c.block_power_denominator,c.decay_denominator)==(1,1,1)
    assert c.objective_gap_bound==c.initial_energy_bound
    q=0
    while F(1,2**q)>tau/2:q+=1
    assert c.original_target_iterations==T*q
    original_blocks=q
    q=0
    while c.initial_energy_bound/F(2**q)>tau/2:q+=1
    assert c.half_bound_blocks==q and c.half_bound_target_iterations==T*q
    half_blocks=q
    q,power=0,F(1)
    while c.initial_energy_bound*power>tau/2:
        q+=1;power*=c.block_bound.value
    assert c.blocks==q and c.target_iterations==T*q
    assert F(c.target_power_numerator,c.target_power_denominator)==power
    assert c.target_iterations<=c.half_bound_target_iterations<=c.original_target_iterations
    old=c.relaxed_horizons
    for field in ('original_target_iterations','half_bound_target_iterations','target_iterations'):
        assert old[field]<=getattr(c,field)
    relaxed,extra=c.relaxed_schedule_work,asdict(c.retargeting_work)
    assert c.metrics.bound_doublings==relaxed['bound_doublings']+extra['original_bound_doublings']
    assert c.source_energy_schedule_checks==relaxed['source_checks']+extra['source_schedule_checks']
    assert c.source_energy_schedule_doublings==relaxed['source_doublings']+extra['source_schedule_doublings']
    assert c.binomial_schedule_checks==relaxed['binomial_checks']+extra['binomial_schedule_checks']
    assert c.binomial_schedule_power_updates==relaxed['binomial_power_updates']+extra['binomial_schedule_power_updates']
    return dict(eta=eta,tau=tau,grid=h,gamma=c.gamma_bound,
                original_blocks=original_blocks,half_blocks=half_blocks,binomial_blocks=q,
                target_iterations=c.target_iterations,target_power=power,
                relaxed_horizons=c.relaxed_horizons,retargeting_work=extra)

def verify_case(case):
    global CONTEXT
    problem=Problem(case)
    oracle=StrictLocalOracle(problem,case.get('reverse',False))
    CONTEXT=dict(candidates=[],observed_source_passes=[],initializations=[],
        construction_oracle_counts=[],hint_zero_graph_access_checks=0,
        release_zero_graph_access_checks=0)
    alpha,rho,epsilon=(case[k] for k in ('alpha','rho','epsilon'))
    answer=SOLVER(oracle,problem.seed,alpha,rho,epsilon)
    CONTEXT['raw_result']=answer
    inner=answer if BACKEND=='fast' else answer.result
    am=answer.adaptive_metrics if BACKEND=='adaptive' else SimpleNamespace(
        source_passes=0,initial_seed_source_evaluations=0,degree_replies=0,
        first_adjacency_entries=0,first_row_requests=0)
    schedule=answer.schedule if BACKEND=='adaptive' else [None]*len(inner.stages)
    assert len(inner.stages)==len(schedule)==len(CONTEXT['candidates'])
    if BACKEND=='mass-grid':
        assert len(inner.stages)==len(answer.mass_hints)==len(CONTEXT['initializations'])
    previous,previous_r={},F(1,problem.degree(problem.seed))
    previous_mass=F(0)
    stage_checks=[]
    global_max_outside=0
    previous_grid=F(1)
    for index,(stage,record,candidate_record) in enumerate(zip(
            inner.stages,schedule,CONTEXT['candidates'])):
        r,delta,h=stage.regularization,stage.delta,stage.grid
        full_source=problem.source(previous)
        maximum=max(full_source.values())
        if BACKEND=='adaptive':
            assert record.previous_regularization==previous_r
            assert record.source_maximum==maximum
            assert full_source[record.maximum_vertex]==maximum
            assert record.proposed_regularization==maximum/(4*alpha)
            assert r==record.chosen_regularization==max(rho,maximum/(4*alpha))
            assert record.target_clamped==(rho>maximum/(4*alpha))
            assert previous_r/4<=record.proposed_regularization<=previous_r/2
        else:
            assert r==max(rho,previous_r/2)
        assert alpha*previous_r<=maximum<=2*alpha*previous_r
        assert previous_r/4<=r<previous_r and rho<=r
        if r!=rho:
            assert r<=previous_r/2
        assert all(0<=v<=4*alpha*r for v in full_source.values())
        outside=bool(index and previous and any(
            v==maximum and i not in previous for i,v in full_source.items()))
        global_max_outside+=int(outside)
        exact=problem.optimum(r)
        eta=1-previous_mass
        eta_star=1-problem.mass(exact)
        assert r<=eta<=4*eta_star
        assert problem.volume(previous)==stage.baseline_volume<=eta/r
        assert stage.source_records<=1+2*stage.baseline_volume
        assert stage.projection_mass_cap==eta and stage.source_mass==alpha*eta
        assert sum((problem.degree(i)*v for i,v in full_source.items()),F(0))==alpha*eta
        assert stage.requested_gap==alpha*delta*delta/8
        expected_delta=alpha*r/2
        if r==rho:
            while expected_delta**2>epsilon*r/2:
                expected_delta/=2
        assert delta==expected_delta
        assert 0<=stage.certified_gap<stage.requested_gap
        assert h.numerator==1 and not h.denominator&(h.denominator-1)
        if BACKEND=='mass-grid':
            hint=answer.mass_hints[index]
            init=CONTEXT['initializations'][index]
            assert hint.regularization==r and hint.mass_cap==eta
            assert hint.baseline_records==len(previous)
            assert hint.previous_grid_denominator==(previous_grid.denominator if index else None)
            # The exact baseline must fit the new grid; an empty or coarser
            # represented baseline does not require nested historical grids.
            assert all(h.denominator%value.denominator==0 for value in previous.values())
            assert all(previous_grid.denominator%value.denominator==0 for value in previous.values())
            assert h<=inner.theta*stage.requested_gap/(256*eta)
            gamma=29*eta*h/inner.theta
            assert init['gamma']==gamma and init['tau']==stage.requested_gap and init['eta']==eta
        else:
            assert h<=previous_grid and h.denominator%previous_grid.denominator==0
            assert inner.theta*stage.requested_gap/512<h<=inner.theta*stage.requested_gap/256
            gamma=29*h/inner.theta
        assert gamma<stage.requested_gap/8 and gamma<=alpha*alpha*r*eta
        assert 2*h<=alpha*r/4 and h<=delta/2
        assert stage.kinetic_volume<=360*eta*stage.iterations/r
        assert stage.kinetic_volume<=stage.kinetic_volume_bound==360*F(stage.iterations)/r
        cm=stage.corrector_metrics
        assert cm['iterations']==stage.iterations and cm['kinetic_volume']==stage.kinetic_volume
        assert cm['rebase_adjacency_entries']==0
        if BACKEND=='mass-grid':
            mg=cm['mass_scaled_grid']
            assert mg['actual_tolerance']==stage.requested_gap and mg['gamma_bound']==gamma
            assert mg['mass_cap_hint']==eta and mg['relaxed_initialization_tolerance']==stage.requested_gap/eta
            assert mg['retargeting_work']==init['retargeting_work']
            assert stage.iterations<=cm['prescribed_target_iterations']==init['target_iterations']
            assert cm['prescribed_target_iterations']<=cm['binomial_block']['half_bound_target_iterations']<=cm['original_target_iterations']
            old,extra=mg['relaxed_schedule_work'],mg['retargeting_work']
            assert cm['bound_doublings']==old['bound_doublings']+extra['original_bound_doublings']
            assert cm['source_energy_schedule_checks']==old['source_checks']+extra['source_schedule_checks']
            assert cm['source_energy_schedule_doublings']==old['source_doublings']+extra['source_schedule_doublings']
            assert cm['binomial_block']['schedule_checks']==old['binomial_checks']+extra['binomial_schedule_checks']
            assert cm['binomial_block']['schedule_power_updates']==old['binomial_power_updates']+extra['binomial_schedule_power_updates']
        cp=cm['checkpoints']
        assert cp['cache_misses']==0
        assert cp['checker_containers']==cp['checker_reclaimed_containers']==6*cp['checkpoints']
        assert cp['checkpoints']==cp['accepted_checkpoints']+cp['rejected_checkpoints']
        assert cp['candidate_volume']==cp['cached_adjacency_entries']
        assert cp['candidate_volume']<=eta*(cp['checkpoints']+360*cp['checkpoint_iteration_sum'])/r
        if cp['checkpoints']:
            assert cp['checkpoint_iteration_sum']<2*stage.iterations
        else:
            assert cp['checkpoint_iteration_sum']==0
        current=problem.unpack(stage.repaired_output)
        candidate=problem.unpack(candidate_record['candidate'])
        assert candidate_record['regularization']==r
        assert candidate_record['certified_gap']==stage.certified_gap
        pre_gap=problem.objective(r,candidate)-problem.objective(r,exact)
        assert 0<=pre_gap<=stage.certified_gap
        assert problem.mass(candidate)<=1
        for i in problem.vertices:
            assert 0<=previous.get(i,0)<=current.get(i,0)<=exact.get(i,0)
            assert exact.get(i,0)-current.get(i,0)<=2*delta
        current_source=problem.source(current)
        assert all(0<=v<=2*alpha*r for v in current_source.values())
        assert alpha*r<=max(current_source.values())<=2*alpha*r
        mass=problem.mass(current)
        assert previous_mass<=mass<=1 and stage.repaired_mass==mass
        assert stage.repaired_volume==problem.volume(current)<=eta_star/r
        assert eta_star<=1-mass<=(1+alpha)*eta_star
        assert all(h.denominator%value.denominator==0 for value in current.values())
        gap=problem.objective(r,current)-problem.objective(r,exact)
        assert 0<=gap<=2*delta**2/r
        if BACKEND=='adaptive':
            sm=record.source_metrics
            assert sm.source_passes==int(index>0)
            assert sm.initial_seed_source_evaluations==int(index==0)
            assert sm.scanned_adjacency_entries==problem.volume(previous)
            assert sm.baseline_input_records==len(previous)
            assert sm.baseline_input_volume==problem.volume(previous)
            assert sm.temporary_records_reclaimed==sm.baseline_input_records+sm.affected_records
            if index:
                observed=CONTEXT['observed_source_passes'][index-1]
                assert observed['maximum']==maximum and observed['vertex']==record.maximum_vertex
                assert observed['metrics']==asdict(sm)
                assert sm.affected_records<=1+2*problem.volume(previous)
                assert sm.maximum_comparisons==sm.affected_records
                assert sm.rational_maximum_constructions==1
        stage_checks.append(dict(index=index,exact_optimum=exact,baseline_source=full_source,
            repaired_source=current_source,source_maximum_outside_nonempty_baseline=outside,
            actual_candidate_gap=pre_gap,actual_repaired_gap=gap,repair_gap_bound=2*delta**2/r,
            eta=eta,eta_star=eta_star,gamma=gamma,
            strengthened_kinetic_bound=360*eta*stage.iterations/r))
        previous,previous_r,previous_mass,previous_grid=current,r,mass,h

    if inner.stages:
        assert inner.stages[-1].regularization==rho
        if BACKEND=='adaptive':
            assert am.source_passes==len(inner.stages)-1
            assert am.initial_seed_source_evaluations==1
        assert sum((1/stage.regularization for stage in inner.stages),F(0))<3/rho
    else:
        assert not schedule and not CONTEXT['observed_source_passes']
        if BACKEND=='adaptive':
            assert all(getattr(am,f.name)==0 for f in fields(am))
        assert inner.objective_gap_bound==0
        assert alpha==1 or rho*problem.degree(problem.seed)>=1
    if BACKEND=='adaptive':
        for f in fields(am):
            if f.name in ('scheduling_records','source_interface_checks'):
                assert getattr(am,f.name)==len(inner.stages)
            else:
                assert getattr(am,f.name)==sum(getattr(rec.source_metrics,f.name) for rec in answer.schedule)
    if BACKEND=='mass-grid':
        hm=answer.mass_hint_metrics
        count=len(inner.stages)
        assert hm.factory_calls==hm.hint_interface_checks==hm.previous_references_released==count
        assert hm.initial_unit_hints==int(count>0)
        assert hm.rational_mass_hints==max(0,count-1)
        records=sum(hint.baseline_records for hint in answer.mass_hints)
        assert hm.baseline_input_records==hm.previous_degree_cache_lookups==hm.baseline_count_conversions==hm.mass_integer_updates==records
        assert CONTEXT['hint_zero_graph_access_checks']==count
        assert CONTEXT['release_zero_graph_access_checks']==int(count>0)
    assert len(CONTEXT['observed_source_passes'])==am.source_passes
    assert oracle.counts['degrees']==inner.metrics.seed_degree_queries+inner.metrics.repair_degree_replies+am.degree_replies+sum(
        s.corrector_metrics['degree_replies'] for s in inner.stages)
    assert oracle.counts['entries']==inner.metrics.repair_first_adjacency_entries+am.first_adjacency_entries+sum(
        s.corrector_metrics['first_adjacency_entries'] for s in inner.stages)
    assert oracle.phases.get('adaptive_source',{}).get('degrees',0)==am.degree_replies
    assert oracle.phases.get('adaptive_source',{}).get('entries',0)==am.first_adjacency_entries
    assert oracle.phases.get('adaptive_source',{}).get('rows',0)==am.first_row_requests
    values=problem.unpack(answer.output)
    exact=problem.optimum(rho)
    assert all(0<=values.get(i,0)<=exact.get(i,0) for i in problem.vertices)
    gap=problem.objective(rho,values)-problem.objective(rho,exact)
    assert 0<=gap<=inner.objective_gap_bound<=epsilon
    independent_bound=problem.independent_gap_bound(rho,values)
    assert gap<=independent_bound
    assert problem.volume(values)<=1/rho and 0<=problem.mass(values)<=1
    assert inner.metrics.final_output_words==3*len(values)
    if inner.stages:
        assert answer.output==inner.stages[-1].repaired_output
    final_source=problem.source(values)
    assert all(0<=v<=2*alpha*rho for v in final_source.values())
    assert sum((problem.degree(i)*v for i,v in final_source.items()),F(0))==alpha*(1-problem.mass(values))
    if problem.star:
        assert set(oracle.row_vertices).issubset({problem.seed})
    return answer,dict(actual_gap=gap,exact_optimum=exact,final_source=final_source,
        independent_subgradient_gap_bound=independent_bound,
        independent_certificate_meets_epsilon=independent_bound<=epsilon,
        stages=stage_checks,source_maxima_outside_nonempty_baseline=global_max_outside,
        analytic_star=problem.star,oracle_counts=oracle.counts,oracle_phases=oracle.phases,
        oracle_degree_vertices=oracle.degree_vertices,oracle_row_vertices=oracle.row_vertices,
        discovered_vertices=sorted(oracle.known),
        complete_candidates=CONTEXT['candidates'],observed_source_passes=CONTEXT['observed_source_passes'],
        initialization_checks=CONTEXT['initializations'],
        factory_hint_zero_graph_access_checks=CONTEXT['hint_zero_graph_access_checks'],
        factory_release_zero_graph_access_checks=CONTEXT['release_zero_graph_access_checks'])


def load_manifest(path):
    raw=json.loads(Path(path).read_text())
    digest=hashlib.sha256(json.dumps(raw,sort_keys=True).encode()).hexdigest()
    if digest!=MANIFEST_HASH or len(raw)!=183:
        raise ValueError('cases.json does not match the fixed audited 183-case manifest')
    cases=[]
    for item in raw:
        case=dict(item)
        for name in ('alpha','rho','epsilon'):
            case[name]=F(case[name])
        if 'rows' in case:
            case['rows']={int(i):tuple(row) for i,row in case['rows'].items()}
        if 'inverse' in case:
            case['inverse']={int(i):v for i,v in case['inverse'].items()}
        cases.append(case)
    names={case['name'] for case in cases}
    if len(names)!=183:
        raise ValueError('Duplicate case names')
    for case in cases:
        if 'compare_to' in case and case['compare_to'] not in names:
            raise ValueError('Missing invariance reference case')
    return cases,raw,digest

            if p.suffix=='.py' and p.is_relative_to(ROOT):
                paths.add(p)
    return {str(p.relative_to(ROOT)):hashlib.sha256(p.read_bytes()).hexdigest()
            for p in sorted(paths)}


def write(target,evidence):
    target=Path(target)
    temporary=target.with_name(target.name+'.tmp')
    temporary.write_text(json.dumps(serialize(evidence),indent=2)+'\n')
    temporary.replace(target)


def main(argv=None):
    global BACKEND,SOLVER,ORIGINAL_SOURCE,CASE_STARTED
    if not __debug__:
        raise RuntimeError('Run validation without Python -O: assertions are required')
    parser=argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--backend',choices=('fast','adaptive','mass-grid'),default='fast')
    parser.add_argument('--smoke',action='store_true',help='Run the fixed five-case smoke subset only')
    parser.add_argument('--check-data',action='store_true',help='Check case transcription/hash without running a solver')
    parser.add_argument('--output',type=Path,help='JSON result path; completed records are saved after every case')
    args=parser.parse_args(argv)
    BACKEND=args.backend
    SOLVER={'fast':solve_fast,'adaptive':solve_adaptive,'mass-grid':solve_mass_scaled_grid}[BACKEND]
    all_cases,raw,digest=load_manifest(Path(__file__).with_name('cases.json'))
    cases=[case for case in all_cases if case['name'] in SMOKE_NAMES] if args.smoke else all_cases
    selected_names={case['name'] for case in cases}
    assert all(case.get('compare_to',case['name']) in selected_names for case in cases)
    target=args.output or Path.cwd()/('validation_'+BACKEND+('_smoke' if args.smoke else '')+'.json')
    hashes=sources()
    evidence=dict(status='running',backend=BACKEND,deterministic=True,
        scope='smoke_subset' if args.smoke else 'full_183_case_manifest',
        full_manifest_completed=False,selected_cases_completed=False,
        manifest_case_count=183,selected_case_count=len(cases),manifest_sha256=digest,
        selected_names=[case['name'] for case in cases],manifest=raw,
        runtime=dict(python=sys.version,executable=sys.executable,platform=platform.platform()),
        resource_guards=GUARDS,alarm_preemption_available=hasattr(signal,'setitimer'),
        source_hashes_before=hashes,completed=[],
        observer_scope='Guard steps, record returned stage candidates, meter adaptive helper calls, and audit mass-grid initialization/hint access; numerical operations unchanged.',
        analytic_star_scope='Seed/hub formulas cover the omitted zero leaf class; no hub row is read.')
    write(target,evidence)
    if args.check_data:
        evidence.update(status='data_verified',solver_cases_run=0,source_hashes_after=sources())
        write(target,evidence)
        print('DATA VERIFIED:',digest)
        return 0
    namespace=SOLVER.__globals__
    key={'fast':'FastDyadicCorrector','adaptive':'AdaptiveFastCorrector','mass-grid':'MassScaledGridCorrector'}[BACKEND]
    original_corrector=namespace[key]
    class GuardCorrector(original_corrector):
        def __init__(self,*args,**kwargs):
            if BACKEND=='mass-grid':
                CONTEXT['construction_oracle_counts'].append(dict(args[0].counts))
            super().__init__(*args,**kwargs)
            if BACKEND=='mass-grid':
                CONTEXT['initializations'].append(initial_checks(self))
        def step(self):
            BUDGET['iterations']+=1
            guard()
            return super().step()
        def run(self):
            candidate=super().run()
            CONTEXT['candidates'].append(dict(regularization=self.rho,candidate=candidate,
                                              certified_gap=self.objective_gap_bound))
            return candidate
    namespace[key]=GuardCorrector
    original_factory=None
    if BACKEND=='mass-grid':
        original_factory=namespace['MassCapHintFactory']
        class ObservedFactory(original_factory):
            def __call__(self,*args,**kwargs):
                before=dict(args[0].counts)
                answer=super().__call__(*args,**kwargs)
                assert before==CONTEXT['construction_oracle_counts'][-1]
                CONTEXT['hint_zero_graph_access_checks']+=1
                return answer
            def release(self):
                oracle=self.previous.oracle if self.previous is not None else None
                before=dict(oracle.counts) if oracle is not None else None
                super().release()
                if oracle is not None:
                    assert before==oracle.counts
                    CONTEXT['release_zero_graph_access_checks']+=1
        namespace['MassCapHintFactory']=ObservedFactory
    if BACKEND=='adaptive':
        ORIGINAL_SOURCE=namespace['source_maximum']
        namespace['source_maximum']=observed_source
    if hasattr(signal,'SIGALRM'):
        signal.signal(signal.SIGALRM,alarm_handler)
    previous_results={}
    case=None
    exit_code=0
    try:
        for number,case in enumerate(cases,1):
            CASE_STARTED=time.monotonic()
            set_alarm(GUARDS['per_case_seconds'])
            guard()
            answer,checks=verify_case(case)
            if 'compare_to' in case:
                old=previous_results[case['compare_to']]
                inverse=case['inverse']
                normalize=lambda out:tuple(sorted((inverse[i],v,d) for i,v,d in out))
                assert normalize(answer.output)==tuple(sorted(old.output))
                assert answer.objective_gap_bound==old.objective_gap_bound
                assert len(answer.stages)==len(old.stages)
                for new_stage,old_stage in zip(answer.stages,old.stages):
                    assert normalize(new_stage.repaired_output)==tuple(sorted(old_stage.repaired_output))
                    for field in ('regularization','delta','requested_gap','certified_gap','iterations','grid','projection_mass_cap'):
                        assert getattr(new_stage,field)==getattr(old_stage,field)
                checks['label_and_reversed_row_invariance']=case['compare_to']
            set_alarm(0)
            previous_results[case['name']]=answer
            evidence['completed'].append(dict(name=case['name'],passed=True,
                elapsed_seconds=time.monotonic()-CASE_STARTED,result=answer,checks=checks))
            CASE_STARTED=None
            evidence['totals']=dict(cases=len(evidence['completed']),iterations=BUDGET['iterations'],
                oracle_entries=BUDGET['entries'],degree_replies=BUDGET['degrees'],
                stages=sum(len(previous_results[name].stages) for name in previous_results))
            write(target,evidence)
            print(f'PASS {number}/{len(cases)} {case["name"]}: '+
                  f'{len(answer.stages)} stages, {sum(s.iterations for s in answer.stages)} iterations',flush=True)
        evidence.update(status='complete_subset' if args.smoke else 'complete',
                        full_manifest_completed=not args.smoke,selected_cases_completed=True)
    except BaseException as error:
        set_alarm(0)
        evidence['status']='incomplete'
        evidence['failure']=dict(case=None if case is None else case['name'],
            exception=repr(error),traceback=traceback.format_exc(),partial_observations=CONTEXT)
        print('INCOMPLETE:',repr(error),file=sys.stderr)
        exit_code=1
    finally:
        namespace[key]=original_corrector
        if BACKEND=='mass-grid':
            namespace['MassCapHintFactory']=original_factory
        if BACKEND=='adaptive':
            namespace['source_maximum']=ORIGINAL_SOURCE
        set_alarm(0)
        CASE_STARTED=None
        evidence['source_hashes_after']=sources()
        evidence['sources_unchanged']=evidence['source_hashes_after']==hashes
        if not evidence['sources_unchanged']:
            evidence['status']='incomplete'
            evidence['full_manifest_completed']=False
            exit_code=1
        evidence['elapsed_seconds']=time.monotonic()-BUDGET['started']
        write(target,evidence)
    print(evidence['status'].upper(),json.dumps(evidence.get('totals',{})),flush=True)
    return exit_code


if __name__=='__main__':
    raise SystemExit(main())
