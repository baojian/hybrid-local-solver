"""Small deterministic AVL map with weighted rank/prefix queries.

This avoids an expected-time dictionary primitive in the local reporter.
Keys have a deterministic total order; no priority or hash is sampled.
"""

from __future__ import annotations

from dataclasses import dataclass


@dataclass
class Node:
    key: object
    data: object
    weight: float = 0.0
    value: float = 0.0
    left: Node | None = None
    right: Node | None = None
    height: int = 1
    size: int = 1
    sum_weight: float = 0.0
    sum_value: float = 0.0


def height(node):
    return 0 if node is None else node.height


def size(node):
    return 0 if node is None else node.size


def sums(node):
    return (0.0, 0.0) if node is None else (node.sum_weight, node.sum_value)


def refresh(node):
    node.height = 1 + max(height(node.left), height(node.right))
    node.size = 1 + size(node.left) + size(node.right)
    lw, lv = sums(node.left)
    rw, rv = sums(node.right)
    node.sum_weight = lw + node.weight + rw
    node.sum_value = lv + node.value + rv
    return node


def rotate_left(node):
    new = node.right
    node.right = new.left
    new.left = refresh(node)
    return refresh(new)


def rotate_right(node):
    new = node.left
    node.left = new.right
    new.right = refresh(node)
    return refresh(new)


def balance(node):
    refresh(node)
    if height(node.left) - height(node.right) > 1:
        if height(node.left.left) < height(node.left.right):
            node.left = rotate_left(node.left)
        return rotate_right(node)
    if height(node.right) - height(node.left) > 1:
        if height(node.right.right) < height(node.right.left):
            node.right = rotate_right(node.right)
        return rotate_left(node)
    return node


class AVLMap:
    def __init__(self):
        self.root = None
        self.visits = 0

    def __len__(self):
        return size(self.root)

    def get(self, key):
        node = self.root
        while node is not None:
            self.visits += 1
            if key == node.key:
                return node.data
            node = node.left if key < node.key else node.right
        return None

    def set(self, key, data, weight=0.0, value=0.0):
        def insert(node):
            self.visits += 1
            if node is None:
                return refresh(Node(key, data, weight, value))
            if key < node.key:
                node.left = insert(node.left)
            elif key > node.key:
                node.right = insert(node.right)
            else:
                node.data, node.weight, node.value = data, weight, value
            return balance(node)

        self.root = insert(self.root)

    def delete(self, key):
        def remove(node, target):
            self.visits += 1
            if node is None:
                raise KeyError(target)
            if target < node.key:
                node.left = remove(node.left, target)
            elif target > node.key:
                node.right = remove(node.right, target)
            elif node.left is None:
                return node.right
            elif node.right is None:
                return node.left
            else:
                successor = node.right
                while successor.left is not None:
                    self.visits += 1
                    successor = successor.left
                node.key, node.data = successor.key, successor.data
                node.weight, node.value = successor.weight, successor.value
                node.right = remove(node.right, successor.key)
            return balance(node)

        self.root = remove(self.root, key)

    def kth_largest(self, k):
        if not 1 <= k <= len(self):
            raise IndexError(k)
        node = self.root
        while node is not None:
            self.visits += 1
            right_size = size(node.right)
            if k == right_size + 1:
                return node.key, node.data
            if k <= right_size:
                node = node.right
            else:
                k -= right_size + 1
                node = node.left
        raise AssertionError("invalid rank traversal")

    def prefix_largest(self, k):
        if not 0 <= k <= len(self):
            raise IndexError(k)
        total_weight = total_value = 0.0
        node = self.root
        while node is not None and k:
            self.visits += 1
            right_size = size(node.right)
            if k <= right_size:
                node = node.right
            else:
                rw, rv = sums(node.right)
                total_weight += rw + node.weight
                total_value += rv + node.value
                k -= right_size + 1
                node = node.left
        return total_weight, total_value

    def sums_above_value(self, threshold):
        node = self.root
        total_weight = total_value = 0.0
        while node is not None:
            self.visits += 1
            if node.key[0] <= threshold:
                node = node.right
            else:
                rw, rv = sums(node.right)
                total_weight += rw + node.weight
                total_value += rv + node.value
                node = node.left
        return total_weight, total_value

    def items(self):
        def walk(node):
            if node is not None:
                self.visits += 1
                yield from walk(node.left)
                yield node.key, node.data
                yield from walk(node.right)

        yield from walk(self.root)

    def above_value(self, threshold):
        def walk(node):
            if node is None:
                return
            self.visits += 1
            if node.key[0] <= threshold:
                yield from walk(node.right)
            else:
                yield from walk(node.left)
                yield node.key, node.data
                yield from walk(node.right)

        yield from walk(self.root)

    def validate(self):
        """Audit helper, deliberately not used in the solver's hot path."""

        def visit(node, low, high):
            if node is None:
                return 0, 0, 0.0, 0.0
            assert low is None or low < node.key
            assert high is None or node.key < high
            lh, ls, lw, lv = visit(node.left, low, node.key)
            rh, rs, rw, rv = visit(node.right, node.key, high)
            assert abs(lh - rh) <= 1
            assert node.height == 1 + max(lh, rh)
            assert node.size == 1 + ls + rs
            assert abs(node.sum_weight - (lw + node.weight + rw)) <= 1e-12 * (
                1 + abs(node.sum_weight)
            )
            assert abs(node.sum_value - (lv + node.value + rv)) <= 1e-12 * (
                1 + abs(node.sum_value)
            )
            return node.height, node.size, node.sum_weight, node.sum_value

        visit(self.root, None, None)
