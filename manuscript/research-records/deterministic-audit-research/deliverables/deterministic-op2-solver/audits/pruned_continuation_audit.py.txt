"""Independent full-objective audit of fixed-degree-pruned continuation."""

from __future__ import annotations

import itertools
import json
import sys
import time
from fractions import Fraction as F
from pathlib import Path

import networkx as nx
from independent_continuation_audit import dot, graph_data, obstacle, product, subtract

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT))
from pruned_rppr import solve


class Oracle:
    def __init__(self, graph):
        self.graph = graph
        self.degree_calls = 0
        self.entries = 0
        self.rows = []

    def degree(self, i):
        self.degree_calls += 1
        return self.graph.degree(i)

    def neighbors(self, i):
        self.rows.append(i)
        for j in sorted(self.graph.neighbors(i)):
            self.entries += 1
            yield j


def check_case(graph, seed, alpha, rho, epsilon, prune, adaptive):
    oracle = Oracle(graph)
    start = time.monotonic()
    result = solve(
        oracle, seed, alpha, rho, epsilon, prune=prune, adaptive_box=adaptive
    )
    elapsed = time.monotonic() - start
    degrees, q, b = graph_data(graph, seed, alpha)
    optimum = obstacle(q, b, alpha, rho)
    output = [F(0)] * len(graph)
    for i, value, degree in result.output:
        assert degree == degrees[i]
        output[i] = value
    assert all(0 <= v <= t for v, t in zip(output, optimum))
    error = subtract(output, optimum)
    slack = [v - s + alpha * rho for v, s in zip(product(q, optimum), b)]
    gap = dot(error, product(q, error), degrees) / 2 + dot(slack, output, degrees)
    assert 0 <= gap <= result.objective_gap_bound <= epsilon
    assert sum(degrees[i] for i, value in enumerate(output) if value) <= 1 / rho
    if prune:
        assert all(degrees[i] <= 1 / rho for i in oracle.rows)
    previous = [F(0)] * len(graph)
    for stage in result.stages:
        exact = obstacle(q, b, alpha, stage.regularization)
        state = [F(0)] * len(graph)
        for i, value, degree in stage.repaired_output:
            assert degree == degrees[i]
            state[i] = value
        assert all(old <= x <= target for old, x, target in zip(previous, state, exact))
        source = subtract(b, product(q, state))
        for i, value in enumerate(source):
            if not prune or degrees[i] <= 1 / rho:
                assert 0 <= value <= 2 * alpha * stage.regularization
        previous = state
    return {
        "n": len(graph),
        "seed": seed,
        "alpha": str(alpha),
        "rho": str(rho),
        "epsilon": str(epsilon),
        "prune": prune,
        "adaptive_box": adaptive,
        "wall_seconds": elapsed,
        "actual_graph_entries": oracle.entries,
        "degree_calls": oracle.degree_calls,
        "maximum_scanned_degree": max((degrees[i] for i in oracle.rows), default=0),
        "stages": len(result.stages),
        "iterations": sum(stage.iterations for stage in result.stages),
        "kinetic_volume": sum(stage.kinetic_volume for stage in result.stages),
        "output_volume": sum(degrees[i] for i, v in enumerate(output) if v),
        "actual_gap": str(gap),
        "gap_bound": str(result.objective_gap_bound),
        "randomized": False,
    }


def main():
    graphs = [
        ("edge", nx.path_graph(2), 0),
        ("path6", nx.path_graph(6), 0),
        ("star6_leaf", nx.star_graph(5), 1),
        ("star6_hub", nx.star_graph(5), 0),
        ("clique5", nx.complete_graph(5), 0),
        ("barbell6", nx.barbell_graph(3, 0), 0),
    ]
    output = ROOT / "results/pruned-continuation-audit.jsonl"
    start = time.monotonic()
    count = 0
    with output.open("w") as stream:
        for (name, graph, seed), alpha, ratio, (prune, adaptive) in itertools.product(
            graphs,
            (F(1, 3), F(1, 16)),
            (F(1, 4), F(1, 16)),
            ((False, False), (True, False), (True, True)),
        ):
            rho = ratio / graph.degree(seed)
            record = check_case(graph, seed, alpha, rho, F(1, 10**8), prune, adaptive)
            record["graph"] = name
            stream.write(json.dumps(record) + "\n")
            stream.flush()
            count += 1
            print(
                json.dumps(
                    {
                        k: v
                        for k, v in record.items()
                        if k not in ("actual_gap", "gap_bound")
                    }
                ),
                flush=True,
            )
    summary = {
        "cases": count,
        "elapsed_seconds": time.monotonic() - start,
        "status": "enumeration_complete",
        "randomized": False,
    }
    output.with_suffix(".completed.json").write_text(json.dumps(summary, indent=2))
    print(json.dumps(summary), flush=True)


if __name__ == "__main__":
    main()
