"""Exact PSD checks for the deterministic boundary-forest certificate."""

from __future__ import annotations

import itertools
import json
import sys
import time
from fractions import Fraction as F
from pathlib import Path

import networkx as nx

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT))
from pruned_rppr.component import forest_spectral_bound


def positive_semidefinite(matrix):
    """Exact rational symmetric elimination, permitting null final blocks."""
    a = [row.copy() for row in matrix]
    n = len(a)
    for k in range(n):
        pivot = a[k][k]
        if pivot < 0:
            return False
        if pivot == 0:
            if any(a[i][k] != 0 for i in range(k + 1, n)):
                return False
            continue
        for i in range(k + 1, n):
            for j in range(i, n):
                a[i][j] -= a[i][k] * a[j][k] / pivot
                a[j][i] = a[i][j]
    return True


def check(graph, killing):
    n = len(graph)
    adjacency = [sorted(graph.neighbors(i)) for i in range(n)]
    degrees = [len(row) + b for row, b in zip(adjacency, killing)]
    alpha = F(1, 3)
    beta = forest_spectral_bound(degrees, adjacency, alpha)
    gap = (beta - alpha) / ((1 - alpha) / 2)
    assert gap >= min(F(b, d) for b, d in zip(killing, degrees))
    assert 0 <= gap <= 1
    shifted = [
        [
            (1 - gap) * degrees[i] if i == j else -F(graph.has_edge(i, j))
            for j in range(n)
        ]
        for i in range(n)
    ]
    assert positive_semidefinite(shifted)
    return gap


def main():
    start = time.monotonic()
    cases = 0
    for graph in nx.graph_atlas_g():
        if not 2 <= len(graph) <= 5 or not nx.is_connected(graph):
            continue
        for killing in itertools.product((0, 1, 3), repeat=len(graph)):
            check(graph, killing)
            cases += 1
    larger = []
    for length in (16, 32, 64):
        graph = nx.path_graph(length)
        for ends in (1, 2):
            killing = [0] * length
            killing[0] = 1
            if ends == 2:
                killing[-1] = 1
            gap = check(graph, killing)
            larger.append(
                {"length": length, "killed_ends": ends, "normalized_gap_bound": str(gap)}
            )
    result = {
        "atlas_psd_checks": cases,
        "larger_path_checks": larger,
        "elapsed_seconds": time.monotonic() - start,
        "all_checks_exact": True,
        "randomized": False,
        "status": "complete",
    }
    (ROOT / "results/forest-spectral-audit.json").write_text(
        json.dumps(result, indent=2)
    )
    print(json.dumps(result), flush=True)


if __name__ == "__main__":
    main()
