"""Exact independent projection and full-objective checks for the fast path."""

from __future__ import annotations

import itertools
import json
import sys
import time
from fractions import Fraction as F
from pathlib import Path

import networkx as nx
from independent_continuation_audit import (
    dot,
    graph_data,
    obstacle,
    product,
    projection,
    solve,
    subtract,
)
from pruned_continuation_audit import Oracle

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT))
from pruned_rppr.component import clipped_projection, solve_fast


def exact_reference(matrix, source, alpha, rho):
    n = len(source)
    rhs = [v - alpha * rho for v in source]
    active = [i for i, v in enumerate(rhs) if v > 0]
    while True:
        x = [F(0)] * n
        if active:
            values = solve(
                [[matrix[i][j] for j in active] for i in active],
                [rhs[i] for i in active],
            )
            assert all(value > 0 for value in values)
            for i, value in zip(active, values):
                x[i] = value
        slack = subtract(product(matrix, x), rhs)
        additions = [i for i in range(n) if i not in active and slack[i] < 0]
        if not additions:
            assert all(v >= 0 for v in slack)
            assert all(v * r == 0 for v, r in zip(x, slack))
            if n <= 6:
                assert x == obstacle(matrix, source, alpha, rho)
            return x
        active = sorted(active + additions)


def leaky_pair():
    graph = nx.Graph()
    graph.add_edge(0, 1)
    graph.add_edge(0, 2)
    graph.add_edge(1, 3)
    for vertex in range(4, 9):
        graph.add_edge(2, vertex)
    for vertex in range(9, 14):
        graph.add_edge(3, vertex)
    return graph


def main():
    start = time.monotonic()
    count = 0
    for degrees, upper in itertools.product(
        ((1, 2, 5), (3, 7, 11), (1, 1, 1, 1)), (F(1, 4), F(1, 7), F(2, 3))
    ):
        for coefficients in itertools.product(
            (F(-2), F(0), F(1, 2), F(1), F(2), F(3)), repeat=len(degrees)
        ):
            raw = [upper * v for v in coefficients]
            expected, _ = projection(raw, degrees, upper, F(1))
            assert clipped_projection(raw, degrees, upper) == expected
            count += 1
    projection_count = count
    rows = []
    graphs = [
        ("edge", nx.path_graph(2), 0),
        ("path6", nx.path_graph(6), 0),
        ("star6_leaf", nx.star_graph(5), 1),
        ("clique5", nx.complete_graph(5), 0),
        ("barbell6", nx.barbell_graph(3, 0), 0),
        ("leaky_pair", leaky_pair(), 0),
    ]
    for (name, graph, seed), alpha, ratio in itertools.product(
        graphs, (F(1, 3), F(1, 16), F(1, 10000)), (F(3, 4), F(1, 4), F(1, 16))
    ):
        rho = ratio / graph.degree(seed)
        epsilon = min(F(1, 10**8), alpha**2 * rho**2 / 100)
        oracle = Oracle(graph)
        then = time.monotonic()
        result = solve_fast(oracle, seed, alpha, rho, epsilon)
        elapsed = time.monotonic() - then
        degrees, matrix, source = graph_data(graph, seed, alpha)
        optimum = exact_reference(matrix, source, alpha, rho)
        output = [F(0)] * len(graph)
        for i, value, degree in result.output:
            assert degree == degrees[i]
            output[i] = value
        assert all(0 <= x <= target for x, target in zip(output, optimum))
        error = subtract(output, optimum)
        slack = [v - b + alpha * rho for v, b in zip(product(matrix, optimum), source)]
        gap = dot(error, product(matrix, error), degrees) / 2 + dot(
            slack, output, degrees
        )
        assert 0 <= gap <= result.objective_gap_bound <= epsilon
        assert rho * result.metrics.prepass_scanned_volume <= 1
        assert all(rho * degrees[i] <= 1 for i in oracle.rows)
        record = {
            "graph": name,
            "alpha": str(alpha),
            "rho": str(rho),
            "epsilon": str(epsilon),
            "route": result.route,
            "actual_gap": str(gap),
            "gap_bound": str(result.objective_gap_bound),
            "wall_seconds": elapsed,
            "original_graph_entries": oracle.entries,
            "prepass_volume": result.metrics.prepass_scanned_volume,
            "component_volume": result.metrics.component_volume,
            "beta": str(result.metrics.spectral_lower_bound),
            "iterations": result.metrics.iterations,
            "prescribed_iterations": result.metrics.prescribed_iterations,
            "fallback_iterations": sum(
                stage.iterations for stage in result.continuation.stages
            )
            if result.continuation
            else 0,
            "randomized": False,
        }
        rows.append(record)
        (ROOT / "results/component-fast-path-audit.json").write_text(
            json.dumps(
                {
                    "projection_cases": projection_count,
                    "solver_cases": rows,
                    "status": "running",
                },
                indent=2,
            )
        )
        print(
            json.dumps(
                {
                    k: v
                    for k, v in record.items()
                    if k not in ("actual_gap", "gap_bound")
                }
            ),
            flush=True,
        )
    final = {
        "projection_cases": projection_count,
        "solver_cases": rows,
        "status": "enumeration_complete",
        "elapsed_seconds": time.monotonic() - start,
        "randomized": False,
    }
    (ROOT / "results/component-fast-path-audit.json").write_text(
        json.dumps(final, indent=2)
    )
    print(
        json.dumps(
            {
                "projection_cases": projection_count,
                "solver_cases": len(rows),
                "elapsed_seconds": final["elapsed_seconds"],
                "status": "enumeration_complete",
            }
        ),
        flush=True,
    )


if __name__ == "__main__":
    main()
