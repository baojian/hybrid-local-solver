% Shared package configuration for the active manuscript.
% Keep package-order constraints here rather than in main.tex.

% Core mathematics and notation. Load amsthm before jmlr2e because the JMLR
% style declares the repository's standard theorem environments.
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{amsthm}
\usepackage{mathtools}
\usepackage{bm}

% Color must be available before jmlr2e loads hyperref.
\usepackage{xcolor}
\usepackage{jmlr2e}
\hypersetup{
    colorlinks=true,
    citecolor=blue,
    linkcolor=black,
    urlcolor=blue
}

% Algorithms, tables, figures, and manuscript structure.
\usepackage{algorithm}
\usepackage{algorithmic}
\usepackage{booktabs}
\usepackage{multirow}
\usepackage{tabularx}
\usepackage{pifont}
\usepackage{enumitem}
\usepackage{microtype}
\usepackage{comment}

% Load cleveref after jmlr2e, which loads hyperref.
\usepackage[capitalize,nameinlink,noabbrev]{cleveref}

% The JMLR style already defines theorem, lemma, proposition, corollary,
% definition, remark, and related environments.
\theoremstyle{definition}
\newtheorem{assumption}[theorem]{Assumption}
\newtheorem{problem}[theorem]{Problem}
\theoremstyle{plain}

\crefname{assumption}{Assumption}{Assumptions}
\crefname{problem}{Problem}{Problems}
