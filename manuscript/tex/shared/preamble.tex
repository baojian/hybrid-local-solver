% Shared package configuration for the active manuscript.
% Keep package-order constraints here rather than in main.tex.

% Core mathematics and notation. Load amsthm before jmlr2e because the JMLR
% style declares the repository's standard theorem environments.
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{amsthm}
\usepackage{mathtools}
\usepackage{bm}

% Color must be available before jmlr2e loads hyperref.
\usepackage{xcolor}
\usepackage{jmlr2e}
\hypersetup{
    colorlinks=true,
    citecolor=blue,
    linkcolor=black,
    urlcolor=blue
}

% Algorithms, tables, figures, and manuscript structure.
\usepackage{algorithm}
\usepackage{algorithmic}
\usepackage{booktabs}
\usepackage{longtable}
\usepackage{multirow}
\usepackage{tabularx}
\usepackage{pifont}
\usepackage{enumitem}
\usepackage{microtype}
\usepackage{comment}

% Load cleveref after jmlr2e, which loads hyperref.
\usepackage[capitalize,nameinlink,noabbrev]{cleveref}

% The JMLR style already defines theorem, lemma, proposition, corollary,
% definition, remark, and related environments. It declares them before
% cleveref is loaded, however, so cleveref only sees their shared `theorem`
% counter and would name every one of them ``Theorem''. Redeclare the
% dependent environments here, after cleveref, preserving jmlr2e's plain style
% and shared counter, so that \cref reports the correct environment name. The
% `theorem` and `example` environments own their counters and are named
% through \crefname instead.
\makeatletter
\let\lemma\@undefined       \let\endlemma\@undefined
\let\proposition\@undefined \let\endproposition\@undefined
\let\remark\@undefined      \let\endremark\@undefined
\let\corollary\@undefined   \let\endcorollary\@undefined
\let\definition\@undefined  \let\enddefinition\@undefined
\let\conjecture\@undefined  \let\endconjecture\@undefined
\let\axiom\@undefined       \let\endaxiom\@undefined
\makeatother
\theoremstyle{plain}
\newtheorem{lemma}[theorem]{Lemma}
\newtheorem{proposition}[theorem]{Proposition}
\newtheorem{remark}[theorem]{Remark}
\newtheorem{corollary}[theorem]{Corollary}
\newtheorem{definition}[theorem]{Definition}
\newtheorem{conjecture}[theorem]{Conjecture}
\newtheorem{axiom}[theorem]{Axiom}

\theoremstyle{definition}
\newtheorem{assumption}[theorem]{Assumption}
\newtheorem{problem}[theorem]{Problem}
\theoremstyle{plain}

\crefname{theorem}{Theorem}{Theorems}
\crefname{example}{Example}{Examples}
\crefname{assumption}{Assumption}{Assumptions}
\crefname{problem}{Problem}{Problems}
