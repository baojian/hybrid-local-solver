The RPPR theorem already yields a PPR approximation through
\cref{prop:rppr-ppr-bridge}.  A stricter two-stage interface is useful for
algorithm design: Stage~I returns only a set, all changing-face state is
discarded, and Stage~II solves one ordinary principal PPR system.  The key
observation is that the retained face need not contain the exact RPPR
support.

For a reachable face $U$, let
\begin{equation}
 \vu_U:=\mQ_{UU}^{-1}\vb_U,
 \label{eq:principal-ppr-system}
\end{equation}
and write $\widetilde\vu_U$ for its zero extension.

\begin{lemma}[Approximate-envelope linearization]
\label{lem:approximate-envelope}
Let
\begin{equation}
 \delta_U:=\max_{i\notin U}
       \frac{(x_\rho^*)_i}{\sqrt{d_i}},
 \label{eq:outside-rppr-amplitude}
\end{equation}
with the maximum equal to zero when $U=[n]$.  Then
\begin{equation}
 \vzero\leq\widetilde\vu_U\leq\vx_0^*,
 \qquad
 \norm{\mD^{-1/2}(\vx_0^*-\widetilde\vu_U)}_\infty
 \leq\rho+\delta_U.
 \label{eq:approximate-envelope-bound}
\end{equation}
\end{lemma}

\begin{proof}
By \cref{eq:rppr-bias}, every coordinate outside $U$ satisfies
\begin{equation}
 (x_0^*)_i\leq(x_\rho^*)_i+\rho\sqrt{d_i}
 \leq(\rho+\delta_U)\sqrt{d_i}.
 \label{eq:outside-ppr-envelope}
\end{equation}
On $U$, put $\vw_U=(\vx_0^*)_U-\vu_U$.  The full and principal equations
give
\begin{equation}
 \mQ_{UU}\vw_U=-\mQ_{U\bar U}(\vx_0^*)_{\bar U}\geq\vzero,
 \label{eq:principal-ppr-difference}
\end{equation}
so inverse positivity gives $\vzero\leq\vu_U\leq(\vx_0^*)_U$.
Using \cref{eq:outside-ppr-envelope}, the off-diagonal sign, and
\cref{eq:constant-vector-inverse-positive},
\begin{align*}
 \mQ_{UU}\vw_U
 &\leq-\mQ_{U\bar U}(\rho+\delta_U)
      \mD_{\bar U}^{1/2}\one\\
 &\leq\mQ_{UU}(\rho+\delta_U)\mD_U^{1/2}\one.
\end{align*}
Multiplication by $\mQ_{UU}^{-1}\geq\vzero$ proves the inside normalized
bound; \cref{eq:outside-ppr-envelope} proves it outside.
\end{proof}

\begin{lemma}[Low face energy is a set certificate]
\label{lem:face-energy-envelope}
Let
\begin{equation}
 \Gamma_U:=F_\rho(\vx^U)-F_\rho(\vx_\rho^*).
 \label{eq:exact-face-gap-transfer}
\end{equation}
Then
\begin{equation}
 \max_{i\notin U}\frac{(x_\rho^*)_i}{\sqrt{d_i}}
 \leq\sqrt{\frac{2\Gamma_U}{\alpha}},
 \label{eq:face-gap-amplitude}
\end{equation}
and
\begin{equation}
 \norm{\mD^{-1/2}(\vx_0^*-\widetilde\vu_U)}_\infty
 \leq\rho+\sqrt{\frac{2\Gamma_U}{\alpha}}.
 \label{eq:face-gap-ppr-envelope}
\end{equation}
\end{lemma}

\begin{proof}
Strong convexity gives
$\Gamma_U\geq(\alpha/2)\norm{\vx^U-\vx_\rho^*}_2^2$.  Since the zero
extension of $\vx^U$ vanishes outside $U$ and every $d_i\geq1$, this proves
\cref{eq:face-gap-amplitude}.  Apply
\cref{lem:approximate-envelope} with the resulting value of $\delta_U$.
\end{proof}

\begin{algorithm}[t]
\caption{Strict discover-once, solve-once PPR}
\label{alg:two-stage-ppr}
\begin{algorithmic}[1]
\STATE Choose positive budgets $\rho,\eta,\tau$ with
       $\rho+\eta+\tau\leq\epsppr$ and set
       $\epsobj=\alpha\eta^2/2$.
\STATE Run \cref{alg:threshold-batch-rppr} with failure budget $\zeta/2$.
       Retain only its certified face $U$; discard the numerical point and
       all Stage~I solver state.
\STATE Solve $\mQ_{UU}\vu_U=\vb_U$ to Euclidean error at most $\tau$ with
       failure budget $\zeta/2$.
\STATE Project the Stage~II vector onto $\R_+^U$, zero-pad it, and return.
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Randomized literal two-stage completion]
\label{thm:two-stage-completion}
Let $0<\epsppr<1$ and choose positive budgets
\begin{equation}
 \rho+\eta+\tau\leq\epsppr,
 \qquad
 \epsobj=\frac{\alpha\eta^2}{2}.
 \label{eq:two-stage-budgets}
\end{equation}
With failure probability at most $\zeta$,
\cref{alg:two-stage-ppr} returns $\widehat\vx$ satisfying
\cref{eq:semantic-ppr-target}.  If Stage~II uses an independent
residual-certified randomized SDD solve, the expected fully charged work is
\begin{equation}
 \softO\!\left(
  \frac{1}{\rho\sqrt\alpha}
  \log\frac1{\alpha\eta^2}\log\frac4\zeta
  +\frac1\rho
  \log\frac1{\sqrt\alpha\tau}\log\frac4\zeta
 \right).
 \label{eq:two-stage-total-work}
\end{equation}
In particular, $\rho=\eta=\tau=\epsppr/3$ gives
$\softO(1/(\epsppr\sqrt\alpha))$ expected work.  If the initial degree test
finds $\rho\geq1/d_v$, the algorithm may return zero and omit Stage~II.
\end{theorem}

\begin{proof}
On a successful Stage~I execution, the proof of \cref{thm:local-rppr}
certifies that the exact retained face has $\Gamma_U\leq\epsobj$.
By \cref{lem:face-energy-envelope}, the exact principal PPR solution then has
semantic error at most $\rho+\eta$.  A Euclidean Stage~II error of at most
$\tau$ contributes at most $\tau$ in the normalized $\ell_\infty$ norm
because $d_i\geq1$.  Projection cannot increase the Euclidean distance from
the exact nonnegative principal solution.  The two failure budgets combine by
a union bound.

For work, use the degree-coordinate SDD matrix in
\cref{eq:degree-scaled-face-sdd}.  The exact principal PPR solution has
energy at most $\alpha$.  A good call at relative energy tolerance
$\sqrt\alpha\tau$ has active residual norm at most $\alpha\tau$.  Accept a
call only after an active-row scan verifies that inequality.  Every accepted
call then has Euclidean solution error at most $\tau$, deterministically,
because $\mQ_{UU}\succeq\alpha\mI$.  Capped independent retries give the
second failure budget.  Since $U\subseteq\mathcal S_\rho^*$ and
$\vol(U)\leq1/\rho$, the supplied SDD work plus
\cref{eq:local-rppr-work} proves \cref{eq:two-stage-total-work}.

In the zero-RPPR regime, \cref{eq:rppr-bias} directly bounds the semantic
error of zero by $\rho\leq\epsppr$.
\end{proof}

\begin{corollary}[The shortest execution stops after Stage~I]
\label{cor:direct-stage-one-return}
Set $\rho=\epsppr/2$ and $\epsobj=\alpha\epsppr^2/8$.  The projected
numerical point already returned by \cref{alg:threshold-batch-rppr} is
semantically $\epsppr$-accurate by \cref{eq:objective-to-ppr}, with expected
work $\softO(1/(\epsppr\sqrt\alpha))$.  Hence Stage~II is a modular set-only
variant, not a mandatory extra cost.
\end{corollary}

This separation clarifies the relation to AESP--CD.  Support-safe admission,
the hard face cap, restart, and the certificate race transfer to an AESP--CD
portfolio.  The theorem itself does not carry the original momentum state
across changing faces; it instead pays for fresh solves on a provably bounded
number of faces.  Practical SOR or APPR tails may be raced with the certified
Stage~II solve, but their empirical behavior is not used in the theorem.
