\textbf{Notations.} Define
$\mQ = \alpha \mI + \frac{1-\alpha}{2}\mL$ where $\mL = \mI - \mD^{-1/2} \mA \mD^{-1/2}$.

In this paper, we consider the following two optimization problems:
\begin{equation}
\min_{\vx \in \R^n}
\left\{
f(\vx)
:=
\frac{1}{2}\langle \vx, \mQ \vx \rangle
- \alpha \langle \mD^{-1/2} \vs, \vx \rangle
\right\},
\label{equ:f(x)}
\end{equation}

\begin{equation}
\min_{\vx \in \R^n}
\left\{
F_\rho(\vx)
:=
f(\vx) + g(\vx)
\right\},
\qquad
\text{where }
g(\vx)
:=
\alpha \rho \left\|\mD^{1/2}\vx\right\|_1.
\label{equ:F(x)}
\end{equation}
