\section{Scope and status}

This standalone note records a proposed Accelerated Evolving Set Process with
local proximal coordinate descent (AESP-CD) for regularized PageRank.  It
separates the shared problem definition, proved implementation facts, and the
remaining outer-locality target.  The coordinate-level theory is closed: the
minimum-norm KKT mass contracts under every exact proximal coordinate update,
including sign changes and zero-coordinate clipping, and the KKT diagnostic
certifies degree-normalized solution error.  These statements give a rigorous
local inner oracle for composite AESP.  They do not bound the KKT mass of an
arbitrary extrapolated Catalyst center.  A relative-accuracy continuation
below proves that, on any fixed certified envelope of degree volume $V$, the
inner work is only $\softO(V)$ per outer stage and hence
$\softO(V/\sqrt\alpha)$ overall.  Taking the envelope to be the optimal RPPR
support gives $V\leq1/\rho$; discovering such an envelope locally without an
oracle remains open.  The same heap implementation does, however, admit an
oracle-free bound in terms of the maximum volume it actually explores; this
reduces the remaining universal question to controlling that explored volume.
Nothing here changes the repository-wide residual decision.

\section{Problem formulation}

\input{../../tex/shared/source_aligned_problem}

This note specializes the shared model to $\alpha\in(0,1/2)$ and solves the
same composite objective $F_\rho$.  Its final, note-scoped diagnostic target is
\begin{equation}
  \frac{R_{\mathrm{kkt},\rho}(x)}{\alpha}\leq\epskkt.
  \label{eq:aesp-cd-target}
\end{equation}
This is not identified with $\epsobj$, $\epspg$, or $\epsppr$.

\section{KKT diagnostic}

Define the note-scoped minimum KKT-violation vector
\begin{equation}
\kappa_{\rho,i}(x)=
\begin{cases}
\nabla_i f(x)+\alpha\rho\sqrt{d_i},&x_i>0,\\
\nabla_i f(x)-\alpha\rho\sqrt{d_i},&x_i<0,\\
\nabla_i f(x)-\operatorname{clip}
  \bigl(\nabla_i f(x),-\alpha\rho\sqrt{d_i},
                    \alpha\rho\sqrt{d_i}\bigr),&x_i=0,
\end{cases}
\label{eq:aesp-cd-kkt-vector}
\end{equation}
and the scalar diagnostic
\begin{equation}
  R_{\mathrm{kkt},\rho}(x)
  :=\norm{D^{-1/2}\kappa_\rho(x)}_\infty.
  \label{eq:aesp-cd-kkt-diagnostic}
\end{equation}
The implication from this diagnostic to a solution-error norm is a required
lemma proved below, but it is not an adopted repository-wide stopping
convention.

\section{AESP outer framework}

At outer iteration $t$, the proposed method approximately minimizes
\begin{equation}
  F_{\rho,t}(z):=F_\rho(z)
  +\frac{\kappa_{\mathrm A}}{2}
       \norm{z-y^{(t-1)}}_2^2,
  \qquad \kappa_{\mathrm A}:=1-2\alpha,
  \label{eq:aesp-cd-subproblem}
\end{equation}
with AESP-scoped momentum
\begin{equation}
  \beta_{\mathrm A}
  :=\frac{\sqrt{1-\alpha}-\sqrt\alpha}
          {\sqrt{1-\alpha}+\sqrt\alpha}.
  \label{eq:aesp-cd-momentum}
\end{equation}
The inner initialization is $z_t^{(0)}=y^{(t-1)}$.

\section{Inner local proximal coordinate descent}

The smooth part of the inner objective has gradient
\begin{equation}
  \nabla f_t(z)
  =(Q+\kappa_{\mathrm A}I)z-b
    -\kappa_{\mathrm A}y^{(t-1)}.
  \label{eq:aesp-cd-inner-gradient}
\end{equation}
Define $\kappa_t(z)$ from \cref{eq:aesp-cd-kkt-vector} by replacing
$\nabla f$ with $\nabla f_t$, and set
\begin{equation}
  C_t(z):=\norm{D^{1/2}\kappa_t(z)}_1.
  \label{eq:aesp-cd-kkt-mass}
\end{equation}

Let
\begin{equation}
  H_t:=Q+\kappa_{\mathrm A}I,
  \qquad
  \mu_t:=\alpha+\kappa_{\mathrm A}.
  \label{eq:aesp-cd-shifted-hessian}
\end{equation}
Then $H_t$ has nonpositive off-diagonal entries and satisfies
\begin{equation}
  \mu_t I\preceq H_t,
  \qquad
  H_tD^{1/2}\one=\mu_tD^{1/2}\one.
  \label{eq:aesp-cd-shifted-properties}
\end{equation}

For coordinate $i$, exact proximal coordinate descent uses
\begin{equation}
  z_i^+
  =\prox_{(\alpha\rho\sqrt{d_i}/\ell)\abs{\cdot}}
     \left(z_i-\frac{\nabla_i f_t(z)}{\ell}\right),
  \qquad
  \ell=(Q+\kappa_{\mathrm A}I)_{ii}
      =\frac{1+\alpha}{2}+\kappa_{\mathrm A}.
  \label{eq:aesp-cd-coordinate-update}
\end{equation}
Only coordinate $i$ and its neighbors require gradient updates, so the shared
work charge for this update is $d_i$.

\section{KKT certificate and local monotonicity}

\begin{lemma}[KKT diagnostic certifies inner solution error]
\label{lem:aesp-cd-kkt-error}
Let $z_t^\star$ be the unique minimizer of $F_{\rho,t}$.  Then every
$z\in\R^n$ satisfies
\begin{equation}
  \norm{D^{-1/2}(z-z_t^\star)}_\infty
  \leq
  \frac{1}{\mu_t}
  \norm{D^{-1/2}\kappa_t(z)}_\infty.
  \label{eq:aesp-cd-inner-error}
\end{equation}
In particular, for the unshifted RPPR objective,
\begin{equation}
  \norm{D^{-1/2}(x-x^\star(\rho))}_\infty
  \leq \frac{R_{\mathrm{kkt},\rho}(x)}{\alpha}.
  \label{eq:aesp-cd-rppr-error}
\end{equation}
\end{lemma}

\begin{proof}
Choose subgradients $\xi\in\partial\norm{D^{1/2}z}_1$ and
$\xi^\star\in\partial\norm{D^{1/2}z_t^\star}_1$ so that
\begin{equation*}
  \kappa_t(z)=H_t(z-z_t^\star)
  +\alpha\rho(\xi-\xi^\star).
\end{equation*}
The coordinate definition of $\kappa_t$ is precisely the minimum-magnitude
valid choice at $z$, while optimality supplies a choice with zero residual at
$z_t^\star$.

Put $e=z-z_t^\star$ and choose $i$ maximizing
$M:=\abs{e_i}/\sqrt{d_i}$.  If $M=0$, the result is immediate.  Otherwise,
the normalized maximum principle and
\cref{eq:aesp-cd-shifted-properties} give
\begin{equation*}
 \frac{\operatorname{sign}(e_i)(H_te)_i}{\sqrt{d_i}}
 \geq \mu_tM.
\end{equation*}
Coordinatewise monotonicity of the weighted absolute-value subdifferential
gives
$e_i(\xi_i-\xi_i^\star)\geq0$.  Hence
\begin{equation*}
 \frac{\abs{\kappa_{t,i}(z)}}{\sqrt{d_i}}
 \geq
 \frac{\operatorname{sign}(e_i)\kappa_{t,i}(z)}{\sqrt{d_i}}
 \geq \mu_tM,
\end{equation*}
which proves \cref{eq:aesp-cd-inner-error}.  Set
$\kappa_{\mathrm A}=0$ for \cref{eq:aesp-cd-rppr-error}.
\end{proof}

\begin{theorem}[Exact KKT-mass decrease]
\label{thm:aesp-cd-mass}
For a sequential proximal coordinate update at coordinate $i$,
\begin{equation}
  C_t(z^+)
  \leq C_t(z)
  -\tau_{\mathrm{cd}}\sqrt{d_i}\abs{\kappa_{t,i}(z)},
  \qquad
  \tau_{\mathrm{cd}}
  :=\frac{2(\alpha+\kappa_{\mathrm A})}
          {1+\alpha+2\kappa_{\mathrm A}}.
  \label{eq:aesp-cd-mass-target}
\end{equation}
Under $\kappa_{\mathrm A}=1-2\alpha$, the displayed coefficient is $2/3$.
\end{theorem}

\begin{proof}
Write $\lambda_i:=\alpha\rho\sqrt{d_i}$ and
$\Delta_i:=z_i^+-z_i$.  Exact scalar proximal minimization makes the new
coordinate KKT violation zero and gives
\begin{equation}
  \kappa_{t,i}(z^+)=0,
  \qquad
  \ell\abs{\Delta_i}\leq\abs{\kappa_{t,i}(z)}.
  \label{eq:aesp-cd-displacement}
\end{equation}
The second inequality follows directly from the three scalar soft-threshold
cases: the update stays on the same nonzero face, reaches zero, or crosses to
the opposite face.  Equality holds in the first case; thresholding can only
shorten the displacement in the latter two.

For every $j\neq i$, the value $z_j$ is unchanged.  As a function of its
smooth gradient, the minimum KKT violation is either a translate
$g\mapsto g\pm\lambda_j$ on a nonzero face or the soft-threshold map
$g\mapsto g-\operatorname{clip}(g,-\lambda_j,\lambda_j)$ at zero.  Each map
is one-Lipschitz.  Therefore
\begin{equation*}
 \abs{\kappa_{t,j}(z^+)}
 \leq
 \abs{\kappa_{t,j}(z)}+\abs{H_{t,ji}\Delta_i}.
\end{equation*}
Only neighbors of $i$ have a nonzero off-diagonal entry, and
\begin{equation*}
 \sum_{j\neq i}\sqrt{d_j}\abs{H_{t,ji}\Delta_i}
 =\frac{1-\alpha}{2}\sqrt{d_i}\abs{\Delta_i}.
\end{equation*}
Using \cref{eq:aesp-cd-displacement}, we obtain
\begin{align*}
 C_t(z^+)
 &\leq C_t(z)-\sqrt{d_i}\abs{\kappa_{t,i}(z)}
       +\frac{1-\alpha}{2}\sqrt{d_i}\abs{\Delta_i}\\
 &\leq C_t(z)
 -\left(1-\frac{1-\alpha}{2\ell}\right)
  \sqrt{d_i}\abs{\kappa_{t,i}(z)}.
\end{align*}
Finally,
\begin{equation*}
 1-\frac{1-\alpha}{2\ell}
 =\frac{2(\alpha+\kappa_{\mathrm A})}
        {1+\alpha+2\kappa_{\mathrm A}}
 =\tau_{\mathrm{cd}}.
\end{equation*}
\end{proof}

\begin{theorem}[Stieltjes-quadratic inner locality]
\label{thm:aesp-cd-stieltjes}
Let
\begin{equation*}
 \Psi(z)=\frac12 z^\top Hz-c^\top z
          +\sum_{j=1}^n\lambda_j\abs{z_j},
 \qquad \lambda_j\geq0,
\end{equation*}
where $H$ is symmetric positive definite and $H_{ij}\leq0$ for $i\neq j$.
Suppose there are $\upsilon\in\R_{++}^n$ and $\mu>0$ such that
$H\upsilon\geq\mu\upsilon$ coordinatewise.  Let $\kappa_\Psi(z)$ be the
minimum-magnitude KKT vector, and define
\begin{equation*}
 C_\upsilon(z):=\sum_j\upsilon_j\abs{\kappa_{\Psi,j}(z)},
 \qquad
 \theta_i:=\frac{(H\upsilon)_i}{H_{ii}\upsilon_i}.
\end{equation*}
Then $0<\mu/H_{ii}\leq\theta_i\leq1$, and an exact proximal update of
coordinate $i$ satisfies
\begin{equation}
 C_\upsilon(z^+)
 \leq C_\upsilon(z)
       -\theta_i\upsilon_i\abs{\kappa_{\Psi,i}(z)}.
 \label{eq:aesp-cd-stieltjes-mass}
\end{equation}
If $z_\Psi^\star$ minimizes $\Psi$, then
\begin{equation}
 \norm{\operatorname{diag}(\upsilon)^{-1}
             (z-z_\Psi^\star)}_\infty
 \leq\frac1\mu
 \norm{\operatorname{diag}(\upsilon)^{-1}
             \kappa_\Psi(z)}_\infty.
 \label{eq:aesp-cd-stieltjes-error}
\end{equation}
\end{theorem}

\begin{proof}
Let $\Delta_i=z_i^+-z_i$.  Exact scalar proximal minimization gives
$\kappa_{\Psi,i}(z^+)=0$ and
$H_{ii}\abs{\Delta_i}\leq\abs{\kappa_{\Psi,i}(z)}$.  The minimum KKT map
at every untouched coordinate is one-Lipschitz in its smooth gradient, so
\begin{align*}
 C_\upsilon(z^+)
 &\leq C_\upsilon(z)-\upsilon_i\abs{\kappa_{\Psi,i}(z)}
       +\sum_{j\neq i}\upsilon_j\abs{H_{ji}\Delta_i}\\
 &\leq C_\upsilon(z)
       -\left(\upsilon_i
         -\frac{\sum_{j\neq i}\upsilon_j\abs{H_{ji}}}{H_{ii}}\right)
        \abs{\kappa_{\Psi,i}(z)}\\
 &=C_\upsilon(z)-\theta_i\upsilon_i
        \abs{\kappa_{\Psi,i}(z)}.
\end{align*}
The last identity uses symmetry and the nonpositive off-diagonal entries.
Those same signs show $\theta_i\leq1$, while
$H\upsilon\geq\mu\upsilon$ gives
$\theta_i\geq\mu/H_{ii}>0$.

For the error certificate, put $e=z-z_\Psi^\star$ and choose $i$ maximizing
$M:=\abs{e_i}/\upsilon_i$.  The weighted maximum principle gives
\begin{equation*}
 \frac{\operatorname{sign}(e_i)(He)_i}{\upsilon_i}
 \geq M\frac{(H\upsilon)_i}{\upsilon_i}\geq\mu M.
\end{equation*}
Monotonicity of each absolute-value subdifferential makes its contribution
nonnegative after multiplication by $\operatorname{sign}(e_i)$.  Therefore
$\abs{\kappa_{\Psi,i}(z)}/\upsilon_i\geq\mu M$, which proves
\cref{eq:aesp-cd-stieltjes-error}.
\end{proof}

The PageRank result is the specialization of
\cref{thm:aesp-cd-stieltjes} with
$H=H_t$, $\upsilon=D^{1/2}\one$, and
$H_t\upsilon=\mu_t\upsilon$.  Since every diagonal entry is $\ell$, one has
$\theta_i=\mu_t/\ell=\tau_{\mathrm{cd}}$ for every coordinate.  Thus the
proved inner-locality mechanism applies not only to RPPR, but also to shifted
RPPR subproblems and, more generally, to separable $\ell_1$ regularization of
any Stieltjes quadratic admitting a positive supersolution.  It does not by
itself control how an outer extrapolation chooses its successive starts.

Thus the composite analogue of the ordinary-PPR residual-mass contraction is
valid for arbitrary signed iterates; no nonnegativity hypothesis is needed.

\section{Inner stopping proposal}

The proposed active set is
\begin{equation}
  S_t^{(k)}
  :=\left\{i:
       \frac{\abs{\kappa_{t,i}(z^{(k)})}}{\sqrt{d_i}}
       \geq\epsin
     \right\}.
  \label{eq:aesp-cd-active-set}
\end{equation}
If an oracle does not expose $m=|E|$, the simple-graph bound
$m\leq n(n-1)/2$ is a correctness-preserving but potentially conservative
choice when calibrating an inner objective certificate.

\begin{theorem}[Local inner-oracle work]
\label{thm:aesp-cd-inner-work}
Starting from $z^{(0)}$, repeatedly update any coordinate in
$S_t^{(k)}$ until this set is empty.  Then the method terminates with
\begin{align}
 \sum_{k=0}^{K_t-1}d_{i_k}
 &\leq
 \frac{C_t(z^{(0)})}{\tau_{\mathrm{cd}}\epsin},
 \label{eq:aesp-cd-work}\\
 \norm{D^{-1/2}(z^{(K_t)}-z_t^\star)}_\infty
 &<\frac{\epsin}{\mu_t}.
 \label{eq:aesp-cd-output-error}
\end{align}
Moreover,
\begin{equation}
 F_{\rho,t}(z^{(K_t)})-F_{\rho,t}(z_t^\star)
 \leq
 \frac{\epsin C_t(z^{(0)})}{2\mu_t}.
 \label{eq:aesp-cd-output-gap}
\end{equation}
Consequently, choosing
\begin{equation}
 \epsin=\frac{2\mu_t\varphi_t}{C_t(z^{(0)})}
 \label{eq:aesp-cd-phi-threshold}
\end{equation}
is a valid local oracle for the AESP inner requirement
$F_{\rho,t}(z)-F_{\rho,t}(z_t^\star)\leq\varphi_t$, with work
\begin{equation}
 \sum_{k=0}^{K_t-1}d_{i_k}
 \leq
 \frac{C_t(z^{(0)})^2}
      {2\tau_{\mathrm{cd}}\mu_t\varphi_t}.
 \label{eq:aesp-cd-gap-work}
\end{equation}
\end{theorem}

\begin{proof}
Every active update has
$\sqrt{d_i}\abs{\kappa_{t,i}(z)}\geq\epsin d_i$.
Sum \cref{eq:aesp-cd-mass-target} and use nonnegativity of $C_t$ to prove
\cref{eq:aesp-cd-work}.  At termination, the KKT diagnostic is below
$\epsin$, so \cref{lem:aesp-cd-kkt-error} proves
\cref{eq:aesp-cd-output-error}.

The function $F_{\rho,t}$ is $\mu_t$-strongly convex and
$\kappa_t(z^{(K_t)})\in\partial F_{\rho,t}(z^{(K_t)})$.  Hence
\begin{equation*}
 F_{\rho,t}(z^{(K_t)})-F_{\rho,t}(z_t^\star)
 \leq\frac{\norm{\kappa_t(z^{(K_t)})}_2^2}{2\mu_t}.
\end{equation*}
At termination and by \cref{thm:aesp-cd-mass},
\begin{align*}
 \norm{\kappa_t(z^{(K_t)})}_2^2
 &\leq
 \norm{D^{-1/2}\kappa_t(z^{(K_t)})}_\infty
 \norm{D^{1/2}\kappa_t(z^{(K_t)})}_1\\
 &<\epsin C_t(z^{(K_t)})
 \leq\epsin C_t(z^{(0)}).
\end{align*}
This proves \cref{eq:aesp-cd-output-gap}; substitution yields
\cref{eq:aesp-cd-gap-work}.
\end{proof}

For the AESP shift $\kappa_{\mathrm A}=1-2\alpha$, one has
$\mu_t=1-\alpha$ and $\tau_{\mathrm{cd}}=2/3$.  Thus
\cref{eq:aesp-cd-gap-work} becomes
\begin{equation}
 \sum_{k=0}^{K_t-1}d_{i_k}
 \leq
 \frac{3C_t(z^{(0)})^2}{4(1-\alpha)\varphi_t}.
 \label{eq:aesp-cd-specialized-work}
\end{equation}

\begin{corollary}[Composite AESP plug-in]
\label{cor:aesp-cd-plugin}
Suppose a composite AESP outer loop accepts its stage-$t$ inner output when
\begin{equation*}
 F_{\rho,t}(z^{(K_t)})-F_{\rho,t}(z_t^\star)\leq\varphi_t
\end{equation*}
and initializes the inner solve at $z_t^{(0)}=y^{(t-1)}$.  If
$C_t(y^{(t-1)})=0$, the initial point is already optimal.  Otherwise,
thresholded exact coordinate descent with
\begin{equation*}
 \varepsilon_{\mathrm{in},t}
 =\frac{2(1-\alpha)\varphi_t}{C_t(y^{(t-1)})}
\end{equation*}
satisfies every such inner-oracle condition, even when the extrapolated
centers are signed.  Across any $T$ outer stages its total degree work obeys
\begin{equation}
 \mathcal W_{\mathrm{inner}}(T)
 \leq\frac{3}{4(1-\alpha)}
       \sum_{t=1}^T
       \frac{C_t(y^{(t-1)})^2}{\varphi_t}.
 \label{eq:aesp-cd-cumulative-interface}
\end{equation}
\end{corollary}

\begin{proof}
Apply \cref{thm:aesp-cd-inner-work} independently at every stage and sum
\cref{eq:aesp-cd-specialized-work}.  The signed-center claim follows because
\cref{thm:aesp-cd-mass} made no sign assumption.
\end{proof}

\Cref{eq:aesp-cd-cumulative-interface} is the precise interface supplied to
the outer acceleration proof.  It removes inner correctness and sign
monotonicity from the open problem: a graph-uniform end-to-end theorem now
requires a bound on the displayed cumulative start-mass functional, or an
outer safeguard that bounds it while retaining accelerated convergence.

\section{Outer-locality continuation: relative accuracy}
\label{sec:aesp-cd-relative}

The absolute-gap interface above is valid but too coarse to preserve the
accelerated locality target.  This section first records the obstruction and
then replaces the raw center by the standard composite Catalyst warm start.

\begin{lemma}[The shift cancels at the raw center]
\label{lem:aesp-cd-center-cancel}
At every outer stage,
\begin{equation}
 \kappa_t(y^{(t-1)})=\kappa_\rho(y^{(t-1)}),
 \qquad
 C_t(y^{(t-1)})
 =\norm{D^{1/2}\kappa_\rho(y^{(t-1)})}_1.
 \label{eq:aesp-cd-center-cancel}
\end{equation}
\end{lemma}

\begin{proof}
Substitution into \cref{eq:aesp-cd-inner-gradient} gives
$\nabla f_t(y^{(t-1)})=Qy^{(t-1)}-b=\nabla f(y^{(t-1)})$.
The coordinatewise minimum-magnitude subgradient choices are consequently
identical.
\end{proof}

There is no dimension-free inequality that upper-bounds the right-hand side
of \cref{eq:aesp-cd-center-cancel} by the current objective gap.  Indeed, for
the one-dimensional Stieltjes objective
\begin{equation*}
 \Psi(w)=\frac{h}{2}w^2+\lambda\abs w,
 \qquad h,\lambda>0,
\end{equation*}
the minimizer is zero, while for $w>0$,
\begin{equation*}
 \kappa_\Psi(w)=hw+\lambda,
 \qquad
 \frac{\abs{\kappa_\Psi(w)}^2}{\Psi(w)-\Psi(0)}
 =\frac{(hw+\lambda)^2}{hw^2/2+\lambda w}
 \longrightarrow\infty
 \quad\text{as }w\downarrow0.
\end{equation*}
Thus a proof that merely combines a decaying outer objective gap with the raw
KKT mass cannot close the cumulative bound.  The discontinuity is caused by
a nonzero point approaching an $\ell_1$ kink; it is not a graph-volume issue.

We next give the relative-accuracy repair.  Let $U\subseteq V$ be a fixed
coordinate envelope, put
\begin{equation*}
 \mathcal V_U:=\vol(U),
\end{equation*}
and restrict every vector and objective to coordinates in $U$.  A matrix-vector
product with $Q_{UU}$ is charged at most $\mathcal V_U$, since scanning the
adjacency lists of $U$ also exposes every internal entry and cut edge.  Write
\begin{equation}
 L_{\mathrm A}:=1+\kappa_{\mathrm A}=2(1-\alpha),
 \qquad
 \chi_{\mathrm A}:=\frac{\alpha}{\alpha+\kappa_{\mathrm A}}
               =\frac{\alpha}{1-\alpha},
 \qquad
 \delta_{\mathrm A}:=
 \frac{\sqrt{\chi_{\mathrm A}}}{2-\sqrt{\chi_{\mathrm A}}}.
 \label{eq:aesp-cd-relative-parameters}
\end{equation}
For a stage center $y\in\R^U$, let $p_U(y)$ minimize the restricted shifted
objective $F_{\rho,t}^U$.  Define the composite proximal warm start
\begin{equation}
 u_U(y):=\prox_{g_{\rho,U}/L_{\mathrm A}}
 \left(y-\frac{\nabla f_{t,U}(y)}{L_{\mathrm A}}\right).
 \label{eq:aesp-cd-smoothed-start}
\end{equation}

\begin{lemma}[Smoothed start-mass bound]
\label{lem:aesp-cd-smoothed-mass}
The restricted minimum KKT mass at \cref{eq:aesp-cd-smoothed-start} satisfies
\begin{equation}
 C_{t,U}(u_U(y))
 \leq 2(1-\alpha)\sqrt{\mathcal V_U}
       \norm{y-p_U(y)}_2.
 \label{eq:aesp-cd-smoothed-mass}
\end{equation}
\end{lemma}

\begin{proof}
Write $u=u_U(y)$ and $H_U=Q_{UU}+\kappa_{\mathrm A}I_U$.  Proximal
optimality supplies a subgradient for which the shifted KKT vector at $u$ is
\begin{equation*}
 (L_{\mathrm A}I_U-H_U)(y-u)=(I_U-Q_{UU})(y-u).
\end{equation*}
The minimum KKT vector can only have smaller coordinate magnitudes.  The
matrix $I-Q$ is entrywise nonnegative and, with
$v=D^{1/2}\one$, satisfies $(I-Q)v=(1-\alpha)v$.  Restriction to $U$ can
only decrease its weighted column sums, so
\begin{equation*}
 C_{t,U}(u)
 \leq(1-\alpha)\norm{D_U^{1/2}(y-u)}_1
 \leq(1-\alpha)\sqrt{\mathcal V_U}\norm{y-u}_2.
\end{equation*}
The forward map in \cref{eq:aesp-cd-smoothed-start} is nonexpansive because
$0\preceq I_U-H_U/L_{\mathrm A}\preceq I_U$, and the proximal map is
nonexpansive.  Since $p_U(y)$ is a fixed point of their composition,
$\norm{u-p_U(y)}_2\leq\norm{y-p_U(y)}_2$.  The triangle inequality now gives
$\norm{y-u}_2\leq2\norm{y-p_U(y)}_2$.
\end{proof}

\begin{lemma}[Greedy KKT-mass contraction in degree work]
\label{lem:aesp-cd-greedy-linear}
Starting from any $z^{(0)}\in\R^U$, choose at each update
\begin{equation*}
 i_k\in\argmax_{i\in U}
       \frac{\abs{\kappa_{t,U,i}(z^{(k)})}}{\sqrt{d_i}}.
\end{equation*}
If $W_K:=\sum_{k=0}^{K-1}d_{i_k}$, then
\begin{equation}
 C_{t,U}(z^{(K)})
 \leq C_{t,U}(z^{(0)})
       \exp\left(-\frac{\tau_{\mathrm{cd}}W_K}{\mathcal V_U}\right).
 \label{eq:aesp-cd-greedy-linear}
\end{equation}
\end{lemma}

\begin{proof}
The principal system satisfies
$H_UD_U^{1/2}\one\geq\mu_tD_U^{1/2}\one$.  Hence
\cref{thm:aesp-cd-stieltjes} gives at least the PageRank contraction
$\tau_{\mathrm{cd}}=\mu_t/\ell$.  Moreover,
\begin{equation*}
 \max_{i\in U}
 \frac{\abs{\kappa_{t,U,i}(z)}}{\sqrt{d_i}}
 \geq\frac{C_{t,U}(z)}{\mathcal V_U}.
\end{equation*}
The chosen update therefore gives
\begin{equation*}
 C_{t,U}(z^+)
 \leq C_{t,U}(z)
       \left(1-\frac{\tau_{\mathrm{cd}}d_i}{\mathcal V_U}\right)
 \leq C_{t,U}(z)
       \exp\left(-\frac{\tau_{\mathrm{cd}}d_i}{\mathcal V_U}\right).
\end{equation*}
Multiplication over the updates proves the claim.
\end{proof}

The greedy key can be maintained in a heap.  A coordinate update changes the
smooth gradient, and hence the KKT key, only at the updated vertex and its
neighbors.  Its data-structure overhead is therefore logarithmic per touched
edge and is hidden by $\softO$; no repeated scan of all of $U$ is required.

For the relative Catalyst condition at center $y$, require
\begin{equation}
 F_{\rho,t}^U(z)-F_{\rho,t}^{U,\star}
 \leq\frac{\delta_{\mathrm A}\kappa_{\mathrm A}}2
             \norm{z-y}_2^2.
 \label{eq:aesp-cd-c2}
\end{equation}
The following theorem makes this condition locally certifiable and bounds
its work.

\begin{theorem}[Local relative-accuracy oracle]
\label{thm:aesp-cd-relative-oracle}
Initialize at $u_U(y)$ and run the greedy updates of
\cref{lem:aesp-cd-greedy-linear} until
\begin{equation}
 C_{t,U}(z)^2
 \leq\delta_{\mathrm A}\kappa_{\mathrm A}\mu_t
       \norm{z-y}_2^2.
 \label{eq:aesp-cd-relative-stop}
\end{equation}
Then \cref{eq:aesp-cd-c2} holds.  Including the proximal warm start, the
degree work of the stage is at most
\begin{equation}
 \mathcal V_U\left[
 2+\frac{1}{2\tau_{\mathrm{cd}}}
 \log_+\left(
  \frac{16(1-\alpha)^2\mathcal V_U}
       {\delta_{\mathrm A}\kappa_{\mathrm A}\mu_t}
 \right)\right].
 \label{eq:aesp-cd-relative-work}
\end{equation}
\end{theorem}

\begin{proof}
The strong-convexity subgradient bound gives, for every $z$,
\begin{equation*}
 F_{\rho,t}^U(z)-F_{\rho,t}^{U,\star}
 \leq\frac{\norm{\kappa_{t,U}(z)}_2^2}{2\mu_t}
 \leq\frac{C_{t,U}(z)^2}{2\mu_t}.
\end{equation*}
Thus \cref{eq:aesp-cd-relative-stop} implies
\cref{eq:aesp-cd-c2}.

Put $r=\norm{y-p_U(y)}_2$.  If $r=0$, the proximal warm start is already
optimal.  Otherwise, it is sufficient to reduce the KKT mass until
\begin{equation*}
 C_{t,U}(z)^2
 \leq\frac{\delta_{\mathrm A}\kappa_{\mathrm A}\mu_t}{4}r^2.
\end{equation*}
Indeed, the preceding gap bound and strong convexity then give
$\norm{z-p_U(y)}_2\leq r/2$, so
$\norm{z-y}_2\geq r/2$ and \cref{eq:aesp-cd-relative-stop} follows.
By \cref{lem:aesp-cd-smoothed-mass}, the squared ratio between the initial
mass and this sufficient target is at most
\begin{equation*}
 \frac{16(1-\alpha)^2\mathcal V_U}
      {\delta_{\mathrm A}\kappa_{\mathrm A}\mu_t}.
\end{equation*}
Now apply \cref{eq:aesp-cd-greedy-linear}.  Crossing the target can overshoot
the corresponding work budget by at most one coordinate cost, which is at
most $\mathcal V_U$.  The proximal warm start costs another
$\mathcal V_U$, yielding \cref{eq:aesp-cd-relative-work}.
\end{proof}

The fixed envelope in \cref{thm:aesp-cd-relative-oracle} is useful for a
graph-uniform corollary, but it is not required to implement the method.  For
a full-graph stage, let $E_t$ be the union of the sparse source support, the
supports of the center and proximal displacement, every coordinate whose KKT
key becomes nonzero, and every coordinate updated before
\cref{eq:aesp-cd-relative-stop} holds.  Put
\begin{equation*}
 \mathcal V_t^{\mathrm{exp}}:=\vol(E_t),
 \qquad
 \mathcal V_{\max}^{\mathrm{exp}}
 :=\max_{1\leq t\leq T}\mathcal V_t^{\mathrm{exp}}.
\end{equation*}

\begin{theorem}[Oracle-free explored-volume bound]
\label{thm:aesp-cd-explored-volume}
Run full-graph strongly convex Catalyst with parameters
\cref{eq:aesp-cd-relative-parameters}.  At every stage, compute the proximal
warm start locally, maintain all nonzero normalized KKT keys in a heap, run
the greedy coordinate updates, and stop by
\cref{eq:aesp-cd-relative-stop}.  Then each inner output satisfies the full
relative condition \cref{eq:aesp-cd-c2}, and the total degree work through
$T$ stages is at most
\begin{equation}
 \sum_{t=1}^T \mathcal V_t^{\mathrm{exp}}
 \left[
  3+\frac{1}{2\tau_{\mathrm{cd}}}
  \log_+\left(
   \frac{16(1-\alpha)^2\mathcal V_t^{\mathrm{exp}}}
        {\delta_{\mathrm A}\kappa_{\mathrm A}\mu_t}
  \right)\right].
 \label{eq:aesp-cd-explored-work}
\end{equation}
Consequently, reaching objective gap $\epsobj$ costs
\begin{equation}
 \softO\left(
   \frac{\mathcal V_{\max}^{\mathrm{exp}}}{\sqrt\alpha}
   \log\frac1{\epsobj}
 \right).
 \label{eq:aesp-cd-explored-rate}
\end{equation}
No support oracle or graph-wide scan is used in this trajectory-dependent
statement.
\end{theorem}

\begin{proof}
For a sparse $z$ and center $y$, a zero coordinate outside
$\supp(z)\cup N(\supp(z))\cup\supp(y)\cup\supp(b)$ has zero smooth gradient
and hence zero minimum KKT key.  Initially, the proximal displacement is
found by scanning the center support, its incident edges, and the sparse
source.  Thereafter an exact coordinate update changes KKT keys only at the
updated coordinate and its neighbors.  Thus the heap contains every nonzero
key and its maximum is the global maximum; newly exposed coordinates are
discovered while scanning already charged incident edges.

The proof of \cref{lem:aesp-cd-smoothed-mass} on the full graph uses
Cauchy--Schwarz only on the support of the proximal displacement, whose
volume is at most $\mathcal V_t^{\mathrm{exp}}$.  At every later update, the
support of the KKT mass also has volume at most
$\mathcal V_t^{\mathrm{exp}}$.  Hence the proofs of
\cref{lem:aesp-cd-greedy-linear,thm:aesp-cd-relative-oracle} apply
retrospectively with this explored volume.  Two local scans initialize the
proximal point and its heap keys, and the last coordinate update may overshoot
the mass budget by one coordinate cost; these give the constant three in
\cref{eq:aesp-cd-explored-work}.

Every stage satisfies the full relative Catalyst condition, so the outer
rate used in \cref{cor:aesp-cd-certified-envelope} applies without
restriction.  Bound every explored volume in
\cref{eq:aesp-cd-explored-work} by
$\mathcal V_{\max}^{\mathrm{exp}}$ and use the
$\bigO(\alpha^{-1/2}\log(1/\epsobj))$ stage count.
\end{proof}

\begin{corollary}[Accelerated work on a certified envelope]
\label{cor:aesp-cd-certified-envelope}
Assume a fixed envelope $U$ contains the optimal RPPR support, so minimizing
$F_\rho$ over $\R^U$ has the same solution and value as the full problem.
Run strongly convex Catalyst on this restricted objective with
\cref{eq:aesp-cd-relative-parameters} and use
\cref{thm:aesp-cd-relative-oracle} at every stage.  To obtain
$F_\rho(x^{(T)})-F_\rho(x^\star)\leq\epsobj$, it is sufficient to take
\begin{equation*}
 T\leq
 \left\lceil\frac{2}{\sqrt{\chi_{\mathrm A}}}
 \log\left(\frac{2(F_\rho(x^{(0)})-F_\rho(x^\star))}{\epsobj}\right)
 \right\rceil.
\end{equation*}
The total degree work is consequently
\begin{equation}
 \softO\left(\frac{\mathcal V_U}{\sqrt\alpha}
                  \log\frac1{\epsobj}\right).
 \label{eq:aesp-cd-certified-envelope-work}
\end{equation}
If an oracle supplies $U=S^\star(\rho)$, then
$\mathcal V_U\leq1/\rho$ and the bound becomes
\begin{equation}
 \softO\left(\frac1{\rho\sqrt\alpha}
                  \log\frac1{\epsobj}\right).
 \label{eq:aesp-cd-support-oracle-work}
\end{equation}
\end{corollary}

\begin{proof}
Condition \cref{eq:aesp-cd-c2} is the relative Catalyst criterion of
\citet{lin2018catalyst}.  With
$\chi_{\mathrm A}=\alpha/(\alpha+\kappa_{\mathrm A})$ and
$\delta_{\mathrm A}=\sqrt{\chi_{\mathrm A}}/(2-\sqrt{\chi_{\mathrm A}})$,
the standard strongly convex Catalyst estimate is
\begin{equation*}
 F_\rho(x^{(T)})-F_\rho(x^\star)
 \leq2\left(1-\frac{\sqrt{\chi_{\mathrm A}}}{2}\right)^T
       (F_\rho(x^{(0)})-F_\rho(x^\star)).
\end{equation*}
Combine this estimate with \cref{eq:aesp-cd-relative-work}.  All factors
other than $\mathcal V_U/\sqrt\alpha$ are logarithmic in the displayed
parameters.  The last claim uses the RPPR support-volume theorem.
\end{proof}

This closes the start-mass and outer-rate interaction on every fixed certified
envelope, while \cref{thm:aesp-cd-explored-volume} gives an oracle-free
trajectory-dependent result.  It also explains why the absolute interface
\cref{eq:aesp-cd-cumulative-interface} loses a power of the final tolerance:
it uses a one-shot $C^2/\varphi$ estimate where the relative oracle instead
uses geometric mass contraction.  The remaining graph-uniform algorithmic
task is no longer to control signed KKT mass.  It is to prove that the
automatically explored envelope has volume $\bigO(1/\rho)$, or to enforce
such an envelope without restarting acceleration after every admission.

\begin{corollary}[Zero-start RPPR coordinate descent]
\label{cor:aesp-cd-zero-start}
For the unshifted objective, $C_\rho(0)\leq\alpha$.  Coordinate descent from
zero, stopped when $R_{\mathrm{kkt},\rho}(x)<\alpha\epskkt$, therefore
returns
\begin{equation*}
 \norm{D^{-1/2}(x-x^\star(\rho))}_\infty<\epskkt
\end{equation*}
with degree work at most
\begin{equation}
 \frac{1+\alpha}{2\alpha\epskkt}.
 \label{eq:aesp-cd-zero-start-work}
\end{equation}
\end{corollary}

\begin{proof}
At zero, the only KKT violations are seed coordinates and
\begin{equation*}
 C_\rho(0)
 =\alpha\sum_i(s_i-\rho d_i)_+
 \leq\alpha\sum_i s_i=\alpha.
\end{equation*}
Apply \cref{thm:aesp-cd-inner-work} with $\kappa_{\mathrm A}=0$,
$\tau_{\mathrm{cd}}=2\alpha/(1+\alpha)$, and
$\epsin=\alpha\epskkt$.
\end{proof}

\section{Implementation record and remaining open question}

Any experiment for this proposal must record the shared graph and work model,
$\alpha$, $\rho$, $\epskkt$, $\epsin$, random seed, stopping formula, and code
version.  It should report both $R_{\mathrm{kkt},\rho}(x)$ and the KKT-mass
decrease diagnostics.

\Cref{thm:aesp-cd-mass,thm:aesp-cd-inner-work,cor:aesp-cd-plugin} close the
inner local-monotonicity and oracle-correctness tasks.
\Cref{thm:aesp-cd-relative-oracle,cor:aesp-cd-certified-envelope} additionally
close the interaction between start mass and the accelerated outer rate on a
fixed certified envelope.  The remaining central task is oracle-free envelope
discovery and expanding-subspace continuation.  Therefore this note does not
yet claim the target $\softO(1/(\rho\sqrt\alpha))$ for a fully implementable
graph-uniform algorithm.  A valid continuation must maintain the certified
$\bigO(1/\rho)$ envelope without destroying acceleration or charge every
expansion only to newly admitted degree volume.
