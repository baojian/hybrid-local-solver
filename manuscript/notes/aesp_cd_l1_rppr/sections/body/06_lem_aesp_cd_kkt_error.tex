\section{KKT certificate and local monotonicity}

\begin{lemma}[KKT diagnostic certifies inner solution error]
\label{lem:aesp-cd-kkt-error}
Let $z_t^\star$ be the unique minimizer of $F_{\rho,t}$.  Then every
$z\in\R^n$ satisfies
\begin{equation}
  \norm{D^{-1/2}(z-z_t^\star)}_\infty
  \leq
  \frac{1}{\mu_t}
  \norm{D^{-1/2}\kappa_t(z)}_\infty.
  \label{eq:aesp-cd-inner-error}
\end{equation}
In particular, for the unshifted RPPR objective,
\begin{equation}
  \norm{D^{-1/2}(x-x^\star(\rho))}_\infty
  \leq \frac{R_{\mathrm{kkt},\rho}(x)}{\alpha}.
  \label{eq:aesp-cd-rppr-error}
\end{equation}
\end{lemma}

\begin{proof}
Choose subgradients $\xi\in\partial\norm{D^{1/2}z}_1$ and
$\xi^\star\in\partial\norm{D^{1/2}z_t^\star}_1$ so that
\begin{equation*}
  \kappa_t(z)=H_t(z-z_t^\star)
  +\alpha\rho(\xi-\xi^\star).
\end{equation*}
The coordinate definition of $\kappa_t$ is precisely the minimum-magnitude
valid choice at $z$, while optimality supplies a choice with zero residual at
$z_t^\star$.

Put $e=z-z_t^\star$ and choose $i$ maximizing
$M:=\abs{e_i}/\sqrt{d_i}$.  If $M=0$, the result is immediate.  Otherwise,
the normalized maximum principle and
\cref{eq:aesp-cd-shifted-properties} give
\begin{equation*}
 \frac{\operatorname{sign}(e_i)(H_te)_i}{\sqrt{d_i}}
 \geq \mu_tM.
\end{equation*}
Coordinatewise monotonicity of the weighted absolute-value subdifferential
gives
$e_i(\xi_i-\xi_i^\star)\geq0$.  Hence
\begin{equation*}
 \frac{\abs{\kappa_{t,i}(z)}}{\sqrt{d_i}}
 \geq
 \frac{\operatorname{sign}(e_i)\kappa_{t,i}(z)}{\sqrt{d_i}}
 \geq \mu_tM,
\end{equation*}
which proves \cref{eq:aesp-cd-inner-error}.  Set
$\kappa_{\mathrm A}=0$ for \cref{eq:aesp-cd-rppr-error}.
\end{proof}

\begin{theorem}[Exact KKT-mass decrease]
\label{thm:aesp-cd-mass}
For a sequential proximal coordinate update at coordinate $i$,
\begin{equation}
  C_t(z^+)
  \leq C_t(z)
  -\tau_{\mathrm{cd}}\sqrt{d_i}\abs{\kappa_{t,i}(z)},
  \qquad
  \tau_{\mathrm{cd}}
  :=\frac{2(\alpha+\kappa_{\mathrm A})}
          {1+\alpha+2\kappa_{\mathrm A}}.
  \label{eq:aesp-cd-mass-target}
\end{equation}
Under $\kappa_{\mathrm A}=1-2\alpha$, the displayed coefficient is $2/3$.
\end{theorem}

\begin{proof}
Write $\lambda_i:=\alpha\rho\sqrt{d_i}$ and
$\Delta_i:=z_i^+-z_i$.  Exact scalar proximal minimization makes the new
coordinate KKT violation zero and gives
\begin{equation}
  \kappa_{t,i}(z^+)=0,
  \qquad
  \ell\abs{\Delta_i}\leq\abs{\kappa_{t,i}(z)}.
  \label{eq:aesp-cd-displacement}
\end{equation}
The second inequality follows directly from the three scalar soft-threshold
cases: the update stays on the same nonzero face, reaches zero, or crosses to
the opposite face.  Equality holds in the first case; thresholding can only
shorten the displacement in the latter two.

For every $j\neq i$, the value $z_j$ is unchanged.  As a function of its
smooth gradient, the minimum KKT violation is either a translate
$g\mapsto g\pm\lambda_j$ on a nonzero face or the soft-threshold map
$g\mapsto g-\operatorname{clip}(g,-\lambda_j,\lambda_j)$ at zero.  Each map
is one-Lipschitz.  Therefore
\begin{equation*}
 \abs{\kappa_{t,j}(z^+)}
 \leq
 \abs{\kappa_{t,j}(z)}+\abs{H_{t,ji}\Delta_i}.
\end{equation*}
Only neighbors of $i$ have a nonzero off-diagonal entry, and
\begin{equation*}
 \sum_{j\neq i}\sqrt{d_j}\abs{H_{t,ji}\Delta_i}
 =\frac{1-\alpha}{2}\sqrt{d_i}\abs{\Delta_i}.
\end{equation*}
Using \cref{eq:aesp-cd-displacement}, we obtain
\begin{align*}
 C_t(z^+)
 &\leq C_t(z)-\sqrt{d_i}\abs{\kappa_{t,i}(z)}
       +\frac{1-\alpha}{2}\sqrt{d_i}\abs{\Delta_i}\\
 &\leq C_t(z)
 -\left(1-\frac{1-\alpha}{2\ell}\right)
  \sqrt{d_i}\abs{\kappa_{t,i}(z)}.
\end{align*}
Finally,
\begin{equation*}
 1-\frac{1-\alpha}{2\ell}
 =\frac{2(\alpha+\kappa_{\mathrm A})}
        {1+\alpha+2\kappa_{\mathrm A}}
 =\tau_{\mathrm{cd}}.
\end{equation*}
\end{proof}

\begin{theorem}[Stieltjes-quadratic inner locality]
\label{thm:aesp-cd-stieltjes}
Let
\begin{equation*}
 \Psi(z)=\frac12 z^\top Hz-c^\top z
          +\sum_{j=1}^n\lambda_j\abs{z_j},
 \qquad \lambda_j\geq0,
\end{equation*}
where $H$ is symmetric positive definite and $H_{ij}\leq0$ for $i\neq j$.
Suppose there are $\upsilon\in\R_{++}^n$ and $\mu>0$ such that
$H\upsilon\geq\mu\upsilon$ coordinatewise.  Let $\kappa_\Psi(z)$ be the
minimum-magnitude KKT vector, and define
\begin{equation*}
 C_\upsilon(z):=\sum_j\upsilon_j\abs{\kappa_{\Psi,j}(z)},
 \qquad
 \theta_i:=\frac{(H\upsilon)_i}{H_{ii}\upsilon_i}.
\end{equation*}
Then $0<\mu/H_{ii}\leq\theta_i\leq1$, and an exact proximal update of
coordinate $i$ satisfies
\begin{equation}
 C_\upsilon(z^+)
 \leq C_\upsilon(z)
       -\theta_i\upsilon_i\abs{\kappa_{\Psi,i}(z)}.
 \label{eq:aesp-cd-stieltjes-mass}
\end{equation}
If $z_\Psi^\star$ minimizes $\Psi$, then
\begin{equation}
 \norm{\operatorname{diag}(\upsilon)^{-1}
             (z-z_\Psi^\star)}_\infty
 \leq\frac1\mu
 \norm{\operatorname{diag}(\upsilon)^{-1}
             \kappa_\Psi(z)}_\infty.
 \label{eq:aesp-cd-stieltjes-error}
\end{equation}
\end{theorem}

\begin{proof}
Let $\Delta_i=z_i^+-z_i$.  Exact scalar proximal minimization gives
$\kappa_{\Psi,i}(z^+)=0$ and
$H_{ii}\abs{\Delta_i}\leq\abs{\kappa_{\Psi,i}(z)}$.  The minimum KKT map
at every untouched coordinate is one-Lipschitz in its smooth gradient, so
\begin{align*}
 C_\upsilon(z^+)
 &\leq C_\upsilon(z)-\upsilon_i\abs{\kappa_{\Psi,i}(z)}
       +\sum_{j\neq i}\upsilon_j\abs{H_{ji}\Delta_i}\\
 &\leq C_\upsilon(z)
       -\left(\upsilon_i
         -\frac{\sum_{j\neq i}\upsilon_j\abs{H_{ji}}}{H_{ii}}\right)
        \abs{\kappa_{\Psi,i}(z)}\\
 &=C_\upsilon(z)-\theta_i\upsilon_i
        \abs{\kappa_{\Psi,i}(z)}.
\end{align*}
The last identity uses symmetry and the nonpositive off-diagonal entries.
Those same signs show $\theta_i\leq1$, while
$H\upsilon\geq\mu\upsilon$ gives
$\theta_i\geq\mu/H_{ii}>0$.

For the error certificate, put $e=z-z_\Psi^\star$ and choose $i$ maximizing
$M:=\abs{e_i}/\upsilon_i$.  The weighted maximum principle gives
\begin{equation*}
 \frac{\operatorname{sign}(e_i)(He)_i}{\upsilon_i}
 \geq M\frac{(H\upsilon)_i}{\upsilon_i}\geq\mu M.
\end{equation*}
Monotonicity of each absolute-value subdifferential makes its contribution
nonnegative after multiplication by $\operatorname{sign}(e_i)$.  Therefore
$\abs{\kappa_{\Psi,i}(z)}/\upsilon_i\geq\mu M$, which proves
\cref{eq:aesp-cd-stieltjes-error}.
\end{proof}

The PageRank result is the specialization of
\cref{thm:aesp-cd-stieltjes} with
$H=H_t$, $\upsilon=D^{1/2}\one$, and
$H_t\upsilon=\mu_t\upsilon$.  Since every diagonal entry is $\ell$, one has
$\theta_i=\mu_t/\ell=\tau_{\mathrm{cd}}$ for every coordinate.  Thus the
proved inner-locality mechanism applies not only to RPPR, but also to shifted
RPPR subproblems and, more generally, to separable $\ell_1$ regularization of
any Stieltjes quadratic admitting a positive supersolution.  It does not by
itself control how an outer extrapolation chooses its successive starts.

\begin{theorem}[Fixed-face Chebyshev solve with a safe Stieltjes output]
\label{thm:aesp-cd-safe-chebyshev-face}
Let $A$ be a fixed active face and consider the interior linear system
\begin{equation*}
 H_Ax=b_A,
 \qquad
 H_A=H_A^\top\succ0,
 \qquad
 (H_A)_{ij}\leq0\quad(i\neq j).
\end{equation*}
Suppose
\begin{equation*}
 \mu_{\rm sp}I\preceq H_A\preceq L_{\rm sp}I,
 \qquad
 H_A\upsilon\geq\mu_\upsilon\upsilon
\end{equation*}
for certified scalars $0<\mu_{\rm sp}\leq L_{\rm sp}$,
$\mu_\upsilon>0$, and a certified vector $\upsilon>0$.  Assume the face
solution $x_A^\star:=H_A^{-1}b_A$ is strictly positive.  Let $y_A\geq0$ be
an optional published checkpoint satisfying
$b_A-H_Ay_A\geq0$ on $\supp(y_A)$; take $y_A=0$ when no checkpoint is
needed.  Starting from any
$z_0$, let $\widehat x_k$ be the degree-$k$ Chebyshev semi-iterate and put
$r_k=b_A-H_A\widehat x_k$.  Then
\begin{equation}
 \norm{r_k}_2
 \leq2\zeta^k\norm{r_0}_2,
 \qquad
 \mathcal K:=\frac{L_{\rm sp}}{\mu_{\rm sp}},
 \qquad
 \zeta:=\frac{\sqrt{\mathcal K}-1}{\sqrt{\mathcal K}+1}.
 \label{eq:aesp-cd-safe-chebyshev-rate}
\end{equation}
Define the a posteriori Stieltjes retraction
\begin{equation}
 \Delta_k:=\max_{i\in A}
   \frac{[-r_{k,i}]_+}{\mu_\upsilon\upsilon_i},
 \qquad
 s_k:=[\widehat x_k-\Delta_k\upsilon]_+,
 \qquad
 \ell_k:=y_A\vee s_k.
 \label{eq:aesp-cd-safe-chebyshev-retraction}
\end{equation}
Then $b_A-H_A\ell_k\geq0$ on $\supp(\ell_k)$ and
$0\leq y_A\leq\ell_k\leq x_A^\star$.  Moreover, with
$\upsilon_{\min}:=\min_i\upsilon_i$,
\begin{equation}
 \norm{x_A^\star-\ell_k}_2
 \leq\left(
  \frac1{\mu_{\rm sp}}
  +\frac{\norm{\upsilon}_2}{\mu_\upsilon\upsilon_{\min}}
 \right)\norm{r_k}_2.
 \label{eq:aesp-cd-safe-chebyshev-output-error}
\end{equation}
If $y_A>0$, the output remains in the same relative interior.  The same is
true without a positive checkpoint if
\begin{equation}
 \min_{i\in A}\frac{\widehat x_{k,i}}{\upsilon_i}>\Delta_k,
 \label{eq:aesp-cd-safe-chebyshev-face-guard}
\end{equation}
holds.
The computation uses $k$ applications of $H_A$ and one face scan.

For $H_A=Q_A+\kappa'I$, with
$\lambda_1(Q_A)\geq\underline\lambda>0$ and $Q_A\preceq I$, choosing
$\kappa'=\Theta(\underline\lambda)$ gives
\begin{equation}
 k=O\!\left(
  \underline\lambda^{-1/2}
  \log\frac{\norm{r_0}_2}{\xi}
 \right)
 \label{eq:aesp-cd-safe-chebyshev-small-shift-work}
\end{equation}
applications to reach a prescribed residual scale $\xi$, up to the standard
condition-number factor inside the logarithm.  If an $H_A$ application costs
$O(\vol(A))$, the degree work is
$\widetilde O(\vol(A)/\sqrt{\underline\lambda})$.
For a principal RPPR face one may take
$\upsilon=D_A^{1/2}\one$ and
$\mu_\upsilon=\alpha+\kappa'$, because deleting exterior Stieltjes
off-diagonal entries only increases $Q_A\upsilon$.
More specifically, if
$b_A=c_A+\kappa'y_A$, $y_A>0$, and
$c_A-Q_Ay_A\geq0$, then the published output is also an original-system
lower certificate:
\begin{equation}
 c_A-Q_A\ell_k
 =(b_A-H_A\ell_k)+\kappa'(\ell_k-y_A)\geq0.
 \label{eq:aesp-cd-safe-chebyshev-original-lower}
\end{equation}
\end{theorem}

\begin{proof}
The Chebyshev estimate in
\cref{eq:aesp-cd-safe-chebyshev-rate} is the standard minimax residual bound
on $[\mu_{\rm sp},L_{\rm sp}]$.  It remains to verify that acceleration has
not destroyed the order certificate.  If $s_{k,i}>0$, then
$s_{k,i}=\widehat x_{k,i}-\Delta_k\upsilon_i$, whereas
$s_{k,j}\geq\widehat x_{k,j}-\Delta_k\upsilon_j$ for every $j$.  The
nonpositive off-diagonal entries therefore give
\begin{align*}
 (H_As_k)_i
 &\leq[H_A(\widehat x_k-\Delta_k\upsilon)]_i\\
 &\leq(H_A\widehat x_k)_i-\mu_\upsilon\Delta_k\upsilon_i
 \leq b_{A,i}.
\end{align*}
Thus $s_k$ is an obstacle subsolution.  The coordinatewise maximum of two
Stieltjes obstacle subsolutions is again a subsolution: at a coordinate
selected from one input, increasing the other coordinates can only decrease
that matrix row.  Hence $\ell_k=y_A\vee s_k$ is a subsolution.  The weighted
maximum principle yields
$0\leq y_A\leq\ell_k\leq x_A^\star$.

The linear-system error obeys
$\norm{x_A^\star-\widehat x_k}_2\leq
\norm{r_k}_2/\mu_{\rm sp}$, while
$\Delta_k\leq
\norm{r_k}_2/(\mu_\upsilon\upsilon_{\min})$.
For $t_k:=\widehat x_k-\Delta_k\upsilon$ one has
\begin{equation*}
 H_A(x_A^\star-t_k)=r_k+\Delta_kH_A\upsilon\geq0,
\end{equation*}
so inverse positivity gives $t_k\leq x_A^\star$.  Taking the positive part
and then the maximum with $y_A$ can only move the point coordinatewise toward
$x_A^\star$.  This proves
\cref{eq:aesp-cd-safe-chebyshev-output-error}.  The strict guard prevents the
positive-part operation from deleting a face coordinate when the checkpoint
itself is not already positive.  The
small-shift work statement follows from
\begin{equation*}
 \mathcal K
 \leq\frac{1+\kappa'}{\underline\lambda+\kappa'}
 =O(\underline\lambda^{-1}).
\end{equation*}
Every Chebyshev multiplication is restricted to the already exposed face;
the theorem makes no claim about discovering that face or surviving an
admission during the iteration.
\end{proof}

\begin{corollary}[Collatz-certified small shift]
\label{cor:aesp-cd-collatz-small-shift}
Let
$M_{\bar\kappa}=\bar\kappa(Q_A+\bar\kappa I)^{-1}$ be entrywise positive
on a connected fixed face, and let $z>0$.  A certified Collatz upper ratio
\begin{equation*}
 U\geq\max_{i\in A}\frac{(M_{\bar\kappa}z)_i}{z_i},
 \qquad U<1,
\end{equation*}
gives the spectral lower bound
\begin{equation}
 \lambda_1(Q_A)\geq
 \underline\lambda_U:=\bar\kappa(U^{-1}-1).
 \label{eq:aesp-cd-collatz-small-shift}
\end{equation}
Consequently \cref{thm:aesp-cd-safe-chebyshev-face} may use
$\kappa'=\Theta(\underline\lambda_U)$ without knowing the Perron eigenpair.
If, for a constant $c>0$, the certificate also satisfies
$1-U\geq c[1-m_1(M_{\bar\kappa})]$, then
$\underline\lambda_U=\Theta(\lambda_1(Q_A))$ and the fixed-face work is
$\widetilde O(\vol(A)/\sqrt{\lambda_1(Q_A)})$.
For constant $\bar\kappa$, this is
$\widetilde O(\vol(A)/q_A)$, where
$q_A^2=\lambda_1/(\bar\kappa+\lambda_1)$.
\end{corollary}

\begin{proof}
Collatz--Wielandt gives
$m_1(M_{\bar\kappa})\leq U$.  Since
$m_1=\bar\kappa/(\bar\kappa+\lambda_1(Q_A))$, inversion gives
\cref{eq:aesp-cd-collatz-small-shift}.  The remaining statements follow from
\cref{eq:aesp-cd-safe-chebyshev-small-shift-work}.
Constructing the certified ratio requires its own resolvent solve and face
scan; the corollary does not treat that work as free.
\end{proof}

\begin{corollary}[Direct accelerated solve after final-face certification]
\label{cor:aesp-cd-final-face-chebyshev}
Suppose the final RPPR support $S^\star$ has been exposed and certified, and
the restricted optimum is in its relative interior.  Apply
\cref{thm:aesp-cd-safe-chebyshev-face} directly to the unshifted system
$Q_{S^\star}x=c_{S^\star}$, using
\begin{equation*}
 \alpha I\preceq Q_{S^\star}\preceq I,
 \qquad
 \upsilon=D_{S^\star}^{1/2}\one,
 \qquad
 Q_{S^\star}\upsilon\geq\alpha\upsilon.
\end{equation*}
Once the raw Chebyshev residual is at most $\xi$, the published safe output
has the error bound in
\cref{eq:aesp-cd-safe-chebyshev-output-error}; reaching that raw scale uses
\begin{equation}
 O\!\left(\frac1{\sqrt\alpha}
   \log\frac{\norm{r_0}_2}{\xi}\right)
 \label{eq:aesp-cd-final-face-chebyshev-work}
\end{equation}
restricted matrix applications and one final certificate scan.  Hence its
degree work is
\begin{equation*}
 \widetilde O\!\left(
   \frac{\vol(S^\star)}{\sqrt\alpha}\right)
 \leq
 \widetilde O\!\left(\frac1{\rho\sqrt\alpha}\right).
\end{equation*}
This is an accelerated local terminal solver once the final face is known;
it does not charge support discovery or prove that a provisional face is
final.
\end{corollary}

\begin{proposition}[Unrestricted polynomial scratch can spill exponentially]
\label{prop:aesp-cd-polynomial-scratch-spill-stop}
Let $T_d$ be a rooted $d$-regular tree, $d\geq3$, with adjacency matrix
$A_d$.  If
$p_k(t)=a_kt^k+\cdots+a_0$ has $a_k\neq0$, then for every vertex $v$ at
distance $k$ from the root $o$,
\begin{equation}
 [p_k(A_d)e_o]_v=a_k\neq0.
 \label{eq:aesp-cd-polynomial-scratch-spill}
\end{equation}
Hence the scratch support contains at least
$d(d-1)^{k-1}$ coordinates.  The same conclusion holds for normalized
adjacency and Stieltjes operators with nonzero edge weights, with the right
side multiplied by the unique path-weight product.

This spill is not merely a nonzero-support artifact.  Put
\begin{equation*}
 B:=\frac{A_d}{2\sqrt{d-1}},\qquad
 c_\alpha:=\frac{1+\alpha}{2},\qquad
 r_\alpha:=\frac{1-\alpha}{2},\qquad
 H:=c_\alpha I-r_\alpha B.
\end{equation*}
Then $\operatorname{spec}(H)\subseteq[\alpha,1]$, and the degree-$k$ Chebyshev residual
polynomial applied to $e_o$ is
\begin{equation*}
 \frac{T_k(B)e_o}{T_k(c_\alpha/r_\alpha)}.
\end{equation*}
Its squared $\ell_2$ mass on the distance-$k$ frontier is exactly
\begin{equation}
 \frac{d}{4(d-1)T_k(c_\alpha/r_\alpha)^2}.
 \label{eq:aesp-cd-chebyshev-frontier-mass}
\end{equation}
For $k=\Theta(1/\sqrt\alpha)$ the denominator is constant-order.  More
sharply, the frontier has $N=d(d-1)^{k-1}$ equal-magnitude entries, while
$\norm{T_k(B)e_o}_2\leq1$.  Any coordinate-sparse approximation retaining
fewer than $N/2$ frontier coordinates therefore has relative error at least
\begin{equation*}
 \sqrt{\frac{d}{8(d-1)}}.
\end{equation*}
Thus every smaller fixed-relative-error representation retains
$\exp(\Theta(1/\sqrt\alpha))$ coordinates.

This wave is compatible with a root-only obstacle optimum.  For any
$\delta>0$, take $x^\star=e_o$ and the sparse right-hand side
\begin{equation*}
 b:=He_o-\delta\sum_{v\sim o}e_v.
\end{equation*}
Then $Hx^\star-b$ is zero at the root and equals $\delta$ at its neighbors,
so the obstacle KKT conditions hold and $\supp(x^\star)=\{o\}$.  The neighbor
part of $b$ contributes zero to the distance-$k$ frontier of $T_k(B)b$ by
tree parity, whereas its root part retains the wave above up to the constant
$c_\alpha$.  Taking $\delta$ sufficiently small, constant-relative-error
full-graph scratch therefore requires
$\exp(\Theta(1/\sqrt\alpha))$ coordinates even for a one-vertex published
support, and cannot be charged only to $\vol(S^\star)/\sqrt\alpha$.
\end{proposition}

\begin{proof}
For $j<k$, no length-$j$ walk reaches a vertex at distance $k$.  At length
$k$ the shortest root--vertex path is unique, so
$(A_d^ke_o)_v=1$.  This proves
\cref{eq:aesp-cd-polynomial-scratch-spill}; weighted edges multiply the
unique path contributions.  More precisely, the leading term
$2^{k-1}B^k$ gives value $1/[2(d-1)^{k/2}]$ at every distance-$k$ vertex.
There are $d(d-1)^{k-1}$ such vertices, proving
\cref{eq:aesp-cd-chebyshev-frontier-mass}.  Moreover
$T_k(c_\alpha/r_\alpha)
=\cosh(k\operatorname{arcosh}((1+\alpha)/(1-\alpha)))$ is constant-order
when $k=\Theta(1/\sqrt\alpha)$.  Every power in $T_k$ has the parity of $k$;
a neighbor of the root is at distance $k-1$ from a depth-$k$ vertex, so its
contribution vanishes.  The displayed KKT residual proves the obstacle claim.
This is a generic Stieltjes obstacle construction; it is not asserted to
preserve the special RPPR right-hand side.
\end{proof}

This is a STOP for exact untruncated full-graph scratch, not for a thresholded
polynomial equipped with a certified truncation-error budget.  Such a budget
is an additional locality mechanism, not a consequence of the terminal
support bound.

\begin{proposition}[Certified-boundary face-by-face restart has quadratic volume]
\label{prop:aesp-cd-face-by-face-volume-stop}
For every $n$ there is an endpoint-seeded RPPR instance on the path $P_n$
whose final support is all of $V(P_n)$.  Consider any nested active-set
protocol which starts at the endpoint, admits only vertices on the currently
exposed graph boundary, and performs a full scan or matrix application on
the current face after every admission.  Before reaching the final face it
must expose the prefixes
\begin{equation*}
 A_1\subset A_2\subset\cdots\subset A_n=V(P_n),
 \qquad |A_j|=j,
\end{equation*}
and therefore pays
\begin{equation}
 \sum_{j=1}^n\Omega(\vol(A_j))=\Omega(n^2).
 \label{eq:aesp-cd-face-by-face-volume-stop}
\end{equation}
In particular, invoking a fresh accelerated restricted solve after every
single admission does not give a linear-volume local solver, even though the
terminal solve on $A_n$ is accelerated.
\end{proposition}

\begin{proof}
Let $x^{(0)}=Q^{-1}b$ be the zero-threshold endpoint solution.  It is strictly
positive on the connected path.  Since
$Q^{-1}(\alpha\rho D^{1/2}\one)=\rho D^{1/2}\one$, choosing
\begin{equation*}
 0<\rho<\min_i\frac{x_i^{(0)}}{\sqrt{d_i}}
\end{equation*}
makes the RPPR optimum
$x^\star=x^{(0)}-\rho D^{1/2}\one$ strictly positive, hence its support is
all of $P_n$.  The graph boundary of a connected endpoint prefix $A_j$ is the
single next vertex.  A boundary-only nested protocol cannot skip it, so it
visits every prefix.  Path volume is $\Theta(j)$, which proves
\cref{eq:aesp-cd-face-by-face-volume-stop}.
\end{proof}

The proposition does not rule out speculative certified envelopes, dynamic
reuse of a Krylov/factorization state, or full-face solves only at
volume-doubling checkpoints.  It shows that at least one such mechanism is
necessary.

The path obstruction is representation-specific rather than informational.
The exact Schur update identifies both the escape on paths and the missing
general-graph primitive.

\begin{proposition}[Incremental Schur response and activation-once paths]
\label{prop:aesp-cd-incremental-schur-response}
Let an old restricted system be $Hx=b$, with $H$ symmetric positive definite
and Stieltjes.  Add a block of coordinates so that
\begin{equation*}
 H_+=\begin{pmatrix}H&-C\\-C^{\mathsf T}&G\end{pmatrix},
 \qquad b_+=\binom b d,
 \qquad C\geq0,
\end{equation*}
and put
\begin{equation*}
 S:=G-C^{\mathsf T}H^{-1}C,
 \qquad \delta:=d+C^{\mathsf T}x.
\end{equation*}
Then the new exact solution and restricted-optimum gain are
\begin{equation}
 t=S^{-1}\delta,\qquad
 x_+=x+H^{-1}Ct,\qquad
 \Delta=\frac12\langle\delta,S^{-1}\delta\rangle.
 \label{eq:aesp-cd-incremental-schur-response}
\end{equation}
If $H_+$ is also Stieltjes and $\delta>0$, the update is coordinatewise
nonnegative.  The only genuinely new old-face operation is therefore the
response $H^{-1}C$; retaining the old solution vector alone does not provide
an incremental algorithm.

On an endpoint path with diagonal $a_i$, off-diagonal $-c_i$, and an
endpoint-source right-hand side satisfying $b_i\leq0$ for $i\geq1$, define
append-only messages
\begin{align*}
 D_0&:=a_0,& y_0&:=b_0,\\
 D_i&:=a_i-\frac{c_{i-1}^2}{D_{i-1}},&
 y_i&:=b_i+\frac{c_{i-1}}{D_{i-1}}y_{i-1}.
\end{align*}
The exact endpoint value on prefix $\{0,\ldots,i\}$ is $y_i/D_i$, so the
next boundary key is
\begin{equation*}
 \delta_{i+1}=b_{i+1}+c_i\frac{y_i}{D_i}.
\end{equation*}
It is decided in $O(1)$ arithmetic per admitted vertex.  As long as the
generated prefixes are interior, the first nonpositive key is the global
obstacle KKT certificate; one reverse substitution then materializes the
final solution in linear work.  Thus all singleton-prefix admissions on an
endpoint-seeded RPPR path cost $O(\vol(S^\star))$, not the quadratic sum in
\cref{prop:aesp-cd-face-by-face-volume-stop}.
\end{proposition}

\begin{proof}
Block elimination gives the first two identities in
\cref{eq:aesp-cd-incremental-schur-response}; completing the restricted
quadratic square gives the gain identity.  Stieltjes inverse positivity and
Schur-complement positivity prove monotonicity.  The path formulas are scalar
$LDL^{\mathsf T}$ elimination.  The last eliminated coordinate has value
$y_i/D_i$, and the unique outside row gives the displayed key.  Each message
is appended once and each record is read once in the terminal reverse pass.
\end{proof}

This exact path algorithm is already developed as the activation-once solver
in the companion incremental-active-set note.  For arbitrary graphs,
\cref{eq:aesp-cd-incremental-schur-response} sharpens the open interface:
one needs an implicit dynamic Schur/SDD representation that applies the old
inverse to new coupling blocks and refreshes all affected boundary queries in
near-linear total local work.  Fixed-rank recycling of old Krylov vectors is
not such an interface, because successive coupling blocks can introduce new
hard spectral directions.

\begin{corollary}[Dynamic-Schur plus doubling checkpoint interface]
\label{cor:aesp-cd-dynamic-schur-doubling-interface}
Suppose a nested-face implementation maintains a compressed restricted
solution and all boundary keys under admissions with total charged work
$\mathcal I(S^\star)$, without materializing the old solution after every
update.  This charge includes adjacency discovery, every affected
boundary-key refresh, admissions after the last rebuild, and terminal
materialization (or a terminal rebuild if the compressed oracle does not
directly recover the exact solution).  Rebuild its accelerated
restricted-solve state only when active
volume has doubled, and assume each rebuild on $A$ costs
$\widetilde O(\vol(A)/\sqrt\alpha)$.  Then support discovery, all rebuilds,
and one final materialization cost
\begin{equation*}
 \mathcal I(S^\star)
 +\widetilde O\!\left(\frac{\vol(S^\star)}{\sqrt\alpha}\right).
\end{equation*}
In particular, a near-linear dynamic Schur/boundary oracle would complete the
support-sensitive accelerated accounting.  The path messages above realize
$\mathcal I(S^\star)=O(\vol(S^\star))$ exactly.
\end{corollary}

\begin{proof}
Between two rebuilds, charge every admission to the declared incremental
oracle.  Rebuild volumes form a doubling sequence, whose sum is less than
twice the final active volume.  Add the one-time final output scan.
\end{proof}

The dynamic response oracle is nonvacuous on forests, and more generally on
a supplied bounded-treewidth decomposition.  The result below isolates its
exact scope: it maintains a declared stream of candidate queries, but does
not discover that stream for free.

\begin{proposition}[Dynamic Schur messages on forests and bounded treewidth]
\label{prop:aesp-cd-dynamic-schur-forest}
Let a currently explored $N$-vertex forest carry an SPD Stieltjes quadratic
on its revealed factors, with every coordinate marked either free or pinned
at zero.  Given a stream consisting
of $F$ factor/edge reveals, $J$ mark toggles, and $Q$ pinned-coordinate KKT
queries, there is an exact dynamic Schur representation using $O(N)$ records
and
\begin{equation}
 O\!\left((N+F+J+Q)\log(2+N)\right)
 \label{eq:aesp-cd-dynamic-schur-forest-work}
\end{equation}
field operations, including one terminal recovery of all free coordinates.
For a queried pinned vertex $v$, it returns the exact marginal
\begin{equation*}
 \frac12\sigma_vt^2-\delta_vt+\mathrm{const},
 \qquad \sigma_v>0,
\end{equation*}
so $\delta_v=(b-Hx)_v$ is its exact admission key.  If $\delta_v>0$,
unpinning $v$ gives $x_v=\delta_v/\sigma_v>0$ and makes all old free
coordinates nondecreasing.  Coordinates in the irreducible free block
coupled to $v$ increase strictly.

More generally, suppose a width-$w$ running-intersection bag tree with $B$
bags is supplied, and every unary/edge factor has a fixed home bag.  The same
interface costs
\begin{equation}
 O\!\left((w+1)^3(F+J+Q)\log(2+B)+(w+1)^3B\right)
 \label{eq:aesp-cd-dynamic-schur-treewidth-work}
\end{equation}
field operations and $O((w+1)^2B)$ records.  Thus, for fixed supplied
treewidth and
$B+F+J+Q=\widetilde O(\vol(S^\star))$, it realizes the
incremental oracle required by
\cref{cor:aesp-cd-dynamic-schur-doubling-interface}.
\end{proposition}

\begin{proof}
Use a balanced top-tree contraction of the forest.  A cluster has at most two
boundary vertices and stores the quadratic obtained after pinning its marked
interior coordinates and Schur-eliminating its free interior coordinates.
To avoid double counting, every edge factor has one edge-leaf record, while
unary factors and pin marks at cluster-boundary vertices are excluded from
that cluster and inserted at the unique lowest cluster where their home
vertex becomes interior.  Ancestors only inherit the resulting summary.  A
join therefore eliminates at most one scalar.
If its assembled part is
\begin{equation*}
 \frac12z^{\mathsf T}Az+z^{\mathsf T}cy+\frac d2y^2
 -f^{\mathsf T}z-gy+\gamma,
\end{equation*}
the updated summary is
\begin{equation*}
 K'=A-cc^{\mathsf T}/d,\qquad
 f'=f-cg/d,\qquad
 \gamma'=\gamma-g^2/(2d).
\end{equation*}
Every pivot is positive because it is a Schur complement of a principal free
block of the SPD matrix.  Standard top-tree expose, link, cut, and mark
updates change $O(\log(2+N))$ clusters.  At root closure, add each remaining
root-boundary unary factor and apply each remaining root-boundary pin mark
once.  Exposing $v$, omitting only the zero mark of the queried vertex, and
eliminating all other free variables gives the displayed marginal.
Inverse positivity proves the nonnegative update after a positive toggle.  A
downward pass recovers all free coordinates once.

For the bag-tree version, a cluster port is the union of its at most two
boundary bags and has size at most $2(w+1)$.  Assign every factor and mark a
home bag containing its full scope, exclude boundary-bag home factors from
the child summary, and add the home factors and marks once when the bag
becomes internal (or once at root closure if it remains a root-boundary bag).
Running intersection guarantees
that the eliminated block is exactly the global constrained principal block.
Each dense join and Schur elimination uses $O((w+1)^3)$ operations; the same
top-tree accounting proves
\cref{eq:aesp-cd-dynamic-schur-treewidth-work}.
\end{proof}

The explicit $Q$ term is essential.  Dynamic Schur messages answer a named
frontier query for the currently revealed quadratic but do not automatically
report every positive frontier key; a future factor reveal may change it.
Even on a star, pinned-leaf residuals can be affine functions
$\delta_i(t)=\varepsilon(it-i^2)$ of the free center value; their maximum has
linearly many breakpoints.  Likewise, eliminating all nonnegative leaves
produces a piecewise-quadratic obstacle value with one breakpoint per leaf,
not a constant-size quadratic summary.  Therefore
\cref{prop:aesp-cd-dynamic-schur-forest} is a literal fixed-mark response
primitive.  An end-to-end forest solver still needs a lazy/kinetic frontier
reporter ensuring $Q=\widetilde O(\vol(S^\star))$; rescanning a persistent
frontier after every admission can make $Q$ quadratic.

The next result records a sufficient condition under which restricted-face
scratch and boundary discovery do have the desired geometric accounting.

\begin{proposition}[Margin-certified geometric face batches]
\label{prop:aesp-cd-geometric-face-batches}
Let $A_0\subset A_1\subset\cdots\subset S^\star$ be nested certified faces,
assume $\{v:c_v>0\}\subseteq A_0$, and let $p_A>0$ denote the exact
restricted solution on $A$.  For
$v\in\partial A$ define
\begin{equation*}
 g_v(A):=c_v-Q_{vA}p_A,
 \qquad
 L_v^\partial:=\sum_{i\in A}(-Q_{vi}).
\end{equation*}
Suppose the safe Chebyshev publication $z_A$ satisfies
$0\leq z_A\leq p_A$ and
$\norm{p_A-z_A}_\infty\leq\varepsilon_A$.  Then
\begin{equation}
 g_v(z_A)\leq g_v(A)
 \leq g_v(z_A)+L_v^\partial\varepsilon_A.
 \label{eq:aesp-cd-boundary-key-interval}
\end{equation}
In particular, $g_v(z_A)>0$ certifies $v\in S^\star$.

Assume that for every proper face $A_j$ there is a batch
$B_j\subseteq\partial A_j$ such that, for fixed $0<\gamma\leq1$,
\begin{equation}
 \vol(B_j)\geq\gamma\vol(A_j),
 \qquad
 g_v(A_j)\geq\tau_jL_v^\partial\quad(v\in B_j),
 \qquad
 \varepsilon_j\leq\frac{\tau_j}{2}.
 \label{eq:aesp-cd-geometric-face-batch-hypothesis}
\end{equation}
Admit every vertex certified positive by the boundary scan.  Then
$\vol(A_{j+1})\geq(1+\gamma)\vol(A_j)$.  If the final face additionally
satisfies
\begin{equation*}
 g_v(z_{S^\star})+L_v^\partial\varepsilon_\star\leq0
 \qquad(v\in\partial S^\star),
\end{equation*}
one final scan certifies termination.  Restricted safe-Chebyshev work and
boundary scans therefore total
\begin{equation}
 \widetilde O\!\left(
   \frac{\vol(S^\star)}{\gamma\sqrt\alpha}\right),
 \label{eq:aesp-cd-geometric-face-batch-work}
\end{equation}
where the logarithms include the certified margins $\tau_j$.
\end{proposition}

\begin{proof}
Because $Q_{vA}\leq0$ entrywise,
\begin{equation*}
 g_v(A)-g_v(z_A)
 =\sum_{i\in A}(-Q_{vi})(p_{A,i}-z_{A,i})
 \in[0,L_v^\partial\varepsilon_A],
\end{equation*}
which proves \cref{eq:aesp-cd-boundary-key-interval}.  If $x_v^\star=0$,
the global obstacle KKT condition and $z_A\leq x_A^\star$ imply
$g_v(z_A)\leq0$; hence a positive lower key is a genuine support
certificate.  Under
\cref{eq:aesp-cd-geometric-face-batch-hypothesis}, every $v\in B_j$ has
$g_v(z_{A_j})\geq\tau_jL_v^\partial/2>0$ and is admitted.  The face volumes
therefore grow geometrically.  Since one restricted solve and scan at face
$A_j$ costs
$\widetilde O(\vol(A_j)/\sqrt\alpha)$, summing the geometric series gives
\cref{eq:aesp-cd-geometric-face-batch-work}.  The displayed upper boundary
certificate at the final face is the complementary KKT condition.
\end{proof}

The growth and margin hypotheses are substantive.  On a path prefix only
the next vertex is visible; an arbitrarily small positive key may neither
double volume nor meet a fixed work-normalized margin, while the following
vertex cannot be certified before that admission.  Thus delayed batching
can merge many already exposed frontier events, but it does not by itself
remove the serial-frontier obstruction in
\cref{prop:aesp-cd-face-by-face-volume-stop}.

There is a different, trajectory-dependent escape: double a coordinate
envelope rather than requiring the certified active face itself to double.

\begin{proposition}[Conditional speculative-envelope doubling]
\label{prop:aesp-cd-speculative-envelope-doubling}
Let $U_0\subset U_1\subset\cdots\subset U_J$ be coordinate envelopes
containing $\{v:c_v>0\}$.  The restricted obstacle minimizer $p_U$ may
have active face $A_U=\supp(p_U)$ strictly smaller than $U$.  Assume a
restricted obstacle primitive publishes $z_U$ with
\begin{equation*}
 0\leq z_U\leq p_U,\qquad
 \norm{p_U-z_U}_\infty\leq\varepsilon_U
\end{equation*}
in
$\widetilde O(\vol(U)/\sqrt{\lambda_{\rm floor}})$ work, including its
active fallback.  Assume a charged exact fallback whose cost is included in
this bound, or a nonzero
boundary-margin promise resolves every interval in
\cref{eq:aesp-cd-boundary-key-interval} in finite work, either with positive
lower endpoint or nonpositive upper endpoint.

Here $\partial U$ means every outside neighbor of $U$.  If all upper
endpoints are nonpositive, the zero extension of $p_U$ is the
global obstacle optimum and the algorithm terminates.  Otherwise, anchor a
speculative graph search at the certified-positive vertices and choose the
next envelope so that
\begin{equation*}
 \vol(U_{j+1})\geq2\vol(U_j),
\end{equation*}
unless all source-containing ambient components reached by the search have
been exhausted; every selected certified-positive anchor must be included in
$U_{j+1}$.  Only the positive support of the next obstacle solve is
published.  Then there is at most one
nondoubling terminal expansion and
\begin{equation}
 J\leq1+\left\lceil\log_2
   \frac{\vol(U_J)}{\vol(U_0)}\right\rceil,\qquad
 \sum_{j=0}^J\vol(U_j)\leq3\vol(U_J).
 \label{eq:aesp-cd-speculative-envelope-volume}
\end{equation}
Consequently discovery, restricted solves, and certificate scans cost
\begin{equation}
 \widetilde O\!\left(
   \frac{\vol(U_J)}{\sqrt{\lambda_{\rm floor}}}
 \right).
 \label{eq:aesp-cd-speculative-envelope-work}
\end{equation}
This is a bound in the final explored volume, not in
$\vol(S^\star)$.
\end{proposition}

\begin{proof}
The boundary interval follows exactly as in
\cref{prop:aesp-cd-geometric-face-batches}.  Nonboundary vertices outside
$U$ have no incident active coordinate and no source, so a closed boundary,
together with the restricted obstacle KKT conditions, is the global KKT
certificate.  Restricted Stieltjes comparison gives
$p_{U_j}\leq p_{U_{j+1}}\leq x^\star$ after zero padding, so publication
remains lower-safe.  Every nonterminal envelope doubles.  If graph search
exhausts all relevant source-containing components before doubling, the
following solve has a closed boundary and is terminal.  The geometric series proves
\cref{eq:aesp-cd-speculative-envelope-volume}; summing the assumed primitive
cost proves \cref{eq:aesp-cd-speculative-envelope-work}.
\end{proof}

\begin{proposition}[RPPR decoy stops support-sensitive blind doubling]
\label{prop:aesp-cd-rppr-speculative-decoy}
Fix $0<\alpha<1$ and
\begin{equation*}
 0<\rho<\bar\rho_\alpha
 :=\frac{(1-\alpha)^2}{\alpha^2+10\alpha+5}.
\end{equation*}
For every sufficiently large $M$, let $G_M$ contain the path
$s-a-u-h$, attach $M-1$ leaves to $h$, and seed $s$.  Thus
\begin{equation*}
 (d_s,d_a,d_u,d_h,d_z)=(1,2,2,M,1).
\end{equation*}
For all $M$ above a $\rho,\alpha$ dependent threshold, the exact RPPR support
is
\begin{equation*}
 S^\star=\{s,a,u\},qquad \vol(S^\star)=5.
\end{equation*}
Nevertheless, the factor-two rule in
\cref{prop:aesp-cd-speculative-envelope-doubling}, even when graph search is
anchored only at a newly certified violation, can be forced to include $h$.
Its explored volume is then at least $M+5$, so
\begin{equation}
 \frac{\vol(U_J)}{\vol(S^\star)}\geq\frac{M+5}{5}\longrightarrow\infty.
 \label{eq:aesp-cd-rppr-speculative-decoy}
\end{equation}
Thus no graph-uniform replacement of explored volume by final support volume
is valid for blind factor-two prefetch.
\end{proposition}

\begin{proof}
Write $b_\alpha=(1-\alpha)/2$ and let
$T=\{s,a,u\}$.  The principal matrix $Q_T$ and the restricted right-hand side
$c_{\rho,T}=\alpha e_s-\alpha\rho\sqrt d_T$ are independent of $M$.  At
$\rho=0$, inverse positivity makes $Q_T^{-1}(\alpha e_s)$ strictly positive.
Put $p_\alpha=(1+\alpha)/2$ and use degree-scaled coordinates
$y=D_T^{-1/2}x$.  Direct elimination on the two prefixes gives scaled
next-boundary keys
\begin{equation*}
 \Gamma_a=-2\alpha\rho+b_\alpha y_s,
 \qquad
 \Gamma_u=-2\alpha\rho+b_\alpha y_a,
\end{equation*}
where
\begin{equation*}
 y_s=\frac{\alpha(1-\rho)}{p_\alpha},
 \qquad
 y_a=\frac{\alpha[b_\alpha(1-\rho)-2p_\alpha\rho]}
 {2p_\alpha^2-b_\alpha^2}.
\end{equation*}
Here $\Gamma_v:=\sqrt{d_v}\,(c_\rho-Qx)_v$, so it has the same sign as the
ordinary KKT key.  The displayed restriction on $\rho$ is exactly the stricter condition
$\Gamma_u>0$ and also implies $\Gamma_a>0$.  Stieltjes inverse positivity
then makes the full restricted solution on $T$ strictly positive.

Let $x_T^\star=Q_T^{-1}c_{\rho,T}$.  At $h$, its exterior KKT key is
\begin{equation*}
 g_h(T)=-\alpha\rho\sqrt M
 +\frac{b_\alpha x_{T,u}^\star}{\sqrt{2M}}<0
\end{equation*}
once $M>b_\alpha x_{T,u}^\star/(\alpha\rho\sqrt2)$.  Every leaf of $h$ has
key $-\alpha\rho<0$.  Hence the zero extension of $x_T^\star$ satisfies the
global KKT conditions and $S^\star=T$.

Starting from $\{s\}$, admitting the certified vertex $a$ grows volume from
$1$ to $3$.  At $\{s,a\}$, the only continuation violation is $u$; admitting
it gives volume $5<2\cdot3$.  The unique outward neighbor of this new anchor
is $h$, so blind graph search toward volume six includes $h$ and immediately
pays its degree $M$, proving
\cref{eq:aesp-cd-rppr-speculative-decoy}.
\end{proof}

The hypothesis of an accelerated
\emph{obstacle} primitive is not supplied by
\cref{thm:aesp-cd-safe-chebyshev-face}: that theorem accelerates a certified
interior linear face, whereas a speculative envelope may contain inactive
coordinates.  Moreover, explored volume cannot in general be charged to
restricted-optimum gain or final active volume.  A prefetched halo may remain
identically zero (zero gain) yet contain an arbitrarily high-degree decoy.
Implementations therefore need a hard degree-volume guard and a fallback;
without an independent bound on $\vol(U_J)$,
\cref{eq:aesp-cd-speculative-envelope-work} is not the desired universal
local theorem.

\begin{corollary}[Strict margins instantiate the obstacle primitive]
\label{cor:aesp-cd-margin-obstacle-primitive}
On a fixed envelope $U$, let
\begin{equation*}
 F_U(x)=\tfrac12\langle x,H_Ux\rangle-\langle b_U,x\rangle
       +I_{x\geq0},
 \qquad \mu I\preceq H_U\preceq LI,
\end{equation*}
and let $p_U$ have support $S$.  Suppose certified margins satisfy
\begin{equation*}
 \min_{i\in S}p_{U,i}\geq m_x>0,\qquad
 \max_{i\in U\setminus S}(b_U-H_Up_U)_i\leq-m_g<0.
\end{equation*}
When $S=U$, omit the dual-margin condition and interpret
$m_g/R_\infty=+\infty$ below.
Assume also that the old padded checkpoint $y_U$ is an obstacle subsolution,
$0\leq y_U\leq p_U$, with $\supp(y_U)\subseteq S$.
Assume $H_U$ is Stieltjes and the certified face $S$ has a known vector
$v_S>0$ and scalar $\mu_v>0$ such that
$H_{SS}v_S\geq\mu_vv_S$ coordinatewise, as required by
\cref{thm:aesp-cd-safe-chebyshev-face}.
Put $R_\infty=\max_i\sum_j|(H_U)_{ij}|$.  Accelerated projected-gradient
scratch to
\begin{equation*}
 \norm{a-p_U}_\infty
 \leq\frac13\min\{m_x,m_g/R_\infty\}
\end{equation*}
identifies $S$: active coordinates exceed $m_x/2$, while inactive
coordinates are below $m_x/2$ and have dual residual below $-m_g/2$.
Running
\cref{thm:aesp-cd-safe-chebyshev-face} only on this certified linear face,
from the old padded lower checkpoint, publishes a safe approximation to
$p_U$.  The combined work is
\begin{equation*}
 \widetilde O\!\left(
   \vol(U)\sqrt{L/\mu}\right),
\end{equation*}
where $m_x,m_g$ enter only logarithmically.  Hence
\cref{prop:aesp-cd-speculative-envelope-doubling} is implementable under
uniform strict active/dual margins.
\end{corollary}

\begin{proof}
Strongly convex accelerated projected gradient reaches the displayed distance
in $O(\sqrt{L/\mu}\log(E_0/\min\{m_x,m_g/R_\infty\}))$ iterations.  Every
active coordinate is then at least $2m_x/3$.  On an inactive coordinate,
$a_i\leq m_x/3$ and
\begin{equation*}
 (b_U-H_Ua)_i
 \leq-m_g+R_\infty\norm{a-p_U}_\infty\leq-2m_g/3.
\end{equation*}
Thus the exact support is certified before the signed linear-face scratch is
published.  The maximum-with-old-checkpoint retraction preserves the lower
certificate, as in \cref{thm:aesp-cd-safe-chebyshev-face}.
\end{proof}

Without strict margins, projected acceleration may leave arbitrarily small
positive inactive coordinates and cannot certify the face at a uniform
accuracy.  This corollary is therefore a nondegenerate conditional GO, not a
resolution of the universal active-set problem.

\begin{proposition}[Cone-preserving residual polynomials cannot accelerate]
\label{prop:aesp-cd-positive-polynomial-stop}
Let $P_k(t)=\sum_{j=0}^k a_jt^j$ be a residual polynomial with
$P_k(1)=1$.  Suppose that $P_k(T)r\geq0$ for every entrywise nonnegative
matrix $T$ of every finite dimension and every $r\geq0$.  Then
$a_j\geq0$ for every $j$, and for $0\leq t\leq1$,
\begin{equation}
 P_k(t)\geq t^k.
 \label{eq:aesp-cd-positive-polynomial-stop}
\end{equation}
Consequently, at $t=1-1/\mathcal K$, reducing the residual by a factor
$\varepsilon$ requires
$k=\Omega(\mathcal K\log(1/\varepsilon))$.  The same lower bound holds for
any product of such cone-preserving restart polynomials.  Thus a
square-root-rate inner solver must permit signed intermediate residuals; the
safe publication map in
\cref{thm:aesp-cd-safe-chebyshev-face} is not merely cosmetic.
\end{proposition}

\begin{proof}
Apply the hypothesis to the nilpotent forward shift $S$ in dimension
$k+1$ and its first basis vector.  The coordinates of
$P_k(S)e_1$ are exactly $a_0,\ldots,a_k$, so all coefficients are
nonnegative.  Since they sum to one and $t^j\geq t^k$ for $j\leq k$,
\cref{eq:aesp-cd-positive-polynomial-stop} follows.  Products preserve
nonnegative coefficients and add total degree.
\end{proof}

Thus the composite analogue of the ordinary-PPR residual-mass contraction is
valid for arbitrary signed iterates; no nonnegativity hypothesis is needed.
