\subsection{Round-025 boundary shielding and the post-full spectral filter}

The discontinuity in
\cref{prop:aesp-cd-retraction-shadowing-stop} concerns the raw map
$\mathcal L$ on arbitrary inputs.  The realized recurrence reaches a support
boundary with substantially more structure.  In particular, the first
retraction containing a newly positive coordinate is one-sided stable.

\begin{lemma}[Support entries are shielded by the previous end residual]
\label{lem:aesp-cd-support-entry-shield}
Consider the actual finite recurrence of
\cref{lem:aesp-cd-finite-inner-identities}.  For $t\geq2$, put
\begin{equation*}
 A_t:=\supp(x_t),\qquad
 E_t:=A_t\setminus A_{t-1},\qquad
 P_t:=A_t\cap A_{t-1},
\end{equation*}
and let
\begin{equation*}
 g_t:=[\zeta_\rho^{\mathrm{low}}(y_t)]_{-,A_t},\qquad
 G_t^{\mathrm{ent}}
 :=\norm{D_{E_t}^{-1/2}g_{t,E_t}}_\infty,
 \qquad
 G_t^{\mathrm{per}}
 :=\norm{D_{P_t}^{-1/2}g_{t,P_t}}_\infty,
\end{equation*}
where the maximum over an empty set is zero.  Then
\begin{equation}
 0\leq g_{t,E_t}\leq\beta_{\mathrm A}a_{t-1,E_t},
 \qquad
 G_t^{\mathrm{ent}}
 \leq\beta_{\mathrm A}
       \norm{D_{E_t}^{-1/2}a_{t-1,E_t}}_\infty
 \leq\beta_{\mathrm A}C_{\mathrm{end},t-1},
 \label{eq:aesp-cd-support-entry-residual-bound}
\end{equation}
and
\begin{equation}
 \alpha\Delta_t=\max\{G_t^{\mathrm{ent}},G_t^{\mathrm{per}}\}.
 \label{eq:aesp-cd-entry-persistent-split}
\end{equation}
Consequently, with exact shifted solves, a newly entered row contributes
zero to its first retraction and cannot be its positive controlling row.  In
the finite recurrence, if
$G_t^{\mathrm{ent}}>G_t^{\mathrm{per}}$, then the entire common correction
is residual-controlled:
\begin{equation}
 \alpha\Delta_t\leq
 \beta_{\mathrm A}C_{\mathrm{end},t-1}.
 \label{eq:aesp-cd-entry-dominant-delta}
\end{equation}
Thus the support-entry jump of the raw map cannot be charged as an
uncontrolled state perturbation along the realized recurrence.
\end{lemma}

\begin{proof}
If $i\in E_t$, support monotonicity gives
$x_{t-1,i}=\ell_{t-2,i}=0$, and hence $s_{t-1,i}=0$.  Since $x_{t,i}>0$,
one also has $s_{t,i}=x_{t,i}>0$, $\lambda_{t,i}=0$, and $a_{t,i}\geq0$.
The exact finite collapse identity
\cref{eq:aesp-cd-finite-inner-collapse} therefore gives
\begin{equation*}
 g_{t,i}
 =\left[-(1+\beta_{\mathrm A})\kappa_{\mathrm A}s_{t,i}
       +\beta_{\mathrm A}a_{t-1,i}
       -(1+\beta_{\mathrm A})a_{t,i}
       -\beta_{\mathrm A}\lambda_{t-1,i}\right]_+
 \leq\beta_{\mathrm A}a_{t-1,i}.
\end{equation*}
Taking the normalized maximum proves the first inequality in
\cref{eq:aesp-cd-support-entry-residual-bound}.  The second follows from
$\norm{D^{-1/2}a}_{\infty}\leq\langle D^{1/2}\one,a\rangle$ in the shared
connected unit-weight graph model.  Splitting the maximum in
\cref{eq:aesp-cd-finite-inner-retraction} over the disjoint union
$A_t=E_t\mathbin{\dot\cup}P_t$ proves
\cref{eq:aesp-cd-entry-persistent-split,eq:aesp-cd-entry-dominant-delta}.
For an exact solve, $a_{t-1}=0$.
\end{proof}

The same finite identity gives a one-sided stability interface on every
support, not only at an entry.  It compares corrections at the same realized
states and therefore does not iterate the ill-conditioned ambient map.

\begin{lemma}[Finite residuals create only polynomial one-sided correction excess]
\label{lem:aesp-cd-one-sided-correction-excess}
For $t\geq2$, define the residual-free same-state driver
\begin{equation*}
 \alpha\bar\Delta_t
 :=\kappa_{\mathrm A}
 \norm{D_{A_t}^{-1/2}
 [\beta_{\mathrm A}s_{t-1}
  -(1+\beta_{\mathrm A})s_t]_+}_\infty,
 \qquad
 \bar r_t:=\min\{\beta_{\mathrm A}d_t,
                  \bar\Delta_tD^{1/2}\one\},
\end{equation*}
and put
\begin{equation*}
 \delta_t:=\frac{\beta_{\mathrm A}}\alpha
 \norm{D_{A_t}^{-1/2}a_{t-1,A_t}}_\infty.
\end{equation*}
Then
\begin{equation}
 \Delta_t\leq\bar\Delta_t+\delta_t,
 \qquad
 0\leq[r_t-\bar r_t]_+
 \leq\delta_tD^{1/2}\one
 \quad\text{coordinatewise}.
 \label{eq:aesp-cd-one-sided-correction-excess}
\end{equation}
If $\vol(A_t)\leq V$ and the combined C2-plus-absolute stop in
\cref{cor:aesp-cd-dual-inner-stop} is imposed at stage $t-1$, then
\begin{equation}
 \norm{[r_t-\bar r_t]_+}_2
 \leq\frac{\beta_{\mathrm A}\mu_t}{\alpha}\xi_{t-1},
 \qquad
 \frac{1+q}{q}\norm{[r_t-\bar r_t]_+}_2
 \leq\frac{1-q}{q}\frac{\mu_t}{\alpha}\xi_{t-1}.
 \label{eq:aesp-cd-one-sided-momentum-excess}
\end{equation}
Thus the additional correction created by a finite end residual has a
one-stage polynomial factor $O(1/(\alpha q))$, rather than the iterated
$(1+1/\alpha)^T$ factor of black-box trajectory shadowing.  The statement is
one-sided: the current residual and boundary multiplier may suppress a
same-state correction, so it is not an absolute trajectory-distance bound
and does not prove finite net packing.
\end{lemma}

\begin{proof}
Coordinatewise subadditivity of the positive part in
\cref{eq:aesp-cd-finite-inner-collapse}, after discarding the two
nonpositive terms containing $a_t$ and $\lambda_{t-1}$, gives
\begin{equation*}
 [\zeta_\rho^{\mathrm{low}}(y_t)]_{-,A_t}
 \leq\kappa_{\mathrm A}
 [\beta_{\mathrm A}s_{t-1}-(1+\beta_{\mathrm A})s_t]_+
 +\beta_{\mathrm A}a_{t-1,A_t}.
\end{equation*}
Taking the normalized maximum proves the bound on $\Delta_t$.  For fixed
$u\geq0$, the scalar map $z\mapsto\min\{u,z\}$ can increase by at most the
increase in $z$; applying this coordinatewise proves
\cref{eq:aesp-cd-one-sided-correction-excess}.  Finally,
$\norm{D^{-1/2}a_{t-1}}_\infty\leq C_{\mathrm{end},t-1}$ and the dual stop
give
\begin{equation*}
 \delta_t\sqrt V
 \leq\frac{\beta_{\mathrm A}}\alpha
       C_{\mathrm{end},t-1}\sqrt V
 \leq\frac{\beta_{\mathrm A}\mu_t}\alpha\xi_{t-1}.
\end{equation*}
Since $\norm{D_{A_t}^{1/2}\one}_2=\sqrt{\vol(A_t)}\leq\sqrt V$, this is the
first inequality in
\cref{eq:aesp-cd-one-sided-momentum-excess}.  The identity
$(1+q)\beta_{\mathrm A}=1-q$ proves the second.
\end{proof}

The preceding residual term can be made harmless at the accelerated scale
without knowing how many coordinates enter.  This is a direct finite-prefix
statement, rather than trajectory shadowing.

\begin{corollary}[Entry-dominated inflation has an affordable finite ledger]
\label{cor:aesp-cd-entry-inflation-ledger}
Assume $\alpha<1/2$, so $0<q<1$, and fix a fresh-gate tolerance
$0<\tau\leq1$.  Put
\begin{equation*}
 \eta_{\mathrm{gate}}:=\frac{2\alpha\tau}{1+\alpha}.
\end{equation*}
Consider any pre-gate prefix of the actual finite recurrence: for every
index in the prefix,
$H_\rho(x_t)>\alpha\tau$.  Fix $\delta>0$ and enforce the locally
maintained end-mass stop
\begin{equation}
 C_{\mathrm{end},t}
 \leq\delta\alpha\eta_{\mathrm{gate}}^2q^2
 \qquad\text{at every finite shifted solve.}
 \label{eq:aesp-cd-entry-end-mass-stop}
\end{equation}
Let
\begin{equation*}
 \mathcal T_{\mathrm{ent}}
 :=\{t\geq2:G_t^{\mathrm{ent}}>G_t^{\mathrm{per}}\}
\end{equation*}
be the entry-dominated stages.  Then every such stage satisfies
\begin{equation}
 \log\max\{1,\gamma_t^{\mathrm{fin}}\}\leq4\delta q,
 \qquad
 \sum_{t\in\mathcal T_{\mathrm{ent}},\,t<T}
 \log\max\{1,\gamma_t^{\mathrm{fin}}\}
 \leq4\delta qT.
 \label{eq:aesp-cd-entry-inflation-bound}
\end{equation}
The bound is independent of the number of support entries.

On an envelope of volume at most $V$, the stop
\cref{eq:aesp-cd-entry-end-mass-stop} can be imposed by the same increasing
greedy heap run as C2.  The degree work for this end-mass requirement alone
at one stage is at most
\begin{equation}
 V\left[3+\frac1{\tau_{\mathrm{cd}}}
 \log_+\left(
 \frac{2(1-\alpha)\sqrt V}
      {\delta\alpha\eta_{\mathrm{gate}}^2q^2}
 \right)\right].
 \label{eq:aesp-cd-entry-polish-work}
\end{equation}
Thus $V\leq1/\rho$ and $T=\widetilde O(1/q)$ keep this residual shielding
within $\widetilde O(1/(\rho q))$ degree work, including the coordinate
updates and their cached rekeys.  It supplies no resource vector by itself,
because the inflation on persistent-row-dominated stages remains open.
\end{corollary}

\begin{proof}
Write $e_t=x^\star-x_t$, $a_q=(1-q)/q$,
$c_q=(1+q)/q$, and $h_t=c_qr_t$.  The finite safe chain gives
$e_t\geq0$ and $0\leq h_t\leq a_qd_t$.  Direct expansion yields
\begin{equation*}
 \mathfrak D_t^{\mathrm{fin}}
 =\norm{h_t}_2^2+2\langle e_t,h_t\rangle
   -2a_q\langle d_t,h_t\rangle
 \leq2\langle e_t,h_t\rangle.
\end{equation*}
At an entry-dominated stage,
\cref{eq:aesp-cd-entry-dominant-delta} and
$r_t\leq\Delta_tD^{1/2}\one$ give
\begin{equation*}
 \langle e_t,h_t\rangle
 \leq c_q\Delta_t\langle D^{1/2}\one,e_t\rangle
 \leq\frac{c_q\beta_{\mathrm A}}\alpha
       C_{\mathrm{end},t-1},
\end{equation*}
where the RPPR mass bound gives
$\langle D^{1/2}\one,e_t\rangle\leq1$.

On the other hand, the row bound used for the terminal gate gives
\begin{equation*}
 H_\rho(x_t)
 \leq\frac{1+\alpha}{2}
       \norm{D^{-1/2}e_t}_\infty.
\end{equation*}
Gate failure therefore implies
$\norm{e_t}_2\geq\eta_{\mathrm{gate}}$.  Strong convexity of the Moreau
envelope gives
$\Phi_t^{\mathrm{fin}}\geq
\mu_{\mathcal E}\eta_{\mathrm{gate}}^2/2$.  If the defect is positive,
$\log\gamma\leq\gamma-1$ and the preceding bounds imply
\begin{equation*}
 \log\gamma_t^{\mathrm{fin}}
 \leq
 \frac{2c_q\beta_{\mathrm A}C_{\mathrm{end},t-1}}
      {\alpha\eta_{\mathrm{gate}}^2}
 \leq4\delta q,
\end{equation*}
using $c_q\leq2/q$, $\beta_{\mathrm A}\leq1$, and
\cref{eq:aesp-cd-entry-end-mass-stop}.  A nonpositive defect contributes
zero, and summation proves
\cref{eq:aesp-cd-entry-inflation-bound}.

For the work claim, the proximal warm-start mass is at most
$2(1-\alpha)\sqrt V$.  The greedy mass contraction used in
\cref{cor:aesp-cd-dual-inner-stop}, with the right-hand side of
\cref{eq:aesp-cd-entry-end-mass-stop} as its target, proves
\cref{eq:aesp-cd-entry-polish-work}.  The heap already stores the exact end
mass; no graph-wide residual scan is introduced.
\end{proof}

After a full correction, the settled-face algebra has a complementary
spectral interpretation.  It explains why the next possible reset is driven
by relatively high Dirichlet modes, but it does not commute with the
coordinatewise positive part well enough to finish the packing theorem.

\begin{proposition}[The two-stage post-full collapse is a high-pass filter]
\label{prop:aesp-cd-post-full-high-pass}
Use exact shifted solves on the settled positive face
$A=S^\star(\rho)$, and suppose the stage-$t$ correction is full:
$\ell_t=x_t$.  Put $e_t=x^\star-x_t$, $H_A=Q_{AA}+\kappa_{\mathrm A}I$,
$M_A=\kappa_{\mathrm A}H_A^{-1}$, and
\begin{equation}
 \sigma_q:=\beta_{\mathrm A}(2+\beta_{\mathrm A})
 =\frac{(1-q)(3+q)}{(1+q)^2}.
 \label{eq:aesp-cd-post-full-sigma}
\end{equation}
If $Q_{AA}e_t\geq0$, then
\cref{lem:aesp-cd-full-correction-separation} gives $r_{t+1}=0$.  Before
the stage-$t+2$ retraction, its exact collapse driver is
\begin{equation}
 \begin{split}
 c_{t+2}
 &:=\beta_{\mathrm A}s_{t+1}
       -(1+\beta_{\mathrm A})s_{t+2}\\
 &=C_Ae_t,
 \qquad
 C_A:=Q_{AA}H_A^{-2}
          (\sigma_qQ_{AA}-\kappa_{\mathrm A}I).
 \end{split}
 \label{eq:aesp-cd-post-full-filter}
\end{equation}
Consequently
\begin{equation}
 \alpha\Delta_{t+2}
 =\kappa_{\mathrm A}
 \norm{D_A^{-1/2}[C_Ae_t]_+}_\infty.
 \label{eq:aesp-cd-post-full-filter-retraction}
\end{equation}
On a $Q_{AA}$ eigenmode of eigenvalue $\lambda$, the filter coefficient is
\begin{equation}
 \psi_q(\lambda)
 =\frac{\lambda(\sigma_q\lambda-\kappa_{\mathrm A})}
        {(\kappa_{\mathrm A}+\lambda)^2}.
 \label{eq:aesp-cd-post-full-filter-eigenvalue}
\end{equation}
In particular, if the nonnegative error $e_t$ is a single eigenmode, that
mode cannot trigger the stage-$t+2$ retraction unless
\begin{equation}
 \frac\lambda{\kappa_{\mathrm A}}
 >\frac1{\sigma_q}
 =\frac{(1+q)^2}{(1-q)(3+q)}.
 \label{eq:aesp-cd-post-full-high-threshold}
\end{equation}
For a general nonnegative error, however, the coordinatewise map
$[C_Ae_t]_+$ mixes spectral modes.  This limitation is already exact on
$K_8$.  Take $q=1/10$, so
\begin{equation*}
 \alpha=\frac1{101},\quad
 \kappa_{\mathrm A}=\frac{99}{101},\quad
 \beta_{\mathrm A}=\frac9{11},\quad
 \sigma_q=\frac{279}{121},
\end{equation*}
and put
$e=(5693/5000,1,\ldots,1)^T$.  The two eigenvalues of $Q$ are
$1/101$ and $407/707$.  Exact decomposition into the constant and
orthogonal modes gives
\begin{equation}
 Qe=\left(\frac{40343}{505000},
          \frac1{10100},\ldots,\frac1{10100}\right)^T>0,
 \qquad
 (C_Ae)_1=\frac{21267}{121000000}>0.
 \label{eq:aesp-cd-post-full-k8-mixing}
\end{equation}
Thus even $Q_{AA}e_t\geq0$ and a small
$\lambda_{\min}(Q_{AA})$ do not make the positive part vanish under the
algebraic hypotheses alone.  This vector is a filter stress test, not a
claimed reachable outer state within this proposition.  Round 026 below
realizes a positive scaling of the same direction from the zero start.  The
present proposition therefore supplies an exact
local/window identity, not a graph-uniform two-step separation, cumulative
$\mathcal J_T^{\mathrm{fin}}$ bound, or solver theorem.
\end{proposition}

\begin{proof}
The first exact solve after the full center gives
$e_{t+1}=M_Ae_t$.  Since $r_{t+1}=0$, its extrapolated center error is
\begin{equation*}
 g_{t+1}=(1+\beta_{\mathrm A})e_{t+1}
             -\beta_{\mathrm A}e_t,
\end{equation*}
and the next exact error is $e_{t+2}=M_Ag_{t+1}$.  Hence
\begin{align*}
 s_{t+1}&=(I-M_A)e_t,\\
 s_{t+2}&=(I-M_A)
       ((1+\beta_{\mathrm A})M_A-\beta_{\mathrm A}I)e_t.
\end{align*}
Substitution gives
\begin{equation*}
 c_{t+2}
 =(I-M_A)
 \left[\sigma_qI-(1+\beta_{\mathrm A})^2M_A\right]e_t.
\end{equation*}
Now $I-M_A=Q_{AA}H_A^{-1}$ and
$\sigma_q-(1+\beta_{\mathrm A})^2=-1$.  Since all factors commute, this is
exactly \cref{eq:aesp-cd-post-full-filter}.  The exact collapse identity
proves \cref{eq:aesp-cd-post-full-filter-retraction}.  Evaluating the
rational matrix function on an eigenvalue gives
\cref{eq:aesp-cd-post-full-filter-eigenvalue}; its sign changes only at
$\lambda=\kappa_{\mathrm A}/\sigma_q$.  If $e_t\geq0$ is one eigenmode, a
nonpositive coefficient makes $[C_Ae_t]_+=0$, proving the threshold.  No
coordinatewise conclusion follows from the signs of the modal coefficients
for a mixed error, which is the stated limitation.
\end{proof}
