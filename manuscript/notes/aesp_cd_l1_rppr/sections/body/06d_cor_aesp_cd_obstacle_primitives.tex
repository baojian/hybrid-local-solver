\begin{theorem}[Margin-free clip-or-pay obstacle retraction]
\label{thm:aesp-cd-obstacle-clip-retraction}
Let
\begin{equation*}
 p=\argmin_{x\geq0}\left\{
   \tfrac12\langle x,Hx\rangle-\langle b,x\rangle\right\},
 \qquad
 H=H^\top\succ0,
 \qquad H_{ij}\leq0\quad(i\neq j).
\end{equation*}
Suppose a certified vector $v>0$ and scalar $\mu_v>0$ satisfy
$Hv\geq\mu_vv$ coordinatewise.  Let $\widehat p\geq0$ be any feasible
scratch approximation with
$\norm{\widehat p-p}_\infty\leq\varepsilon$.  Define
\begin{equation}
 \tau_i:=\frac{\widehat p_i}{v_i},\qquad
 R_i:=\frac{(H\widehat p-b)_i}{(Hv)_i},\qquad
 \Delta:=\max_i\bigl[\min\{\tau_i,R_i\}\bigr]_+,
 \qquad
 s:=[\widehat p-\Delta v]_+.
 \label{eq:aesp-cd-obstacle-clip-pay}
\end{equation}
Then $s$ is an obstacle subsolution and is order-safe:
\begin{equation}
 s\geq0,\qquad b_i-(Hs)_i\geq0\quad(i\in\supp(s)),
 \qquad 0\leq s\leq p.
 \label{eq:aesp-cd-obstacle-clip-pay-safe}
\end{equation}
If $y$ is any previously published obstacle subsolution with
$0\leq y\leq p$, then $z:=y\vee s$ has the same properties and
$\norm{p-z}_\infty\leq\norm{p-s}_\infty$.

Put $v_{\min}:=\min_i v_i$, $v_{\max}:=\max_i v_i$, and
$\norm{H}_\infty:=\max_i\sum_j|H_{ij}|$.  The retraction obeys the explicit
margin-free bounds
\begin{align}
 \Delta
 &\leq \frac{\varepsilon}{v_{\min}}
   \max\left\{1,\frac{\norm{H}_\infty}{\mu_v}\right\},
 \label{eq:aesp-cd-obstacle-clip-pay-delta}\\
 \norm{p-z}_\infty
 &\leq
 \left[1+\frac{v_{\max}}{v_{\min}}
   \max\left\{1,\frac{\norm{H}_\infty}{\mu_v}\right\}\right]\varepsilon.
 \label{eq:aesp-cd-obstacle-clip-pay-error}
\end{align}
Thus large inactive dual residuals do not force a large common shift: an
inactive coordinate may instead pay through its small primal ratio $\tau_i$.
The publication costs one matrix application and one scan.
\end{theorem}

\begin{proof}
Let $t=\widehat p-\Delta v$.  If $s_i>0$, then
$s_i=t_i$ and $\Delta<\tau_i$.  Since
$\Delta\geq\min\{\tau_i,R_i\}$, including when that minimum is negative,
one must have $R_i\leq\Delta$.  Moreover $s\geq t$, and increasing
coordinates other than $i$ can only decrease row $i$ of a Stieltjes matrix.
Consequently
\begin{equation*}
 (Hs)_i\leq(Ht)_i=(H\widehat p)_i-\Delta(Hv)_i\leq b_i.
\end{equation*}
This proves the subsolution condition.

For completeness, let $I=\{i:s_i>p_i\}$.  On $I$, the subsolution
condition and obstacle KKT give $[H(s-p)]_I\leq0$.  On $I^c$ one has
$s-p\leq0$, so $H_{I,I^c}(s-p)_{I^c}\geq0$.  Hence
$H_{II}(s-p)_I\leq0$.  A principal Stieltjes inverse is nonnegative,
contradicting $(s-p)_I>0$ unless $I=\varnothing$.  Thus $s\leq p$.
The coordinatewise maximum of two obstacle subsolutions is again a
subsolution: on a row selected from either input, increasing the other
coordinates only decreases that row.  This proves the assertions for $z$.

It remains to bound the shift without a complementarity margin.  If
$p_i=0$, feasibility and the error bound give
$\tau_i\leq\varepsilon/v_{\min}$.  If $p_i>0$, obstacle KKT gives
$(Hp-b)_i=0$, and therefore
\begin{equation*}
 [R_i]_+
 \leq\frac{|[H(\widehat p-p)]_i|}{\mu_vv_{\min}}
 \leq\frac{\norm{H}_\infty\varepsilon}{\mu_vv_{\min}}.
\end{equation*}
Taking the maximum proves
\cref{eq:aesp-cd-obstacle-clip-pay-delta}.  Finally
$0\leq p-z\leq p-s$ coordinatewise and
$\norm{p-s}_\infty\leq\varepsilon+\Delta v_{\max}$, which proves
\cref{eq:aesp-cd-obstacle-clip-pay-error}.
\end{proof}

\begin{lemma}[One positive load makes every subsolution rooted and connected]
\label{lem:aesp-cd-point-source-subsolution-connected}
Let the nonzero off-diagonal graph of the Stieltjes matrix $H$ be connected,
and suppose $b_o>0$ while $b_i\leq0$ for $i\neq o$.  Every nonzero obstacle
subsolution $z$ has connected support containing $o$.  In particular, both
the obstacle minimizer and every nonzero clip-or-pay publication in
\cref{thm:aesp-cd-obstacle-clip-retraction} have one rooted component.
\end{lemma}

\begin{proof}
If a connected component $C$ of $\supp(z)$ does not contain $o$, then the
subsolution inequalities on $C$ read
\begin{equation*}
 H_{CC}z_C\leq b_C\leq0;
\end{equation*}
coordinates outside $C$ make no contribution because they are zero or are
separated from $C$ in the support graph.  Principal Stieltjes inverse
positivity gives $z_C\leq0$, a contradiction.  Thus every positive component
contains $o$, and there can be at most one.  The exact minimizer is nonzero
because the directional derivative at zero along $e_o$ is $-b_o<0$.
\end{proof}

\begin{corollary}[Margin-free accelerated obstacle primitive]
\label{cor:aesp-cd-margin-free-obstacle-primitive}
In the setting of
\cref{prop:aesp-cd-speculative-envelope-doubling}, suppose on every fixed
envelope $U$ that
\begin{equation*}
 \mu I\preceq H_U\preceq LI,
 \qquad H_Uv_U\geq\mu_vv_U>0,
\end{equation*}
with certified parameters.  Accelerated projected-gradient scratch, followed
by the publication in
\cref{thm:aesp-cd-obstacle-clip-retraction}, implements the assumed
restricted obstacle primitive in
\begin{equation*}
 \widetilde O\!\left(\vol(U)\sqrt{L/\mu}\right)
\end{equation*}
work.  No active-coordinate or strict-complementarity margin is required.
For a principal RPPR envelope, $L\leq1$, $\mu\geq\alpha$, and one may take
$v_U=D_U^{1/2}\one$ with $H_Uv_U\geq\alpha v_U$, giving
$\widetilde O(\vol(U)/\sqrt\alpha)$ work.

This removes the active-set margin hypothesis from the obstacle solve in
\cref{prop:aesp-cd-speculative-envelope-doubling}.  It does not control the
final explored volume, nor does it by itself decide an exactly zero boundary
key with finite precision; the boundary-interval fallback in that proposition
remains explicit.
\end{corollary}

\begin{proof}
Projected acceleration keeps every scratch iterate feasible and reaches a
certified $\ell_\infty$ error $\varepsilon$ after
$O(\sqrt{L/\mu}\log(E_0/\varepsilon))$ matrix applications, for example by
driving the standard strongly-convex objective-gap bound below
$\mu\varepsilon^2/2$.  The retraction error amplification in
\cref{eq:aesp-cd-obstacle-clip-pay-error} changes only the logarithmic target.
The RPPR specialization uses the principal Stieltjes spectral bounds and the
same positive supersolution as
\cref{thm:aesp-cd-safe-chebyshev-face}.
\end{proof}

\begin{corollary}[Strict margins instantiate the obstacle primitive]
\label{cor:aesp-cd-margin-obstacle-primitive}
On a fixed envelope $U$, let
\begin{equation*}
 F_U(x)=\tfrac12\langle x,H_Ux\rangle-\langle b_U,x\rangle
       +I_{x\geq0},
 \qquad \mu I\preceq H_U\preceq LI,
\end{equation*}
and let $p_U$ have support $S$.  Suppose certified margins satisfy
\begin{equation*}
 \min_{i\in S}p_{U,i}\geq m_x>0,\qquad
 \max_{i\in U\setminus S}(b_U-H_Up_U)_i\leq-m_g<0.
\end{equation*}
When $S=U$, omit the dual-margin condition and interpret
$m_g/R_\infty=+\infty$ below.
Assume also that the old padded checkpoint $y_U$ is an obstacle subsolution,
$0\leq y_U\leq p_U$, with $\supp(y_U)\subseteq S$.
Assume $H_U$ is Stieltjes and the certified face $S$ has a known vector
$v_S>0$ and scalar $\mu_v>0$ such that
$H_{SS}v_S\geq\mu_vv_S$ coordinatewise, as required by
\cref{thm:aesp-cd-safe-chebyshev-face}.
Put $R_\infty=\max_i\sum_j|(H_U)_{ij}|$.  Accelerated projected-gradient
scratch to
\begin{equation*}
 \norm{a-p_U}_\infty
 \leq\frac13\min\{m_x,m_g/R_\infty\}
\end{equation*}
identifies $S$: active coordinates exceed $m_x/2$, while inactive
coordinates are below $m_x/2$ and have dual residual below $-m_g/2$.
Running
\cref{thm:aesp-cd-safe-chebyshev-face} only on this certified linear face,
from the old padded lower checkpoint, publishes a safe approximation to
$p_U$.  The combined work is
\begin{equation*}
 \widetilde O\!\left(
   \vol(U)\sqrt{L/\mu}\right),
\end{equation*}
where $m_x,m_g$ enter only logarithmically.  Hence
the exact face itself, rather than only a lower-safe approximation, is
certified under uniform strict active/dual margins.
\end{corollary}

\begin{proof}
Strongly convex accelerated projected gradient reaches the displayed distance
in
\begin{equation*}
 O\!\left(\sqrt{L/\mu}\,
 \log\!\frac{E_0}{\min\{m_x,m_g/R_\infty\}}\right)
\end{equation*}
iterations.  Every
active coordinate is then at least $2m_x/3$.  On an inactive coordinate,
$a_i\leq m_x/3$ and
\begin{equation*}
 (b_U-H_Ua)_i
 \leq-m_g+R_\infty\norm{a-p_U}_\infty\leq-2m_g/3.
\end{equation*}
Thus the exact support is certified before the signed linear-face scratch is
published.  The maximum-with-old-checkpoint retraction preserves the lower
certificate, as in \cref{thm:aesp-cd-safe-chebyshev-face}.
\end{proof}

Without strict margins, projected acceleration may leave arbitrarily small
positive inactive scratch coordinates and cannot certify the exact face at a
uniform accuracy.  The clip-or-pay publication nevertheless deletes every
unsafe coordinate without paying its entire dual residual.  Thus strict
margins remain useful for face identification, not for lower-safe obstacle
publication.

\begin{proposition}[Projected acceleration can spill outside the true support]
\label{prop:aesp-cd-accelerated-support-spill}
Even for a two-coordinate Stieltjes obstacle problem, one exact critically
tuned projected Nesterov step can activate a coordinate outside the support of
the true minimizer.  Hence
\cref{cor:aesp-cd-margin-free-obstacle-primitive} makes the published output
local, but it does not make unrestricted accelerated scratch support-local.
\end{proposition}

\begin{proof}
Take
\begin{equation*}
 H=\begin{pmatrix}601/676&-45/169\\-45/169&61/169\end{pmatrix},
 \qquad b=\begin{pmatrix}1\\-1/3\end{pmatrix}.
\end{equation*}
The spectrum is $\{1,1/4\}$ and the obstacle minimizer is
$p=(676/601,0)^\top$; its second KKT slack is $61/1803>0$.
With unit step, $x_0=0$, and critical momentum
$\beta=(1-\sqrt{1/4})/(1+\sqrt{1/4})=1/3$, the first projected-gradient
point is $x_1=(1,0)^\top<p$ on its support.  Extrapolation gives
$y_1=x_1+\beta(x_1-x_0)=(4/3,0)^\top$, and the next projected point is
\begin{equation*}
 x_2=[y_1-(Hy_1-b)]_+
 =\begin{pmatrix}194/169\\11/507\end{pmatrix}.
\end{equation*}
Thus $(x_2)_2>0=p_2$.  This stops the simplest strategy of running the full
accelerated scratch by exposing only the final support.  It is not a lower
bound against restricted envelopes, implicit scratch, or a different
localized acceleration.
\end{proof}

\begin{proposition}[Cone-preserving residual polynomials cannot accelerate]
\label{prop:aesp-cd-positive-polynomial-stop}
Let $P_k(t)=\sum_{j=0}^k a_jt^j$ be a residual polynomial with
$P_k(1)=1$.  Suppose that $P_k(T)r\geq0$ for every entrywise nonnegative
matrix $T$ of every finite dimension and every $r\geq0$.  Then
$a_j\geq0$ for every $j$, and for $0\leq t\leq1$,
\begin{equation}
 P_k(t)\geq t^k.
 \label{eq:aesp-cd-positive-polynomial-stop}
\end{equation}
Consequently, at $t=1-1/\mathcal K$, reducing the residual by a factor
$\varepsilon$ requires
$k=\Omega(\mathcal K\log(1/\varepsilon))$.  The same lower bound holds for
any product of such cone-preserving restart polynomials.  Thus a
square-root-rate inner solver must permit signed intermediate residuals; the
safe publication map in
\cref{thm:aesp-cd-safe-chebyshev-face} is not merely cosmetic.
\end{proposition}

\begin{proof}
Apply the hypothesis to the nilpotent forward shift $S$ in dimension
$k+1$ and its first basis vector.  The coordinates of
$P_k(S)e_1$ are exactly $a_0,\ldots,a_k$, so all coefficients are
nonnegative.  Since they sum to one and $t^j\geq t^k$ for $j\leq k$,
\cref{eq:aesp-cd-positive-polynomial-stop} follows.  Products preserve
nonnegative coefficients and add total degree.
\end{proof}

Thus the composite analogue of the ordinary-PPR residual-mass contraction is
valid for arbitrary signed iterates; no nonnegativity hypothesis is needed.

\begin{proposition}[All-positive batches are not radius bounded]
\label{prop:aesp-cd-all-positive-radius-stop}
There is an eight-vertex unweighted single-seed RPPR instance whose exact
support has radius one, but the canonical restricted-solve rule that admits
every currently strict-positive exterior key takes five nonempty batches.
Thus neither support radius nor BFS layers bound the number of exact batch
resolves.
\end{proposition}

\begin{proof}
Take seed $0$, $\alpha=1/200$, $\rho=23/800$, and edges
\begin{equation*}
 \{01,02,03,04,05,06,07,12,13,17,24,26,34,45,47,56,57,67\}.
\end{equation*}
In degree-unscaled coordinates use
$\widehat H=((1+\alpha)/2)D-((1-\alpha)/2)A$ and
$\widehat\ell=\alpha e_0-\alpha\rho d$.  Starting from $U_0=\{0\}$,
exact restricted solves and simultaneous admission of all positive exterior
keys give
\begin{equation*}
 \{3\},\quad\{1\},\quad\{2\},\quad\{6\},\quad\{5\},\quad\varnothing.
\end{equation*}
Every restricted solution is strictly interior.  The five positive key
margins are, in order,
\begin{equation*}
 \frac{5013}{37520000},\quad
 \frac{344981}{8088200000},\quad
 \frac{49777913}{4439456576000},\quad
 \frac{169932330613}{61418243753760000},\quad
 \frac{44816054369}{84444409556591680}.
\end{equation*}
The terminal support is $\{0,1,2,3,5,6\}$, every nonseed member is adjacent
to $0$, and the remaining keys at $4,7$ are strictly negative.  Hence
$R_\star=1$ while the batch count is five.  This finite witness stops a
radius-layer count, not every possible amortized batch analysis.
\end{proof}
