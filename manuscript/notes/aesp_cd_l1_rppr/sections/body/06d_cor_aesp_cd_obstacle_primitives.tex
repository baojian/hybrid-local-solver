\begin{theorem}[Margin-free clip-or-pay obstacle retraction]
\label{thm:aesp-cd-obstacle-clip-retraction}
Let
\begin{equation*}
 p=\argmin_{x\geq0}\left\{
   \tfrac12\langle x,Hx\rangle-\langle b,x\rangle\right\},
 \qquad
 H=H^\top\succ0,
 \qquad H_{ij}\leq0\quad(i\neq j).
\end{equation*}
Suppose a certified vector $v>0$ and scalar $\mu_v>0$ satisfy
$Hv\geq\mu_vv$ coordinatewise.  Let $\widehat p\geq0$ be any feasible
scratch approximation with
$\norm{\widehat p-p}_\infty\leq\varepsilon$.  Define
\begin{equation}
 \tau_i:=\frac{\widehat p_i}{v_i},\qquad
 R_i:=\frac{(H\widehat p-b)_i}{(Hv)_i},\qquad
 \Delta:=\max_i\bigl[\min\{\tau_i,R_i\}\bigr]_+,
 \qquad
 s:=[\widehat p-\Delta v]_+.
 \label{eq:aesp-cd-obstacle-clip-pay}
\end{equation}
Then $s$ is an obstacle subsolution and is order-safe:
\begin{equation}
 s\geq0,\qquad b_i-(Hs)_i\geq0\quad(i\in\supp(s)),
 \qquad 0\leq s\leq p.
 \label{eq:aesp-cd-obstacle-clip-pay-safe}
\end{equation}
If $y$ is any previously published obstacle subsolution with
$0\leq y\leq p$, then $z:=y\vee s$ has the same properties and
$\norm{p-z}_\infty\leq\norm{p-s}_\infty$.

Put $v_{\min}:=\min_i v_i$, $v_{\max}:=\max_i v_i$, and
$\norm{H}_\infty:=\max_i\sum_j|H_{ij}|$.  The retraction obeys the explicit
margin-free bounds
\begin{align}
 \Delta
 &\leq \frac{\varepsilon}{v_{\min}}
   \max\left\{1,\frac{\norm{H}_\infty}{\mu_v}\right\},
 \label{eq:aesp-cd-obstacle-clip-pay-delta}\\
 \norm{p-z}_\infty
 &\leq
 \left[1+\frac{v_{\max}}{v_{\min}}
   \max\left\{1,\frac{\norm{H}_\infty}{\mu_v}\right\}\right]\varepsilon.
 \label{eq:aesp-cd-obstacle-clip-pay-error}
\end{align}
Thus large inactive dual residuals do not force a large common shift: an
inactive coordinate may instead pay through its small primal ratio $\tau_i$.
The publication costs one matrix application and one scan.
\end{theorem}

\begin{proof}
Let $t=\widehat p-\Delta v$.  If $s_i>0$, then
$s_i=t_i$ and $\Delta<\tau_i$.  Since
$\Delta\geq\min\{\tau_i,R_i\}$, including when that minimum is negative,
one must have $R_i\leq\Delta$.  Moreover $s\geq t$, and increasing
coordinates other than $i$ can only decrease row $i$ of a Stieltjes matrix.
Consequently
\begin{equation*}
 (Hs)_i\leq(Ht)_i=(H\widehat p)_i-\Delta(Hv)_i\leq b_i.
\end{equation*}
This proves the subsolution condition.

For completeness, let $I=\{i:s_i>p_i\}$.  On $I$, the subsolution
condition and obstacle KKT give $[H(s-p)]_I\leq0$.  On $I^c$ one has
$s-p\leq0$, so $H_{I,I^c}(s-p)_{I^c}\geq0$.  Hence
$H_{II}(s-p)_I\leq0$.  A principal Stieltjes inverse is nonnegative,
contradicting $(s-p)_I>0$ unless $I=\varnothing$.  Thus $s\leq p$.
The coordinatewise maximum of two obstacle subsolutions is again a
subsolution: on a row selected from either input, increasing the other
coordinates only decreases that row.  This proves the assertions for $z$.

It remains to bound the shift without a complementarity margin.  If
$p_i=0$, feasibility and the error bound give
$\tau_i\leq\varepsilon/v_{\min}$.  If $p_i>0$, obstacle KKT gives
$(Hp-b)_i=0$, and therefore
\begin{equation*}
 [R_i]_+
 \leq\frac{|[H(\widehat p-p)]_i|}{\mu_vv_{\min}}
 \leq\frac{\norm{H}_\infty\varepsilon}{\mu_vv_{\min}}.
\end{equation*}
Taking the maximum proves
\cref{eq:aesp-cd-obstacle-clip-pay-delta}.  Finally
$0\leq p-z\leq p-s$ coordinatewise and
$\norm{p-s}_\infty\leq\varepsilon+\Delta v_{\max}$, which proves
\cref{eq:aesp-cd-obstacle-clip-pay-error}.
\end{proof}

\begin{lemma}[One positive load makes every subsolution rooted and connected]
\label{lem:aesp-cd-point-source-subsolution-connected}
Let the nonzero off-diagonal graph of the Stieltjes matrix $H$ be connected,
and suppose $b_o>0$ while $b_i\leq0$ for $i\neq o$.  Every nonzero obstacle
subsolution $z$ has connected support containing $o$.  In particular, both
the obstacle minimizer and every nonzero clip-or-pay publication in
\cref{thm:aesp-cd-obstacle-clip-retraction} have one rooted component.
\end{lemma}

\begin{proof}
If a connected component $C$ of $\supp(z)$ does not contain $o$, then the
subsolution inequalities on $C$ read
\begin{equation*}
 H_{CC}z_C\leq b_C\leq0;
\end{equation*}
coordinates outside $C$ make no contribution because they are zero or are
separated from $C$ in the support graph.  Principal Stieltjes inverse
positivity gives $z_C\leq0$, a contradiction.  Thus every positive component
contains $o$, and there can be at most one.  The exact minimizer is nonzero
because the directional derivative at zero along $e_o$ is $-b_o<0$.
\end{proof}

\begin{proposition}[Point-source RPPR is not a linear-PPR superlevel set]
\label{prop:aesp-cd-point-source-superlevel-stop}
Even with one positive load, the obstacle support need not be contained in
the positive support of the corresponding unconstrained shifted PPR vector.
Specifically, on the three-vertex path with source $0$,
$\alpha=1/5$, and $\rho=5/21$, the degree-unscaled Hessian, shifted load,
and unconstrained vector are
\begin{equation*}
 H=\begin{pmatrix}
  3/5&-2/5&0\\ -2/5&6/5&-2/5\\0&-2/5&3/5
 \end{pmatrix},\qquad
 b=\begin{pmatrix}16/105\\-2/21\\-1/21\end{pmatrix},\qquad
 H^{-1}b=\begin{pmatrix}8/35\\-4/105\\-11/105\end{pmatrix},
\end{equation*}
whereas the obstacle minimizer is
\begin{equation*}
 p=\begin{pmatrix}38/147\\1/147\\0\end{pmatrix}.
\end{equation*}
Thus computing ordinary point-source PPR and taking its positive
$\rho$-superlevel component cannot replace the RPPR obstacle solve or its
boundary reporter.
\end{proposition}

\begin{proof}
Direct multiplication gives the displayed unconstrained solve.  On the face
$\{0,1\}$, direct inversion gives the displayed positive vector, its two
active equations are exact, and its inactive KKT slack at vertex $2$ is
$11/245>0$.  Hence it is the unique obstacle minimizer.  Moreover the
root-only restricted solution has exterior key $2/315>0$ at vertex $1$, so
the extra obstacle coordinate is not an artifact of choosing an unreachable
face.
\end{proof}

\begin{proposition}[Ordinary PPR screens every source support only at the
classical \(\alpha\rho\) scale]
\label{prop:aesp-cd-point-source-ppr-screening}
Let \(y^0\) be ordinary PPR for any probability source $s$ in
degree-unscaled coordinates,
\begin{equation*}
 \widehat H y^0=\alpha s,
 \qquad d^\top y^0=1,
 \qquad p=\frac{1+\alpha}{2}.
\end{equation*}
Then every active RPPR coordinate satisfies
\begin{equation}
 i\in S^\star(\rho)
 \quad\Longrightarrow\quad
 y_i^0>\frac{\alpha}{p}\rho,
 \qquad
 \vol\!\left\{i:y_i^0>\frac{\alpha}{p}\rho\right\}
 <\frac{p}{\alpha\rho}.
 \label{eq:aesp-cd-point-source-ppr-screening}
\end{equation}
Thus exact PPR supplies a graph-universal support-containing envelope, but
its guaranteed volume is on the classical \(1/(\alpha\rho)\) scale, not the
desired \(1/(\rho\sqrt\alpha)\) scale.

The factor \(\alpha/p\) cannot be replaced by any graph-universal
\(\omega(\alpha)\) factor.  For integers \(1\leq M\leq L\), let
\(G_{M,L}\) contain an edge \(ou\), a clique \(C\) of size \(L\), and edges
from \(u\) to \(M\) vertices of \(C\).  Use the point source \(o\), and put
\begin{equation*}
 \tau_M:=\frac{c}{p(M+1)+c},
 \qquad c=\frac{1-\alpha}{2}.
\end{equation*}
For every fixed \(M,L\), all \(\rho<\tau_M\) sufficiently close to
\(\tau_M\) have exact RPPR support \(\{o,u\}\).  Nevertheless
\begin{equation}
 \lim_{M\to\infty}\lim_{L\to\infty}
 \frac{y_{u}^{0,(M,L)}}{\tau_M}
 =\frac{\alpha}{p}.
 \label{eq:aesp-cd-point-source-ppr-screening-sharp}
\end{equation}
In particular, a putative screening rule
\(S^\star(\rho)\subseteq\{i:y_i^0\geq C\rho\sqrt\alpha\}\), with a positive
graph-independent \(C\), is false as \(\alpha\downarrow0\).  This is a STOP
for discovering the accelerated-size RPPR envelope by thresholding ordinary
PPR alone; it is not a lower bound against a dynamic obstacle reporter that
uses the changing-face residuals.
\end{proposition}

\begin{proof}
Let \(S=S^\star(\rho)\), and define
\begin{equation*}
 a_S:=\widehat H_{SS}^{-1}\alpha s_S,
 \qquad
 t_S:=\widehat H_{SS}^{-1}\alpha d_S.
\end{equation*}
The active solution is \(y_S^\star=a_S-\rho t_S>0\).  Because
\(y^0_{\bar S}\geq0\) and \(\widehat H_{S\bar S}\leq0\),
\begin{equation*}
 \widehat H_{SS}y_S^0
 =\alpha s_S-\widehat H_{S\bar S}y_{\bar S}^0
 \geq\alpha s_S,
\end{equation*}
so principal inverse positivity gives \(y_S^0\geq a_S\).  Also
\begin{equation*}
 (t_S)_i
 =\alpha\sum_{j\in S}
   (\widehat H_{SS}^{-1})_{ij}d_j
 \geq\alpha(\widehat H_{SS}^{-1})_{ii}d_i
 \geq\frac{\alpha d_i}{(\widehat H)_{ii}}
 =\frac{\alpha}{p}.
\end{equation*}
Here the middle inequality is the standard Schur-complement bound
\((B^{-1})_{ii}\geq1/B_{ii}\) for \(B\succ0\).
Therefore \(y_i^0\geq(a_S)_i>\rho(t_S)_i\geq\alpha\rho/p\).
Summing \(d_i y_i^0\) over the superlevel set proves
\cref{eq:aesp-cd-point-source-ppr-screening}.

For the sharp family, the root-only solution has
\(y_o=\alpha(1-\rho)/p\), and the exterior key at \(u\) is
\begin{equation*}
 -\alpha\rho(M+1)+cy_o.
\end{equation*}
It vanishes exactly at \(\rho=\tau_M\).  At that parameter every clique key
is strictly negative.  Hence continuity and Stieltjes inverse positivity
show that, immediately below \(\tau_M\), \(u\) is positive while all clique
vertices remain inactive.

For ordinary PPR, eliminate the clique in the four equitable classes
\(\{o\},\{u\},N_C(u)\), and \(C\setminus N_C(u)\).  Alternatively, note that
the clique principal block is at least \(\alpha(L-1)I\).  Its load from \(u\)
has norm \(c y_u^0\sqrt M\), so for fixed \(M\) its feedback
\(c\sum_{v\in N_C(u)}y_v^0\) tends to zero as \(L\to\infty\).  The limiting
two equations are
\begin{equation*}
 py_o^0-cy_u^0=\alpha,
 \qquad
 -cy_o^0+p(M+1)y_u^0=0.
\end{equation*}
Consequently
\begin{equation*}
 \lim_{L\to\infty}y_u^{0,(M,L)}
 =\frac{\alpha c}{p^2(M+1)-c^2},
\end{equation*}
and division by \(\tau_M\), followed by \(M\to\infty\), proves
\cref{eq:aesp-cd-point-source-ppr-screening-sharp}.
\end{proof}

\begin{proposition}[Point-source homotopy breakpoints mix but can reorder]
\label{prop:aesp-cd-point-source-homotopy-reorder}
Fix a face $S\ni o$ and write its point-source parametric solve as
\begin{equation*}
 x_S(\rho)=a_S-\rho h_S,\qquad
 a_S=H_{SS}^{-1}\alpha e_o,\qquad
 h_S=H_{SS}^{-1}\alpha d_S.
\end{equation*}
For an exterior vertex $v$, put
\begin{equation*}
 A_v=-H_{vS}a_S\geq0,\qquad
 B_v=\alpha d_v-H_{vS}h_S>0,\qquad
 g_v(\rho)=A_v-\rho B_v,\qquad
 \tau_v=A_v/B_v.
\end{equation*}
Suppose one exterior vertex $w$ is added.  Then for every remaining exterior
vertex $v$ there is a scalar $\gamma_v\geq0$ such that
\begin{equation}
 (A'_v,B'_v)=(A_v,B_v)+\gamma_v(A_w,B_w),\qquad
 \tau'_v=
 \frac{B_v\tau_v+\gamma_vB_w\tau_w}{B_v+\gamma_vB_w}.
 \label{eq:aesp-cd-point-source-homotopy-mix}
\end{equation}
Thus if $w$ is the current largest breakpoint, every surviving breakpoint
moves monotonically toward, but never past, $\tau_w$.

This one-dimensional structure does not make a stale priority queue exact:
breakpoint order can reverse.  On the six-vertex graph with
$\alpha=2/7$ and edges
\begin{equation*}
 01,03,04,12,13,23,25,34,45,
\end{equation*}
the exact decreasing-$\rho$ point-source trace starts
$\{0\}\to\{0,1,4\}\to\{0,1,3,4\}$.  Immediately before admitting $3$,
\begin{equation*}
 \tau_5=\frac{25}{1838}>\frac{25}{2517}=\tau_2,
\end{equation*}
whereas immediately afterwards
\begin{equation*}
 \tau'_2=\frac{975}{53696}>
 \frac{1025}{64318}=\tau'_5.
\end{equation*}
Hence point-source homotopy compresses each update to a nonnegative
rank-one mixture, but a reporter must still maintain a dynamic maximum under
nonuniform mixtures rather than merely lazily validating one fixed heap
order.
\end{proposition}

\begin{proof}
Let
\begin{equation*}
 q=-H_{SS}^{-1}H_{Sw}\geq0,\qquad
 \sigma=H_{ww}-H_{wS}H_{SS}^{-1}H_{Sw}>0.
\end{equation*}
Block elimination on the enlarged face gives
\begin{equation*}
 a'_w=A_w/\sigma,\quad a'_S=a_S+qa'_w,\qquad
 h'_w=B_w/\sigma,\quad h'_S=h_S+qh'_w.
\end{equation*}
Therefore
\cref{eq:aesp-cd-point-source-homotopy-mix} holds with
\begin{equation*}
 \gamma_v=\frac{-H_{vS}q-H_{vw}}{\sigma}\geq0.
\end{equation*}
The second identity in
\cref{eq:aesp-cd-point-source-homotopy-mix} is then a weighted average.
Direct rational elimination on the displayed graph gives first the tied
breakpoints $\tau_1=\tau_4=5/96$, then the unique winner
$\tau_3=185/4231$, and the four before/after values displayed above.
\end{proof}

\begin{lemma}[Canonical ground-state normalization of a connected principal face]
\label{lem:aesp-cd-point-source-ground-state-normalization}
Let \(0<\alpha<1\), let \(A\) be a nonempty connected vertex set, and in degree-unscaled
coordinates, put
\begin{equation*}
 H_A=\frac{1+\alpha}{2}D_A-\frac{1-\alpha}{2}A_A,
 \qquad
 h_A:=H_A^{-1}\alpha d_A.
\end{equation*}
Here \(D_A\) retains the ambient graph degrees and \(d_A=D_A\one\).
Then \(0<h_A\leq\one\) coordinatewise.  Define the diagonal mass
\begin{equation}
 W_A:=\operatorname{diag}\!\left(\frac{d_i}{h_{A,i}}:i\in A\right).
 \label{eq:aesp-cd-point-source-ground-mass}
\end{equation}
The following identities and Loewner bounds are exact:
\begin{equation}
 H_Ah_A=\alpha W_Ah_A,\qquad
 \alpha W_A\preceq H_A\preceq W_A.
 \label{eq:aesp-cd-point-source-ground-sandwich}
\end{equation}
Consequently
\(W_A^{-1/2}H_AW_A^{-1/2}\) is an irreducible SPD Stieltjes
matrix with spectrum in \([\alpha,1]\) and known positive ground eigenpair
\begin{equation*}
 \left(\alpha,\;W_A^{1/2}h_A\right).
\end{equation*}
For every shift \(\kappa>0\), the resolvent
\begin{equation*}
 M_A^{(h)}:=\kappa(H_A+\kappa W_A)^{-1}W_A
\end{equation*}
therefore has known Perron pair
\begin{equation*}
 M_A^{(h)}h_A=\frac{\kappa}{\kappa+\alpha}h_A
\end{equation*}
and all its generalized modes lie in
\([\kappa/(\kappa+1),\,\kappa/(\kappa+\alpha)]\).
Equivalently, the shifted system has \(W_A\)-condition number at most
\begin{equation*}
 \frac{1+\kappa}{\alpha+\kappa};
\end{equation*}
for \(\kappa=1\) this is below two, uniformly over the proper face.

On a point-source \(\rho\)-homotopy face, \(h_A\) is exactly the slope
response already maintained by
\cref{thm:aesp-cd-point-source-ratio-pivot}; its elimination record recovers
the normalization with no larger asymptotic exact-field work.  Alternatively
it costs one certified fixed-face solve.

This removes the need to estimate the lowest proper-face eigenpair.  It does
not by itself extend the full-face common-cap master: that proof uses the
specific cap direction \(\one\) and the original \(D\)-geometry.  What the
lemma supplies unconditionally is the exact ground data for a separately
proved \(W_A\)-geometric, correction-free Perron tail; an eigengap or an
observable alignment certificate remains an additional hypothesis.
\end{lemma}

\begin{proof}
The inverse of the irreducible Stieltjes matrix \(H_A\) is strictly positive,
so \(h_A>0\).  If \(b_A\) is the vector of ambient edges leaving \(A\) and
\(c=(1-\alpha)/2\), then
\begin{equation*}
 H_A\one=\alpha d_A+c b_A.
\end{equation*}
Thus \(H_A(\one-h_A)=cb_A\geq0\), and inverse positivity gives
\(h_A\leq\one\).  The definition of \(W_A\) now gives
\(W_Ah_A=d_A\) and hence the first identity in
\cref{eq:aesp-cd-point-source-ground-sandwich}.

On the full graph,
\(H=\alpha D+c(D-A)\preceq D\), because the normalized Laplacian has
spectrum in \([0,2]\).  Principal restriction preserves the inequality, and
\(h_A\leq\one\) gives \(D_A\preceq W_A\).  Hence \(H_A\preceq W_A\).
The symmetric matrix \(W_A^{-1/2}H_AW_A^{-1/2}\) is irreducible Stieltjes
and has the strictly positive eigenvector \(W_A^{1/2}h_A\) at eigenvalue
\(\alpha\).  Perron--Frobenius applied to a positive shifted inverse says
that this is its simple smallest eigenvalue.  This proves the lower Loewner
bound and the spectral sandwich.  Spectral mapping gives the resolvent
claim.
\end{proof}

\begin{theorem}[Exact point-source ratio-pivot support homotopy]
\label{thm:aesp-cd-point-source-ratio-pivot}
For the point-source RPPR obstacle problem at a target \(\rho>0\), the exact
support can be found by a monotone Schur-complement elimination that admits
each support vertex once.  More precisely, after the root has entered, keep
the exterior Schur complement
\begin{equation*}
 K^S
 =H_{\bar S\bar S}-H_{\bar SS}H_{SS}^{-1}H_{S\bar S}
\end{equation*}
and the breakpoint pairs \((A_v,B_v)\) from
\cref{prop:aesp-cd-point-source-homotopy-reorder}.  Repeatedly choose a
largest ratio \(\tau_w=A_w/B_w\).  If \(\tau_w\leq\rho\), stop.  Otherwise
admit \(w\), and pivot
\begin{align}
 A'_v&=A_v-\frac{K^S_{vw}}{K^S_{ww}}A_w,&
 B'_v&=B_v-\frac{K^S_{vw}}{K^S_{ww}}B_w,
 \label{eq:aesp-cd-ratio-pivot-pair}\\
 K^{S\cup\{w\}}_{uv}
 &=K^S_{uv}-\frac{K^S_{uw}K^S_{wv}}{K^S_{ww}}.
 \label{eq:aesp-cd-ratio-pivot-schur}
\end{align}
Exact ties may be pivoted as one batch or in a fixed symbolic order.
At termination \(S=\supp(x_\rho^\star)\), and
\(x_{\rho,S}^\star=a_S-\rho h_S>0\).

This is a local explicit-fill algorithm.  Scan the adjacency row of a vertex
only when it is pivoted; point-source connectedness ensures that an unseen
vertex with no edge to \(S\) cannot enter first.  Let \(\delta_w\) be the
number of current nonzero Schur neighbors of \(w\), including newly exposed
original neighbors, just before its pivot.  With a heap for the ratios and
sparse explicit fill, discovery, the elimination record, and terminal
back-substitution cost
\begin{equation}
 \widetilde O\!\left(
   \vol(S)+\sum_{w\in S}(1+\delta_w)^2
 \right)
 \label{eq:aesp-cd-ratio-pivot-work}
\end{equation}
exact field operations.  In particular, homotopy elimination width
\(\max_w\delta_w\leq\omega\) gives
\(\widetilde O(\omega^2\vol(S))\) discovery.  Once the face is certified,
\cref{cor:aesp-cd-final-face-chebyshev} adds
\(\widetilde O(\vol(S)/\sqrt\alpha)\) terminal work.  Thus bounded-width
point-source traces satisfy the desired product-scale bound.
\end{theorem}

\begin{proof}
For \(\rho\geq1/d_o\), every load coordinate is nonpositive and the obstacle
solution is zero.  Just below \(1/d_o\), the root is the first active
coordinate.  On any later face \(S\), the active solution is
\(x_S(\rho)=a_S-\rho h_S\), and \(h_S\geq0\).  If \(w\) is the largest
breakpoint, block elimination gives
\begin{equation*}
 x'_w(\rho)
 =\frac{A_w-\rho B_w}{K^S_{ww}}
 =\frac{B_w(\tau_w-\rho)}{K^S_{ww}},
 \qquad
 x'_S(\rho)=x_S(\rho)-H_{SS}^{-1}H_{Sw}x'_w(\rho).
\end{equation*}
Therefore \(w\) becomes positive immediately below \(\tau_w\), and every
old active coordinate is nondecreasing.  The Schur complement of a
Stieltjes matrix is Stieltjes, so
\(-K^S_{vw}/K^S_{ww}\geq0\).
\Cref{eq:aesp-cd-ratio-pivot-pair,eq:aesp-cd-ratio-pivot-schur} are precisely
the block-elimination identities.  At the first event below the target,
every exterior key is
\begin{equation*}
 g_v(\rho)=B_v(\tau_v-\rho)\leq0,
\end{equation*}
while all admitted coordinates are positive.  These are the global obstacle
KKT conditions, proving correctness.

For locality, a vertex not yet adjacent to \(S\) has \(A_v=0\), hence cannot
beat a positive breakpoint.  Scanning the selected row reveals every new
original frontier incidence before its pivot.  Updating the Schur clique
costs \(O((1+\delta_w)^2)\), pair and heap updates cost only a soft-linear
amount in \(\delta_w\), and each selected original row costs \(d_w\).
Summing gives
\cref{eq:aesp-cd-ratio-pivot-work}; the stored elimination multipliers give a
back-substitution with no larger charge.
\end{proof}

\begin{corollary}[Fixed-target scalar residual pivoting]
\label{cor:aesp-cd-point-source-fixed-target-pivot}
At one fixed target \(\rho<1/d_o\), the two breakpoint vectors are not
needed.  On a current face define the exterior restricted residual
\begin{equation*}
 g^S_{\bar S}
 :=c_{\rho,\bar S}-H_{\bar SS}H_{SS}^{-1}c_{\rho,S}
 =A_{\bar S}-\rho B_{\bar S}.
\end{equation*}
Starting with \(S=\{o\}\), choose \emph{any} row \(w\notin S\) with
\(g^S_w>0\), admit it, and update
\begin{align*}
 g^{S\cup\{w\}}_v
 &=g^S_v-\frac{K^S_{vw}}{K^S_{ww}}g^S_w,\\
 K^{S\cup\{w\}}_{uv}
 &=K^S_{uv}-\frac{K^S_{uw}K^S_{wv}}{K^S_{ww}}.
\end{align*}
When every remaining residual is nonpositive, the restricted solution is the
exact obstacle solution.  Every admitted coordinate stays strictly positive,
so each final-support row is admitted at most once.  In the point-source
problem, any unseen nonroot row with no original edge to \(S\) has
\(g^S_v=c_{\rho,v}=-\alpha\rho d_v<0\).  Hence the same row-on-first-pivot
discovery and explicit-fill work bound
\cref{eq:aesp-cd-ratio-pivot-work} applies without a global initial scan.

More generally, suppose a finite reporter gives certified intervals
\begin{equation*}
 g_v^-\leq g^S_v\leq g_v^+.
\end{equation*}
It may safely pivot any row with \(g_v^->0\).  If instead
\begin{equation}
 max_{v\notin S}\frac{[g_v^+]_+}{d_v}
 \leq\alpha\epskkt,
 \label{eq:aesp-cd-fixed-target-residual-stop}
\end{equation}
then the current restricted solution already meets the note's finite KKT
target.  Otherwise it refines only the gray certificate.  Thus a fixed-target
implementation needs one-sided row-residual accuracy
\(O(\alpha\epskkt d_v)\), not exact support, exact breakpoint order, or even
the separate vectors \(A\) and \(B\).
\end{corollary}

\begin{proof}
Let \(x_S=H_{SS}^{-1}c_{\rho,S}\).  If \(g^S_w>0\), block elimination gives
\begin{equation*}
 x'_w=\frac{g^S_w}{K^S_{ww}}>0,
 \qquad
 x'_S=x_S-H_{SS}^{-1}H_{Sw}x'_w\geq x_S>0,
\end{equation*}
because every principal inverse is nonnegative and \(H_{Sw}\leq0\).  The
displayed residual and Schur recurrences are the same block-elimination
identity, and their multipliers are nonnegative.  Induction preserves strict
interiority.  At termination, active equations and exterior nonpositive
residuals are exactly the global obstacle KKT conditions.  The point-source
load proves the locality claim.  Finally, active residuals vanish, so
\cref{eq:aesp-cd-fixed-target-residual-stop} is precisely the normalized KKT
diagnostic bound.
\end{proof}

\begin{lemma}[The support-safe frontier universe is output-linear]
\label{lem:aesp-cd-point-source-frontier-universe}
Let \(S_0\subset S_1\subset\cdots\subset S^\star\) be any nested
support-safe trace for a source distribution with $T=\supp(s)$, and let
\begin{equation*}
 \partial S:=\{v\notin S:\ A_{vS}\one>0\}.
\end{equation*}
Every original-graph row that can ever occur on a frontier belongs to
\(S^\star\cup\partial S^\star\).  After including the source rows that must
be candidate records before they are adjacent to the current support, the
whole persistent candidate universe is contained in
\(T\cup S^\star\cup\partial S^\star\), and, on a graph without isolated
vertices,
\begin{equation}
 \left|T\cup S^\star\cup\partial S^\star\right|
 \leq |T|+|S^\star|+|E(S^\star,V\setminus S^\star)|
 \leq |T|+2\vol(S^\star).
 \label{eq:aesp-cd-point-source-frontier-universe}
\end{equation}
Moreover all incidences from the trace to this universe are revealed by
scanning each admitted adjacency row once, at total work
\(\vol(S^\star)\).  An exterior row need not have its own adjacency list
scanned: it suffices to keep its identifier, degree, and accumulated
contributions from already admitted neighbors.

Thus sparse-source discovery has only an output-linear number of persistent
boundary records.  Its remaining difficulty is not candidate enumeration,
but updating their accumulated keys when old active coordinates increase.
This statement uses the standard local-access convention that a named
vertex degree is available in \(O(1)\) work; otherwise even deciding whether
to retain a high-degree exterior row can cost its ambient degree.
\end{lemma}

\begin{proof}
For every \(t\), \(S_t\subseteq S^\star\), so every vertex adjacent to
\(S_t\) lies in \(S^\star\cup\partial S^\star\).  Map each vertex of
\(\partial S^\star\) to one incident cut edge.  The map is injective because
a cut edge has only one endpoint outside \(S^\star\), proving the first
inequality in \cref{eq:aesp-cd-point-source-frontier-universe}.  Also
\(|S^\star|\leq\vol(S^\star)\) and the number of cut edges is at most
\(\vol(S^\star)\).  Finally, every incidence is first seen while scanning
the row of its endpoint in \(S^\star\), and those row lengths sum to
\(\vol(S^\star)\).
\end{proof}

\begin{proposition}[Point-source RPPR is a dissipative divisible sandpile]
\label{prop:aesp-cd-point-source-dissipative-sandpile}
In degree-unscaled coordinates \(y=D^{-1/2}x\), put
\begin{equation*}
 \widehat H=D^{1/2}QD^{1/2}=pD-cA,
 \qquad
 p=\frac{1+\alpha}{2},
 \qquad
 c=\frac{1-\alpha}{2},
 \qquad
 \ell_\rho=\alpha e_o-\alpha\rho d.
\end{equation*}
In the obstacle/odometer language reviewed by
\href{https://arxiv.org/abs/1903.06263}{Ruszel}, the point-source RPPR vector
is the odometer of the following killed continuous toppling process.  At a
state \(y\geq0\), write
\(r=\ell_\rho-\widehat Hy\).  A row \(i\) with \(r_i>0\) may topple by
\begin{equation}
 y_i\leftarrow y_i+\delta,
 \qquad
 0<\delta\leq\frac{r_i}{p d_i}.
 \label{eq:aesp-cd-point-source-toppling}
\end{equation}
The full choice sets \(r_i\) to zero, sends \(c\delta\) to every neighbor,
and dissipates the fraction
\begin{equation*}
 1-\frac{c}{p}=\frac{2\alpha}{1+\alpha}
\end{equation*}
of the removed residual mass.  Every legal toppling sequence is
coordinatewise bounded by \(y^\star=D^{-1/2}x^\star\), and every fair
sequence of full topplings converges monotonically to \(y^\star\), where
fair means that no row whose residual stays positive is ignored forever.

Equivalently, \(y^\star\) obeys the least-action identity
\begin{equation}
 y^\star
 =\min\{z\geq0:\widehat Hz\geq\ell_\rho\}
 \quad\text{in the coordinatewise order.}
 \label{eq:aesp-cd-point-source-least-action}
\end{equation}
Thus the point-source reduction really does supply an Abelian rooted growth
process.  It does not by itself accelerate it: if full local topplings are
performed only while \(r_i/d_i>\alpha\epskkt\), their charged adjacency work
is at most
\begin{equation}
 \frac{p}{\alpha\epskkt},
 \label{eq:aesp-cd-point-source-toppling-work}
\end{equation}
which is the classical \(1/(\alpha\epskkt)\) rather than the desired
\(1/(\sqrt\alpha\epskkt)\) scale.  The fixed-target Schur pivot in
\cref{cor:aesp-cd-point-source-fixed-target-pivot} is the corresponding
exact block-toppling operation.
\end{proposition}

\begin{proof}
The obstacle KKT conditions say precisely that \(y^\star\geq0\),
\(\widehat Hy^\star\geq\ell_\rho\), with equality on
\(\supp(y^\star)\).  Let \(z\geq0\) be any stabilizer and put
\(T=\{i:y_i^\star>z_i\}\).  If \(T\neq\varnothing\), then every row in
\(T\) belongs to \(\supp(y^\star)\), so
\begin{equation*}
 [\widehat H(y^\star-z)]_T\leq0.
\end{equation*}
On \(T^c\), \(y^\star-z\leq0\), while
\(\widehat H_{T,T^c}\leq0\).  Hence
\(\widehat H_{TT}(y^\star-z)_T\leq0\).  Principal Stieltjes inverse
positivity contradicts \((y^\star-z)_T>0\), proving
\cref{eq:aesp-cd-point-source-least-action}.

Suppose inductively that a legal state \(y\leq y^\star\) topples row \(i\).
The safety of positive-residual admission gives \(y_i^\star>0\), and thus
\begin{equation*}
 r_i=[\widehat H(y^\star-y)]_i
 \leq p d_i(y_i^\star-y_i),
\end{equation*}
because the off-diagonal terms are nonpositive.  Therefore
\cref{eq:aesp-cd-point-source-toppling} preserves \(y\leq y^\star\).
Every legal sequence is monotone and bounded.  Under fairness, its limit can
have no positive residual row: otherwise a later full toppling has an amount
bounded away from zero.  The limit is therefore a stabilizer, and least
action identifies it with \(y^\star\).

For the work statement, put \(\mathcal R=\sum_i[r_i]_+\).  A full toppling
removes \(r_i=p d_i\delta\) at \(i\), while its neighbors can gain at most
\(c d_i\delta\) in positive residual.  Hence
\begin{equation*}
 \mathcal R_{\rm new}-\mathcal R_{\rm old}
 \leq-\frac{\alpha}{p}r_i.
\end{equation*}
Initially \(\mathcal R\leq\alpha\).  While
\(r_i>\alpha\epskkt d_i\), one unit of charged work \(d_i\) is therefore
paid by more than
\((\alpha^2\epskkt/p)d_i\) of potential decrease.  Summing proves
\cref{eq:aesp-cd-point-source-toppling-work}.
\end{proof}

\begin{lemma}[Point-source positive-residual mass is nonincreasing]
\label{lem:aesp-cd-point-source-residual-mass}
Along any exact fixed-target pivot trace from
\cref{cor:aesp-cd-point-source-fixed-target-pivot}, put
\begin{equation*}
 \mathcal P_S:=\sum_{v\notin S}[g^S_v]_+.
\end{equation*}
Then \(\mathcal P_S\) never increases and, immediately after the root
enters,
\begin{equation}
 \mathcal P_{\{o\}}\leq\frac{1-\alpha}{1+\alpha}\alpha
 \leq\alpha.
 \label{eq:aesp-cd-point-source-residual-mass}
\end{equation}
More sharply, if \(w\) is the next pivot, then
\begin{equation}
 \mathcal P_{S\cup\{w\}}
 \leq\mathcal P_S-\frac{\alpha}{p}g^S_w,
 \qquad p=\frac{1+\alpha}{2}.
 \label{eq:aesp-cd-point-source-residual-mass-drop}
\end{equation}
Consequently every exact fixed-target pivot order satisfies the global
injection budget
\begin{equation}
 \sum_{\text{pivots }w}g^S_w\leq\frac{1-\alpha}{2}.
 \label{eq:aesp-cd-point-source-pivot-injection}
\end{equation}
Consequently, at every time the number of exterior rows violating the finite
KKT threshold
\begin{equation*}
 g^S_v/d_v>\alpha\epskkt
\end{equation*}
is at most \(1/\epskkt\).  This is a simultaneous sparsity bound for
finite-significant rows.  It does not by itself bound how often a compressed
Schur reporter must update or rediscover those rows over the full trace.
\end{lemma}

\begin{proof}
The Schur complement \(K^S\) is SDDM.  Its row sums satisfy the stronger
point-source-independent killing bound
\begin{equation*}
 K^S\one\geq\alpha d_{\bar S}.
\end{equation*}
Indeed, from \(H\one=\alpha d\) and block elimination,
\begin{align*}
 K^S\one_{\bar S}
 &=H_{\bar S\bar S}\one_{\bar S}
   -H_{\bar SS}H_{SS}^{-1}H_{S\bar S}\one_{\bar S}\\
 &=\alpha d_{\bar S}
   -H_{\bar SS}H_{SS}^{-1}\alpha d_S
 \geq\alpha d_{\bar S}.
\end{align*}
Moreover \(K^S_{ww}\leq H_{ww}=p d_w\).  Therefore, for a pivot \(w\),
\begin{equation*}
 \gamma_v:=-K^S_{vw}/K^S_{ww}\geq0,
 \qquad
 \sum_{v\notin S\cup\{w\}}\gamma_v
 =1-\frac{(K^S\one)_w}{K^S_{ww}}
 \leq1-\frac{\alpha}{p}.
\end{equation*}
Since \(g'_v=g_v+\gamma_vg_w\) and \(g_w>0\),
\begin{equation*}
 \sum_{v\ne w}[g'_v]_+
 \leq\sum_{v\ne w}[g_v]_++g_w\sum_{v\ne w}\gamma_v
 \leq\sum_v[g_v]_+-\frac{\alpha}{p}g_w.
\end{equation*}
At the root-only face, with
\(p=(1+\alpha)/2\), \(c=(1-\alpha)/2\),
\begin{equation*}
 x_o=\frac{\alpha(1-\rho d_o)}{p d_o},
 \qquad
 g_v=-\alpha\rho d_v+cA_{vo}x_o.
\end{equation*}
Dropping the negative loads and summing the \(d_o\) root incidences gives
\begin{equation*}
 \mathcal P_{\{o\}}
 \leq c d_ox_o
 =\frac{c}{p}\alpha(1-\rho d_o)
 \leq\frac{1-\alpha}{1+\alpha}\alpha.
\end{equation*}
Summing the sharpened decrease and substituting this initial mass gives
\begin{equation*}
 \sum_w g^S_w
 \leq\frac{p}{\alpha}\mathcal P_{\{o\}}
 \leq\frac{1-\alpha}{2},
\end{equation*}
which proves \cref{eq:aesp-cd-point-source-pivot-injection}.
Every finite-significant row contributes more than
\(\alpha\epskkt d_v\geq\alpha\epskkt\) to this mass, proving the count.
\end{proof}

\begin{proposition}[A support pivot injects a killed hitting-probability column]
\label{prop:aesp-cd-point-source-hitting-column}
Work in the degree-unscaled coordinates of
\cref{prop:aesp-cd-point-source-dissipative-sandpile}.  Let \(S\) be a
current exact face, let \(w\notin S\) have \(g_w^S>0\), and put
\begin{equation*}
 \Delta_w:=\frac{g_w^S}{K^S_{ww}},
 \qquad
 h^{S,w}:=-\widehat H_{SS}^{-1}\widehat H_{Sw}.
\end{equation*}
The exact restricted solution after admitting \(w\) is
\begin{equation}
 y^{S\cup\{w\}}_w=\Delta_w,
 \qquad
 y^{S\cup\{w\}}_S=y^S+\Delta_w h^{S,w}.
 \label{eq:aesp-cd-point-source-hitting-update}
\end{equation}
For the killed transition matrix
\(\mathsf P=(c/p)D^{-1}A\), the response column is exactly
\begin{equation}
 h^{S,w}=(I-\mathsf P_{SS})^{-1}\mathsf P_{Sw}.
 \label{eq:aesp-cd-point-source-hitting-column}
\end{equation}
Hence \(h^{S,w}_i\) is the probability that a walk started at \(i\in S\)
hits \(w\) before it is killed or first visits any other vertex of
\(V\setminus S\).  In particular \(0\leq h^{S,w}\leq\one\), and the column
obeys the exact killed-flux identity
\begin{equation}
 \alpha\sum_{i\in S}d_i h^{S,w}_i
 +c\sum_{\substack{i\in S,\ v\notin S\\ i\sim v}}h^{S,w}_i
 =c\,|E(S,\{w\})|.
 \label{eq:aesp-cd-point-source-hitting-flux}
\end{equation}
The time-reversed occupation vector
\begin{equation}
 q^{S,w}:=\mathsf P_{wS}(I-\mathsf P_{SS})^{-1}
 \quad\text{satisfies}\quad
 q^{S,w}_i=\frac{d_i}{d_w}h^{S,w}_i,
 \qquad
 \sum_{i\in S}q^{S,w}_i\leq\frac c\alpha.
 \label{eq:aesp-cd-point-source-hitting-occupation}
\end{equation}
Thus \(q^{S,w}_i\) is the expected number of visits to \(i\) during the
killed excursion that starts at \(w\), first steps into \(S\), and stops on
killing or its next exterior visit.

Consequently the finite-level reporter of
\cref{cor:aesp-cd-finite-level-reporter} reduces to the following precise
dynamic primitive.  For each active \(i\), retain its slack to the next
geometric level.  On a pivot \((w,\Delta_w)\), report exactly the rows with
\begin{equation}
 \Delta_w h^{S,w}_i>\operatorname{slack}_i,
 \label{eq:aesp-cd-point-source-hitting-event}
\end{equation}
advance their levels, and update the incident boundary accumulators from
\cref{lem:aesp-cd-point-source-frontier-universe}.  Exhaustive event reports
give the finite KKT certificate without a separation promise and use only a
support-linear number of boundary records.  What remains open on a general
graph is an online, compressed implementation of
\cref{eq:aesp-cd-point-source-hitting-event} at total
\(\widetilde O(\vol(S^\star)/\sqrt\alpha)\) work.  Ordinary walk simulation
does not supply that rate by itself: the geometric killing horizon has mean
\(\Theta(1/\alpha)\), and cone-preserving walk polynomials are subject to
\cref{prop:aesp-cd-positive-polynomial-stop}.
\end{proposition}

\begin{proof}
The block equations for the enlarged restricted solve give
\cref{eq:aesp-cd-point-source-hitting-update}.  Since
\begin{equation*}
 \widehat H_{SS}=pD_S(I-\mathsf P_{SS}),
 \qquad
 -\widehat H_{Sw}=pD_S\mathsf P_{Sw},
\end{equation*}
they also give \cref{eq:aesp-cd-point-source-hitting-column}.  Expanding the
Neumann series enumerates paths that stay in \(S\) until their final step to
\(w\); killing or a different exterior visit ends the trial.  This proves
the probability interpretation and \(0\leq h\leq\one\).

Finally use \(\widehat H=\alpha D+cL\), sum
\(\widehat H_{SS}h=cA_{Sw}\) over \(S\), and cancel every internal edge.
Each cut edge contributes \(c h_i\), while the right-hand side counts the
edges from \(S\) to \(w\).  This is
\cref{eq:aesp-cd-point-source-hitting-flux}.  Reversibility gives
\begin{equation*}
 d_i[(I-\mathsf P_{SS})^{-1}]_{ij}
 =d_j[(I-\mathsf P_{SS})^{-1}]_{ji},
 \qquad
 d_w\mathsf P_{wj}=d_j\mathsf P_{jw}.
\end{equation*}
Substitution proves the coordinate identity in
\cref{eq:aesp-cd-point-source-hitting-occupation}; its mass bound follows
from \cref{eq:aesp-cd-point-source-hitting-flux}.  The event-reduction
statement then follows directly from
\cref{eq:aesp-cd-point-source-hitting-update} and the exhaustive finite-level
certificate.
\end{proof}

\begin{lemma}[A residual certificate gives simultaneous response intervals]
\label{lem:aesp-cd-point-source-response-residual-certificate}
In the setting of
\cref{prop:aesp-cd-point-source-hitting-column}, let
\(\widetilde h\in\mathbb R^S\) approximate \(h^{S,w}\), and define its
degree-unscaled response residual
\begin{equation*}
 r:=cA_{Sw}-\widehat H_{SS}\widetilde h.
\end{equation*}
Then every old coordinate satisfies
\begin{equation}
 \left|h_i^{S,w}-\widetilde h_i\right|
 \leq\frac{\norm{D_S^{-1/2}r}_2}{\alpha\sqrt{d_i}},
 \qquad i\in S.
 \label{eq:aesp-cd-point-source-response-residual-certificate}
\end{equation}
Consequently, for a pivot amount \(\Delta_w>0\), the computable condition
\begin{equation}
 \Delta_w\norm{D_S^{-1/2}r}_2\leq\alpha\eta
 \label{eq:aesp-cd-point-source-response-residual-target}
\end{equation}
puts every exact increment in the simultaneous interval
\begin{equation*}
 \Delta_wh_i^{S,w}
 \in\left[
  \Delta_w\widetilde h_i-\frac{\eta}{\sqrt{d_i}},
  \Delta_w\widetilde h_i+\frac{\eta}{\sqrt{d_i}}
 \right].
\end{equation*}
If successive pivot solves use targets \(\eta_t\) with
\(\sum_t\eta_t\leq\eta_{\rm tot}\), their accumulated coordinatewise
response uncertainty is at most
\(\eta_{\rm tot}/\sqrt{d_i}\).  A summable schedule such as
\(\eta_t=\Theta(\delta_{\rm fin}/(t+1)^2)\) therefore costs only additional
logarithmic solver accuracy and can be absorbed by the finite KKT hysteresis.
It does not locate the coordinates whose intervals cross a retained level;
that exhaustive output-sensitive step remains the interface in
\cref{cor:aesp-cd-point-source-route-output-interface}.
\end{lemma}

\begin{proof}
Put \(e=h^{S,w}-\widetilde h\).  Since
\(\widehat H_{SS}e=r\), the degree-normalized vector
\(z=D_S^{1/2}e\) obeys
\begin{equation*}
 Q_{SS}z=D_S^{-1/2}r,
 \qquad Q_{SS}\succeq\alpha I.
\end{equation*}
Thus
\begin{equation*}
 \norm{z}_2\leq\alpha^{-1}\norm{D_S^{-1/2}r}_2,
 \qquad
 |e_i|=|z_i|/\sqrt{d_i},
\end{equation*}
which proves
\cref{eq:aesp-cd-point-source-response-residual-certificate}.  Multiply by
\(\Delta_w\), use
\cref{eq:aesp-cd-point-source-response-residual-target}, and sum the interval
radii over the pivot sequence.
For any boundary row \(v\), the resulting key uncertainty is at most
\begin{equation*}
 c\sum_{\substack{i\in S\\i\sim v}}\frac{\eta_{\rm tot}}{\sqrt{d_i}}
 \leq c\,d_v\eta_{\rm tot}.
\end{equation*}
Thus taking
\(\eta_{\rm tot}\leq\delta_{\rm fin}=\alpha\epskkt/(2c)\) consumes at most
half of the terminal KKT band, which justifies the stated finite-hysteresis
interface.
\end{proof}

\begin{lemma}[A support pivot has a square-root-scale response radius]
\label{lem:aesp-cd-point-source-hitting-radius}
In the setting of
\cref{prop:aesp-cd-point-source-hitting-column}, put
\begin{equation*}
 \lambda_\alpha:=\frac{1-\sqrt\alpha}{1+\sqrt\alpha}.
\end{equation*}
For \(i\in S\), let
\(r=\operatorname{dist}_{G[S\cup\{w\}]}(i,w)\).  If \(r\geq2\), then
\begin{equation}
 \sqrt{\frac{d_i}{d_w}}\,h_i^{S,w}
 \leq \frac2\alpha\lambda_\alpha^{r-1}.
 \label{eq:aesp-cd-point-source-hitting-radius}
\end{equation}
Moreover every legal pivot for any source distribution satisfies
\begin{equation}
 \Delta_w h_i^{S,w}
 \leq \frac2{\alpha\sqrt{d_id_w}}
       \lambda_\alpha^{r-1}
 \leq \frac2\alpha\lambda_\alpha^{r-1}.
 \label{eq:aesp-cd-point-source-hitting-event-radius}
\end{equation}
Consequently, if all retained coordinate-level slacks are at least
\(\delta>0\), every event in
\cref{eq:aesp-cd-point-source-hitting-event} lies within graph distance
\begin{equation}
 R_{\rm hit}(\delta)
 :=1+\left\lceil
   \frac{\log_+(2/(\alpha\delta))}
        {\log(1/\lambda_\alpha)}
 \right\rceil
 =\widetilde O(1/\sqrt\alpha)
 \label{eq:aesp-cd-point-source-hitting-event-locality}
\end{equation}
of the new pivot.  For the finite-level reporter,
\(\delta=\delta_{\rm fin}=\alpha\epskkt/(2c)\), so one may use
\begin{equation*}
 R_{\rm hit}(\delta_{\rm fin})
 =O\!\left(
  1+\frac1{\sqrt\alpha}
  \log\!\left(2+\frac{c}{\alpha^2\epskkt}\right)
 \right).
\end{equation*}

Thus the killed-walk mean horizon \(\Theta(1/\alpha)\) is not the relevant
deterministic information radius: finite changing-face events are confined
to the accelerated \(1/\sqrt\alpha\) scale.  This still does not prove the
desired work bound.  A radius-\(R_{\rm hit}\) ball can have arbitrarily large
volume, and many pivot neighborhoods can overlap; an implementation must
use the support-linear frontier universe and emit only actual level events,
not scan every such ball after every pivot.
\end{lemma}

\begin{proof}
Use the degree-normalized principal matrix
\(Q_{SS}=D_S^{-1/2}\widehat H_{SS}D_S^{-1/2}\).  Its spectrum lies in
\([\alpha,1]\), and the response equation gives
\begin{equation*}
 \frac{D_S^{1/2}h^{S,w}}{\sqrt{d_w}}
 =-Q_{SS}^{-1}Q_{Sw}.
\end{equation*}
The standard degree-\(k\) Chebyshev inverse polynomial on
\([\alpha,1]\) satisfies
\begin{equation*}
 \|Q_{SS}^{-1}-p_k(Q_{SS})\|_2
 \leq\frac2\alpha\lambda_\alpha^{k+1}.
\end{equation*}
Take \(k=r-2\).  Graph sparsity makes the \(i\)-th coordinate of
\(p_k(Q_{SS})Q_{Sw}\) zero, while
\(\|Q_{Sw}\|_2\leq\|Qe_w\|_2\leq1\).  This proves
\cref{eq:aesp-cd-point-source-hitting-radius}.

The enlarged restricted value at \(w\) is \(\Delta_w\), and nested
Stieltjes comparison gives
\(\Delta_w\leq y_w^\star\leq y_w^0\), where \(y^0\) is ordinary PPR for
the same source distribution in degree-unscaled coordinates.  Since
\(d^\top y^0=1\), one has \(\Delta_w\leq1/d_w\).  Multiplying the first
bound by \(\Delta_w\) proves
\cref{eq:aesp-cd-point-source-hitting-event-radius}.  Solving its final
upper bound for a contribution larger than \(\delta\), and using
\(\log(1/\lambda_\alpha)\geq2\sqrt\alpha\), proves the remaining claims.
\end{proof}

\begin{corollary}[A route-output-sensitive event locator closes point-source
discovery]
\label{cor:aesp-cd-point-source-route-output-interface}
Use the standard finite reduction
\(\rho=\epskkt=\epsppr/2\).  Suppose a dynamic locator maintains the exact
nested point-source pivot state and, on pivot \(w\), emits every coordinate
level crossed by \(\Delta_w h^{S,w}\).  Assume its total charged work is
\begin{equation}
\widetilde O\!\left(
  \sum_{w\in S^\star}d_w
  +
  \sum_{(w,i,\ell)\in\mathcal E_{\rm lev}}
   \bigl\{1+\operatorname{dist}_{G[S\cup\{w\}]}(w,i)\bigr\}
 \right),
 \label{eq:aesp-cd-point-source-route-output-interface}
\end{equation}
where \(\mathcal E_{\rm lev}\) records one item for each crossed geometric
level, and where maintaining the principal response state, rejected queries,
and locating an empty output are all included in this charge.  Then the
finite reporter, exact support-safe discovery, and final obstacle
publication use
\begin{equation*}
 \widetilde O\!\left(
  \frac{\vol(S^\star)}{\sqrt\alpha}\right)
 \subseteq
 \widetilde O\!\left(
  \frac1{\rho\sqrt\alpha}\right)
\end{equation*}
work.  The final certified-face Chebyshev solve has the same order.
\end{corollary}

\begin{proof}
\Cref{cor:aesp-cd-finite-level-reporter} bounds the number of level records by
\begin{equation*}
 |\mathcal E_{\rm lev}|
 =\widetilde O(\vol(S^\star))
\end{equation*}
when \(\rho=\epskkt\).  Every such record has distance at most
\(R_{\rm hit}(\delta_{\rm fin})=\widetilde O(1/\sqrt\alpha)\) by
\cref{lem:aesp-cd-point-source-hitting-radius}.  Also
\(\sum_{w\in S^\star}d_w=\vol(S^\star)\).
Substitution into
\cref{eq:aesp-cd-point-source-route-output-interface}, followed by the
support cap \(\vol(S^\star)\leq1/\rho\), proves the claim.
\end{proof}

\begin{corollary}[The same route-output locator closes sparse-source discovery]
\label{cor:aesp-cd-sparse-source-route-output-interface}
Let $s\geq0$, $\one^\top s=1$, and $T=\supp(s)$.  Under the standard
finite reduction $\rho=\epskkt=\epsppr/2$, initialize one candidate record
for every source row and use the exact nested positive-residual pivot state
for the joint sparse-source objective.  Suppose the dynamic locator in
\cref{cor:aesp-cd-point-source-route-output-interface} supplies the same
exhaustive coordinate-level events and the same route-output charge, including
empty reports and component mergers.  Then discovery and the final
certified-envelope solve use
\begin{equation}
 |T|+\widetilde O\!\left(
  \frac{\vol(S^\star)}{\sqrt\alpha}
 \right)
 \subseteq
 |T|+\widetilde O\!\left(
  \frac1{\rho\sqrt\alpha}
 \right)
 \label{eq:aesp-cd-sparse-source-route-output-interface}
\end{equation}
work.  Thus the unresolved locator interface is strong enough to recover the
original additive sparse-source target; point-source specialization removes
multi-root bookkeeping but does not weaken the value of a solution to that
interface.
\end{corollary}

\begin{proof}
In degree-unscaled coordinates a boundary row has key
\begin{equation*}
 g_v=P_v-T_v,
 \qquad
 P_v=c\sum_{i\in S\cap N(v)}y_i,
 \qquad
 T_v=\alpha(\rho d_v-s_v).
\end{equation*}
Reading the sparse source initializes all $T$-records in $O(|T|)$ work.
Rows with $T_v<0$ are strict-positive pivots already at zero.  For every
remaining row $0\leq T_v\leq\alpha\rho d_v$.  With the rounding parameters
of \cref{cor:aesp-cd-finite-level-reporter}, a named row satisfies
\begin{equation*}
 \bar P_v\leq P_v
 \leq(1+\eta_{\rm fin})\bar P_v+c d_v\delta_{\rm fin}.
\end{equation*}
Hence $\bar P_v>T_v$ is a safe pivot, while $\bar P_v\leq T_v$ gives
\begin{equation*}
 g_v
 \leq\eta_{\rm fin}T_v+c d_v\delta_{\rm fin}
 \leq\alpha\epskkt d_v.
\end{equation*}
This is the same exhaustive finite KKT dichotomy as in the point-source
proof.  Ordinary PPR for any probability source obeys $d^\top y^0=1$, so
every active coordinate is at most $1/d_i$; the number of geometric level
records is unchanged.  The response column, its residual certificate, and
its square-root event radius depend only on the current principal matrix and
the pivot amount.  The generalized argument in
\cref{lem:aesp-cd-point-source-hitting-radius} bounds that amount for any
source distribution.  Finally
\cref{prop:aesp-cd-sparse-source-radius} confines all components to the same
source-set radius, while the support cap gives
\(\vol(S^\star)\leq1/\rho\).  Applying the assumed route-output charge proves
\cref{eq:aesp-cd-sparse-source-route-output-interface}.
\end{proof}

\paragraph{Dynamic-inverse scope.}
The hypothesis in
\cref{cor:aesp-cd-point-source-route-output-interface} is stronger than a
named-coordinate dynamic solve.  The
\href{https://arxiv.org/abs/1906.10530}{dynamic spectral vertex sparsifier}
of Durfee, Gao, Goranci, and Peng supports energy-accurate named solution
queries on bounded-degree graphs, but its stated update/query bound is
sublinear in the ambient \(n\), not output-local in \(\vol(S^\star)\), and
does not provide exhaustive one-sided level events.  General
\href{https://arxiv.org/abs/1905.05067}{dynamic matrix-inverse}
data structures and their conditional lower bounds treat much broader
element/column update streams.  They neither instantiate the monotone
Stieltjes locator above nor prove it impossible.  The remaining question is
therefore a specialized dynamic Green-function reporting problem, not an
ordinary static linear solve and not a settled dynamic-inverse lower bound.
Likewise, the nearly-linear
\href{https://arxiv.org/abs/1810.02348}{M-matrix solvers} of Ahmadinejad,
Jambulapati, Saberi, and Sidford are nearly linear in the supplied input
matrix size.  They solve the ambient linear-algebra primitive but do not
discover an unknown output-local principal envelope or emit all one-sided
level events, so their theorem does not by itself instantiate the locator.

\begin{proposition}[Exact support pivots form a source-independent
energy-orthogonal response basis]
\label{prop:aesp-cd-point-source-orthogonal-pivots}
Consider any exact nested fixed-target trace
\(\varnothing=S_0\subset S_1\subset\cdots\subset S_J=S^\star\), with one
positive-residual pivot \(w_j\) at a time.  Zero-pad every restricted
solution and put
\begin{equation*}
 z_j:=y^{S_{j+1}}-y^{S_j}.
\end{equation*}
Then the nonzero increments are pairwise \(\widehat H\)-orthogonal:
\begin{equation}
 z_i^\top\widehat H z_j=0\quad(i\ne j),
 \qquad
 \|z_j\|_{\widehat H}^2
 =K^{S_j}_{w_jw_j}\Delta_{w_j}^2
 =g^{S_j}_{w_j}\Delta_{w_j}.
 \label{eq:aesp-cd-point-source-pivot-orthogonality}
\end{equation}
Their energies telescope exactly through the restricted objective
\(\widehat F(y)=\tfrac12y^\top\widehat Hy-\ell_\rho^\top y\):
\begin{equation}
 \frac12\sum_{j=0}^{J-1}\|z_j\|_{\widehat H}^2
 =\widehat F(0)-\widehat F(y^\star)
 =\frac12\ell_\rho^\top y^\star
 \leq\frac{\alpha}{2}.
 \label{eq:aesp-cd-point-source-pivot-energy}
\end{equation}
Under the point source $s=e_o$, the last bound sharpens to
\(\alpha/(2d_o)\).
Because every increment is coordinatewise nonnegative, its degree-weighted
variation also telescopes without cancellation:
\begin{equation}
 \sum_{j=0}^{J-1}\sum_i d_i(z_j)_i
 =d^\top y^\star\leq d^\top y^0=1.
 \label{eq:aesp-cd-point-source-pivot-variation}
\end{equation}
Moreover, for every coordinate \(i\), Bessel's inequality gives the sharp
leverage ledger
\begin{equation}
 \sum_{j:z_j\ne0}
 \frac{(z_j)_i^2}{\|z_j\|_{\widehat H}^2}
 \leq e_i^\top\widehat H^{-1}e_i
 \leq\frac1{\alpha d_i}.
 \label{eq:aesp-cd-point-source-pivot-bessel}
\end{equation}
In fact, after restricting to the terminal support, the increments are a
complete basis and give the exact inverse factorization
\begin{equation}
 \widehat H_{S^\star S^\star}^{-1}
 =\sum_{j:z_j\ne0}
   \frac{(z_j)_{S^\star}(z_j)_{S^\star}^\top}
        {\norm{z_j}_{\widehat H}^2}.
 \label{eq:aesp-cd-pivot-inverse-factorization}
\end{equation}
Thus equality holds in the first leverage bound when $i\in S^\star$ and
the inverse there is interpreted as the terminal principal inverse.

Thus the changing-face response stream has bounded total energy and no
energy-direction repetition.  The remaining finite-level task can be
viewed as coordinate heavy-hitter reporting in this orthogonal stream.
This ledger alone is not an algorithm: constructing or sketching a new
dense response \(z_j\) without solving its killed-harmonic column is exactly
the interface left by
\cref{prop:aesp-cd-point-source-hitting-column}.
Equivalently, the missing primitive is an online thresholded
inverse-Cholesky-column reporter for an initially unknown terminal principal
matrix.  A named-coordinate inverse query does not enumerate its threshold
crossings, while an ordinary sparse factorization assumes that this terminal
matrix is already supplied.
\end{proposition}

\begin{proof}
The enlarged restricted equations give
\begin{equation*}
 [\widehat H z_j]_{S_j}=0.
\end{equation*}
Every earlier increment is supported on \(S_j\); symmetry therefore makes
it orthogonal to \(z_j\).  The block formula from
\cref{eq:aesp-cd-point-source-hitting-update} gives the norm identity in
\cref{eq:aesp-cd-point-source-pivot-orthogonality}.  Completing the square
on each nested restricted subspace gives
\begin{equation*}
 \widehat F(y^{S_j})-\widehat F(y^{S_{j+1}})
 =\frac12\|z_j\|_{\widehat H}^2,
\end{equation*}
so the sum telescopes.  At the obstacle optimum,
\(\widehat Hy^\star=\ell_\rho\) on its positive support, whence
\(-\widehat F(y^\star)=\ell_\rho^\top y^\star/2\).
For a general probability source, put $b=\alpha D^{-1/2}s$.  Obstacle
comparison and the negative threshold term give
\begin{equation*}
 \ell_\rho^\top y^\star
 \leq (\alpha s)^\top y^0
 =b^\top Q^{-1}b
 \leq\frac{\norm{b}_2^2}{\alpha}
 \leq\alpha,
\end{equation*}
where $y^0$ is ordinary PPR in the same coordinates and
\(\norm{b}_2\leq\alpha\norm{s}_1=\alpha\).  For $s=e_o$, one may instead
use
\(\ell_\rho^\top y^\star\leq\alpha y_o^0\leq\alpha/d_o\), giving the
stated sharper constant.  Monotonicity of the nested restricted solves,
\(y^\star\leq y^0\), and $d^\top y^0=1$ also prove
\cref{eq:aesp-cd-point-source-pivot-variation}.

Finally normalize the nonzero \(z_j\)'s in the \(\widehat H\)-inner product.
They are orthonormal, while coordinate evaluation has Riesz representer
\(\widehat H^{-1}e_i\).  Bessel gives the first bound in
\cref{eq:aesp-cd-point-source-pivot-bessel}; the second follows from
\(\widehat H\succeq\alpha D\).
There are exactly $|S^\star|$ nonzero increments, and in admission order
their restrictions to $S^\star$ form a triangular matrix with nonzero
diagonal.  Consequently they are a basis of $\mathbb R^{S^\star}$.  If $Z$
is the matrix of these restricted columns and
$E=\operatorname{diag}(\norm{z_j}_{\widehat H}^2)$, then
\begin{equation*}
 Z^\top\widehat H_{S^\star S^\star}Z=E.
\end{equation*}
Invertibility of $Z$ gives
$\widehat H_{S^\star S^\star}^{-1}=ZE^{-1}Z^\top$, which is
\cref{eq:aesp-cd-pivot-inverse-factorization}.
\end{proof}

\begin{proposition}[Schur pivot multipliers are killed first-exit laws]
\label{prop:aesp-cd-point-source-harmonic-exit}
Let $S$ be any eliminated face, $E=V\setminus S$, and
\begin{equation*}
 \mathsf P:=\frac{c}{p}D^{-1}A,
 \qquad
 p=\frac{1+\alpha}{2},
 \qquad c=\frac{1-\alpha}{2}.
\end{equation*}
Thus \(\mathsf P\) is the transition matrix of the ordinary random walk
followed at every step with survival probability $c/p$.  Its trace kernel
on $E$ is
\begin{equation}
 \mathsf T^S
 :=\mathsf P_{EE}
   +\mathsf P_{ES}(I-\mathsf P_{SS})^{-1}\mathsf P_{SE}.
 \label{eq:aesp-cd-point-source-trace-kernel}
\end{equation}
Then the exterior Schur complement is exactly
\begin{equation}
 K^S=pD_E(I-\mathsf T^S).
 \label{eq:aesp-cd-point-source-trace-schur}
\end{equation}
For a fixed-target pivot $w\in E$, every multiplier in the residual update
has the probabilistic representation
\begin{equation}
 \gamma_v=-\frac{K^S_{vw}}{K^S_{ww}}
 =\frac{\mathsf T^S_{wv}}{1-\mathsf T^S_{ww}}
 \qquad(v\in E\setminus\{w\}).
 \label{eq:aesp-cd-point-source-harmonic-exit}
\end{equation}
Equivalently, start the killed walk at $w$, ignore every return to $w$,
and stop at the first different exterior vertex; \(\gamma_v\) is the
probability of stopping at $v$, with the missing probability assigned to
killing.  In particular
\begin{equation*}
 \sum_{v\ne w}\gamma_v\leq\frac{c}{p}
 =1-\frac{\alpha}{p}.
\end{equation*}

Consequently the exact block reporter is a monotone transport process whose
implicit update can be sampled without materializing Schur fill.  Literal
walk simulation is not yet the desired acceleration: because every original
transition kills with probability \(\alpha/p\), one such exit sample can
take $O(p/\alpha)$ expected transitions.  The remaining algorithmic
question is whether signed/accelerated response data can provide the same
one-sided threshold reports in \(\widetilde O(1/\sqrt\alpha)\), amortized
over the point-source injection budget
\cref{eq:aesp-cd-point-source-pivot-injection}.
\end{proposition}

\begin{proof}
The degree-unscaled Hessian factors as
\begin{equation*}
 \widehat H=pD(I-\mathsf P).
\end{equation*}
Taking its Schur complement over $S$ gives
\begin{align*}
 K^S
 &=pD_E\bigl[
   I-\mathsf P_{EE}
   -\mathsf P_{ES}(I-\mathsf P_{SS})^{-1}\mathsf P_{SE}
  \bigr],
\end{align*}
which is \cref{eq:aesp-cd-point-source-trace-schur}.  The bracketed series
in \cref{eq:aesp-cd-point-source-trace-kernel} enumerates one trace step:
either the walk moves directly between exterior vertices or it makes a
positive-length excursion through $S$.  Symmetry of $K^S$ and its
$w$-row give
\begin{equation*}
 -\frac{K^S_{vw}}{K^S_{ww}}
 =-\frac{K^S_{wv}}{K^S_{ww}}
 =\frac{\mathsf T^S_{wv}}{1-\mathsf T^S_{ww}}.
\end{equation*}
Summing the geometric series of repeated trace returns to $w$ proves the
first-exit interpretation.  Since a trace excursion must survive its first
original transition, \(\sum_v\mathsf T^S_{wv}\leq c/p\); hence
\begin{equation*}
 \frac{\sum_{v\ne w}\mathsf T^S_{wv}}
      {1-\mathsf T^S_{ww}}
 \leq\frac{c/p-\mathsf T^S_{ww}}
           {1-\mathsf T^S_{ww}}
 \leq\frac cp.
\end{equation*}
The expected-transition statement follows from geometric killing.
\end{proof}

\paragraph{Relation to principal-pivoting theory.}
The linear number of exact support pivots is consistent with the broader
K-matrix LCP theorem of
\href{https://arxiv.org/abs/0807.1249}{Foniok, Fukuda, G\"artner, and
L\"uthi}, which proves linear pivot-path length for K-matrix LCPs.  The
point-source content used here is the explicit decreasing-\(\rho\) ratio
parameterization, the sparse-fill locality ledger, and the finite KKT
interface below; the note does not claim that linear pivot count itself is
new.

\begin{corollary}[Finite KKT stopping band for ratio homotopy]
\label{cor:aesp-cd-ratio-pivot-finite-stop}
On every point-source homotopy face and every exterior vertex,
\begin{equation}
 \alpha d_v\leq B_v\leq\frac{1+\alpha}{2}d_v.
 \label{eq:aesp-cd-ratio-pivot-slope-band}
\end{equation}
Consequently, if a reporter certifies
\begin{equation*}
 \max_{v\notin S}\tau_v\leq\rho+\eta,
\end{equation*}
then the current restricted point at the target \(\rho\) is a lower
subsolution satisfying
\begin{equation}
 R_{\mathrm{kkt},\rho}(x_S(\rho))
 =\max_{v\notin S}\frac{[B_v(\tau_v-\rho)]_+}{d_v}
 \leq\frac{1+\alpha}{2}\eta.
 \label{eq:aesp-cd-ratio-pivot-finite-kkt}
\end{equation}
Thus it suffices to take
\(\eta=2\alpha\epskkt/(1+\alpha)\) for the note's terminal requirement.
An approximate Schur or ratio reporter needs a certified upper envelope only
to this additive scale; arbitrarily small breakpoint gaps do not force exact
support recovery.
\end{corollary}

\begin{proof}
Nonnegativity of \(H_{SS}^{-1}\) gives \(h_S\geq0\).  The full vector
\(\one\) is a supersolution for the derivative system because
\begin{equation*}
 H_{SS}\one
 =\alpha d_S+\frac{1-\alpha}{2}A_{S,\bar S}\one
 \geq\alpha d_S.
\end{equation*}
Hence \(0\leq h_S\leq\one\), and with
\(c=(1-\alpha)/2\),
\begin{equation*}
 B_v=\alpha d_v+c\sum_{u\in S\cap N(v)}(h_S)_u
 \in[\alpha d_v,(\alpha+c)d_v].
\end{equation*}
Since \(\alpha+c=(1+\alpha)/2\), this proves
\cref{eq:aesp-cd-ratio-pivot-slope-band}.  Active rows have zero KKT
residual, while in the degree-unscaled coordinates an inactive positive key
\(g_v\) contributes \(g_v/d_v\) to
\(R_{\mathrm{kkt},\rho}\).  Substituting
\(g_v=B_v(\tau_v-\rho)\) proves
\cref{eq:aesp-cd-ratio-pivot-finite-kkt}.
\end{proof}

\begin{corollary}[Certified interval interface for finite ratio pivoting]
\label{cor:aesp-cd-ratio-pivot-interval-interface}
On a current point-source homotopy face, suppose a reporter returns rigorous
coordinatewise intervals
\begin{equation*}
 0\leq A_v^-\leq A_v\leq A_v^+,
 \qquad
 0<B_v^-\leq B_v\leq B_v^+
 \qquad(v\notin S).
\end{equation*}
Put
\begin{equation*}
 \tau_v^-:=A_v^-/B_v^+,
 \qquad
 \tau_v^+:=A_v^+/B_v^-.
\end{equation*}
Then the following three-way interface is safe.
\begin{enumerate}
\item If some \(\tau_v^->\rho\), that row has a strictly positive actual
  exterior key at the target and may be pivoted.
\item If \(\max_{v\notin S}\tau_v^+\leq\rho+\eta\), the current restricted
  point satisfies the finite KKT guarantee
  \begin{equation*}
   R_{\mathrm{kkt},\rho}(x_S(\rho))
   \leq\frac{1+\alpha}{2}\eta.
  \end{equation*}
\item Otherwise the reporter may refine the gray rows or rebuild its
  certificate; no unsupported pivot or terminal declaration is made.
\end{enumerate}
Thus exact breakpoint ordering is not part of the required approximate
Schur contract.

For a concrete sufficient scale, suppose
\begin{equation*}
 |\widehat A_v-A_v|\leq\epsilon_A d_v,
 \qquad
 |\widehat B_v-B_v|\leq\epsilon_B d_v,
 \qquad
 \epsilon_B\leq\alpha/4,
\end{equation*}
and intersect the corresponding centered intervals with the universal bands
\(A_v\geq0\) and \(B_v\geq\alpha d_v\).  Every homotopy breakpoint obeys
\(0\leq\tau_v\leq1/d_o\leq1\), and hence
\begin{equation}
 0\leq\tau_v^+-\tau_v
 \leq\frac{4(\epsilon_A+\epsilon_B)}{\alpha}.
 \label{eq:aesp-cd-ratio-pivot-interval-width}
\end{equation}
In particular, normalized pair radii
\(\epsilon_A+\epsilon_B\leq\alpha\eta/4\) are sufficient for an additive
\(\eta\) ratio upper certificate.
\end{corollary}

\begin{proof}
The interval ratios enclose \(\tau_v=A_v/B_v\).  The first claim follows
from
\(g_v(\rho)=B_v(\tau_v-\rho)>0\), and adding any strictly violated row is a
legal monotone obstacle pivot.  The second claim is
\cref{cor:aesp-cd-ratio-pivot-finite-stop}.

The root enters at \(1/d_o\).  Every later breakpoint is a nonnegative
weighted average of an old breakpoint and the admitted breakpoint by
\cref{eq:aesp-cd-point-source-homotopy-mix}; hence all are in
\([0,1/d_o]\).  A centered certified interval can overshoot the true
\(A_v\) and undershoot the true \(B_v\) by at most
\(2\epsilon_A d_v\) and \(2\epsilon_Bd_v\), respectively.  Since
\(A_v\leq B_v\), \(B_v\geq\alpha d_v\), and
\(B_v-2\epsilon_Bd_v\geq\alpha d_v/2\), direct subtraction gives
\begin{equation*}
 \frac{A_v+2\epsilon_A d_v}{B_v-2\epsilon_Bd_v}
 -\frac{A_v}{B_v}
 \leq\frac{4(\epsilon_A+\epsilon_B)}{\alpha},
\end{equation*}
which proves \cref{eq:aesp-cd-ratio-pivot-interval-width}.
\end{proof}

\begin{proposition}[Two-sided spectral Schur approximation is not a ratio certificate]
\label{prop:aesp-cd-spectral-schur-ratio-stop}
For every \(0<\varepsilon<1\), the SPD Stieltjes matrices
\begin{equation*}
 K_\varepsilon=
 \begin{pmatrix}1&-\varepsilon\\-\varepsilon&1\end{pmatrix},
 \qquad
 \widetilde K=I
\end{equation*}
satisfy
\begin{equation*}
 (1-\varepsilon)\widetilde K
 \preceq K_\varepsilon
 \preceq(1+\varepsilon)\widetilde K.
\end{equation*}
Nevertheless, take current breakpoint pairs
\((A_w,B_w)=(1,1)\) and \((A_v,B_v)=(0,b)\), \(b>0\), and pivot \(w\).
The exact Schur update gives
\begin{equation*}
 (A'_v,B'_v)=(\varepsilon,b+\varepsilon),
 \qquad \tau'_v=\frac{\varepsilon}{b+\varepsilon}>0,
\end{equation*}
whereas using \(\widetilde K\) literally gives
\((\widetilde A'_v,\widetilde B'_v)=(0,b)\) and
\(\widetilde\tau'_v=0\).
Therefore an ordinary two-sided spectral vertex sparsifier or approximate
Schur complement cannot by itself be interpreted as the certified one-sided
ratio envelope in
\cref{cor:aesp-cd-ratio-pivot-interval-interface}.  It must be accompanied
by a conversion to rowwise pair-error intervals (at scale
\(O(\alpha\eta d_v)\)), or by another directly certified reporter.  This is
an interface obstruction, not a lower bound against spectral sparsification
or randomized approximate elimination.
\end{proposition}

\begin{proof}
The two eigenvalues of \(K_\varepsilon\) are
\(1-\varepsilon\) and \(1+\varepsilon\).  The exact elimination multiplier
is \(-K_{vw}/K_{ww}=\varepsilon\), whereas the multiplier read from
\(\widetilde K\) is zero.  Substitution into
\cref{eq:aesp-cd-ratio-pivot-pair} gives the displayed pairs and ratios.
\end{proof}

\paragraph{Relation to spectral elimination.}
Nearly-linear
\href{https://arxiv.org/abs/1605.02353}{approximate Gaussian elimination}
and
\href{https://arxiv.org/abs/1506.08204}{spectral vertex sparsification}
produce sparse energy approximations, while
\href{https://arxiv.org/abs/1906.10530}{dynamic spectral vertex
sparsifiers} additionally support terminal changes and energy/solution
queries.  Their stated guarantees are spectral or energy-norm guarantees,
not simultaneous one-sided intervals for every ratio \(A_v/B_v\).  The
preceding proposition shows that such an interval conversion is a genuine
additional proof obligation; it does not say that these techniques cannot be
adapted to provide one.

\begin{corollary}[Margin-free accelerated obstacle primitive]
\label{cor:aesp-cd-margin-free-obstacle-primitive}
In the setting of
\cref{prop:aesp-cd-speculative-envelope-doubling}, suppose on every fixed
envelope $U$ that
\begin{equation*}
 \mu I\preceq H_U\preceq LI,
 \qquad H_Uv_U\geq\mu_vv_U>0,
\end{equation*}
with certified parameters.  Accelerated projected-gradient scratch, followed
by the publication in
\cref{thm:aesp-cd-obstacle-clip-retraction}, implements the assumed
restricted obstacle primitive in
\begin{equation*}
 \widetilde O\!\left(\vol(U)\sqrt{L/\mu}\right)
\end{equation*}
work.  No active-coordinate or strict-complementarity margin is required.
For a principal RPPR envelope, $L\leq1$, $\mu\geq\alpha$, and one may take
$v_U=D_U^{1/2}\one$ with $H_Uv_U\geq\alpha v_U$, giving
$\widetilde O(\vol(U)/\sqrt\alpha)$ work.

This removes the active-set margin hypothesis from the obstacle solve in
\cref{prop:aesp-cd-speculative-envelope-doubling}.  It does not control the
final explored volume, nor does it by itself decide an exactly zero boundary
key with finite precision; the boundary-interval fallback in that proposition
remains explicit.
\end{corollary}

\begin{proof}
Projected acceleration keeps every scratch iterate feasible and reaches a
certified $\ell_\infty$ error $\varepsilon$ after
$O(\sqrt{L/\mu}\log(E_0/\varepsilon))$ matrix applications, for example by
driving the standard strongly-convex objective-gap bound below
$\mu\varepsilon^2/2$.  The retraction error amplification in
\cref{eq:aesp-cd-obstacle-clip-pay-error} changes only the logarithmic target.
The RPPR specialization uses the principal Stieltjes spectral bounds and the
same positive supersolution as
\cref{thm:aesp-cd-safe-chebyshev-face}.
\end{proof}

\begin{corollary}[Strict margins instantiate the obstacle primitive]
\label{cor:aesp-cd-margin-obstacle-primitive}
On a fixed envelope $U$, let
\begin{equation*}
 F_U(x)=\tfrac12\langle x,H_Ux\rangle-\langle b_U,x\rangle
       +I_{x\geq0},
 \qquad \mu I\preceq H_U\preceq LI,
\end{equation*}
and let $p_U$ have support $S$.  Suppose certified margins satisfy
\begin{equation*}
 \min_{i\in S}p_{U,i}\geq m_x>0,\qquad
 \max_{i\in U\setminus S}(b_U-H_Up_U)_i\leq-m_g<0.
\end{equation*}
When $S=U$, omit the dual-margin condition and interpret
$m_g/R_\infty=+\infty$ below.
Assume also that the old padded checkpoint $y_U$ is an obstacle subsolution,
$0\leq y_U\leq p_U$, with $\supp(y_U)\subseteq S$.
Assume $H_U$ is Stieltjes and the certified face $S$ has a known vector
$v_S>0$ and scalar $\mu_v>0$ such that
$H_{SS}v_S\geq\mu_vv_S$ coordinatewise, as required by
\cref{thm:aesp-cd-safe-chebyshev-face}.
Put $R_\infty=\max_i\sum_j|(H_U)_{ij}|$.  Accelerated projected-gradient
scratch to
\begin{equation*}
 \norm{a-p_U}_\infty
 \leq\frac13\min\{m_x,m_g/R_\infty\}
\end{equation*}
identifies $S$: active coordinates exceed $m_x/2$, while inactive
coordinates are below $m_x/2$ and have dual residual below $-m_g/2$.
Running
\cref{thm:aesp-cd-safe-chebyshev-face} only on this certified linear face,
from the old padded lower checkpoint, publishes a safe approximation to
$p_U$.  The combined work is
\begin{equation*}
 \widetilde O\!\left(
   \vol(U)\sqrt{L/\mu}\right),
\end{equation*}
where $m_x,m_g$ enter only logarithmically.  Hence
the exact face itself, rather than only a lower-safe approximation, is
certified under uniform strict active/dual margins.
\end{corollary}

\begin{proof}
Strongly convex accelerated projected gradient reaches the displayed distance
in
\begin{equation*}
 O\!\left(\sqrt{L/\mu}\,
 \log\!\frac{E_0}{\min\{m_x,m_g/R_\infty\}}\right)
\end{equation*}
iterations.  Every
active coordinate is then at least $2m_x/3$.  On an inactive coordinate,
$a_i\leq m_x/3$ and
\begin{equation*}
 (b_U-H_Ua)_i
 \leq-m_g+R_\infty\norm{a-p_U}_\infty\leq-2m_g/3.
\end{equation*}
Thus the exact support is certified before the signed linear-face scratch is
published.  The maximum-with-old-checkpoint retraction preserves the lower
certificate, as in \cref{thm:aesp-cd-safe-chebyshev-face}.
\end{proof}

Without strict margins, projected acceleration may leave arbitrarily small
positive inactive scratch coordinates and cannot certify the exact face at a
uniform accuracy.  The clip-or-pay publication nevertheless deletes every
unsafe coordinate without paying its entire dual residual.  Thus strict
margins remain useful for face identification, not for lower-safe obstacle
publication.

\begin{proposition}[Projected acceleration can spill outside the true support]
\label{prop:aesp-cd-accelerated-support-spill}
Even for a two-coordinate Stieltjes obstacle problem, one exact critically
tuned projected Nesterov step can activate a coordinate outside the support of
the true minimizer.  Hence
\cref{cor:aesp-cd-margin-free-obstacle-primitive} makes the published output
local, but it does not make unrestricted accelerated scratch support-local.
\end{proposition}

\begin{proof}
Take
\begin{equation*}
 H=\begin{pmatrix}601/676&-45/169\\-45/169&61/169\end{pmatrix},
 \qquad b=\begin{pmatrix}1\\-1/3\end{pmatrix}.
\end{equation*}
The spectrum is $\{1,1/4\}$ and the obstacle minimizer is
$p=(676/601,0)^\top$; its second KKT slack is $61/1803>0$.
With unit step, $x_0=0$, and critical momentum
$\beta=(1-\sqrt{1/4})/(1+\sqrt{1/4})=1/3$, the first projected-gradient
point is $x_1=(1,0)^\top<p$ on its support.  Extrapolation gives
$y_1=x_1+\beta(x_1-x_0)=(4/3,0)^\top$, and the next projected point is
\begin{equation*}
 x_2=[y_1-(Hy_1-b)]_+
 =\begin{pmatrix}194/169\\11/507\end{pmatrix}.
\end{equation*}
Thus $(x_2)_2>0=p_2$.  This stops the simplest strategy of running the full
accelerated scratch by exposing only the final support.  It is not a lower
bound against restricted envelopes, implicit scratch, or a different
localized acceleration.
\end{proof}

\begin{proposition}[Cone-preserving residual polynomials cannot accelerate]
\label{prop:aesp-cd-positive-polynomial-stop}
Let $P_k(t)=\sum_{j=0}^k a_jt^j$ be a residual polynomial with
$P_k(1)=1$.  Suppose that $P_k(T)r\geq0$ for every entrywise nonnegative
matrix $T$ of every finite dimension and every $r\geq0$.  Then
$a_j\geq0$ for every $j$, and for $0\leq t\leq1$,
\begin{equation}
 P_k(t)\geq t^k.
 \label{eq:aesp-cd-positive-polynomial-stop}
\end{equation}
Consequently, at $t=1-1/\mathcal K$, reducing the residual by a factor
$\varepsilon$ requires
$k=\Omega(\mathcal K\log(1/\varepsilon))$.  The same lower bound holds for
any product of such cone-preserving restart polynomials.  Thus a
square-root-rate inner solver must permit signed intermediate residuals; the
safe publication map in
\cref{thm:aesp-cd-safe-chebyshev-face} is not merely cosmetic.
\end{proposition}

\begin{proof}
Apply the hypothesis to the nilpotent forward shift $S$ in dimension
$k+1$ and its first basis vector.  The coordinates of
$P_k(S)e_1$ are exactly $a_0,\ldots,a_k$, so all coefficients are
nonnegative.  Since they sum to one and $t^j\geq t^k$ for $j\leq k$,
\cref{eq:aesp-cd-positive-polynomial-stop} follows.  Products preserve
nonnegative coefficients and add total degree.
\end{proof}

Thus the composite analogue of the ordinary-PPR residual-mass contraction is
valid for arbitrary signed iterates; no nonnegativity hypothesis is needed.

\begin{proposition}[A point source can require linearly many radius-one batches]
\label{prop:aesp-cd-fan-linear-batches}
For every even $m\geq4$ there is an unweighted $(m+1)$-vertex point-source
RPPR instance for which every vertex is adjacent to the source, while the
canonical exact restricted-solve rule that admits all currently
strict-positive exterior keys takes at least $m/2-1$ nonempty batches.
Consequently, specializing the end-to-end contract to $s=e_o$ removes the
multi-source merge problem but does not bound the number of exact face
changes by the support radius, even up to a graph-independent factor.
\end{proposition}

\begin{proof}
Let vertex $0$ be adjacent to every vertex of the path
$1-2-\cdots-m$; thus the whole fan has radius one from $0$.  Put
\begin{equation*}
 \rho=\frac1{4m-2},\qquad
 U_k=\{0,1,\ldots,k,m-k+1,\ldots,m\}.
\end{equation*}
We first take the continuous limit $\alpha\downarrow0$.  In the
degree-unscaled coordinates, divide every proper-face solution and every
exterior key by $\alpha$, then multiply the equations by two.  The limiting
restricted system has Hessian $D-A$ and load $2(e_0-\rho d)$.

At $U_0=\{0\}$ the root value is
\begin{equation*}
 r_0=\frac{2(1-\rho m)}m.
\end{equation*}
The two path endpoints have key $r_0-4\rho>0$, whereas every other
path vertex has key $r_0-6\rho<0$; after a common positive denominator their
numerators are respectively $2(m-2)$ and $-4$.

For $k\geq1$, symmetry gives one root value $r_k$ and two identical arm
vectors $u\in\mathbb R^k$.  Let $M_k$ be tridiagonal with first diagonal
entry $2$, all remaining diagonal entries $3$, and off-diagonal entries
$-1$, and let $d_k=(2,3,\ldots,3)^\top$.  Writing
\begin{align*}
 A_k&=\mathbf1^\top M_k^{-1}\mathbf1,&
 B_k&=\mathbf1^\top M_k^{-1}d_k,\\
 a_k&=(M_k^{-1}\mathbf1)_k,&
 b_k&=(M_k^{-1}d_k)_k,
\end{align*}
the arm and root equations are
\begin{equation*}
 M_ku=r_k\mathbf1-2\rho d_k,\qquad
 r_k=\frac{2\{1-\rho(m+2B_k)\}}{m-2A_k}.
\end{equation*}
If $F_j$ denotes the Fibonacci sequence with $F_0=0,F_1=1$, elementary
continuant induction gives
\begin{equation*}
 \det M_k=F_{2k+1},\quad
 a_k=\frac{F_{2k}}{F_{2k+1}},\quad
 b_k=\frac{3F_{2k}-1}{F_{2k+1}},\quad
 A_k=k-a_k,\quad B_k=3k-1-b_k.
\end{equation*}
Let
$\Delta_k=(4m-2)(m-2A_k)>0$; positivity follows also from the Schur
complement of the proper principal Laplacian.  Every unseen vertex not next
to an arm has the same exterior key, while the two vertices immediately next
to the arms have one additional contribution $u_k$.  Substitution of the
four continuant identities gives the exact signs
\begin{equation}
 g_k^{\rm far}=r_k-6\rho
   =-\frac{4}{F_{2k+1}\Delta_k}<0,\qquad
 g_k^{\rm next}=r_k+u_k-6\rho
   =\frac{2(m-2k-2)}{F_{2k+1}\Delta_k}.
 \label{eq:aesp-cd-fan-linear-batch-keys}
\end{equation}
Hence for $1\leq k<m/2-1$ exactly the next two path vertices have
positive keys.  Every restricted solution along this trace is strictly
interior: the root-only solution is positive, and after a strict-positive
batch the new solution equals the zero-padded old solution plus the inverse
of a connected proper principal Stieltjes matrix applied to a nonzero
nonnegative residual.

Return now to
$\widehat H_\alpha=((1+\alpha)/2)D-((1-\alpha)/2)A$.
For each of the finitely many proper faces above, the solution divided by
$\alpha$ and its exterior keys divided by $\alpha$ are continuous at
$\alpha=0$.  All active-coordinate and key inequalities used through
$U_{m/2-1}$ are strict by
\cref{eq:aesp-cd-fan-linear-batch-keys}.  Therefore some
$\alpha_m>0$ preserves the batches
\begin{equation*}
 \{1,m\},\{2,m-1\},\ldots,
 \{m/2-1,m/2+2\}.
\end{equation*}
There are $m/2-1$ of them, and whatever later support is selected remains
inside the radius-one fan.
\end{proof}

\begin{corollary}[The fan batch stop disappears at the matched finite target]
\label{cor:aesp-cd-fan-finite-stop}
On the fan family in
\cref{prop:aesp-cd-fan-linear-batches}, for every \(0<\alpha<1\) the
root-only restricted point already satisfies
\begin{equation*}
 \frac{R_{\mathrm{kkt},\rho}(x_{\{0\}})}{\alpha}<\rho,
 \qquad \rho=\frac1{4m-2}.
\end{equation*}
Hence the linear exact-support batch count is not a lower bound when the
finite diagnostic is chosen at the matched scale
\(\epskkt\geq\rho\), including the usual
\(\rho=\epskkt=\varepsilon/2\) PPR reduction.  The fan remains a strict
STOP for exact face recovery and exact-positive batch accounting, but not for
the requested finite PPR approximation.
\end{corollary}

\begin{proof}
Write \(p=(1+\alpha)/2\) and \(c=(1-\alpha)/2\).  The root-only value is
\begin{equation*}
 x_0=\frac{\alpha(1-\rho m)}{pm}.
\end{equation*}
Every exterior vertex is adjacent to the root, and the largest normalized
key occurs at a path endpoint of degree two.  Therefore
\begin{equation*}
 \frac{R_{\mathrm{kkt},\rho}(x_{\{0\}})}{\alpha}
 \leq
 \left[-\rho+\frac{c(1-\rho m)}{2pm}\right]_+.
\end{equation*}
Since \(c(1-\rho m)<c\leq p<4pm\rho\) for
\(\rho=1/(4m-2)\), the expression inside the positive part is strictly less
than \(\rho\).
\end{proof}

\begin{proposition}[Matched finite-gap fan still forces linearly many re-solves]
\label{prop:aesp-cd-fan-finite-linear-batches}
For every even \(m\geq4\), take the same radius-one fan, but set
\begin{equation*}
 \alpha=\frac14,
 \qquad
 \rho=\tau=\frac1{10m}.
\end{equation*}
At each current face, solve the restricted system exactly and admit every
exterior row satisfying the finite-significance test
\begin{equation*}
 g_v/d_v>\alpha\tau.
\end{equation*}
The admitted batches are exactly
\begin{equation*}
 \{1,m\},\{2,m-1\},\ldots,\{m/2,m/2+1\}.
\end{equation*}
Thus even the matched finite target
\(\rho=\tau=\varepsilon/2\), with \(\varepsilon=1/(5m)\), can require
\(m/2\) radius-one batches.  Any implementation that independently solves
and scans the whole current face in every batch pays \(\Omega(m^2)\) work,
whereas the desired point-source bound at this fixed \(\alpha\) is
\(\widetilde O(m)\).  This is a lower bound for the repeated-full-solve
implementation, not for dynamic factor reuse, principal pivoting, or another
reporter.
\end{proposition}

\begin{proof}
Put \(p=5/8\), \(c=3/8\), and \(t=m\rho=1/10\).  Scale a restricted
solution by writing \(x=(\alpha/m)z\).  On the symmetric face
\begin{equation*}
 U_k=\{0,1,\ldots,k,m-k+1,\ldots,m\},
\end{equation*}
let \(z_0\) be the root coordinate and let \(u\in\mathbb R^k\) be either
arm.  If \(T_k\) has first diagonal \(2p\), remaining diagonal \(3p\), and
off-diagonal \(-c\), then
\begin{equation}
 T_ku=cz_0\one-t(2,3,\ldots,3)^\top,
 \qquad
 pz_0-\frac{2c}{m}\one^\top u=1-t.
 \label{eq:aesp-cd-fan-finite-arm}
\end{equation}
At \(k=0\), \(z_0=(1-t)/p=36/25\).  Monotonicity of every strict-positive
face extension therefore gives \(z_0\geq36/25\) inductively.

We also have \(z_0<8/5\) on every proper \(U_k\).  To see this, fix the
trial root value \(8/5\) and let \(\bar u\) solve the first equation in
\cref{eq:aesp-cd-fan-finite-arm}.  Its right-hand side is \(2/5\) in the
first row and \(3/10\) thereafter.  The constant vector \((4/5)\one\) is a
strict supersolution, so \(0<\bar u_i<4/5\).  For \(k\geq2\), summing the
arm rows gives
\begin{equation*}
 \one^\top\bar u
 =\frac{4k}{15}+\frac4{45}+\frac2{9}\bar u_1
   -\frac13\bar u_k
 <\frac{4(k+1)}{15};
\end{equation*}
the case \(k=1\) follows directly from \(T_1=[5/4]\).  Since
\(k\leq m/2-1\), this is less than \(2m/15\).  Hence at the trial value the
left side of the root equation is
\begin{equation*}
 p\frac85-\frac{2c}{m}\one^\top\bar u
 >1-\frac1{10}=1-t.
\end{equation*}
Its slope in \(z_0\) is the positive root Schur complement divided by \(m\),
so the actual root value is strictly below \(8/5\).

For an unseen path-interior vertex define the scaled residual
\(\Gamma_v:=m g_v/\alpha\).  The finite test is
\(\Gamma_v>t d_v\).  A vertex not immediately next to an arm has
\begin{equation*}
 \Gamma^{\rm far}=-3t+cz_0<3t,
\end{equation*}
because \(z_0<8/5\).  At \(k=0\), each path endpoint instead satisfies
\begin{equation*}
 \Gamma^{\rm end}=-2t+cz_0>2t.
\end{equation*}
For \(k\geq1\), the constant vector \((4/25)\one\) is a subsolution of the
arm equation at every \(z_0\geq36/25\): its row images are at most the
corresponding right-hand sides, with a strict inequality in at least one
row.  Irreducibility gives \(u_k>4/25\).  Therefore each of the two next
vertices has
\begin{equation*}
 \Gamma^{\rm next}=-3t+c(z_0+u_k)>3t.
\end{equation*}
Exactly those two rows pass the finite test.  Induction proves all \(m/2\)
batches and strict interiority.  Finally every face contains the degree-\(m\)
root, so \(m/2\) separate linear solves or boundary scans cost
\(\Omega(m^2)\).
\end{proof}

\begin{proposition}[All-positive batches are not radius bounded]
\label{prop:aesp-cd-all-positive-radius-stop}
There is an eight-vertex unweighted single-seed RPPR instance whose exact
support has radius one, but the canonical restricted-solve rule that admits
every currently strict-positive exterior key takes five nonempty batches.
Thus neither support radius nor BFS layers bound the number of exact batch
resolves.
\end{proposition}

\begin{proof}
Take seed $0$, $\alpha=1/200$, $\rho=23/800$, and edges
\begin{equation*}
 \{01,02,03,04,05,06,07,12,13,17,24,26,34,45,47,56,57,67\}.
\end{equation*}
In degree-unscaled coordinates use
$\widehat H=((1+\alpha)/2)D-((1-\alpha)/2)A$ and
$\widehat\ell=\alpha e_0-\alpha\rho d$.  Starting from $U_0=\{0\}$,
exact restricted solves and simultaneous admission of all positive exterior
keys give
\begin{equation*}
 \{3\},\quad\{1\},\quad\{2\},\quad\{6\},\quad\{5\},\quad\varnothing.
\end{equation*}
Every restricted solution is strictly interior.  The five positive key
margins are, in order,
\begin{equation*}
 \frac{5013}{37520000},\quad
 \frac{344981}{8088200000},\quad
 \frac{49777913}{4439456576000},\quad
 \frac{169932330613}{61418243753760000},\quad
 \frac{44816054369}{84444409556591680}.
\end{equation*}
The terminal support is $\{0,1,2,3,5,6\}$, every nonseed member is adjacent
to $0$, and the remaining keys at $4,7$ are strictly negative.  Hence
$R_\star=1$ while the batch count is five.  This finite witness stops a
radius-layer count, not every possible amortized batch analysis.
\end{proof}
