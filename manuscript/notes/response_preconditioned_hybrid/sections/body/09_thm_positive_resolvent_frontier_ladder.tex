\begin{theorem}[Positive shifted-resolvent frontier ladder]
\label{thm:positive-resolvent-frontier-ladder}
Use the anchor $A$, frontier $F$, face
$T=A\mathbin{\dot\cup}F$, lift $\mathcal U_F$, and frontier Schur
complement $K_F$ of
\cref{lem:two-sided-harmonic-group-sketch}.  Fix
$\eta\in(0,1/8)$ and define
\begin{equation*}
 t_0:=\eta\sqrt\alpha,
 \qquad
 t_{j+1}:=(1+\eta)t_j,
 \qquad
 J:=\min\{j:t_j\geq\eta^{-1}\}.
\end{equation*}
For $0\leq j<J$, put
\begin{equation}
 \sigma_j:=t_{j+1}^2,
 \qquad
 \omega_j:=\frac2\pi(t_{j+1}-t_j),
 \qquad
 r_\eta(x):=\sum_{j=0}^{J-1}\frac{\omega_j}{x+\sigma_j}.
 \label{eq:positive-resolvent-ladder}
\end{equation}
Then all weights and shifts are positive,
\begin{equation}
 J=O\!\left(
  \eta^{-1}\left(\log\frac1\alpha+\log\frac1\eta\right)
 \right),
 \label{eq:positive-resolvent-rung-count}
\end{equation}
and, uniformly for $x\in[\alpha,1]$,
\begin{equation}
 (1-4\eta)x^{-1/2}
 \leq r_\eta(x)\leq x^{-1/2}.
 \label{eq:positive-resolvent-spectral-sandwich}
\end{equation}
The same quadrature has the accelerated work-mass ledgers
\begin{align}
 \sum_{j=0}^{J-1}\frac{\omega_j}{\alpha+\sigma_j}
 &\leq\frac1{\sqrt\alpha},
 \label{eq:positive-resolvent-work-mass}\\
 \sum_{j=0}^{J-1}
  \omega_j\frac{1+\sigma_j}{\alpha+\sigma_j}
 &\leq\frac1{\sqrt\alpha}
   +\frac{2(1+\eta)}{\pi\eta}.
 \label{eq:positive-resolvent-coordinate-mass}
\end{align}

Let $E_F:\mathbb R^F\to\mathbb R^T$ be the coordinate embedding and
define the sparse shifted face matrices
\begin{equation}
 M_j:=Q_{TT}+\sigma_jE_FE_F^T.
 \label{eq:shifted-face-matrix}
\end{equation}
Block elimination gives the exact realization
\begin{equation}
 M_j^{-1}E_F
 =\mathcal U_F(K_F+\sigma_jI_F)^{-1},
 \qquad
 \mathcal U_Fr_\eta(K_F)
 =\sum_{j=0}^{J-1}\omega_jM_j^{-1}E_F.
 \label{eq:positive-resolvent-block-realization}
\end{equation}
Consequently, for every exterior group $C\subseteq V\setminus T$ and
$R_C:=D_C^{-1/2}Q_{CV}\mathcal U_F$,
\begin{equation}
 (1-4\eta)^2\norm{R_CK_F^{-1/2}}_F^2
 \leq\norm{R_Cr_\eta(K_F)}_F^2
 \leq\norm{R_CK_F^{-1/2}}_F^2.
 \label{eq:positive-resolvent-group-sandwich}
\end{equation}
Thus rescaling the rational group mass by $(1-4\eta)^{-2}$ gives a
one-sided constant-factor upper oracle.

There is also a monotone accelerated coordinate ledger.  Put
$s_F:=(\sqrt{d_u})_{u\in F}$.  Then $K_F$ is Stieltjes,
$K_Fs_F\geq\alpha s_F$ coordinatewise, and
$(K_F)_{uu}\leq1$.  For $g\geq0$, solve all systems
\begin{equation*}
 (K_F+\sigma_jI)y_j=\omega_jg
\end{equation*}
by exact residual coordinate settlements, settling $u$ only when its current
nonnegative residual $p_{j,u}$ satisfies
$p_{j,u}/\sqrt{d_u}\geq\tau$.  If settling $u$ is charged $d_u$ units of
Schur-coordinate work, the total charge over every rung is at most
\begin{equation}
 \frac{\langle s_F,g\rangle}{\tau}
 \left(
  \frac1{\sqrt\alpha}
  +\frac{2(1+\eta)}{\pi\eta}
 \right).
 \label{eq:positive-resolvent-coordinate-work}
\end{equation}
For signed $g$, the same conclusion applies separately to $g^+$ and $g^-$.
This is an unconditional $\alpha^{-1/2}$ settlement count in the
Schur-coordinate model; graph work still has to charge the implementation of
one dense Schur coordinate.
\end{theorem}

\begin{proof}
Use the Stieltjes integral
\begin{equation*}
 x^{-1/2}=\frac2\pi\int_0^\infty\frac{dt}{x+t^2}.
\end{equation*}
The function in the integral is decreasing.
\Cref{eq:positive-resolvent-ladder} is its right-endpoint rectangle rule on
$[t_0,t_J]$, so it never exceeds the integral.  On a geometric interval,
\begin{equation*}
 \frac{x+t_{j+1}^2}{x+t_j^2}\leq(1+\eta)^2.
\end{equation*}
Hence the relative rectangle-rule loss is at most
$1-(1+\eta)^{-2}\leq2\eta$.  The omitted lower and upper tails have relative
mass at most $2\eta/\pi$ each, because
$t_0=\eta\sqrt\alpha\leq\eta\sqrt x$ and
$t_J\geq\eta^{-1}\geq\sqrt x/\eta$.  This proves
\cref{eq:positive-resolvent-spectral-sandwich}; the geometric-grid length
proves
\cref{eq:positive-resolvent-rung-count}.

Evaluating the right rule at $x=\alpha$ proves
\cref{eq:positive-resolvent-work-mass}.  Also
\begin{equation*}
 \sum_j\omega_j
 =\frac2\pi(t_J-t_0)
 \leq\frac{2(1+\eta)}{\pi\eta}.
\end{equation*}
Using
$\frac{1+\sigma_j}{\alpha+\sigma_j}
 \leq 1+\frac1{\alpha+\sigma_j}$ and
\cref{eq:positive-resolvent-work-mass} proves
\cref{eq:positive-resolvent-coordinate-mass}.

The Schur complement of $M_j$ onto $F$ is
$K_F+\sigma_jI_F$.  The block inverse column is therefore
\begin{equation*}
 M_j^{-1}E_F
 =\begin{pmatrix}-Q_{AA}^{-1}Q_{AF}\\I_F\end{pmatrix}
  (K_F+\sigma_jI_F)^{-1},
\end{equation*}
which is
\cref{eq:positive-resolvent-block-realization}.  Functional calculus in
\cref{eq:positive-resolvent-spectral-sandwich}, followed by squaring,
congruence by $R_C$, and tracing proves
\cref{eq:positive-resolvent-group-sandwich}.

It remains to prove the coordinate ledger.  Let
$s_T:=(\sqrt{d_u})_{u\in T}$.  Since $Q\sqrt d=\alpha\sqrt d$ and $T$ is
principal, $Q_{TT}s_T\geq\alpha s_T$.  Put
$h_A:=-Q_{AA}^{-1}Q_{AF}s_F$.  The $A$ block of the preceding inequality and
inverse positivity give $0\leq h_A\leq s_A$.  Because $Q_{FA}\leq0$,
\begin{equation*}
 K_Fs_F=Q_{FF}s_F+Q_{FA}h_A
 \geq Q_{FF}s_F+Q_{FA}s_A
 \geq\alpha s_F.
\end{equation*}
Schur complements of Stieltjes matrices are Stieltjes, and
$K_F\preceq Q_{FF}\preceq I$, proving the other two claims.

For rung $j$, use the residual potential
$\langle s_F,p_j\rangle$.  A coordinate settlement of $u$ by
$\Delta=p_{j,u}/((K_F)_{uu}+\sigma_j)$ preserves residual nonnegativity and
decreases the potential by
\begin{equation*}
 \Delta\bigl((K_F+\sigma_jI)s_F\bigr)_u
 \geq
 \tau d_u\frac{\alpha+\sigma_j}{1+\sigma_j}.
\end{equation*}
The initial potential is
$\omega_j\langle s_F,g\rangle$.  Sum the resulting work bounds and apply
\cref{eq:positive-resolvent-coordinate-mass} to obtain
\cref{eq:positive-resolvent-coordinate-work}.
\end{proof}

\begin{corollary}[Delayed anchor debt is an exact sparse invariant]
\label{cor:shifted-resolvent-anchor-debt}
Fix one nonnegative source $g\in\mathbb R^F$.  For each rung, let
$x_j\in\mathbb R_{\geq0}^T$ be any partial state with nonnegative full residual
\begin{equation}
 d_j:=\omega_jE_Fg-M_jx_j\geq0.
 \label{eq:shifted-resolvent-debt}
\end{equation}
Then the rational frontier lift has the exact decomposition
\begin{equation}
 \mathcal U_Fr_\eta(K_F)g
 =\sum_{j=0}^{J-1}x_j
  +\sum_{j=0}^{J-1}M_j^{-1}d_j.
 \label{eq:shifted-resolvent-debt-invariant}
\end{equation}
Every term is nonnegative.  A full-system coordinate settlement preserves
\cref{eq:shifted-resolvent-debt-invariant}, touches only one row of the sparse
matrix $Q_{TT}$, and emits exterior response only through cut incidences of
that row.  No target harmonic row is needed.

More explicitly, if $d_j=(d_{j,A},d_{j,F})$, exact settlement of the current
anchor debt adds $Q_{AA}^{-1}d_{j,A}$ to the anchor state and replaces the
frontier residual by
\begin{equation}
 \overline d_{j,F}
 :=d_{j,F}-Q_{FA}Q_{AA}^{-1}d_{j,A}\geq0.
 \label{eq:shifted-resolvent-schur-debt}
\end{equation}
This is precisely the residual of the shifted Schur system
$(K_F+\sigma_jI)y_j=\omega_jg$.  Thus the positive resolvent ladder represents
a dense Schur coordinate settlement by sparse local propagation plus delayed
nonnegative anchor debt.  The remaining graph-universal obligation is to
aggregate those debt settlements at the Schur-coordinate bound
\cref{eq:positive-resolvent-coordinate-work}; it is no longer the algebraic
construction of target Gaussian harmonic rows.  Literal row-by-row cleanup
does not suffice, as \cref{prop:one-edge-shifted-debt-obstruction} proves.
\end{corollary}

\begin{proof}
By definition,
$M_jx_j+d_j=\omega_jE_Fg$.  Apply $M_j^{-1}$, sum over $j$, and use
\cref{eq:positive-resolvent-block-realization} to prove
\cref{eq:shifted-resolvent-debt-invariant}.  Each $M_j$ is Stieltjes, so its
inverse is nonnegative.  A coordinate residual settlement zeros the selected
coordinate and adds nonnegative residual to its graph neighbors; the diagonal
shift creates no new incidence.  This proves the locality and positivity
claims.

For the final statement, add $Q_{AA}^{-1}d_{j,A}$ to the anchor block.  The
anchor residual becomes zero, while the frontier residual changes by
$-Q_{FA}Q_{AA}^{-1}d_{j,A}$, proving
\cref{eq:shifted-resolvent-schur-debt}.  Inverse positivity and
$Q_{FA}\leq0$ show that this residual remains nonnegative.  Block elimination
then identifies it with the shifted Schur residual.
\end{proof}

\begin{proposition}[Literal sparse debt settlement restores the repeated
factor]
\label{prop:one-edge-shifted-debt-obstruction}
Fix $\eta\in(0,1/8)$ and consider the first rung of
\cref{thm:positive-resolvent-frontier-ladder}.  On the one-edge graph
$o\mathord{-}v$, take $A=\{o\}$ and $F=\{v\}$.  Put
\begin{equation*}
 a:=\frac{1+\alpha}{2},
 \qquad
 h:=\frac{1-\alpha}{2},
 \qquad
 c_\eta:=(1+\eta)^2\eta^2.
\end{equation*}
Then
\begin{equation}
 M_0=
 \begin{pmatrix}a&-h\\-h&a+c_\eta\alpha\end{pmatrix},
 \qquad
 \omega_0=\frac{2\eta^2}{\pi}\sqrt\alpha.
 \label{eq:one-edge-first-shifted-rung}
\end{equation}
Start from zero with residual $d^{(0)}=\omega_0e_v$, and implement the rung
only by exact coordinate residual settlements on the sparse matrix $M_0$.
For $0<\alpha\leq1/10$, reducing
$\|d^{(N)}\|_\infty$ to at most $\omega_0/4$ requires
\begin{equation}
 N\geq \frac{\log 2}{5\alpha}
 \label{eq:one-edge-sparse-shifted-lower-bound}
\end{equation}
row settlements.  Every settlement costs one adjacency-row touch.

In contrast, the Schur complement of $M_0$ onto $F$ is the scalar
$K_F+c_\eta\alpha$, so one exact Schur-coordinate settlement makes its
residual zero.  Thus the $O(1/\sqrt\alpha)$ ledger in
\cref{eq:positive-resolvent-coordinate-work} cannot be transferred to graph
work merely by replacing every dense Schur-coordinate operation with literal
sparse exact pushes and postponing the induced anchor residual.  Some
additional aggregation primitive---for example elimination, block response,
accelerated transport, or dynamic sparsification---is necessary.
\end{proposition}

\begin{proof}
At every time exactly one residual coordinate is nonzero, so every
nontrivial exact coordinate schedule is forced to alternate between $v$ and
$o$.  Define
\begin{equation*}
 \mu:=\frac{h}{a+c_\eta\alpha},
 \qquad
 \lambda:=\frac{h^2}{a(a+c_\eta\alpha)}.
\end{equation*}
Direct substitution in the exact coordinate update gives, for every $k\geq0$,
\begin{equation}
 d^{(2k)}=\omega_0\lambda^ke_v,
 \qquad
 d^{(2k+1)}=\omega_0\mu\lambda^ke_o.
 \label{eq:one-edge-shifted-alternation}
\end{equation}
For $\eta<1/8$, $c_\eta<1/50$.  If $\alpha\leq1/10$, then
$\mu\geq1/2$.  Moreover,
\begin{equation*}
 1-\lambda
 =\frac{\alpha+a c_\eta\alpha}{a(a+c_\eta\alpha)}
 \leq5\alpha\leq\frac12,
\end{equation*}
and hence
\begin{equation*}
 -\log\lambda
 \leq\frac{1-\lambda}{\lambda}
 \leq10\alpha.
\end{equation*}
If termination occurs after $2k$ settlements,
\cref{eq:one-edge-shifted-alternation} implies $\lambda^k\leq1/4$.
If it occurs after $2k+1$ settlements, $\mu\geq1/2$ implies
$\lambda^k\leq1/2$.  In either case the displayed logarithmic bound yields
\cref{eq:one-edge-sparse-shifted-lower-bound}.

Finally, block elimination gives the scalar frontier operator
$K_F+c_\eta\alpha$.  An exact coordinate settlement in a scalar system is
the exact solve, proving the separation.  The proposition concerns this
literal sparse realization only; it is not a lower bound for richer response
or block primitives.
\end{proof}

\begin{theorem}[Aggregate fixed-face debt flush with signed recurrence and
certified exterior]
\label{thm:aggregate-chebyshev-debt-flush}
Work in the exact-real algebraic-cell model.  Fix throughout the operation the
face and its partition $T=A\mathbin{\dot\cup}F$, the exposed blocks
$Q_{TT}$ and $Q_{WT}$, where
\begin{equation*}
 W:=\{v\notin T:Q_{vT}\neq0\},
\end{equation*}
and the complete positive shifted ladder
$\{(\sigma_j,\omega_j,M_j):0\leq j<J\}$ of
\cref{thm:positive-resolvent-frontier-ladder}.  No vertex moves between
$A,F,W$ during this operation, and no rung, shift, or weight changes.

For the scalar primitive, let additive nonnegative debt fragments
$d_j^{(\ell)}\in\mathbb R_{\geq0}^T$ arrive on rung $j$ and put
\begin{align}
 \bar d_j&:=\sum_\ell d_j^{(\ell)},
 &H&:=\sum_{j=0}^{J-1}\norm{\bar d_j}_2,
 &\mathcal C_{\rm frag}
 &:=\sum_{j,\ell}\operatorname{nnz}(d_j^{(\ell)}).
 \label{eq:aggregate-debt-scalar-data}
\end{align}
Thus $\mathcal C_{\rm frag}$ charges every incoming coordinate read, write,
and aggregation; repeated writes to the same aggregate coordinate are not
discarded from the ledger.  The exact correction is
\begin{equation}
 z:=\sum_{j=0}^{J-1}M_j^{-1}\bar d_j\geq0.
 \label{eq:aggregate-debt-exact-correction}
\end{equation}
If $H=0$, take $z=\widetilde z=0$.  If $H>0$, then for every declared
energy budget $\delta>0$ there are Chebyshev semi-iterates
$\widetilde z_j$, computed only by sparse $M_j$-matrix--vector products,
such that, for $\widetilde z:=\sum_j\widetilde z_j$,
\begin{equation}
 \norm{z-\widetilde z}_{Q_{TT}}\leq\delta.
 \label{eq:aggregate-debt-energy-certificate}
\end{equation}
Let $m_j$ be the number of products used on rung $j$, with $m_j=0$ when
$\bar d_j=0$ and $m_j\geq1$ otherwise, and put
$\mathfrak M_1:=\sum_jm_j$.  The allocation in the
proof gives
\begin{equation}
 \mathfrak M_1
 =O\!\left(
  J+
  \frac{J+(1+\eta)^2/\eta^2}{\sqrt\alpha}
  \log\!\left(2+\frac{H}{\delta\sqrt\alpha}\right)
 \right).
 \label{eq:aggregate-debt-chebyshev-products}
\end{equation}

For $r$ nonnegative columns, fragments
$D_j^{(\ell)}\in\mathbb R_{\geq0}^{T\times r}$ aggregate to
$\bar D_j$.  Write $\bar d_{j,q}$ for column $q$, set
\begin{equation}
 H_q:=\sum_j\norm{\bar d_{j,q}}_2,
 \qquad
 Z_q:=\sum_jM_j^{-1}\bar d_{j,q},
 \qquad
 \mathfrak M_r:=\sum_{q=1}^r\sum_jm_{j,q},
 \qquad
 \mathcal C_{\rm frag}^{(r)}
 :=\sum_{j,\ell}\operatorname{nnz}(D_j^{(\ell)}),
 \label{eq:aggregate-debt-multicolumn-data}
\end{equation}
and give column $q$ its own budget $\delta_q>0$ when $H_q>0$.
Then
\begin{align}
 \norm{Z_q-\widetilde Z_q}_{Q_{TT}}&\leq\delta_q,
 \label{eq:aggregate-debt-multicolumn-error}\\
 \mathfrak M_r
 &\leq\sum_{q:H_q>0}
 O\!\left(
  J+\frac{J+(1+\eta)^2/\eta^2}{\sqrt\alpha}
  \log\!\left(2+\frac{H_q}{\delta_q\sqrt\alpha}\right)
 \right).
 \label{eq:aggregate-debt-multicolumn-products}
\end{align}
There is no free block-matrix multiplication: its exact-cell charge is the
sum over all $r$ columns.  For signed source columns, split every fragment
before aggregation as
$D_j^{(\ell)}=D_j^{(\ell),+}-D_j^{(\ell),-}$ and run the two nonnegative
streams separately.  With budgets $\delta_q/2$ for each sign,
\begin{equation}
 Z_q:=Z_q^+-Z_q^-,
 \qquad
 \widetilde Z_q:=\widetilde Z_q^+-\widetilde Z_q^-,
 \qquad
 \norm{Z_q-\widetilde Z_q}_{Q_{TT}}\leq\delta_q.
 \label{eq:aggregate-debt-signed-split}
\end{equation}
Its product count is
$\mathfrak M_{2r}:=\sum_{q,j}(m_{j,q}^++m_{j,q}^-)$, namely the
right side of \cref{eq:aggregate-debt-multicolumn-products} applied to the
$2r$ sign streams with their actual masses $H_q^\pm$.  In particular, it is
at most twice the uniform $r$-column bound evaluated with the half-budgets
$\delta_q/2$ and the corresponding sign-stream masses.  No cancellation
between the two debt streams is credited before certification.  Its fragment coordinate is
\begin{equation*}
 \mathcal C_{\rm frag}^{(\pm)}
 :=\sum_{j,\ell}
 \left(
  \operatorname{nnz}(D_j^{(\ell),+})
  +\operatorname{nnz}(D_j^{(\ell),-})
 \right);
\end{equation*}
when one signed entry is split canonically, exactly one of its two sign cells
is nonzero, but separately supplied positive and negative fragments are all
charged.

For every exterior label and every output column, the exact normalized dual
response satisfies the mathematical interval
\begin{equation}
 -\frac{Q_{vT}\widetilde Z_q}{\sqrt{d_v}}-\delta_q
 \ \leq
 -\frac{Q_{vT}Z_q}{\sqrt{d_v}}
 \ \leq
 -\frac{Q_{vT}\widetilde Z_q}{\sqrt{d_v}}+\delta_q,
 \qquad v\notin T.
 \label{eq:aggregate-debt-boundary-interval}
\end{equation}
The same formula covers the signed split through
\cref{eq:aggregate-debt-signed-split}.  The simultaneous quantifier in
\cref{eq:aggregate-debt-boundary-interval} is a mathematical certificate;
it does not materialize a single interval center, find a violating label, or
implement \textsc{BoundaryBounds} for free.

Here is one exhaustively charged realization of the stronger, non-output-
sensitive operation \textsc{AllBoundaryFlush}.  Scan every row of $T$, then
query every newly revealed incident boundary degree in a second adjacency
batch, retain the exposed record, and process the $p$ nonnegative streams sequentially
($p=r$ for nonnegative columns and $p=2r$ for a signed split), retain the $r$
signed or unsigned output accumulators, scan the stored cut once per output
column, materialize and test both endpoints for every $v\in W$, and emit all
of those intervals plus one certificate record per output column.  Let
$e_\partial(T)$ be the number of cut incidences, let $I_{\rm frag}$ be the
number of externally ordered fragment-arrival stages (zero when fragments
are generated internally), and let $\mathcal C_{\rm frag}$ count all $p$
streams.  In the shared order, this implementation has
\begin{equation}
 \begin{aligned}
 &(C_{\rm adj},R_{\rm adj},R_{\rm int},C_{\rm pre},C_{\rm ctl},C_{\rm rec},
 C_{\rm resp},M_{\rm pers},M_{\rm tmp},C_{\rm mat},C_{\rm emit})\\
 =\bigl(&O(\operatorname{cvol}(T)),O(1),I_{\rm frag}+1,0,
 O(\mathcal C_{\rm frag}+\mathfrak M_p+pJ+r|W|),\\
 &O(\mathfrak M_p\operatorname{cvol}(T)),
 O(r(e_\partial(T)+|W|)),\\
 &O(\operatorname{cvol}(T)+\mathcal C_{\rm frag}+pJ+r|T|+|W|),
 O(|T|+|W|),\\
 &O(\mathfrak M_p|T|+r(|T|+|W|)),
 \Theta(r(|W|+1))\bigr).
 \end{aligned}
 \label{eq:aggregate-debt-eleven-vector}
\end{equation}
At most two adjacency rounds first reveal the rows and cut labels and then
query the newly known boundary degrees.  The $C_{\rm ctl}$ coordinate includes
ladder/coefficient generation,
fragment aggregation, stopping tests, interval arithmetic, dictionary
membership, and threshold/band classification.  Sparse Chebyshev products
and their vector recurrences are in $C_{\rm rec}$; cut contractions and
normalized response-query computation are in $C_{\rm resp}$.  The
$C_{\rm mat}$ coordinate explicitly retains all full-face recurrence writes,
the $r$ accumulated corrections, and the materialized interval endpoints.
The stated sequential schedule gives the displayed peak workspace; a block
schedule may use $O(p|T|)$ temporary cells instead.  The sole final
interaction is the all-boundary certificate reply.  It emits no terminal
PPR/RPPR vector.  If the caller co-requests the materialized correction
$\widetilde Z$, add $r|T|$ cells to $C_{\rm emit}$; any separate
\textsc{Finalize} stage and its output are additional.
Here ``test'' or ``validate'' means checking the certified endpoints against
the declared gate and transition band.  Exact refinement of a label whose
interval still intersects the band is a separate response/control,
materialization, and possibly interaction charge.

Thus the scalar fixed-face recurrence work is
\begin{equation}
 \widetilde O_\eta\!\left(
  \frac{\operatorname{cvol}(T)}{\sqrt\alpha}
  +\mathcal C_{\rm frag}
 \right),
 \label{eq:aggregate-debt-flush-work}
\end{equation}
and the corresponding $r$-column and signed-split bounds replace the first
term by respectively $r$ and at most $2r$ times that sparse-sweep term, with
the actual $\mathfrak M_r$ or $\mathfrak M_{2r}$ authoritative.  The cut,
interval, materialization, memory, round, and emission coordinates remain as
displayed in \cref{eq:aggregate-debt-eleven-vector}; they are not hidden by
the soft-order scalar summary.  No coefficient-bit, rounding-stability,
finite-precision, online face-mutation, output-sensitive partial-flush, or
terminal PPR/RPPR guarantee follows from this theorem.

The interior Chebyshev recurrence may be signed, but only the certified
intervals---never its intermediate signs---may control admission.  In
particular, on the one-edge instance of
\cref{prop:one-edge-shifted-debt-obstruction}, a scalar Chebyshev flush reduces
the first-rung residual by a factor four in $O(1/\sqrt\alpha)$ complete
two-row sparse sweeps.  Literal nonnegative exact-coordinate settlement still
requires $\Omega(1/\alpha)$ individual row touches.
\end{theorem}

\begin{proof}
Every shifted face matrix obeys
\begin{equation}
 \alpha I_T\preceq M_j\preceq(1+\sigma_j)I_T.
 \label{eq:aggregate-debt-shifted-spectrum}
\end{equation}
Put $\kappa_j:=(1+\sigma_j)/\alpha$ and
$\theta_j:=(\sqrt{\kappa_j}-1)/(\sqrt{\kappa_j}+1)$.  The minimax
Chebyshev residual polynomial $p_m$, normalized by $p_m(0)=1$, satisfies
\begin{equation}
 \norm{p_m(M_j)h}_2
 \leq2\theta_j^m\norm{h}_2
 \leq2\exp\!\left(-\frac{2m}{\sqrt{\kappa_j}}\right)\norm{h}_2
 \label{eq:aggregate-debt-chebyshev-residual}
\end{equation}
for every $h$.  Since $p_m(0)=1$, the polynomial
$q_{m-1}(x):=(1-p_m(x))/x$ has degree $m-1$.  Define
\begin{equation*}
 \widetilde z_j:=q_{m-1}(M_j)\bar d_j
\end{equation*}
This vector is computed by the standard Chebyshev semi-iteration without applying an
inverse.

Skip every rung with $\bar d_j=0$.  On the remaining rungs allocate
\begin{equation*}
 \delta_j:=\delta\frac{\norm{\bar d_j}_2}{H}.
\end{equation*}
Choose $m_j$ so that the right side of
\cref{eq:aggregate-debt-chebyshev-residual} is at most
$\delta_j\sqrt\alpha$.  This requires
\begin{equation*}
 m_j=O\!\left(
  1+\sqrt{\frac{1+\sigma_j}{\alpha}}
  \log\!\left(2+\frac{H}{\delta\sqrt\alpha}\right)
 \right).
\end{equation*}
For $e_j:=M_j^{-1}\bar d_j-\widetilde z_j$, the residual certificate and
\cref{eq:aggregate-debt-shifted-spectrum} give
\begin{equation*}
 \norm{e_j}_{Q_{TT}}
 \leq\norm{e_j}_{M_j}
 \leq\frac{\norm{M_je_j}_2}{\sqrt\alpha}
 \leq\delta_j.
\end{equation*}
Summing and using $\sum_j\delta_j=\delta$ proves
\cref{eq:aggregate-debt-energy-certificate}.  Each $M_j$ is Stieltjes, so
\cref{eq:aggregate-debt-exact-correction} is nonnegative even though the
Chebyshev semi-iterates need not be.

It remains to sum the product counts.  Since
$\sqrt{1+\sigma_j}=\sqrt{1+t_{j+1}^2}\leq1+t_{j+1}$ and the grid is
geometric,
\begin{equation*}
 \sum_{j=0}^{J-1}\sqrt{1+\sigma_j}
 \leq J+\sum_{k=1}^{J}t_k
 \leq J+\frac{(1+\eta)^2}{\eta^2}.
\end{equation*}
Here minimality of $J$ gives $t_J<(1+\eta)/\eta$.  This proves
\cref{eq:aggregate-debt-chebyshev-products}.  One product by $M_j$ scans the
sparse rows of $Q_{TT}$ and adds only the diagonal shift on $F$, proving
\cref{eq:aggregate-debt-flush-work} and the stated source-column charge.

Apply \cref{cor:boundary-leverage-one} on the face $T$ to the total error in
\cref{eq:aggregate-debt-energy-certificate}.  It gives, simultaneously as a
mathematical statement,
\begin{equation*}
 \frac{|Q_{vT}(z-\widetilde z)|}{\sqrt{d_v}}
 \leq\delta,
\end{equation*}
which is \cref{eq:aggregate-debt-boundary-interval}.  Applying the scalar
construction independently to every column proves
\cref{eq:aggregate-debt-multicolumn-error,eq:aggregate-debt-multicolumn-products};
the triangle inequality proves \cref{eq:aggregate-debt-signed-split}.  The
explicit scans, scalar operations, stored cells, and emitted records described
in the statement give \cref{eq:aggregate-debt-eleven-vector} coordinate by
coordinate.

Finally, on the one-edge first rung,
\cref{eq:aggregate-debt-shifted-spectrum} has condition number
$O(1/\alpha)$.  Taking $m=O(1/\sqrt\alpha)$ in
\cref{eq:aggregate-debt-chebyshev-residual} makes the Euclidean residual at
most one quarter of its initial value.  One matrix--vector product touches
both sparse rows, which is the complete-sweep accounting asserted above and
is compatible with the individual-row lower bound in
\cref{prop:one-edge-shifted-debt-obstruction}.
\end{proof}

\begin{corollary}[Geometric one-flush ledger]
\label{cor:geometric-aggregate-debt-flush}
Suppose fixed-face debt intervals end on nested faces
$T_0\subsetneq\cdots\subsetneq T_K$ satisfying
$\operatorname{cvol}(T_{k+1})\geq
(1+\gamma)\operatorname{cvol}(T_k)$ for fixed $\gamma>0$.  Aggregate every
coordinate fragment during interval $k$ on one fixed partition
$T_k=A_k\mathbin{\dot\cup}F_k$ and its fixed ladder, and invoke
\cref{thm:aggregate-chebyshev-debt-flush} once at its end, with fixed
$\eta$, a fixed number $r$ of output columns, and logarithmically bounded
certificate parameters.  Let $p=r$ for nonnegative columns and $p=2r$ for a
signed positive/negative split.  If $\mathcal C_{\rm frag}^{\Sigma}$ is the
total charged number of incoming fragment-coordinate writes, then all sparse
recurrences and fragment aggregation cost
\begin{equation}
 \widetilde O_{\eta,\gamma}\!\left(
  \frac{p\operatorname{cvol}(T_K)}{\sqrt\alpha}
  +\mathcal C_{\rm frag}^{\Sigma}
 \right).
 \label{eq:geometric-aggregate-debt-flush-work}
\end{equation}
For the named \textsc{AllBoundaryFlush} realization, expose each newly
admitted row once, keep the cut dictionary incrementally, and let
$I_{\rm frag}^{\Sigma}$ be the total number of external fragment-arrival
stages.  If $\mathfrak M_{p,k}$ is the product count at checkpoint $k$, put
\begin{align*}
 \mathcal W_{\rm rec}
 &:=\sum_{k=0}^K\mathfrak M_{p,k}\operatorname{cvol}(T_k),
 &\mathcal W_{\rm mat}
 &:=\sum_{k=0}^K\mathfrak M_{p,k}|T_k|,\\
 W_\Sigma&:=\sum_{k=0}^K|W_k|,
 &E_\Sigma&:=\sum_{k=0}^Ke_\partial(T_k),
\end{align*}
where $W_k$ is the current graph boundary.  Its complete vector is
\begin{equation}
 \begin{aligned}
 &(C_{\rm adj},R_{\rm adj},R_{\rm int},C_{\rm pre},C_{\rm ctl},C_{\rm rec},
 C_{\rm resp},M_{\rm pers},M_{\rm tmp},C_{\rm mat},C_{\rm emit})\\
 =\bigl(&O(\operatorname{cvol}(T_K)),O(K+1),
 I_{\rm frag}^{\Sigma}+K+1,0,\\
 &O(\mathcal C_{\rm frag}^{\Sigma}
      +\textstyle\sum_k\mathfrak M_{p,k}+pJ(K+1)+rW_\Sigma),
 O(\mathcal W_{\rm rec}),O(r(E_\Sigma+W_\Sigma)),\\
 &O(\operatorname{cvol}(T_K)+\mathcal C_{\rm frag}^{\max}
       +pJ+r|T_K|+\max_k|W_k|),\\
 &O(|T_K|+\max_k|W_k|),
 O(\mathcal W_{\rm mat}+r\textstyle\sum_k(|T_k|+|W_k|)),
 \Theta(r(W_\Sigma+K+1))\bigr).
 \end{aligned}
 \label{eq:geometric-aggregate-debt-eleven-vector}
\end{equation}
Here $\mathcal C_{\rm frag}^{\max}$ is the largest unflushed interval
storage.  Geometric growth bounds
$W_\Sigma,E_\Sigma,\sum_k|T_k|$ by
$O_\gamma(\operatorname{cvol}(T_K))$.  Hence the full all-boundary scans fit
the product scale, but they occur only at these checkpoints.

This is an exact-real implementation theorem only for intervals whose support
decisions are already fixed.  It gives no finite-precision or bit bound and
does not make the online gate sleep safely.  Between checkpoints the remaining
alternatives are an output-sensitive partial-flush reporter or a different
proved support-safe trace.  The next proposition rules out literal geometric
sleep even on a path.  Individual debt pieces and the one-edge
alternating-push factor are no longer separate fixed-face obstructions.
\end{corollary}

\begin{proof}
Sum the scalar-stream bounds in
\cref{eq:aggregate-debt-eleven-vector} and use the geometric-series bound
$\sum_k\operatorname{cvol}(T_k)\leq
(1+1/\gamma)\operatorname{cvol}(T_K)$.  Fragment writes are disjoint charged
events and add directly.  Every boundary label and cut incidence is bounded by
the charged volume of its current face, so the same geometric sum proves the
remaining claims and every coordinate of
\cref{eq:geometric-aggregate-debt-eleven-vector}.
\end{proof}

\begin{proposition}[A frozen notched sun obstructs eager exact slack vectors]
\label{prop:notched-sun-eager-slack-vector-obstruction}
Work in the exact-real algebraic-cell model.  Fix $n\geq5$ and
$\eta\in(0,1/8)$.  Use the notched-double-sun graph of
\cref{prop:notched-double-sun-loss-rank}, but not its prescribed changing-face
trace.  Freeze instead
\begin{equation*}
 T=A\mathbin{\dot\cup}F,
 \qquad
 A=\{u_1,\ldots,u_n\},
 \qquad
 F=\{f_1,\ldots,f_n\},
 \qquad
 W=\{w_1,\ldots,w_n\}.
\end{equation*}
Thus $\operatorname{cvol}(T)=7n+2$, the exposed cut from $T$ to $W$ has
exactly the $n$ incidences $u_iw_i$, and the normalized anchor--report block
has rank $n$.  Fix the complete ladder of
\cref{thm:positive-resolvent-frontier-ladder}.  Put
\begin{equation*}
 \vartheta:=\frac{1-\alpha}{2},
 \qquad
 c_\eta:=(1+\eta)^2\eta^2,
 \qquad
 \sigma_0=c_\eta\alpha,
 \qquad
 M:=M_0=Q_{TT}+\sigma_0E_FE_F^T.
\end{equation*}
During the fixed-face epoch, event $k$ contributes the single nonnegative
first-rung fragment $e_{f_k}$ and no fragment on another rung.  Define its
exact normalized exterior response column
\begin{equation}
 h^{(k)}:=-D_W^{-1/2}Q_{WT}M^{-1}e_{f_k}.
 \label{eq:notched-sun-frozen-response-column}
\end{equation}
Every entry of every $h^{(k)}$ is strictly positive for every
$\alpha\in(0,1)$.  For fixed $n$ and $\eta$, as
$\vartheta\downarrow0$,
\begin{align}
 h_k^{(k)}
 &=\frac{\vartheta^2}{d_{u_k}(1+c_\eta)}
   +O_{n,\eta}(\vartheta^3),
 \label{eq:notched-sun-frozen-matching-response}\\
 h_i^{(k)}&=O_{n,\eta}(\vartheta^3),
 \qquad i\neq k.
 \label{eq:notched-sun-frozen-off-response}
\end{align}

Use the strict gate and transition band
\begin{equation}
 g:=\vartheta^{5/2},
 \qquad
 m:=\frac12g,
 \label{eq:notched-sun-frozen-gate-band}
\end{equation}
and initialize all report demands at zero.  There is
$\alpha_{n,\eta}^{\rm eager}<1$ such that, whenever
$\alpha\in(\alpha_{n,\eta}^{\rm eager},1)$, the cumulative response
\begin{equation*}
 y^{[k]}:=\sum_{\ell=1}^kh^{(\ell)}
\end{equation*}
has the exact delta-only trace
\begin{equation}
 \{i:y_i^{[k]}>g\}=\{1,\ldots,k\},
 \qquad
 y_i^{[k]}\leq g-m\quad(i>k).
 \label{eq:notched-sun-frozen-delta-trace}
\end{equation}
Thus event $k$ emits exactly the newly crossing label $w_k$, and every other
unreported label remains outside the transition band.  This is a fixed-face
debt/query trace.  It is not an RPPR load, KKT admission chronology, or
support-evolution claim.

Now name the representation and query policy
\textsc{EagerExactSlack}.  For every certified nonnegative stream and every
not-yet-emitted label, it stores one separately addressed exact cell equal to
the current response, eagerly adds the new exact column
\cref{eq:notched-sun-frozen-response-column} to all of those cells, tests
each resulting slack against \cref{eq:notched-sun-frozen-gate-band}, and only
then returns the delta labels plus a certificate that all unreported labels
are at most $g-m$.  Since every response-column entry is nonzero, event $k$
must read and overwrite at least $n-k+1$ exact response cells per stream.
Consequently its epoch-internal exact response-state charge is
\begin{equation}
 p\sum_{k=1}^n(n-k+1)
 =\frac{p n(n+1)}2,
 \label{eq:notched-sun-eager-state-write-lower-bound}
\end{equation}
where $p=r$ for $r$ nonnegative output columns and $p=2r$ when $r$ signed
columns are certified by positive/negative splitting.  The scalar charge is
the case $p=r=1$.

The fragment and output interfaces can be kept equally explicit.  For $r$
nonnegative columns, supply one unit fragment in every column at every event,
so $\mathcal C_{\rm frag}^{\Sigma}=rn$.  For signed column $q$, separately
supply $2e_{f_k}$ to its positive stream and $e_{f_k}$ to its negative
stream.  Their signed difference is again $e_{f_k}$, but certification uses
both streams and credits no cancellation before their two error bounds are
combined; hence $\mathcal C_{\rm frag}^{\Sigma}=2rn$.  In both cases the
external delta interface emits the $r$ records $(q,w_k)$ and one
all-remaining-safe certificate per output column at event $k$, for
$C_{\rm emit}=\Theta(rn)$.  The signed example is deliberately chosen so the
final combined gate has the same trace while the two certified streams both
change densely.

There is a literal exact-cell implementation matching the dense scale of
\cref{eq:notched-sun-eager-state-write-lower-bound}.  Eliminate the $F$
leaves from $M$.  The remaining anchor matrix is a weighted cycle with one
chord and a diagonal correction.  Break the cycle and chord into a constant-
rank update of a tridiagonal path factor; exact tridiagonal elimination plus
the constant-size Schur correction uses $\Theta(n)$ cells and gives one exact
source solve, leaf recovery, and complete $W$ contraction in $\Theta(n)$
cells.  Process the $p$ streams sequentially and retain all exact slack cells.
Including one initial scan of the frozen face, a second degree-query round for
the newly exposed report labels, all $n$ fragment/query interactions, ladder
generation, membership, exact solves, slack tests, state writes, and delta
replies, its complete vector is
\begin{equation}
 \begin{aligned}
 &(C_{\rm adj},R_{\rm adj},R_{\rm int},C_{\rm pre},C_{\rm ctl},C_{\rm rec},
 C_{\rm resp},M_{\rm pers},M_{\rm tmp},C_{\rm mat},C_{\rm emit})\\
 =\bigl(&\Theta(n),2,n,\Theta(n+J),
 \Theta(\mathcal C_{\rm frag}^{\Sigma}+pn^2+pJ),0,
 \Theta(pn^2),\\
 &\Theta(pn+n+J),O(n+p),\Theta(pn^2),\Theta(rn)\bigr).
 \end{aligned}
 \label{eq:notched-sun-eager-eleven-vector}
\end{equation}
Here $C_{\rm resp}$ charges every exact factor application and cut
contraction, while $C_{\rm mat}$ separately charges the materialized response
columns and overwritten exact slack-vector cells.  Persistent memory contains
the cyclic factor, ladder, membership state, and all $p$ slack vectors; the
displayed scratch bound uses a sequential stream schedule.  No terminal
PPR/RPPR vector is emitted.

One aggregate call after all fragments still has the scalar recurrence bound
of \cref{eq:aggregate-debt-flush-work}; for the present near-one-$\alpha$
family it is $\widetilde O_\eta(n+\mathcal C_{\rm frag}^{\Sigma})$ before the
separately charged all-boundary realization.  Any beginning or ending
\textsc{AllBoundaryFlush} remains charged by
\cref{eq:geometric-aggregate-debt-eleven-vector} and may still be scheduled
only at geometric face checkpoints.  It does not pay the $n$ intervening
delta replies.  Therefore \textsc{EagerExactSlack} fails the Round-013 GO
test: it spends $\Theta(pn^2)$ between checkpoints for only $\Theta(rn)$
delta/certificate output.  This STOP is only for the named eager, separately
addressed exact-vector representation and query policy.  It does not refute
an implicit cyclic transfer, a packed coded query, a scale-truncated state,
on-demand exact validation, a sparsifier, or any output-sensitive partial
flush.  No coefficient-bit, rounding-stability, finite-precision, or word-RAM
conclusion follows.
\end{proposition}

\begin{proof}
Every $w_i$ has degree one and is adjacent only to $u_i$, so
\begin{equation}
 h_i^{(k)}
 =\frac{\vartheta}{\sqrt{d_{u_i}}}
   (M^{-1})_{u_i f_k}.
 \label{eq:notched-sun-frozen-response-entry}
\end{equation}
The frozen graph on $T$ is connected and $M$ is an irreducible SPD Stieltjes
matrix.  Its inverse is strictly positive, proving the first claim.

At $\vartheta=0$, $\alpha=1$ and
\begin{equation*}
 M(0)=\operatorname{diag}(I_A,(1+c_\eta)I_F).
\end{equation*}
The entries of $M^{-1}$ are analytic for all sufficiently small
$\vartheta$.  The only first-order anchor--petal entry in column $f_k$ is
\begin{equation*}
 (M^{-1})_{u_k f_k}
 =\frac{\vartheta}{\sqrt{d_{u_k}}(1+c_\eta)}
  +O_{n,\eta}(\vartheta^2),
\end{equation*}
whereas $(M^{-1})_{u_i f_k}=O_{n,\eta}(\vartheta^2)$ for $i\neq k$.
Substitution in \cref{eq:notched-sun-frozen-response-entry} proves
\cref{eq:notched-sun-frozen-matching-response,eq:notched-sun-frozen-off-response}.

The finite set of matching expansions implies, after shrinking
$\vartheta>0$ if needed, that every $h_k^{(k)}>\vartheta^{5/2}$.  The finite
set of off-matching expansions similarly gives
\begin{equation*}
 \sum_{\ell\leq k}h_i^{(\ell)}
 \leq\frac12\vartheta^{5/2}
 \qquad(i>k).
\end{equation*}
All terms are positive.  Induction on $k$ now proves
\cref{eq:notched-sun-frozen-delta-trace}, including strictness at the gate
and separation from the transition band.

Under \textsc{EagerExactSlack}, every not-yet-emitted stored cell changes by
the strictly positive quantity $h_i^{(k)}$ (or its positive stream multiple).
An exact separately addressed cell cannot retain its old value, so the policy
performs at least one response-state read and one overwrite for every such
cell.  Summing the $n-k+1$ live cells proves
\cref{eq:notched-sun-eager-state-write-lower-bound}.  The fragment and output
counts follow directly from their declared interfaces.

For completeness, eliminating one degree-one $F$ row changes only its
incident anchor diagonal.  The remaining graph is the anchor cycle plus its
single chord.  Removing the wrap edge and chord leaves a tridiagonal path;
restoring them is a constant-rank symmetric update.  Exact path elimination,
a constant number of path solves, and a constant-size Schur solve therefore
construct the factor and apply it in $\Theta(n)$ exact cells.  Recovering the
leaves and contracting the $n$ exposed report incidences is also
$\Theta(n)$.  Counting these operations and the eager state invariant gives
\cref{eq:notched-sun-eager-eleven-vector}.  The final scope paragraph follows
because the lower bound uses the named invariant, not an information or
oracle restriction shared by the excluded alternatives.
\end{proof}
