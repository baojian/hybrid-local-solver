\section{Rung one: the monotone classes pay $\Theta(1/(\alpha\epsppr))$}
\label{sec:ladder-monotone-lower-bound}

The whole rung rests on one scalar, the \emph{relative unsettled value at the
seed}:
\begin{equation}
    \Psi:=\frac{\pi_{c}-z_{c}}{\pi_{c}}=\frac{e_{c}}{\pi_{c}},
    \qquad
    \Psi(0)=1 .
    \label{eq:ladder-potential}
\end{equation}
In the mass scale of \eqref{eq:ladder-mass-dictionary} this is
$\Psi=\pr(r_{\mathrm{mass}})_{c}/\pi_{c}$, the potential of the source
package \texttt{i2d}. The $z$-scale form used here makes its two properties
exact and graph-free, and --- see \Cref{rem:ladder-phi-eliminated} --- removes
an instance constant from the general-graph statement.

\begin{lemma}[Locality of the potential; \statusproveddraft]
\label{lem:ladder-locality}
Every $\push(u,\delta)$ changes $\Psi$ by $-\delta/\pi_{c}$ if $u=c$ and by
zero otherwise. In particular $\Psi$ is invariant under every operation not
based at the seed, for either sign of $\delta$.
\end{lemma}

\begin{proof}
By \eqref{eq:ladder-primitive}, $z_{c}$ changes only at operations based at
$c$, and then by exactly $\delta$; $\pi_{c}$ is a constant of the instance.
\end{proof}

\begin{lemma}[Multiplicative step under $r\ge0$; \statusproveddraft]
\label{lem:ladder-kappa-step}
Suppose $r\ge0$. Then $r_{c}\le(\galpha/\pi_{c})\,e_{c}$. Consequently, with
\begin{equation}
    \kappa:=\frac{\galpha}{\pi_{c}},
    \qquad\text{on the star}\qquad
    \kappa_{+}=\frac{4\alpha}{(1+\alpha)^{2}}<1,
    \label{eq:ladder-kappa}
\end{equation}
every $\Mpm$-feasible operation multiplies $\Psi$ by at least $1-\kappa$, and
$\Psi$ decreases only at seed-based operations with $\delta>0$. Moreover
$\Psi\ge0$ throughout.
\end{lemma}

\begin{proof}
Since $e=H^{-1}r$ and $r\ge0$,
\[
    e_{c}=\sum_{u}(H^{-1})_{cu}r_{u}
    \ \ge\ (H^{-1})_{cc}\,r_{c}
    \ =\ \frac{\pi_{c}}{\galpha}\,r_{c},
\]
discarding nonnegative off-diagonal terms by \ref{it:ladder-f2} and using
\ref{it:ladder-f3}. Feasibility in $\Mpm$ requires $r_{c}\ge0$ after the
operation, hence $\delta\le r_{c}$ at a seed-based operation, so by
\Cref{lem:ladder-locality}
\[
    -\Delta\Psi=\frac{\delta}{\pi_{c}}\le\frac{r_{c}}{\pi_{c}}
    \le\frac{\galpha}{\pi_{c}}\cdot\frac{e_{c}}{\pi_{c}}=\kappa\Psi .
\]
Operations with $\delta\le0$, or based away from $c$, do not decrease $\Psi$
by \Cref{lem:ladder-locality}. Nonnegativity of $\Psi$ is
$e=H^{-1}r\ge0$, which is \ref{it:ladder-f2}. On the star,
$\kappa_{+}=\galpha/\pi_{c}=\bigl(2\alpha/(1+\alpha)\bigr)\big/
\bigl((1+\alpha)/2\bigr)=4\alpha/(1+\alpha)^{2}$.
\end{proof}

\begin{lemma}[Terminal smallness of the potential; \statusproveddraft]
\label{lem:ladder-terminal}
Suppose a run stops with output $\widehat z=z+Mr$, $M\in\OB$, $r\ge0$, and
$\errop(\widehat z)\le\epsppr$. Then:
\begin{enumerate}[label=\textup{(\roman*)},leftmargin=3em]
    \item\label{it:ladder-terminal-general} on every graph,
        \[
            \norm{r}_{1}\Bigl(\frac{1}{\galpha}-B\Bigr)\le\epsppr\vol(V),
            \qquad
            \Psi_{T}\le\frac{\epsppr d_{c}+B\norm{r}_{1}}{\pi_{c}};
        \]
    \item\label{it:ladder-terminal-star} on the star,
        \[
            \Psi_{T}\le\frac{\norm{r}_{1}}{\galpha}
            \le\frac{\epsppr\vol(V)}{1-\galpha B}.
        \]
\end{enumerate}
\end{lemma}

\begin{proof}
\ref{it:ladder-terminal-general} Summing $(\pi-z-Mr)_{u}\le\epsppr d_{u}$
over $u$ and using \ref{it:ladder-f1} for $\sum_{u}e_{u}=\norm{r}_{1}/\galpha$
together with $\inner{\one}{Mr}\le B\norm{r}_{1}$ gives the first display.
For the second, the $u=c$ coordinate of the guarantee gives
$e_{c}\le\epsppr d_{c}+(Mr)_{c}\le\epsppr d_{c}+B\norm{r}_{1}$, since every
entry of row $c$ of $M$ is at most its own column mass.
\ref{it:ladder-terminal-star} On the star, reversibility gives
$\pr(e_{l})_{c}=m\pi_{l}=(1-\alpha)/2$, so in the mass-scale form
$\Psi=\inner{\phi}{r_{\mathrm{mass}}}$ the weights satisfy
$\phi_{l}=\calpha<1=\phi_{c}$; hence
$\Psi_{T}\le\norm{r_{\mathrm{mass}}}_{1}=\norm{r}_{1}/\galpha$, and the first
display of \ref{it:ladder-terminal-general} bounds $\norm{r}_{1}$.
\end{proof}

\begin{theorem}[Class lower bound for $\Mpm$, hence for $\Mpos$;
\statusproveddraft]
\label{thm:ladder-monotone-lower-bound}
Take the centre-seeded star \eqref{eq:ladder-hard-instance} with
$\epsppr\le1/16$ and $\alpha\in(0,1)$. Every member of $\Mpm$ --- arbitrary
adaptive, randomized or batched policy, arbitrary stopping rule --- whose
output $z+Mr$ with $M\in\OB$ and $\galpha B\le1-\epsppr\vol(V)-\tau$ for some
$\tau>0$ satisfies $\errop\le\epsppr$, performs
\begin{equation}
    N_{c}^{+}\ \ge\ \frac{\ln(1/\Psi_{T})}{-\ln(1-\kappa_{+})}
    \quad\text{seed operations with }\delta>0,
    \qquad
    \Work\ \ge\ m\,N_{c}^{+},
    \label{eq:ladder-monotone-count}
\end{equation}
with $\Psi_{T}$ bounded by \Cref{lem:ladder-terminal}\ref{it:ladder-terminal-star}.
For the plain output $\widehat z=z$ one has
$\Psi_{T}\le2\epsppr m/(1+\alpha)\le1/4$ and
\begin{equation}
    N_{c}^{+}\ \ge\
    \frac{\ln\bigl((1+\alpha)/(2\epsppr m)\bigr)}{-\ln(1-\kappa_{+})},
    \qquad
    \Work\ \ge\ m\,N_{c}^{+}.
    \label{eq:ladder-monotone-count-plain}
\end{equation}
\end{theorem}

\begin{proof}
$\Psi(0)=1$. By \Cref{lem:ladder-locality,lem:ladder-kappa-step}, after any
operation sequence containing $N_{c}^{+}$ positive seed operations,
$\Psi\ge(1-\kappa_{+})^{N_{c}^{+}}$: each positive seed operation retains at
least the factor $1-\kappa_{+}$, and no other operation decreases $\Psi$.
Combining with the terminal bound and taking logarithms gives
\eqref{eq:ladder-monotone-count}; each seed operation is charged $d_{c}=m$.
For $B=0$, \Cref{lem:ladder-terminal}\ref{it:ladder-terminal-general} at the
single coordinate $c$ gives $\Psi_{T}\le\epsppr m/\pi_{c}
=2\epsppr m/(1+\alpha)$ directly, which is at most $1/4$ because
$\epsppr m\le1/8$.
\end{proof}

\begin{corollary}[Explicit constants; \statusproveddraft]
\label{cor:ladder-monotone-constants}
Using $-\ln(1-\kappa)\le\kappa/(1-\kappa)$ and the identity
$(1-\kappa_{+})(1+\alpha)^{2}=(1-\alpha)^{2}$, the plain-output case of
\Cref{thm:ladder-monotone-lower-bound} gives
\begin{equation}
    \Work\ \ge\ \frac{\ln4\,(1-\alpha)^{2}}{64\,\alpha\,\epsppr}
    \quad\text{for all }\epsppr\le\tfrac{1}{16},
    \qquad
    \Work\ \ge\ \frac{\ln\bigl(4(1+\alpha)\bigr)(1-\alpha)^{2}}
                     {32\,\alpha\,\epsppr}
    \label{eq:ladder-monotone-constants}
\end{equation}
the second on the dyadic grid where $\epsppr m=1/8$ exactly. Numerically
$\Work\cdot\alpha\cdot\epsppr\ge0.0190$ in general and $\ge0.0397$ on the
dyadic grid for $\alpha\le1/16$, approaching $\ln4/64=0.0217$ and
$\ln4/32=0.0433$ as $\alpha\to0$.
\end{corollary}

\begin{theorem}[General graphs, with no instance constant;
\statusproveddraft]
\label{thm:ladder-monotone-general}
For any connected $G$, any seed $v$, and the plain output $\widehat z=z$,
every member of $\Mpm$ achieving $\errop\le\epsppr$ performs
\begin{equation}
    N_{v}^{+}\ \ge\
    \frac{\ln\bigl(\pi_{v}/(\epsppr d_{v})\bigr)}
         {-\ln\bigl(1-\galpha/\pi_{v}\bigr)},
    \qquad
    \Work\ \ge\ d_{v}N_{v}^{+},
    \label{eq:ladder-monotone-general}
\end{equation}
which is $\Omega\bigl((\pi_{v}/\alpha)\ln(\pi_{v}/(\epsppr d_{v}))\bigr)$
seed-operation mass for small $\alpha$, and is informative exactly when
$\pi_{v}>\epsppr d_{v}$, i.e.\ when the seed itself lies in
$S_{\epsppr}$.
\end{theorem}

\begin{proof}
\Cref{lem:ladder-locality}, \Cref{lem:ladder-kappa-step} and
\Cref{lem:ladder-terminal}\ref{it:ladder-terminal-general} used nothing about
the star.
\end{proof}

\begin{remark}[The instance constant $\Phi$ is gone from the main line]
\label{rem:ladder-phi-eliminated}
The mass-scale ancestor of \Cref{thm:ladder-monotone-general} (\texttt{i2d}
Proposition~2.6) carried the instance constant
$\Phi=\max_{u}d_{u}\pi_{u}/(d_{v}\pi_{v})$, because it bounded $\Psi_{T}$ by
$\Phi\norm{r_{\mathrm{mass}}}_{1}$. The $z$-scale restatement uses only the
seed coordinate of the guarantee and is therefore $\Phi$-free. What remains
of that question --- the general-graph analogue of
\Cref{lem:ladder-terminal}\ref{it:ladder-terminal-star}, needed for outputs
$z+Mr$ with $B$ near the escape threshold --- is
\Cref{open:ladder-phi} in \Cref{sec:ladder-manuscript-relation}.
\end{remark}

\begin{remark}[The separating axiom is $r\ge0$, not the sign of the step]
\label{rem:ladder-axiom}
\Cref{lem:ladder-kappa-step} is the only place where $r\ge0$ enters, and it
enters twice: it caps the feasible seed step by $\delta\le r_{c}$, and it
converts $r_{c}$ into $\Psi$, so that the seed cannot hold residual out of
proportion to its own unsettled value. Anti-pushes only raise $\Psi$ and must
then be repaid at the same multiplicative rate, which is why $\Mpm$ gains
nothing over $\Mpos$; the pump adversary of \texttt{i2d} measures a factor
$3.7$ \emph{worse}, never better. Over-relaxation with $\omega>1$ breaks
$\delta\le r_{c}$ precisely by letting $r_{c}$ go negative, and
\Cref{sec:ladder-signed-lower-bound} shows that this is worth exactly
$\sqrt\alpha$.
\end{remark}
