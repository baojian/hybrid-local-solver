The half-ratio interface itself has a simpler two-step form.  Combining it
with the finite-$q$ source split makes the remaining uniform estimates
explicit.

\begin{lemma}[Two-step cone and folded source ledger; proved reduction]
\label{lem:terminal-modal-static-two-step-ledger}
The local conditions
\eqref{eq:terminal-modal-static-local-half-ratios} are equivalent to
\begin{align}
 d_{m-1}+\frac14d_m&\ge0,
 &d_{m-2}-\frac18d_m&\ge0,
 \label{eq:terminal-modal-static-two-step-frontier}\\
 d_j-\frac14d_{j+2}&\ge0,
 &&0\le j\le m-3.
 \label{eq:terminal-modal-static-two-step-cone}
\end{align}

There is an exact source formula for the rows $0\le j\le m-6$.  Put
$\overline D_n=\bar c_n-\bar c_{n-1}/2$, where $\bar c_n$ is the rescaled
correction from \eqref{eq:terminal-modal-finite-rescaling}, and let
$L_k^\circ$ be the positive correction kernel from
\eqref{eq:terminal-modal-correction-positive-green}, with $L_0^\circ=0$.
For a vector $z$ on the reflected line, write
\[
 \mathcal Q_jz=z(j)-\frac14z(j+2),
 \qquad
 \mathcal R_{m,j,n}z
 =2\mathcal Q_jB_0
 (L_{m-1-n}^\circ-L_{m-n}^\circ)z.
\]
Let $e_a^{\rm ev}$ be the even pair of point masses at $\pm a$ and set
\[
 q_n^{\rm ev}=e_{n-2}^{\rm ev}+2e_{n-1}^{\rm ev}-3e_n^{\rm ev}.
\]
At $n=2$ the coincident seed masses are combined, as in
\eqref{eq:terminal-modal-finite-source-packets}.  If $\mathcal B_{m,j}$ is
the value of $2\mathcal Q_jB_0(\overline D_{m-1}-\overline D_m)$ for the
exact initial pair \eqref{eq:terminal-modal-finite-initial-values} but the
constant inputs $E_n=-3/10$, $F_n=9/40$, then
\begin{equation}
 \frac{d_j-d_{j+2}/4}{q^3\delta^{m-1}}
 =\mathcal B_{m,j}
 +\sum_{n=2}^{m-1}\frac{\varepsilon_n}{4}
   \mathcal R_{m,j,n}q_n^{\rm ev}
 +\sum_{n=2}^{m-1}\mu_n
   \mathcal R_{m,j,n}e_n^{\rm ev}.
 \label{eq:terminal-modal-static-folded-source-ledger}
\end{equation}
Moreover, writing $\mu_n=-q\nu_n$, the exact scalar source obeys
\begin{equation}
 \frac25<\nu_n<\frac{43}{75}
 \qquad(2\le n\le m-1).
 \label{eq:terminal-modal-static-mass-window}
\end{equation}
The following three folded-kernel inequalities would therefore prove
\eqref{eq:terminal-modal-static-two-step-cone} on these rows with strict
margin:
\begin{align}
 \mathcal B_{m,j}&\ge-\frac q{64},
 \label{eq:terminal-modal-static-base-kernel-target}\\
 \sum_{n=2}^{m-1}\frac{\varepsilon_n}{4}
  \mathcal R_{m,j,n}q_n^{\rm ev}&\ge-\frac q{16},
 \label{eq:terminal-modal-static-derivative-kernel-target}\\
 -\sum_{n=2}^{m-1}\nu_n
  \mathcal R_{m,j,n}e_n^{\rm ev}&\ge\frac3{16}.
 \label{eq:terminal-modal-static-mass-kernel-target}
\end{align}
The base bound and the five direct frontier rows left by $j>m-6$ remain open;
the mass bound is proved below.  The derivative bound with the displayed
$1/16$ constant is in fact false, as proved at the end of this section.  A
weaker lower bound $-q/8$ would still leave a positive aggregate margin, but
that replacement also remains open.  The formula
\eqref{eq:terminal-modal-static-folded-source-ledger} is an exact reduction,
not evidence that any summand has a pointwise sign.
\end{lemma}

\begin{proof}
By definition,
\[
 E_1=\frac{d_{m-1}+d_m/4}{q^3},\qquad
 F_2-\frac12E_1=\frac{d_{m-2}-d_m/8}{q^3}.
\]
For $r\ge3$, with $j=m-r$, the adjacent terms cancel exactly:
\[
 F_r-\frac12F_{r-1}
 =\frac{d_j-d_{j+2}/4}{q^3}.
\]
This proves the equivalence.

Equation \eqref{eq:terminal-modal-finite-rescaled-target} gives
\[
 \frac{\mathcal T_m}{q^3}
 =\delta^{m-2}(\overline D_{m-1}-\overline D_m).
\]
Apply the interior identity
\eqref{eq:terminal-modal-static-temporal-cone-interior}, followed by
$\mathcal Q_j$.  In the exact source split
\eqref{eq:terminal-modal-finite-source-packets}, a source inserted at time
$n$ reaches $\overline D_N$ through $L_{N-n}^\circ$.  Thus its temporal
difference at $N=m$ is
$(L_{m-1-n}^\circ-L_{m-n}^\circ)$, including $n=m-1$ because
$L_0^\circ=0$.  Superposition of the exact initial/constant-input trajectory,
the derivative packet, and the newest-row mass proves
\eqref{eq:terminal-modal-static-folded-source-ledger}.  The range
$j\le m-6$ is precisely where both $d_j$ and $d_{j+2}$ use the proved
interior/seed smoothing identities.

It remains to prove the lower half of
\eqref{eq:terminal-modal-static-mass-window}; the upper half is
\eqref{eq:terminal-modal-finite-mu-bound}.  From
\eqref{eq:terminal-modal-finite-mu-M},
\[
 \nu_n=\delta^{-(n-1)}\mathfrak m_n,qquad
 \mathfrak m_n=
 \frac{\delta\wp_{n-1}}4R_n
 +\frac{\delta s_n+2\eta}{10\eta},qquad
 R_n=\frac{2\eta-s_n}{q}.
\]
Here $u=\chi^n\le\chi^2$ and
\eqref{eq:terminal-modal-finite-mass-ratio} gives $R_n\ge q$.  Indeed,
after setting $u=\chi^2$, the difference from the required lower bound
factors as
\[
 \frac{2(1-\chi^4)}{1+\chi^4}-(4q-2q^3)
 =\frac{2q(q-1)(q+1)(q^4+5q^2-2)}{q^4+6q^2+1}>0.
\]
Since $s_n\le1$,
\begin{align*}
 \mathfrak m_n-\frac25
 &=R_n\left(\frac{\delta\wp_{n-1}}4-\frac q{10\eta}\right)
   -\frac{qs_n}{10\eta}\\
 &\ge q\left(\frac{\delta\wp_{n-1}}4-\frac{1+q}{10\eta}\right)>0.
\end{align*}
The last inequality uses $\wp_{n-1}>9/5$ from
\cref{lem:terminal-modal-frontier-bounds},
$(1+q)/(10\eta)=1/(5\delta)$, and $9\delta^2>4$.
Finally $\delta^{-(n-1)}>1$, proving $\nu_n>2/5$.

If the three displayed kernel targets hold, the right side of
\eqref{eq:terminal-modal-static-folded-source-ledger} is at least
$-q/64-q/16+3q/16=7q/64>0$.  This proves the stated conditional
implication while leaving the kernel estimates themselves open.
\end{proof}

\begin{lemma}[Folded mass kernel; proved with an exact finite certificate]
\label{lem:terminal-modal-static-mass-kernel}
For every $m\ge64$ and $0\le j\le m-6$, the mass term in
\eqref{eq:terminal-modal-static-folded-source-ledger} obeys
\begin{equation}
 -\sum_{n=2}^{m-1}\nu_n
  \mathcal R_{m,j,n}e_n^{\rm ev}>\frac3{16}.
 \label{eq:terminal-modal-static-mass-kernel}
\end{equation}
Thus \eqref{eq:terminal-modal-static-mass-kernel-target} is unconditional.
\end{lemma}

\begin{proof}
We first expose the exact one-dimensional atoms.  Write
\[
 a_k(r)=\bigl[B_0(L_{k-1}^\circ-L_k^\circ)\bigr](r).
\]
Pascal's identity applied to
\eqref{eq:terminal-modal-finite-l-coefficients} gives
\begin{equation}
 a_k(r)=-2^{-(k+1)}\binom{k-1}{|r|-1}
 \quad(1\le |r|\le k),
 \label{eq:terminal-modal-static-a-atom}
\end{equation}
with $a_k(0)=0$ for $k\ge2$ and $a_1(0)=-1/2$.  We use the convention that a
binomial coefficient is zero outside its natural range.  The exceptional
central atom is never encountered below because $D:=m-j\ge6$.

Reindex the source time by $k=m-n$.  Reflection and
\eqref{eq:terminal-modal-static-a-atom} give the exact split
\begin{align}
 \mathcal R_{m,j,m-k}e_{m-k}^{\rm ev}
 &=R_D^+(k)+R_{2m-D}^-(k),
 \label{eq:terminal-modal-static-R-split}\\
 R_D^\pm(k)
 &=2^{-k}\left[
 -\binom{k-1}{|k-D|-1}
 +\frac14\binom{k-1}{|k-D\pm2|-1}
 \right].
 \label{eq:terminal-modal-static-R-atoms}
\end{align}
Put
\[
 S_{m,j}=\sum_{n=2}^{m-1}\mathcal R_{m,j,n}e_n^{\rm ev},
 \qquad
 P_{m,j}=\sum_{n=2}^{m-1}
 \left(\mathcal R_{m,j,n}e_n^{\rm ev}\right)_+,
 \qquad
 E_{m,j}=S_{m,j}+\frac12.
\]

The signed sum has a useful maximum principle.  Let $M=m-3$,
$p_a=2^{-M}\binom Ma$, $F_t=\sum_{a\le t}p_a$, and
\[
 A_j=\sum_{a=0}^jp_a(-1/2)^{j-a}.
\]
Direct summation of \eqref{eq:terminal-modal-static-R-atoms} gives, for
$1\le j\le m-4$,
\begin{align}
 E_{m,j}={}&-\frac12F_{j-4}
 +\frac12p_{j-2}+\frac38p_{j-1}+\frac14p_j
 -\frac3{32}p_{j+1}-\frac1{16}p_{j+2}\notag\\
 &-\frac5{64}A_j-\frac58(-1/2)^{m+j}.
 \label{eq:terminal-modal-static-E-binomial}
\end{align}
Here $F_t=0$ and $p_t=0$ for negative indices.  The binomial recurrences for
$p$ and $F$, together with
$A_j^{(M+1)}=(A_j^{(M)}+A_{j-1}^{(M)})/2$, make every term telescope to
\begin{align}
 E_{m+1,j}&=\frac12(E_{m,j}+E_{m,j-1}),
 &&2\le j\le m-4,\notag\\
 E_{m+1,1}&=\frac12(E_{m,1}+E_{m,0})-2^{-m},\qquad
 E_{m+1,m-3}
 =\frac12(E_{m,m-3}+E_{m,m-4})-\frac18.
 \label{eq:terminal-modal-static-E-smoothing}
\end{align}
The parity term is essential here: its values at $(M+1,j)$, $(M,j)$, and
$(M,j-1)$ satisfy the same identity.  At the seed, direct substitution gives
\[
 E_{m,0}
 =-\frac{3M}{32\,2^M}
  -\frac{\binom M2}{16\,2^M}
  -\frac5{64\,2^M}
  -\frac58(-1/2)^m+\frac1{2^{m-2}}<0
 \qquad(m\ge64).
\]

The exact dyadic certificate in \texttt{verify.py} checks at $m=64$ that
\begin{equation}
 \max_{\substack{0\le j\le m-6\\\text{or }m-j\in\{4,5\}}}E_{64,j}
 =\frac{102332589933520061}{2^{62}}<\frac1{45},
 \label{eq:terminal-modal-static-E-base}
\end{equation}
attained at $j=28$.  The two auxiliary rows $D=4,5$ and the favorable
$-1/8$ boundary defect in
\eqref{eq:terminal-modal-static-E-smoothing} propagate the moving upper
boundary.  To anchor that defect, direct substitution in
\eqref{eq:terminal-modal-static-R-atoms}, including the exceptional central
atom in \eqref{eq:terminal-modal-static-a-atom}, gives
\[
 E_{m,m-3}
 =-\frac14+\frac1{2^M}
 \left\{\frac12\binom M3+\binom M2+\frac78M+\frac34\right\}
 <-\frac18.
\]
The last inequality holds at $M=61$, and every term in braces grows by a
factor less than two when $M$ increases.  Induction therefore proves
\begin{equation}
 E_{m,j}<\frac1{45}
 \qquad(m\ge64,\ 0\le j\le m-6).
 \label{eq:terminal-modal-static-E-uniform}
\end{equation}
The same exact base check gives $E_{64,64-D}<-1/10$ for $3\le D\le19$.
Equation \eqref{eq:terminal-modal-static-E-smoothing}, now read at fixed
$D\ge5$, averages the old $D-1$ and $D$ rows.  The explicit $D=3$ bound and
the favorable $D=4$ defect anchor the induction, so
\begin{equation}
 E_{m,m-D}<-\frac1{10}
 \qquad(m\ge64,\ 6\le D\le19).
 \label{eq:terminal-modal-static-E-frontier}
\end{equation}

It remains to control the positive lobe.  Define the infinite one-branch
lobe masses
\[
 P_D^\pm=\sum_{k\ge1}(R_D^\pm(k))_+.
\]
Since $(x+y)_+\le x_++y_+$, truncating the two branches in
\eqref{eq:terminal-modal-static-R-split} gives
\begin{equation}
 P_{m,j}\le P_D^++P_{2m-D}^-.
 \label{eq:terminal-modal-static-P-branches}
\end{equation}
The atom ratio in \eqref{eq:terminal-modal-static-R-atoms} identifies each
positive lobe exactly.  An integer-only dyadic enumeration for
$6\le D\le511$, included in \texttt{verify.py}, proves
\begin{align}
 P_D^+&\le\frac{113}{1024} &&(D\ge6),\notag\\
 P_D^+&<\frac9{512} &&(D\ge20),\notag\\
 P_D^-&<\frac1{2048} &&(D\ge64)
 \label{eq:terminal-modal-static-P-one-branch}
\end{align}
through that finite range.  This is an exact certificate over the closed
binomial atoms, not a floating screen.

For completeness, a short analytic tail covers every $D\ge512$.  If a
positive atom in \eqref{eq:terminal-modal-static-R-atoms} uses the shifted
binomial index $a$ at horizon $N=k-1$, then its coefficient exceeds four
times the unshifted coefficient.  The adjacent-binomial ratio (and the same
conclusion when the unshifted coefficient vanishes) implies
\[
 \min\{a,N-a\}\le\frac{N+2}{3}.
\]
For the plus branch, shifted support gives
$N\ge(D-4)/2\ge254$.  For the early minus lobe, the ratio inequality forces
$N\ge(3D-2)/4\ge383$.  Thus every finite-lobe horizon is at least $160$ and
the smaller index is at most $27N/80$.  The standard entropy bound and the
exact integer inequality
\[
 80^{80}<2^{75}27^{27}53^{53}
\]
give
\begin{equation}
 \Pr[\operatorname{Bin}(N,1/2)\le27N/80]
 \le2^{-N/16}.
 \label{eq:terminal-modal-static-entropy-tail}
\end{equation}
For $P_D^+$ there are fewer than $2D$ candidates: shifted support excludes
$k<(D-2)/2$, while the adjacent ratio excludes $k\ge2D$.  Therefore
\[
 P_D^+\le\frac D4\,2^{-(D-4)/32}<\frac9{512}.
\]
For $R_D^-$, all positive atoms before its persistent tail have $k\le D$;
their ratio forces $N\ge(3D-2)/4$, so their sum is at most
$D2^{-(3D-2)/64}/8$.  For $k\ge D+3$, the two binomial indices become
$D$ and $D+2$.  Writing $u=N-D$, positivity is exactly
\[
 u(u-1)>4(D+1)(D+2).
\]
It fails at $u=2D+3$ and holds at $u=2D+4$, so the persistent tail starts
at $K=3D+5$ and has $N=K-1=3D+4$.  The negative-binomial identity
\[
 \sum_{k=K}^{\infty}2^{-k}\binom{k-1}{D+2}
 =\Pr[\operatorname{Bin}(K-1,1/2)\le D+2]
\]
bounds its positive part by $2^{-(3D+4)/16}/4$.  Thus
\[
 P_D^-\le\frac D8\,2^{-(3D-2)/64}
 +\frac14\,2^{-(3D+4)/16}<\frac1{2048}.
\]
The displayed right sides decrease for $D\ge512$: their successive ratios
are bounded using
$(513/512)^{32}<2$ and $(513/512)^{64}<8$.  At $D=512$, the plus bound is
less than $128/2^{15}=1/256$, while the two minus terms sum to less than
$64/2^{23}+1/(4\cdot2^{96})<1/2048$.  These are exact powers-of-two
comparisons.  This proves
\eqref{eq:terminal-modal-static-P-one-branch} for every $D$.

Because $2m-D=m+j\ge m\ge64$, the one-branch bounds imply
\begin{align}
 P_{m,j}&<\frac1{50},&&D\ge20,\label{eq:terminal-modal-static-P-bulk}\\
 P_{m,j}&<\frac9{80},&&6\le D\le19.
 \label{eq:terminal-modal-static-P-frontier}
\end{align}
Indeed $9/512+1/2048=37/2048<1/50$, while
$113/1024+1/2048=227/2048<9/80$.
Combining \eqref{eq:terminal-modal-static-E-uniform} and
\eqref{eq:terminal-modal-static-P-bulk} yields
\[
 E_{m,j}+\frac{13}{30}P_{m,j}
 <\frac1{45}+\frac{13}{1500}
 =\frac{139}{4500}<\frac1{32}
 \qquad(D\ge20).
\]
For $6\le D\le19$,
\eqref{eq:terminal-modal-static-E-frontier} and
\eqref{eq:terminal-modal-static-P-frontier} make the left side negative.
Consequently
\begin{equation}
 -S_{m,j}-\frac{13}{30}P_{m,j}>\frac{15}{32}.
 \label{eq:terminal-modal-static-combined-kernel}
\end{equation}

Finally write $\nu_n=2/5+\theta_n$.  The proved mass window gives
$0<\theta_n<13/75$, so
\[
 -\sum_n\nu_nR_n
 \ge-\frac25S_{m,j}-\frac{13}{75}P_{m,j}
 =\frac25\left(-S_{m,j}-\frac{13}{30}P_{m,j}\right)
 >\frac3{16}.
\]
This is exactly \eqref{eq:terminal-modal-static-mass-kernel}.
\end{proof}

\begin{lemma}[Exact asymptotic obstruction to the $1/16$ derivative target]
\label{lem:terminal-modal-static-derivative-stop}
Fix an integer $D\ge6$, put $j=m-D$, and let
\[
 \mathcal V_{m,D}
 =\sum_{n=2}^{m-1}\frac{\varepsilon_n}{4}
   \mathcal R_{m,m-D,n}q_n^{\rm ev}.
\]
Along the named family $q=1/(16m)$,
\begin{equation}
 \lim_{m\to\infty}\frac{\mathcal V_{m,D}}q
 =-\frac{f'(1/16)}4
 \left\{\frac{20}{3}+\frac43(-1/2)^D\right\},
 \qquad
 f(x)=\frac{t^2-10t+3}{10\sqrt t(1+t^2)}+\frac3{10},
 \quad t=e^{-2x}.
 \label{eq:terminal-modal-static-derivative-limit}
\end{equation}
In particular, at $D=6$ the limit is strictly less than $-1/16$.
Consequently
\eqref{eq:terminal-modal-static-derivative-kernel-target} fails for all
sufficiently large $m$.
\end{lemma}

\begin{proof}
Keep only the nonreflected branch for the moment and use the atom
$a_k$ from \eqref{eq:terminal-modal-static-a-atom}.  With
$(c_0,c_1,c_2)=(-3,2,1)$, its response at distance $D$ is
\begin{equation}
 Q_D(k)=2\sum_{s=0}^2c_s\left\{
 a_k(k-D+s)-\frac14a_k(k-D+s+2)\right\}.
 \label{eq:terminal-modal-static-derivative-Q}
\end{equation}
For $h\ge2$, split $a_k(k-h)$ before and after $k=h$.  The late part is a
negative-binomial mass, while the early parts have generating functions
\[
 \sum_{h\ge2}x^h\sum_{k<h}
 2^{-(k+1)}\binom{k-1}{h-k-1}
 =\frac{x^2}{4(1-x/2-x^2/2)},
\]
and, after weighting by $k$,
\[
 \sum_{h\ge2}x^h\sum_{k<h}
 k2^{-(k+1)}\binom{k-1}{h-k-1}
 =\frac{x^2}{4(1-x/2-x^2/2)^2}.
\]
Extracting coefficients gives the exact moments
\begin{align}
 \sum_{k\ge1}a_k(k-h)
 &=-\frac23-\frac1{12}(-1/2)^{h-2},
 \label{eq:terminal-modal-static-a-zero-moment}\\
 \sum_{k\ge1}k\,a_k(k-h)
 &=-\frac{30h+26}{27}
   -\frac{3h+1}{27}(-1/2)^h.
 \label{eq:terminal-modal-static-a-first-moment}
\end{align}
Indeed, the unweighted late mass is $1/2$ and its first moment is $h+1$;
for the weighted early term one uses
\[
 \frac{x^2}{4(1-x)^2(1+x/2)^2}
 =-\frac4{27(1-x)}+\frac1{9(1-x)^2}
  -\frac2{27(1+x/2)}+\frac1{9(1+x/2)^2}.
\]
Substitution of
\eqref{eq:terminal-modal-static-a-zero-moment}--
\eqref{eq:terminal-modal-static-a-first-moment} in
\eqref{eq:terminal-modal-static-derivative-Q} yields
\begin{equation}
 \sum_{k\ge1}Q_D(k)=0,
 \qquad
 \sum_{k\ge1}kQ_D(k)
 =\frac{20}{3}+\frac43(-1/2)^D.
 \label{eq:terminal-modal-static-Q-moments}
\end{equation}

For fixed $k$, the exact identity
\eqref{eq:terminal-modal-finite-epsilon-identity} implies
\[
 \varepsilon_{m-k}\longrightarrow f(1/16),qquad
 \frac{\varepsilon_{m-k}-\varepsilon_m}{q}
 \longrightarrow-kf'(1/16).
\]
The binomial atoms in \eqref{eq:terminal-modal-static-derivative-Q} have a
summable $k$-weighted geometric tail for fixed $D$.  Hence dominated
convergence and the zero moment in
\eqref{eq:terminal-modal-static-Q-moments} give the direct-branch limit in
\eqref{eq:terminal-modal-static-derivative-limit}.  The omitted reflected
branch has distance $2m-D$.  Over the literal range $k\le m-2$, its support
lies within $O_D(1)$ of $k=m$ and every atom is
$O_D(m^{D+2}2^{-m})$; it therefore vanishes after division by $q$.  The
truncated tail of the direct branch obeys the same conclusion.

It remains to certify the strict comparison at $D=6$.  Differentiation gives
\[
 f'(x)=\frac{A(t)}{10\sqrt t(1+t^2)^2},qquad
 A(t)=t^4-30t^3+12t^2+10t+3.
\]
At $x=1/16$, alternating Taylor bounds give
$7/8<t=e^{-1/8}<113/128<(47/50)^2$.  On this interval $A$ is positive and
decreasing: $A''$ is decreasing and negative at $7/8$, and
$A'(7/8)=-4509/128<0$.  Therefore the exact rational comparison
\[
 f'(1/16)>
 \frac{A(113/128)}{10(47/50)(1+(113/128)^2)^2}
 =\frac{1539492805}{39945178223}>\frac4{107}
\]
holds.  At $D=6$, the first moment in
\eqref{eq:terminal-modal-static-Q-moments} is $107/16$.  Hence the limit is
strictly less than
\[
 -\frac{107}{64}\frac4{107}=-\frac1{16}.
\]
This proves the STOP.
\end{proof}
