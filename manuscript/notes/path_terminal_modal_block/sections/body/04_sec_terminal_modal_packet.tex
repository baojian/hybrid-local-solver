\section{The binomial packet and phase-uniform anti-cancellation}
\label{sec:terminal-modal-packet}

Define the ideal entry packet
\begin{equation}
 g_j=-\frac{\alpha}{2}(1-q)^m b_{m,j},
 \qquad b_{m,j}=2^{-m}\binom mj.
 \label{eq:terminal-modal-packet}
\end{equation}
Its center height is
\begin{equation}
 \frac{\max_j|g_j|}{q^{5/2}}
 \longrightarrow 2e^{-1/16}\sqrt{\frac2\pi}.
 \label{eq:terminal-modal-packet-height}
\end{equation}

\begin{lemma}[Exact packet coefficients; proved here]
\label{lem:terminal-modal-packet-transform}
For $1\le h\le m-1$, the weighted cosine coefficient of $g$ is
\begin{equation}
 \widehat g_h=-\frac{\alpha(1-q)^m}{m}
 \left[
  \cos^m\phi_h\cos\frac{h\pi}{2}
  -2^{-m-1}(1+(-1)^h)
 \right].
 \label{eq:terminal-modal-packet-transform}
\end{equation}
In particular, for $h=2s$,
\begin{equation}
 G_{2s}:=\widehat g_{2s}
 =-\frac{\alpha(1-q)^m}{m}
 \left[(-1)^s\cos^m\frac{s\pi}{m}-2^{-m}\right].
 \label{eq:terminal-modal-even-packet}
\end{equation}
For $s=o(\sqrt m)$,
$|G_{2s}|\sim e^{-1/16}\alpha/m=16e^{-1/16}q^3$.
\end{lemma}

\begin{proof}
If $J\sim\operatorname{Binomial}(m,1/2)$, then
\[
 \mathbb E e^{itJ}=e^{imt/2}\cos^m(t/2),
 \qquad
 \mathbb E\cos(tJ)=\cos(mt/2)\cos^m(t/2).
\]
The degree weights double every interior binomial mass and leave the two
endpoint masses single.  Taking $t=h\pi/m$, subtracting those endpoint
copies, multiplying by $-\alpha(1-q)^m/2$, and dividing by $N_h=m$ gives
\eqref{eq:terminal-modal-packet-transform}.  The even-mode specialization and
asymptotic follow immediately.
\end{proof}

\begin{lemma}[Exact position/velocity propagators; proved here]
\label{lem:terminal-modal-propagators}
Put $L=(I+P_{\rm row})/2$ and
$s_k=(1-q)^{-k}r_k^x$.  Every projection-inactive full-face trajectory obeys
\begin{equation}
 s_{k+1}=2Ls_k-Ls_{k-1}.
 \label{eq:terminal-modal-scaled-recurrence}
\end{equation}
Let $K_0=I,K_1=L$ and $J_0=0,J_1=I$ be the two fundamental solutions of
this recurrence.  On the mode with $\theta_h=h\pi/m$, their multipliers are
\begin{align}
 K_k(\theta_h)&=\cos^k\frac{\theta_h}{2}
                         \cos\frac{k\theta_h}{2},
 \label{eq:terminal-modal-position-kernel}\\
 J_k(\theta_h)&=\cos^{k-1}\frac{\theta_h}{2}
 \frac{\sin(k\theta_h/2)}{\sin(\theta_h/2)},
 \qquad k\ge1,
 \label{eq:terminal-modal-velocity-kernel}
\end{align}
with the second expression continued at $\theta_h=0$.  Both kernels are
entrywise nonnegative on the endpoint path.  The row sums of $K_k$ and $J_k$
are respectively $1$ and $k$.

More precisely, if $\mathsf T_j$ is the Chebyshev polynomial, then
\begin{align}
 K_k&=2^{-k}\sum_{j=0}^k\binom{k}{j}\mathsf T_j(P_{\rm row}),
 \label{eq:terminal-modal-position-expansion}\\
 J_k&=2^{-(k-1)}\sum_{a,b=0}^{k-1}\binom{k-1}{b}
 \mathsf T_{|k-1-a-b|}(P_{\rm row}).
 \label{eq:terminal-modal-velocity-expansion}
\end{align}
Consequently, with
\begin{equation}
 c=r_0^x-g,\qquad
 w=\frac{r_1^x}{1-q}-Lr_0^x,
 \label{eq:terminal-modal-propagator-data}
\end{equation}
one has the exact decomposition
\begin{equation}
 r_k^x=(1-q)^k\left(K_kg+K_kc+J_kw\right).
 \label{eq:terminal-modal-propagator-decomposition}
\end{equation}
For $1\le h\le m-1$, the mode-$h$ coefficient of $w$ is
$\cos\phi_h\sin\phi_hD_h=q\cos^2\phi_hV_h$.
\end{lemma}

\begin{proof}
Since $B=(1-q^2)L$, rescaling
\eqref{eq:terminal-modal-second-order} gives
\eqref{eq:terminal-modal-scaled-recurrence}.  Solving its scalar recurrence
at $L$-eigenvalue $\cos^2(\theta_h/2)$ gives
\cref{eq:terminal-modal-position-kernel,eq:terminal-modal-velocity-kernel}.
The position expansion follows by taking the real part of
$2^{-k}(1+e^{i\theta})^k$.  For the velocity expansion, multiply
\[
 \frac{\sin(kx)}{\sin x}=\sum_{a=0}^{k-1}e^{i(k-1-2a)x}
\]
by the binomial expansion of $\cos^{k-1}x$, set $x=\theta/2$, and take real
parts.

To interpret the operator expansions, extend a path vector evenly to the cycle
$\mathbb Z/(2m)$.  On the endpoint-weighted cosine basis, multiplication by
$\cos(j\theta_h)$, equivalently $\mathsf T_j(P_{\rm row})$, is exactly the
average of the two folded shifts by $\pm j$.  It is positivity preserving and
has row sum one, including all endpoint reflections and aliases.  The
coefficient sums in \eqref{eq:terminal-modal-position-expansion} and
\eqref{eq:terminal-modal-velocity-expansion} are $1$ and $k$.  This proves the
kernel claims.  The fundamental-solution initial conditions give
\eqref{eq:terminal-modal-propagator-decomposition}.  The formula for $w_h$
follows from \eqref{eq:terminal-modal-quadratures}, and its second form follows
from \cref{lem:terminal-modal-entry-velocity}.
\end{proof}

\begin{lemma}[Ideal packet remains nonpositive; proved here]
\label{lem:terminal-modal-ideal-sign}
The ideal residual trajectory $r_k^{\rm id}=(1-q)^kK_kg$ is coordinatewise
nonpositive at every integer time $k\ge0$.
\end{lemma}

\begin{proof}
The packet $g$ is coordinatewise nonpositive, while
\cref{lem:terminal-modal-propagators} shows that $K_k$ is entrywise
nonnegative.
\end{proof}

The stationary ideal above discards a useful entry-velocity cancellation.
The same packet can instead be split into two directed waves.  This gives an
exact finite-cycle reduction of the positive-residual part of the regime
problem.

\begin{lemma}[Directed packet and folded velocity reduction; proved here]
\label{lem:terminal-modal-directed-packet}
Extend path vectors evenly to the cycle $\mathbb Z/(2m)$.  On that cycle put
\[
 \mathcal A_\pm=\frac12(I+\mathcal S_\pm),\qquad
 L=\frac12(\mathcal A_++\mathcal A_-).
\]
Let $b^+$ equal $b_{m,j}$ on $1\le j\le m-1$, equal
$b_{m,0}/2$ and $b_{m,m}/2$ at the two endpoints, and vanish on the other
arc.  Put $b^-(j)=b^+(-j)$ and
\[
 g^\pm=-\frac{\alpha}{2}(1-q)^m b^\pm.
\]
Then $g^{\rm ev}=g^++g^-$.  Define
\begin{equation}
 w^{\rm dir}=(\mathcal A_+-L)g^+
             +(\mathcal A_--L)g^-.
 \label{eq:terminal-modal-directed-velocity}
\end{equation}
For every $k\ge0$,
\begin{equation}
 K_kg^{\rm ev}+J_kw^{\rm dir}
 =\mathcal A_+^kg^++\mathcal A_-^kg^-\le0
 \quad\hbox{coordinatewise}.
 \label{eq:terminal-modal-directed-sign}
\end{equation}

Let $c=r_0^x-g$ and let $w$ be the literal velocity source in
\eqref{eq:terminal-modal-propagator-data}, both evenly extended.  For the
proved proper-prefix chronology and every $m\ge64$,
\begin{equation}
 c\le0.
 \label{eq:terminal-modal-entry-correction-sign}
\end{equation}
If $u=w-w^{\rm dir}$, the projection-inactive candidate therefore satisfies
\begin{equation}
 \max_j r_{k,j}^x
 \le (1-q)^k\left(1+\frac{k-1}{2m}\right)
       \norm{u_+}_{1,D},
 \qquad k\ge1.
 \label{eq:terminal-modal-directed-positive-bound}
\end{equation}
Here $u_+=\max(u,0)$ coordinatewise and the degree-weighted path norm equals
the ordinary $\ell_1$ norm of its even extension.
\end{lemma}

\begin{proof}
The half endpoint weights make $b^++b^-$ exactly the even extension of the
path binomial packet.  The commuting operators obey
\[
 \mathcal A_++\mathcal A_-=2L,\qquad
 \mathcal A_+\mathcal A_-=L.
\]
Thus each sequence $\mathcal A_\pm^kg^\pm$ solves
\eqref{eq:terminal-modal-scaled-recurrence}.  Its values at times zero and
one are respectively $g^\pm$ and $\mathcal A_\pm g^\pm$.  The definition
\eqref{eq:terminal-modal-directed-velocity} matches those two initial values
with the $K/J$ fundamental solution, proving
\eqref{eq:terminal-modal-directed-sign}.  Both directed waves are
nonpositive because $\mathcal A_\pm$ are Markov kernels and $g^\pm\le0$.

We next prove \eqref{eq:terminal-modal-entry-correction-sign}.  At the last
proper prefix let $R_m$ be the average-point residual from
\eqref{eq:terminal-modal-chronology-residuals}.  Zero-extend
$G_{m-1}^{\rm id}$ to the moving frontier and set
\[
 R_m^{\rm id}
 =-\frac{2}{1+q}\left(G_m^{\rm id}-c_1G_{m-1}^{\rm id}\right).
\]
On every shared row, the split
\eqref{eq:terminal-modal-prefix-sign-split} and the finite-$q$ correction
sign give $R_m\le R_m^{\rm id}$.  At the frontier, the proof of
\cref{lem:terminal-modal-chronology-reduction} gives
\[
 R_m(m-1)\le
 -\frac{30946901}{157368320}\frac{q^3}{1+q}.
\]
On the other hand,
\[
 \frac{2G_m^{\rm id}(m-1)}{q^3}
 =\frac{32m(1-q)^{m-1}}{2^m}<\frac18
 \qquad(m\ge64),
\]
where $32m/2^m$ is decreasing for $m\ge64$ (and the factor
$(1-q)^{m-1}$ is at most one),
and $30946901/157368320>1/8$.  Hence the frontier inequality
$R_m\le R_m^{\rm id}$ holds as well.

The post-step residual on the last proper face is $B_mR_m$.  Direct Pascal
arithmetic gives
\[
 (B_mR_m^{\rm id})(j)=g_j\quad(1\le j<m),\qquad
 (B_mR_m^{\rm id})(0)=g_0+g_m\le g_0.
\]
Since $B_m$ is entrywise nonnegative, every old row of the full-face entry
residual is at most $g_j$.  At the newly admitted degree-one endpoint the
entry residual is $q^3/5-\eta\sigma_m$.  The raw-frontier lower bound gives
\[
 \frac{\eta\sigma_m-q^3/5}{q^3}
 \ge \frac{6985811}{9835520}>\frac18,
\]
whereas
$|g_m|/q^3=8m(1-q)^m/2^m<1/8$; here too $8m/2^m$ is decreasing from
$m=64$ onward.  Thus the new row is also at most $g_m$,
which proves $c=r_0^x-g\le0$.

It remains to prove the kernel factor in
\eqref{eq:terminal-modal-directed-positive-bound}.  On the infinite line,
the Laurent coefficient of $J_k$ at displacement $r$ is
\begin{equation}
 j_k(r)=
 \Pr\{\operatorname{Bin}(k-1,1/2)\ge |r|\}\le1.
 \label{eq:terminal-modal-J-tail}
\end{equation}
This follows either from \eqref{eq:terminal-modal-velocity-expansion} or by
summing
$\mathcal A_+^{k-t}\mathcal A_-^{t-1}$ over $1\le t\le k$.
Folding to the $2m$-cycle replaces $j_k(r)$ by
$\sum_{\ell\in\mathbb Z}j_k(r+2m\ell)$.  If
$Z\sim\operatorname{Bin}(k-1,1/2)$, the number of points in
$r+2m\mathbb Z$ lying in $[-Z,Z]$ is at most $1+Z/m$.  Taking expectations
therefore gives
\begin{equation}
 \max_r\sum_{\ell\in\mathbb Z}j_k(r+2m\ell)
 \le1+\frac{k-1}{2m}.
 \label{eq:terminal-modal-J-alias}
\end{equation}
Finally, the exact decomposition is
\[
 r_k^x=(1-q)^k\left(
 \mathcal A_+^kg^++\mathcal A_-^kg^-+K_kc+J_ku\right).
\]
The first two terms and $K_kc$ are nonpositive, while $J_k$ is
nonnegative.  Applying \eqref{eq:terminal-modal-J-alias} to $u_+$ proves
\eqref{eq:terminal-modal-directed-positive-bound}.
\end{proof}

Since $q=1/(16m)$, the time-dependent factor in
\eqref{eq:terminal-modal-directed-positive-bound} has the uniform estimate
\begin{equation}
 (1-q)^k\left(1+\frac{k-1}{2m}\right)
 \le e^{-qk}(1+8qk)\le 8e^{-7/8}.
 \label{eq:terminal-modal-directed-damping}
\end{equation}
This removes both the apparent factor $k$ and all repeated-reflection losses.
The remaining regime interface has a further exact spatial reduction.

\begin{lemma}[Static directed remainder certificate; proved here]
\label{lem:terminal-modal-static-remainder}
Put $\delta=1-q$ and $b=qr_0^v$.  In the notation of the last proper-prefix
step, for $0\le j<m$ and at the new endpoint respectively,
\begin{align}
 b_j&=(B_mR_m)_j-\delta A_m(j),
 &b_m&=\eta(qP_m-\sigma_m).
 \label{eq:terminal-modal-static-b}
\end{align}
Define the lower binomial packet
\begin{equation}
 \gamma_j=-\frac{q^2}{2}\delta^m2^{-(m-2)}
 \binom{m-2}{j-1}\quad(1\le j<m),
 \qquad \gamma_j=0\quad(j\notin\{1,\ldots,m-1\}),
 \label{eq:terminal-modal-lower-packet}
\end{equation}
and the path vector
\begin{equation}
 e_0=-\frac{\gamma_1}{2},\qquad
 e_j=\frac{\gamma_{j-1}-\gamma_{j+1}}4\ (1\le j<m),
 \qquad e_m=\frac{\gamma_{m-1}}2.
 \label{eq:terminal-modal-directed-deconvolution}
\end{equation}
If $d=b-e$, then the directed remainder in
\cref{lem:terminal-modal-directed-packet} is exactly
\begin{equation}
 u=w-w^{\rm dir}=Ld.
 \label{eq:terminal-modal-static-remainder}
\end{equation}
Consequently the still-open static target
$\norm{u_+}_{1,D}\le21q^3/80$ is exactly the finite inequality
\begin{equation}
 \left[\frac{d_0+d_1}{2}\right]_+
 +2\sum_{j=1}^{m-1}
 \left[\frac{d_{j-1}+2d_j+d_{j+1}}4\right]_+
 +\left[\frac{d_{m-1}+d_m}{2}\right]_+
 \le\frac{21}{80}q^3.
 \label{eq:terminal-modal-static-finite-target}
\end{equation}

On the infinite line the coefficient of $J_kL$ at displacement $r$ is
\begin{equation}
 h_k(r)=2^{-(k+1)}\sum_{a=0}^{k-1}
 \binom{k+1}{k+r-a}
 =\Pr\{r+1\le Z\le k+r\},
 \quad Z\sim\operatorname{Bin}(k+1,1/2),
 \label{eq:terminal-modal-JL-kernel}
\end{equation}
where out-of-range binomial coefficients vanish.  On the $2m$-cycle these
coefficients are folded modulo $2m$.  Thus every proposed early bound on
$J_ku$ is also an explicit finite signed convolution inequality in $d$.
\end{lemma}

\begin{proof}
Let $e_m^{\rm out}$ be the error after the last proper-prefix step and let
$\iota$ denote zero extension to the full path.  The transported full-entry
errors satisfy
\[
 e_0^x=\iota e_m^{\rm out}
 -(p_{m+1}^\star-\iota p_m^\star),\qquad
 e_0^v=\iota e_m^{\rm out}
 +\frac\delta q(\iota e_m^{\rm out}-\iota e_m).
\]
Applying the full operator and using the endpoint displacement identity from
\cref{lem:terminal-modal-source-scalar-reduction} gives
$b_j=(B_mR_m)_j-\delta A_m(j)$ on the old rows.  At the new endpoint the
same expression is
\[
 (q^3/5-\eta\sigma_m)+(\eta P_m-q^3/5)-\delta\eta P_m
 =\eta(qP_m-\sigma_m),
\]
which proves \eqref{eq:terminal-modal-static-b}.

For the packet identity, work on the $2m$-cycle and let $\gamma^+$ be
\eqref{eq:terminal-modal-lower-packet} on the positive arc and zero on the
other arc; let $\gamma^-$ be its reflection.  With
$\mathcal D=(\mathcal S_+-\mathcal S_-)/4$, Pascal's identity gives
\[
 L\gamma^+=g^++a,\qquad L\gamma^-=g^-+a,
\]
where the same endpoint vector $a$ occurs in both equations.  Since
$w^{\rm dir}=\mathcal Dg^+-\mathcal Dg^-$ and $L$ commutes with
$\mathcal D$,
\[
 w^{\rm dir}=L(\mathcal D\gamma^+-\mathcal D\gamma^-)=Le.
\]
Restricting the cycle vector to the path gives exactly
\eqref{eq:terminal-modal-directed-deconvolution}, including its two endpoint
halves.  The first full-face step gives $w=qLr_0^v=Lb$, hence
\eqref{eq:terminal-modal-static-remainder}.  Expanding the endpoint-path
stencil of $L=(I+P_{\rm row})/2$ gives
\eqref{eq:terminal-modal-static-finite-target}.

Finally, factor $J_kL$ on the line as the sum of
$\mathcal A_+^{k-a}\mathcal A_-^{a+1}$ over $0\le a<k$ and expand the two
binomials.  Vandermonde's identity gives the first expression in
\eqref{eq:terminal-modal-JL-kernel}; the second is the same finite binomial
sum.  Even extension folds it modulo $2m$.
\end{proof}

The signed mass and the endpoint tail can in fact be separated uniformly.
The next lemma leaves only a local half-ratio inequality for the last-prefix
profile.  It is important that this half-ratio is not asserted below.

\begin{lemma}[Static endpoint-tail reduction; proved here]
\label{lem:terminal-modal-static-tail-reduction}
Assume $m\ge64$ and retain $d$ from
\cref{lem:terminal-modal-static-remainder}.  Put
\[
 \beta=\frac{d_m}{q^3}.
\]
Then
\begin{equation}
 \frac{\langle Ld,\one\rangle_D}{q^3}<\frac3{50},
 \qquad -\frac{57}{200}<\beta<0.
 \label{eq:terminal-modal-static-mass-endpoint}
\end{equation}
Define the endpoint geometric comparison vector by
\begin{equation}
 \frac{h_m}{q^3}=\beta,
 \qquad
 \frac{h_{m-r}}{q^3}
 =-\frac{\beta}{4}\left(-\frac12\right)^{r-1}
 \quad(1\le r\le m).
 \label{eq:terminal-modal-static-tail-comparator}
\end{equation}
If
\begin{equation}
 d_j\ge h_j\qquad(0\le j\le m),
 \label{eq:terminal-modal-static-tail-dominance}
\end{equation}
then the static target
\eqref{eq:terminal-modal-static-finite-target} holds strictly.

There is a concrete local sufficient condition for
\eqref{eq:terminal-modal-static-tail-dominance}.  Write
\[
 T_r=\frac{d_{m-r}}{q^3},\qquad
 E_1=T_1+\frac\beta4,
 \qquad F_r=T_r+\frac12T_{r-1}\quad(2\le r\le m).
\]
It is enough to prove
\begin{equation}
 E_1\ge0,\qquad F_2\ge\frac12E_1,\qquad
 F_r\ge\frac12F_{r-1}\quad(3\le r\le m).
 \label{eq:terminal-modal-static-local-half-ratios}
\end{equation}
Thus the open static certificate has been reduced to the explicit local
inequalities in \eqref{eq:terminal-modal-static-local-half-ratios}; no
asymptotic tail replacement is being made.
\end{lemma}

\begin{proof}
Let $\mathsf M(z)=\langle z,\one\rangle_D$.  The deconvolution vector
$e$ in \eqref{eq:terminal-modal-directed-deconvolution} has
$\mathsf M(e)=0$, and $L$ is $D$-stationary.  Hence
\begin{equation}
 \mathsf M(Ld)=\mathsf M(d)=\mathsf M(b).
 \label{eq:terminal-modal-static-signed-mass-identity}
\end{equation}
Put $p_n=P_n/q^2$, where $P_n$ is the proper-prefix frontier optimum.
Weighted row summation of the exact transported-velocity update, followed by
telescoping the optimum displacements, gives the exact scalar identity
\begin{equation}
 \frac{\mathsf M(b)}{q^3}
 =-\delta^m\frac{q^2(5-q)}{5(1+q^2)}
  +\frac{\eta}{1+q}\sum_{n=1}^m
       q\delta^{m-n}p_n.
 \label{eq:terminal-modal-static-signed-mass-formula}
\end{equation}
For completeness, the initial seed term in this telescoping sum is
$-q^2(5-q)/(5(1+q^2))$; at transition $n$ the uncancelled frontier
displacement is $\eta q p_n/(1+q)$.  These are respectively the first and
second terms on the right of
\eqref{eq:terminal-modal-static-signed-mass-formula}, after the intervening
factors $\delta$ are applied.  This is also a direct row-by-row derivation of
the displayed identity.

We bound the remaining sum with rational constants.  Set $q_0=1/1024$ and
\begin{equation}
 \mathsf p_q(t)=\frac{2}{1+t^2}\left\{
 \frac{t(1+t/5)}{1-q}+\frac{t-1/5}{1+q}\right\}.
 \label{eq:terminal-modal-static-frontier-function}
\end{equation}
Then $p_n=\mathsf p_q(\chi^n)$.  The function is increasing in both $t$
and $q$ on the present range: its $q$-derivative has the sign of
\[
 \frac{t(1+t/5)}{(1-q)^2}
 -\frac{t-1/5}{(1+q)^2}>0,
\]
and its $t$-derivative is positive by
\cref{lem:terminal-modal-frontier-bounds}.  Split $1\le n\le m$ at
$\lfloor m/4\rfloor,\lfloor m/2\rfloor,\lfloor3m/4\rfloor$.  Bernoulli's
inequality gives the four upper bounds
\[
 \begin{gathered}
 p_n\le p_0,p_1,p_2,p_3
 \quad\hbox{on the four blocks, respectively},\\
 p_i=\mathsf p_{q_0}(t_i),\\
 (t_0,t_1,t_2,t_3)=\left(1,\frac{32}{33},
 \frac{16}{17},\frac{32}{35}\right).
 \end{gathered}
\]
Indeed, in the second through fourth blocks one has respectively
$n>m/4,m/2,3m/4$, and
$\chi^{-n}\ge1+2nq$; the floor endpoints therefore only strengthen the
three strict $t$-bounds above.
If $x_i=\delta^{m-\lfloor im/4\rfloor}$, then
\[
 x_1\le\frac{64}{67},\qquad x_2\le\frac{32}{33},
 \qquad x_3\le\frac{64}{65}.
\]
The third alternating binomial partial sum, with $mq=1/16$ and
$q\le q_0$, also gives
\[
 \delta^m\ge z:=1-\frac1{16}
 +\frac{16^{-2}-q_0/16}{2}
 -\frac{16^{-3}-3q_0/16^2+2q_0^2/16}{6}
 =\frac{15760245}{16777216}.
\]
Telescoping each block and using $\eta/(1+q)<1/2$ now yields
\begin{align*}
 \frac{\mathsf M(b)}{q^3}
 &<\frac12\left\{p_3-p_0z
 +(p_0-p_1)\frac{64}{67}
 +(p_1-p_2)\frac{32}{33}
 +(p_2-p_3)\frac{64}{65}\right\}\\
 &=\frac{213106150684112952120801}
 {3552503277561938585600000}<\frac3{50}.
\end{align*}

It remains to control the last coordinate.  Put
$S_n=\sigma_n/q^3$ and
\begin{equation}
 \varepsilon_n=S_n-\left(\frac32p_n-\frac25\right).
\end{equation}
The exact frontier recurrence is
\[
 S_n=\frac{p_n+\eta S_{n-1}-\eta p_{n-1}/2-1/5}{1+q}
 \quad(n\ge2),qquad S_1=\frac1q-\frac15.
\]
Consequently
\begin{align}
 \varepsilon_n
 &=\frac{\eta\varepsilon_{n-1}+Q_n}{1+q},
 \notag\\
 Q_n&=\eta(p_{n-1}-p_n)
 -q\left(\frac32+\frac q2\right)p_n
 +\frac{2q}{5}+\frac{q^2}{5}.
 \label{eq:terminal-modal-static-endpoint-error}
\end{align}
From \eqref{eq:terminal-modal-static-frontier-function},
$\partial_t\mathsf p_q(t)<6/5$ on $7/8\le t\le1$, while
$\chi^{n-1}-\chi^n<3q$.  Thus
$0<p_{n-1}-p_n<4q$.  Together with
$p_n\ge3488/1921$ this makes $Q_n<0$; explicitly,
\[
 \frac{Q_n}{q}<\frac{12}{5}+\frac1{5120}
 -\frac{5232}{1921}<0.
\]
Since $\varepsilon_1<1/q$ and $\eta/(1+q)<1/2$,
\begin{equation}
 \varepsilon_m<\frac{1}{q\,2^{m-1}}.
 \label{eq:terminal-modal-static-endpoint-error-bound}
\end{equation}
Moreover $\chi^{-m}\ge1+2mq\ge9/8$, so, by monotonicity,
\[
 p_m\le \mathsf p_{q_0}(8/9)
 =\frac{1474594816}{760216875}.
\]
Because $e_m=-q^2\delta^m/2^m$,
\[
 \beta=\eta(p_m-S_m)+\frac{\delta^m}{q\,2^m}.
\]
The raw-frontier lower bound from
\cref{lem:terminal-modal-chronology-reduction}, now evaluated at $q_0$,
also gives
\[
 S_m-p_m\ge
 \frac{(1-q_0)(5+q_0)}{4(1+q_0)}\frac{3488}{1921}
 -\frac15-\frac{33}{16}
 =\frac{797707}{252035200}.
\]
Since $\eta>1/3$ and
$16m/2^m\le1024/2^{64}<797707/(3\cdot252035200)$, this proves
$\beta<0$.  Finally,
\begin{align*}
 -\beta
 &\le-\frac{b_m}{q^3}
 =\eta\left(\frac{p_m}{2}-\frac25+\varepsilon_m\right)\\
 &<\frac12\left(
 \frac{1474594816}{2\cdot760216875}-\frac25\right)
 +\frac{8m}{2^{m-1}}<\frac{57}{200},
\end{align*}
where the last term is decreasing for $m\ge64$ and the last inequality is
an exact rational check at $m=64$.

We now prove the conditional tail implication.  A direct endpoint-stencil
calculation gives
\begin{align*}
 \frac{(Lh)_m}{q^3}&=\frac{3\beta}{8},
 &\frac{(Lh)_{m-1}}{q^3}&=\frac{5\beta}{32},\\
 \frac{(Lh)_{m-r}}{q^3}
 &=-\frac{\beta}{64}\left(-\frac12\right)^{r-2}
 &&(2\le r\le m-1).
\end{align*}
The seed endpoint is the corresponding half-stencil.  Summing the negative
terms with endpoint degree one and interior degree two gives
\begin{equation}
 \frac{\langle (Lh)_-,\one\rangle_D}{q^3}
 \le(-\beta)\left(\frac{17}{24}
 +\frac{1}{48\,2^{m-2}}\right).
 \label{eq:terminal-modal-static-tail-negative-mass}
\end{equation}
Here $z_-=[-z]_+$ coordinatewise.
The exponentially small correction is attained only when the seed truncates
a negative term.  If $d\ge h$, positivity of $L$ gives $Ld\ge Lh$ and hence
the negative mass of $Ld$ is no larger than the right side above.  Therefore
\begin{align*}
 \frac{\langle(Ld)_+,\one\rangle_D}{q^3}
 &=\frac{\mathsf M(Ld)}{q^3}
   +\frac{\langle(Ld)_-,\one\rangle_D}{q^3}\\
 &<\frac3{50}+\frac{57}{200}\frac{17}{24}
 +\frac{57}{200\cdot48\,2^{m-2}}
 <\frac{419}{1600}+\frac1{1600}=\frac{21}{80}.
\end{align*}

Finally let $x_r=(d_{m-r}-h_{m-r})/q^3$.  Then $x_0=0$,
$x_1=E_1$, and
$x_r=F_r-x_{r-1}/2$ for $r\ge2$.  Under
\eqref{eq:terminal-modal-static-local-half-ratios}, induction gives
$0\le x_{r-1}\le F_{r-1}$ and hence $x_r\ge0$.  This proves the stated
local sufficient condition.
\end{proof}

This certificate isolates the remaining sign issue.  Positivity and
$D$-stationarity alone give only
$\norm{(Ld)_+}_{1,D}\le\norm{d_+}_{1,D}$; the deterministic screen measures
the latter ratio near $0.437$, far above $21/80$.  The sharper measured
$0.260$ value uses cancellation between a positive bulk and an alternating
terminal tail of $d$.  The signed mass and endpoint size are now controlled
uniformly by \cref{lem:terminal-modal-static-tail-reduction}; the remaining
static obstruction is precisely the local half-ratio family
\eqref{eq:terminal-modal-static-local-half-ratios}.  The present chronology
bounds do not yet prove those inequalities, so neither
\eqref{eq:terminal-modal-static-finite-target} nor the corresponding early
convolution inequality is promoted here.

The remaining regime interface is now the static certificate, its early
signed convolution bound, and the position estimate: prove the finite
inequality above and its early-time counterpart, then combine them with a
lower bound on the projection-inactive position candidate strong
enough to show both raw positivity and
\eqref{eq:terminal-modal-unclipped-target}.  The deterministic screen
measures the former ratio near $0.26$, but that measurement is not used as a
bound.  In particular, \eqref{eq:terminal-modal-directed-positive-bound}
alone does not prove the regime lemma because the first full-face position
margin is only of order $q^2$.

\begin{lemma}[Uniform cosine-square floor; proved here]
\label{lem:terminal-modal-cosine-square}
For every integer $H\ge8$ and every real $x$,
\begin{equation}
 \sum_{s=1}^{H}\cos^2(sx)\ge\frac{H}{16}.
 \label{eq:terminal-modal-cosine-square}
\end{equation}
\end{lemma}

\begin{proof}
If $\sin x=0$, the sum equals $H$.  Otherwise, the finite sum identity gives
\[
 \sum_{s=1}^H\cos^2(sx)
 =\frac H2+\frac{\sin(Hx)\cos((H+1)x)}{2\sin x}.
\]
If $|\sin x|\ge2/H$, the right side is at least $H/4$.  Otherwise let
$\xi$ be the distance from $x$ to $\pi\mathbb Z$.  Since
$\sin(\pi/H)\ge2/H$, one has $\xi<\pi/H$.  For
$1\le s\le\lfloor H/4\rfloor$, $s\xi\le\pi/4$, so each corresponding
square is at least $1/2$.  Since
$\lfloor H/4\rfloor\ge H/8$ for $H\ge8$, the result follows.
\end{proof}

The next two statements are the remaining open interfaces after the separate
proper-prefix chronology.  They are displayed as lemmas to freeze the needed
inequalities, but are not asserted as proved.

\begin{lemma}[Entry position and velocity profiles; open]
\label{lem:terminal-modal-packet}
Assume the conclusion of
\cref{lem:terminal-modal-admission-chronology}.
Put
\begin{equation}
 H_m=\left\lfloor\sqrt{\frac{m}{64\log(16m)}}\right\rfloor.
 \label{eq:terminal-modal-bandwidth}
\end{equation}
For all sufficiently large $m$, the literal full-face entry coefficients
satisfy, for $1\le s\le H_m$,
\begin{equation}
 \frac m\alpha|C_{2s}-G_{2s}|\le\frac1{256},
 \qquad
 \frac m\alpha\left|V_{2s}+\frac{G_{2s}}5\right|\le\frac14.
 \label{eq:terminal-modal-entry-profiles}
\end{equation}
\end{lemma}

The second profile is deliberately centered at $-G_{2s}/5$, not at zero.
The screen below indicates a nonvanishing smooth correction, so an $o(1)$
profile error is neither needed nor suggested.

\begin{lemma}[Profiles imply the quadrature target; proved here]
\label{lem:terminal-modal-profile-implies-quadrature}
For all sufficiently large $m$, the bounds in
\eqref{eq:terminal-modal-entry-profiles}, together with inactivity of the
first full-face projection, imply
\begin{equation}
 |C_{2s}-G_{2s}|+|D_{2s}|\le\frac{|G_{2s}|}{64},
 \qquad 1\le s\le H_m.
 \label{eq:terminal-modal-entry-target}
\end{equation}
\end{lemma}

\begin{proof}
Put $g_s=m|G_{2s}|/\alpha$.  Uniformly on the stated band,
$\cos^m(s\pi/m)-2^{-m}\ge14/15$ for all sufficiently large $m$.
Also $(1-q)^m\ge1-mq=15/16$.  Thus $g_s\ge7/8$.
By \cref{lem:terminal-modal-entry-velocity} and $\cot x\le1/x$,
\[
 \frac{|C_{2s}-G_{2s}|+|D_{2s}|}{|G_{2s}|}
 \le \frac1{256g_s}
 +q\cot\frac{s\pi}{m}\left(\frac15+\frac1{4g_s}\right)
 \le\frac1{224}+\frac{17}{560\pi}<\frac1{64}.
\]
This is exactly \eqref{eq:terminal-modal-entry-target}.
\end{proof}

\begin{lemma}[Projection and global-envelope regime; open]
\label{lem:terminal-modal-regime}
Put
\begin{equation}
 K_m=\left\lfloor\frac1{8q}\log\frac1q\right\rfloor.
 \label{eq:terminal-modal-horizon}
\end{equation}
For every $1\le k\le K_m$, the raw point
$y_{k-1}-r(y_{k-1})$ is strictly positive, and the resulting full-face state
satisfies
\begin{equation}
 \min_j p_{k,j}>\delta_k.
 \label{eq:terminal-modal-unclipped-target}
\end{equation}
\end{lemma}

This statement preserves both nonlinearities that the full-face linear
recurrence omits: the coordinate projection and the moving global correction.
The separate proper-prefix chronology lemma preserves the same issues during
support growth.

\begin{theorem}[Conditional terminal logarithmic block; proved here]
\label{thm:terminal-modal-conditional-log}
Assume the conclusions of
\cref{lem:terminal-modal-admission-chronology,lem:terminal-modal-packet,lem:terminal-modal-regime}.
Then, for
all sufficiently large $m$, the named execution has no terminal one-sided
certificate during its first $K_m$ full-face steps.  Hence its terminal block
has length
\begin{equation}
 \Omega\!\left(q^{-1}\log\frac1q\right).
 \label{eq:terminal-modal-conditional-log}
\end{equation}
This conclusion is for the named literal recurrence only.
\end{theorem}

\begin{proof}
By \cref{lem:terminal-modal-profile-implies-quadrature}, the open entry
profiles give \eqref{eq:terminal-modal-entry-target}.
For $1\le s\le H_m$, let $A_s=|G_{2s}|$.  Uniformly on this band and for all
sufficiently large $m$,
\begin{equation}
 \frac{\alpha}{3m}\le A_s\le\frac{2\alpha}{m}.
 \label{eq:terminal-modal-amplitude-bounds}
\end{equation}
Indeed, $(1-q)^m\ge1-mq=15/16$, while
$m(s\pi/m)^2\le\pi^2/(64\log(16m))$ and
$\log\cos z\ge-z^2$ for sufficiently small $z$.

Fix $1\le k\le K_m$.  For $h=2s$, the phase in
\cref{eq:terminal-modal-solution} is $\phi_{2s}=s\pi/m$.  Moreover,
\begin{equation}
 \cos^{k}\frac{H_m\pi}{m}
 \ge \exp\!\left(-\frac{\pi^2}{32}\right)>\frac12.
 \label{eq:terminal-modal-spatial-damping}
\end{equation}
Apply \cref{lem:terminal-modal-cosine-square} at
$x=k\pi/m$.  The ideal part of the band coefficient vector has Euclidean norm
at least
\[
 (1-q)^k\cdot\frac{\alpha}{3m}\cdot\frac12
 \sqrt{\frac{H_m}{16}}
 =\frac{\alpha(1-q)^k\sqrt{H_m}}{24m}.
\]
By \eqref{eq:terminal-modal-entry-target}, its perturbation has norm at most
\[
 (1-q)^k\cdot\frac{2\alpha}{m}\cdot
 \frac{\sqrt{H_m}}{64}
 =\frac{\alpha(1-q)^k\sqrt{H_m}}{32m}.
\]
The reverse triangle inequality therefore gives
\begin{equation}
 \left(\sum_{s=1}^{H_m}a_{2s,k}^2\right)^{1/2}
 \ge\frac{\alpha(1-q)^k\sqrt{H_m}}{96m}.
 \label{eq:terminal-modal-band-energy}
\end{equation}
Parseval and \cref{eq:terminal-modal-popoviciu} imply
\begin{equation}
 R_k\ge
 \frac{\sqrt2\alpha(1-q)^k\sqrt{H_m}}{96m}
 =\frac{\sqrt2}{6}q^3(1-q)^k\sqrt{H_m}.
 \label{eq:terminal-modal-range-lower}
\end{equation}

Because $q\le1/32$,
\[
 (1-q)^{K_m}
 \ge \exp\!\left(-\frac{qK_m}{1-q}\right)
 \ge q^{4/31}.
\]
Also
$\sqrt{H_m}=\Omega(q^{-1/4}(\log(1/q))^{-1/4})$.  Consequently the ratio of
the right side of \eqref{eq:terminal-modal-range-lower} to $q^3$ diverges
like a positive constant times
$q^{-15/124}(\log(1/q))^{-1/4}$.  It is therefore greater than $1/5$ for all
sufficiently small $q$.  The open regime lemma permits
\cref{lem:terminal-modal-range}, which proves non-certification through
$K_m$.
\end{proof}

\begin{remark}[Why one mode is insufficient]
For each fixed $h$, the ideal coefficient is only $\Theta(q^3)$, exactly the
certificate scale, and it can pass through zero.  The gain comes from a band
of $H_m$ orthogonal modes and the phase-uniform
\cref{lem:terminal-modal-cosine-square}; it is not a surviving-mode argument.
The conservative proof constant $1/8$ is not intended to predict the measured
first-certificate constant, which is closer to
$\frac12\log(1/q)$ over the screened sizes.
\end{remark}
